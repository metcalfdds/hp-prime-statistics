// Wait for the finger to lift -- never for more than a second (a stuck
// touch reading must not freeze the program: Lessons Learned 7).
ag_Drain()
BEGIN
  LOCAL drT;
  drT := TICKS;
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Empty the key buffer.
ag_KDrain()
BEGIN
  LOCAL kdK;
  REPEAT kdK := GETKEY; UNTIL kdK == -1;
END;

// Drain, wait for a key OR a tap, drain.  Returns the key, or -1 for a
// tap (its x and y left in agTapX, agTapY).  Holds a page for as long as
// it takes (WAIT(-1) would give up after about a minute).
ag_WaitAny()
BEGIN
  LOCAL waK, waM, waM1;
  ag_KDrain();
  ag_Drain();
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  agTapX := -1; agTapY := -1;
  IF SIZE(waM1) > 0 THEN
    agTapX := waM1(1); agTapY := waM1(2); waK := -1;
  END;
  ag_Drain();
  ag_KDrain();
  RETURN waK;
END;

// One event for a view: a key, or -1 for a touch.  For a touch, the press
// is left in agTapX, agTapY and the last point before the finger lifted in
// agRelX, agRelY, so a drag from one tile to another can be read.  The
// GETKEY + MOUSE polling loop of ChiTable: MOUSE sees a tap on the
// soft-key strip like any other tap (y >= 218).  The finger is followed
// for at most 5 s, then the capped drain (a stuck reading cannot hang it).
ag_Poll()
BEGIN
  LOCAL poK, poM, poM1, poT;
  // drain first: a key left by a CHOOSE, INPUT or MSGBOX must not end
  // the wait before the page is seen, and a finger still down from a
  // menu tap must not count as a tap on the page (Lessons Learned 7;
  // beta 1a, second review item 7)
  ag_KDrain();
  ag_Drain();
  REPEAT
    poK := GETKEY;
    poM := MOUSE;
    poM1 := poM(1);
  UNTIL poK > -1 OR SIZE(poM1) > 0;
  agTapX := -1; agTapY := -1; agRelX := -1; agRelY := -1;
  IF SIZE(poM1) > 0 THEN
    agTapX := poM1(1); agTapY := poM1(2);
    agRelX := agTapX; agRelY := agTapY; poK := -1;
    poT := TICKS;
    REPEAT
      poM := MOUSE;
      poM1 := poM(1);
      IF SIZE(poM1) > 0 THEN agRelX := poM1(1); agRelY := poM1(2); END;
    UNTIL SIZE(poM1) == 0 OR TICKS - poT > 5000;
    ag_Drain();
  ELSE
    ag_KDrain();
  END;
  RETURN poK;
END;

// A message box, then both drains (a MSGBOX opened from a soft key must
// not leave a key or a finger behind for the next page).
ag_Msg(msS)
BEGIN
  MSGBOX(msS);
  ag_KDrain();
  ag_Drain();
END;

// Fixed decimals, rounded first: STRING(x, 2, d) prints SCIENTIFIC when
// 0 < |x| < 10^-d (Lessons Learned 3), and a rounded -0 prints as 0.
// beta 1a: from 1E11 up the number is written by hand as "1.234E11"
// (STRING turns scientific from 1E12, with the device's own E glyph).
ag_Fx(fxV, fxD)
BEGIN
  LOCAL fxR, fxE;
  fxR := ROUND(fxV, fxD);
  IF fxR == 0 THEN fxR := 0; END;
  IF ABS(fxR) >= 1E11 THEN
    fxE := FLOOR(LOG(ABS(fxR)));
    RETURN STRING(ROUND(fxR/10^fxE, 3), 2, 3) + "E" + STRING(fxE, 2, 0);
  END;
  RETURN STRING(fxR, 2, fxD);
END;

// A whole number as text.
ag_I(itV)
BEGIN
  RETURN ag_Fx(itV, 0);
END;

// A number with as many decimals as its size deserves (about 4 digits).
ag_Gn(gnV)
BEGIN
  LOCAL gnA, gnD;
  gnA := ABS(gnV);
  gnD := 4;
  IF gnA >= 1 THEN gnD := 3; END;
  IF gnA >= 10 THEN gnD := 2; END;
  IF gnA >= 1000 THEN gnD := 1; END;
  IF gnA >= 100000 THEN gnD := 0; END;
  RETURN ag_Fx(gnV, gnD);
END;

// A kappa or an ICC: three decimals.
ag_K3(k3V)
BEGIN
  RETURN ag_Fx(k3V, 3);
END;

// An ICC that can run far below -1 (an average-measure form): "< -999"
// beyond that, so a row never overflows.
ag_KS(ksV)
BEGIN
  IF ksV < -999 THEN RETURN "< -999"; END;
  RETURN ag_K3(ksV);
END;

// "p = 0.0670", "p < 0.0001" or "p > 0.9999" (never "p = < ..." and
// never "1.0000": FitCheck's form).
ag_PE(peP)
BEGIN
  IF peP < 0.0001 THEN RETURN "p < 0.0001"; END;
  IF peP > 0.99995 THEN RETURN "p > 0.9999"; END;
  RETURN "p = " + ag_Fx(peP, 4);
END;

// A value of 0 to 1 as a weight: "1", "0", or ".67" (fits a narrow tile).
ag_W2(w2V)
BEGIN
  LOCAL w2S;
  IF w2V >= 0.995 THEN RETURN "1"; END;
  IF w2V < 0.005 THEN RETURN "0"; END;
  w2S := ag_Fx(w2V, 2);
  RETURN MID(w2S, 2);
END;

// A "nice" step, 1, 2 or 5 times a power of ten, at least the value
// handed in (Heckbert), for axis ticks.
ag_Nice(nnV)
BEGIN
  LOCAL nnE, nnF;
  IF nnV <= 0 THEN RETURN 1; END;
  nnE := FLOOR(LOG(nnV));
  nnF := nnV/10^nnE;
  IF nnF <= 1 THEN RETURN 10^nnE; END;
  IF nnF <= 2 THEN RETURN 2*10^nnE; END;
  IF nnF <= 5 THEN RETURN 5*10^nnE; END;
  RETURN 10^(nnE + 1);
END;

// White blended toward the colour (r, g, b) by a share f of 0 to 1:
// the tile shading (more counts, deeper colour).
ag_Mix(mxF, mxR, mxG, mxB)
BEGIN
  LOCAL mxS;
  mxS := MAX(0, MIN(1, mxF));
  RETURN RGB(IP(255 - mxS*(255 - mxR)), IP(255 - mxS*(255 - mxG)), IP(255 - mxS*(255 - mxB)));
END;

// Landis & Koch's (1977) words for ranges of kappa: THEIR stated
// benchmark.  Shown only beside that citation and the words "a
// convention, not a test".
ag_LK(lkV)
BEGIN
  IF lkV < 0 THEN RETURN "poor"; END;
  IF lkV <= 0.2 THEN RETURN "slight"; END;
  IF lkV <= 0.4 THEN RETURN "fair"; END;
  IF lkV <= 0.6 THEN RETURN "moderate"; END;
  IF lkV <= 0.8 THEN RETURN "substantial"; END;
  RETURN "almost perfect";
END;

// Log-gamma, Lanczos g = 7 (LNGAMMA is CAS-only).  StudRange beta 15d's
// sr_LGAM via AnovaLab: accurate to about 1E-13 for x >= 0.5.
// 0.918938533204673 is 0.5*LN(2*pi).
ag_LGam(lgX)
BEGIN
  LOCAL lgC, lgA, lgT, lgJ, lgY;
  lgC := {676.5203681218851, -1259.1392167224028,
      771.32342877765313, -176.61502916214059,
      12.507343278686905, -0.13857109526572012,
      9.9843695780195716E-6, 1.5056327351493116E-7};
  lgY := lgX - 1; lgA := 0.99999999999980993;
  FOR lgJ FROM 1 TO 8 DO
    lgA := lgA + lgC(lgJ)/(lgY + lgJ);
  END;
  lgT := lgY + 7.5;
  RETURN 0.918938533204673 + (lgY + 0.5)*LN(lgT) - lgT + LN(lgA);
END;

// The continued fraction inside the incomplete beta (Numerical Recipes
// 6.4, modified Lentz): StatDist v2.7f's st_betacf via AnovaLab.
ag_BetaCF(bcA, bcB, bcX)
BEGIN
  LOCAL bcQab, bcQap, bcQam, bcC, bcD, bcH;
  LOCAL bcM, bcM2, bcAa, bcDel, bcGo;
  bcQab := bcA + bcB; bcQap := bcA + 1; bcQam := bcA - 1;
  bcC := 1; bcD := 1 - bcQab*bcX/bcQap;
  IF ABS(bcD) < 1E-300 THEN bcD := 1E-300; END;
  bcD := 1/bcD; bcH := bcD;
  bcM := 0; bcGo := 1;
  WHILE bcGo == 1 AND bcM < 300 DO
    bcM := bcM + 1; bcM2 := 2*bcM;
    bcAa := bcM*(bcB - bcM)*bcX/((bcQam + bcM2)*(bcA + bcM2));
    bcD := 1 + bcAa*bcD;
    IF ABS(bcD) < 1E-300 THEN bcD := 1E-300; END;
    bcC := 1 + bcAa/bcC;
    IF ABS(bcC) < 1E-300 THEN bcC := 1E-300; END;
    bcD := 1/bcD; bcH := bcH*bcD*bcC;
    bcAa := -(bcA + bcM)*(bcQab + bcM)*bcX/((bcA + bcM2)*(bcQap + bcM2));
    bcD := 1 + bcAa*bcD;
    IF ABS(bcD) < 1E-300 THEN bcD := 1E-300; END;
    bcC := 1 + bcAa/bcC;
    IF ABS(bcC) < 1E-300 THEN bcC := 1E-300; END;
    bcD := 1/bcD; bcDel := bcD*bcC; bcH := bcH*bcDel;
    IF ABS(bcDel - 1) < 0.00000000001 THEN bcGo := 0; END;
  END;
  RETURN bcH;
END;

// Regularized incomplete beta I_x(a, b), with the symmetry
// I_x(a,b) = 1 - I_(1-x)(b,a) where the fraction is slow.
ag_BetaI(biA, biB, biX)
BEGIN
  LOCAL biL;
  IF biX <= 0 THEN RETURN 0; END;
  IF biX >= 1 THEN RETURN 1; END;
  biL := ag_LGam(biA + biB) - ag_LGam(biA) - ag_LGam(biB) + biA*LN(biX) + biB*LN(1 - biX);
  IF biX < (biA + 1)/(biA + biB + 2) THEN
    RETURN EXP(biL)*ag_BetaCF(biA, biB, biX)/biA;
  END;
  RETURN 1 - EXP(biL)*ag_BetaCF(biB, biA, 1 - biX)/biB;
END;

// A probability kept inside (0, 1) before any _ICDF (an ICDF at 0 or 1
// returns infinity with no error: Lessons Learned 3).
ag_Clp(cpP)
BEGIN
  RETURN MIN(MAX(cpP, 1E-12), 0.999999999999);
END;

// P(F > f) for F(a, b), computed as a LOWER tail (12-digit reals: 1 - CDF
// is exactly 0 below about 5E-13).  Whole df: the mirror P(F(a,b) > f) =
// P(F(b,a) < 1/f) with the built-in FISHER_CDF.  Fractional df: z =
// b/(a f + b) is Beta(b/2, a/2), so P(F > f) = I_z(b/2, a/2).
ag_FUp(fuF, fuA, fuB)
BEGIN
  IF fuF <= 0 THEN RETURN 1; END;
  IF FP(fuA) == 0 AND FP(fuB) == 0 THEN RETURN FISHER_CDF(fuB, fuA, 1/fuF); END;
  RETURN ag_BetaI(fuB/2, fuA/2, fuB/(fuA*fuF + fuB));
END;

// Right-tail F quantile: the x with P(F(n, d) > x) = a.  Whole df: the
// mirror 1 / FISHER_ICDF(d, n, a), never FISHER_ICDF(n, d, 1 - a).  A
// fractional df (ICC(2,1)'s Satterthwaite df) is "Invalid input" for the
// built-in, so then: P(F > x) = I_y(d/2, n/2) with y = d/(d + n x), which
// rises with y.  Newton's method on y (the derivative is the Beta(d/2,
// n/2) density), started from the quantile at the rounded df and kept
// inside a shrinking bracket (a step that leaves it is a halving), stops
// when y moves by less than 1E-12 of itself: about 3 to 6 incomplete-beta
// calls, against 60 for plain halving.  Above 1000 df the df is rounded
// to a whole number (the quantile moves by less than 1E-4 of itself;
// verify_AgreeLab_beta1.py).
ag_FQ(fqA, fqN, fqD)
BEGIN
  LOCAL fqLo, fqHi, fqI, fqY, fqNn, fqDd;
  LOCAL fqA2, fqB2, fqLB, fqG, fqP, fqT;
  LOCAL fqGo;
  fqNn := fqN; fqDd := fqD;
  IF FP(fqNn) <> 0 AND fqNn > 1000 THEN fqNn := ROUND(fqNn, 0); END;
  IF FP(fqDd) <> 0 AND fqDd > 1000 THEN fqDd := ROUND(fqDd, 0); END;
  IF FP(fqNn) == 0 AND FP(fqDd) == 0 THEN RETURN 1/FISHER_ICDF(fqDd, fqNn, ag_Clp(fqA)); END;
  fqA2 := fqDd/2; fqB2 := fqNn/2;
  fqLB := ag_LGam(fqA2 + fqB2) - ag_LGam(fqA2) - ag_LGam(fqB2);
  fqT := 1/FISHER_ICDF(MAX(1, ROUND(fqDd, 0)), MAX(1, ROUND(fqNn, 0)), ag_Clp(fqA));
  fqY := fqDd/(fqDd + fqNn*fqT);
  IF fqY <= 0 OR fqY >= 1 THEN fqY := 0.5; END;
  fqLo := 0; fqHi := 1; fqI := 0; fqGo := 1;
  WHILE fqGo == 1 AND fqI < 60 DO
    fqI := fqI + 1;
    fqG := ag_BetaI(fqA2, fqB2, fqY) - fqA;
    IF fqG < 0 THEN fqLo := fqY; ELSE fqHi := fqY; END;
    fqP := EXP(fqLB + (fqA2 - 1)*LN(fqY) + (fqB2 - 1)*LN(1 - fqY));
    fqT := (fqLo + fqHi)/2;
    IF fqP > 0 THEN
      fqT := fqY - fqG/fqP;
      IF fqT <= fqLo OR fqT >= fqHi THEN fqT := (fqLo + fqHi)/2; END;
    END;
    IF ABS(fqT - fqY) <= 1E-12*fqY THEN fqGo := 0; END;
    fqY := fqT;
  END;
  RETURN fqDd*(1 - fqY)/(fqNn*fqY);
END;

// Right-tail t quantile: P(T > t) = a, as -STUDENT_ICDF(df, a), never
// STUDENT_ICDF(df, 1 - a).
ag_TQ(tqA, tqD)
BEGIN
  RETURN -STUDENT_ICDF(tqD, ag_Clp(tqA));
END;

// The agreement weight of cell (i, j) in a k x k table: wt 1 linear,
// 1 - |i - j|/(k - 1) (Cicchetti & Allison 1971); wt 2 quadratic,
// 1 - (i - j)^2/(k - 1)^2 (Fleiss & Cohen 1973).  1 on the diagonal.
ag_Wt(wtI, wtJ, wtK, wtW)
BEGIN
  IF wtW == 1 THEN RETURN 1 - ABS(wtI - wtJ)/(wtK - 1); END;
  RETURN 1 - (wtI - wtJ)^2/(wtK - 1)^2;
END;

// Weighted kappa (Cohen 1968) for both weightings, index 1 linear and
// 2 quadratic, from the row and column shares ag_TCompute left.
//   po_w = sum w_ij p_ij;  pe_w = sum w_ij p_i. p_.j;
//   kappa_w = (po_w - pe_w)/(1 - pe_w).
//   SE (Fleiss, Cohen & Everitt 1969), with wr_i = sum_j p_.j w_ij and
//   wc_j = sum_i p_i. w_ij:
//     var  = [sum p_ij (w_ij - (wr_i + wc_j)(1 - kw))^2
//             - (kw - pe_w(1 - kw))^2] / (N (1 - pe_w)^2)
//     var0 = [sum p_i. p_.j (w_ij - (wr_i + wc_j))^2 - pe_w^2]
//             / (N (1 - pe_w)^2)   (under H0, for the z test)
ag_WCompute()
BEGIN
  LOCAL wcW, wcI, wcJ, wcK, wcPo, wcPe;
  LOCAL wcWr, wcWc, wcS, wcS0, wcKw, wcV;
  LOCAL wcN, wcPij;
  wcK := agK; wcN := agTN;
  agWOK := 1;
  FOR wcW FROM 1 TO 2 DO
    wcPo := 0; wcPe := 0;
    wcWr := MAKELIST(0, wcI, 1, wcK); wcWc := MAKELIST(0, wcI, 1, wcK);
    FOR wcI FROM 1 TO wcK DO
      FOR wcJ FROM 1 TO wcK DO
        wcV := ag_Wt(wcI, wcJ, wcK, wcW);
        wcPo := wcPo + wcV*agT((wcI - 1)*wcK + wcJ)/wcN;
        wcPe := wcPe + wcV*agRw(wcI)*agCl(wcJ);
        wcWr(wcI) := wcWr(wcI) + agCl(wcJ)*wcV;
        wcWc(wcJ) := wcWc(wcJ) + agRw(wcI)*wcV;
      END;
    END;
    agWPo(wcW) := wcPo; agWPe(wcW) := wcPe;
    IF wcPe > 0.999999999 THEN
      agWOK := 0;
      agWK(wcW) := 0; agWSE(wcW) := 0; agWSE0(wcW) := 0; agWZ(wcW) := 0; agWP(wcW) := -1;
    ELSE
      wcKw := (wcPo - wcPe)/(1 - wcPe);
      wcS := 0; wcS0 := 0;
      FOR wcI FROM 1 TO wcK DO
        FOR wcJ FROM 1 TO wcK DO
          wcV := ag_Wt(wcI, wcJ, wcK, wcW);
          wcPij := agT((wcI - 1)*wcK + wcJ)/wcN;
          wcS := wcS + wcPij*(wcV - (wcWr(wcI) + wcWc(wcJ))*(1 - wcKw))^2;
          wcS0 := wcS0 + agRw(wcI)*agCl(wcJ)*(wcV - (wcWr(wcI) + wcWc(wcJ)))^2;
        END;
      END;
      wcS := wcS - (wcKw - wcPe*(1 - wcKw))^2;
      wcS0 := wcS0 - wcPe^2;
      agWK(wcW) := wcKw;
      agWSE(wcW) := SQRT(MAX(0, wcS)/(wcN*(1 - wcPe)^2));
      agWSE0(wcW) := SQRT(MAX(0, wcS0)/(wcN*(1 - wcPe)^2));
      IF agWSE0(wcW) > 0 THEN
        agWZ(wcW) := wcKw/agWSE0(wcW);
        agWP(wcW) := NORMALD_CDF(0, 1, -agWZ(wcW));
      ELSE
        agWZ(wcW) := 0; agWP(wcW) := -1;
      END;
    END;
  END;
  RETURN agWOK;
END;

// Cohen's kappa for the k x k table agT (rows rater A, columns rater B).
//   po = sum of the diagonal shares; pe = sum of row share x column share;
//   kappa = (po - pe)/(1 - pe)  (Cohen 1960).
//   SE (Fleiss, Cohen & Everitt 1969, large-sample, NOT under H0), with
//   p_ij the cell shares, p_i. the row and p_.j the column shares:
//     var = [ sum_i p_ii (1 - (p_i. + p_.i)(1 - kappa))^2
//           + (1 - kappa)^2 sum_(i<>j) p_ij (p_.i + p_j.)^2
//           - (kappa - pe(1 - kappa))^2 ] / (N (1 - pe)^2)
//   SE under H0 (for the z test):
//     var0 = [pe + pe^2 - sum_i p_i. p_.i (p_i. + p_.i)] / (N (1 - pe)^2)
//   kappa max (Cohen 1960, p 42): po max = sum_i min(p_i., p_.i), the most
//   agreement the margins allow.  PABAK = (k po - 1)/(k - 1) (Byrt et al.
//   1993; 2 po - 1 for a 2 x 2).  2 x 2 only: prevalence index (a - d)/N,
//   bias index (b - c)/N.
// Sets agTOK: 1 computed, 0 empty table, 2 kappa undefined (pe = 1).
ag_TCompute()
BEGIN
  LOCAL tcI, tcJ, tcN, tcS, tcS0, tcA;
  LOCAL tcK, tcPij;
  tcK := agK;
  agRw := MAKELIST(0, tcI, 1, tcK); agCl := MAKELIST(0, tcI, 1, tcK);
  tcN := 0;
  FOR tcI FROM 1 TO tcK DO
    FOR tcJ FROM 1 TO tcK DO
      tcA := agT((tcI - 1)*tcK + tcJ);
      tcN := tcN + tcA;
      agRw(tcI) := agRw(tcI) + tcA;
      agCl(tcJ) := agCl(tcJ) + tcA;
    END;
  END;
  agTN := tcN; agTOK := 0; agWOK := 0;
  agPo := 0; agPe := 0; agPoMx := 0; agKa := 0; agSE := 0; agSE0 := 0;
  agZ := 0; agP := 1; agLo := 0; agHi := 0; agKmx := 0; agPab := 0;
  agPI := 0; agBI := 0;
  IF tcN <= 0 THEN RETURN 0; END;
  FOR tcI FROM 1 TO tcK DO
    agRw(tcI) := agRw(tcI)/tcN;
    agCl(tcI) := agCl(tcI)/tcN;
  END;
  FOR tcI FROM 1 TO tcK DO
    agPo := agPo + agT((tcI - 1)*tcK + tcI)/tcN;
    agPe := agPe + agRw(tcI)*agCl(tcI);
    agPoMx := agPoMx + MIN(agRw(tcI), agCl(tcI));
  END;
  IF tcK == 2 THEN
    agPI := (agT(1) - agT(4))/tcN;
    agBI := (agT(2) - agT(3))/tcN;
  END;
  agPab := (tcK*agPo - 1)/(tcK - 1);
  IF agPe > 0.999999999 THEN agTOK := 2; RETURN 0; END;
  agKa := (agPo - agPe)/(1 - agPe);
  agKmx := (agPoMx - agPe)/(1 - agPe);
  tcS := 0;
  FOR tcI FROM 1 TO tcK DO
    FOR tcJ FROM 1 TO tcK DO
      tcPij := agT((tcI - 1)*tcK + tcJ)/tcN;
      IF tcI == tcJ THEN
        tcS := tcS + tcPij*(1 - (agRw(tcI) + agCl(tcI))*(1 - agKa))^2;
      ELSE
        tcS := tcS + (1 - agKa)^2*tcPij*(agCl(tcI) + agRw(tcJ))^2;
      END;
    END;
  END;
  tcS := tcS - (agKa - agPe*(1 - agKa))^2;
  agSE := SQRT(MAX(0, tcS)/(tcN*(1 - agPe)^2));
  tcS0 := agPe + agPe^2;
  FOR tcI FROM 1 TO tcK DO
    tcS0 := tcS0 - agRw(tcI)*agCl(tcI)*(agRw(tcI) + agCl(tcI));
  END;
  agSE0 := SQRT(MAX(0, tcS0)/(tcN*(1 - agPe)^2));
  agLo := agKa - 1.95996398454*agSE;
  agHi := agKa + 1.95996398454*agSE;
  // beta 1a (second review item 6): the large-sample SE, CI and p are
  // given only when N >= 10, every row and column total is above 0, and
  // SE > 0; the CI is clipped to [-1, 1] (agTClp = 1 says it was)
  agTInf := 1; agTWhy := ""; agTClp := 0;
  FOR tcI FROM 1 TO tcK DO
    IF agRw(tcI) <= 0 OR agCl(tcI) <= 0 THEN agTInf := 0; agTWhy := "an empty margin"; END;
  END;
  IF agSE <= 0 THEN agTInf := 0; agTWhy := "SE = 0"; END;
  IF tcN < 10 THEN agTInf := 0; agTWhy := "N < 10"; END;
  IF agLo < -1 THEN agLo := -1; agTClp := 1; END;
  IF agHi > 1 THEN agHi := 1; agTClp := 1; END;
  IF agSE0 > 0 THEN
    agZ := agKa/agSE0;
    agP := NORMALD_CDF(0, 1, -agZ);
  ELSE
    agZ := 0; agP := -1;
  END;
  agTOK := 1;
  ag_WCompute();
  RETURN 1;
END;

// Fleiss' kappa (Fleiss 1971) for agF: N subjects (rows) x q categories,
// n_ij = how many of the m raters put subject i in category j.
//   p_j = sum_i n_ij / (N m);  P_i = (sum_j n_ij^2 - m)/(m(m - 1)), the
//   share of subject i's rater pairs that agree;  P-bar = mean P_i;
//   Pe = sum p_j^2;  kappa = (P-bar - Pe)/(1 - Pe).
//   Per category: kappa_j = 1 - sum_i n_ij (m - n_ij) / (N m (m-1) p_j q_j).
//   SE under H0 only (Fleiss, Nee & Landis 1979):
//     var0 = 2/(N m (m-1)) x [(sum p_j q_j)^2 - sum p_j q_j (q_j - p_j)]
//            / (sum p_j q_j)^2;   SE0(kappa_j) = sqrt(2/(N m (m-1))).
//   kappa_j is set to 99 where p_j is 0 or 1 (undefined).
// Sets agFOK: 1 computed, 2 undefined (Pe = 1: every rating in one category).
ag_FCompute()
BEGIN
  LOCAL fcI, fcJ, fcN, fcQ, fcM, fcA;
  LOCAL fcS, fcSpq, fcT, fcPq, fcV;
  fcN := agFN; fcQ := agFQ; fcM := agFM;
  agFPj := MAKELIST(0, fcJ, 1, fcQ);
  agFPi := MAKELIST(0, fcI, 1, fcN);
  agFKj := MAKELIST(99, fcJ, 1, fcQ);
  agFOK := 0; agFK := 0; agFSE0 := 0; agFZ := 0; agFP := 1;
  agFPb := 0; agFPe := 0;
  FOR fcI FROM 1 TO fcN DO
    fcS := 0;
    FOR fcJ FROM 1 TO fcQ DO
      fcA := agF((fcI - 1)*fcQ + fcJ);
      agFPj(fcJ) := agFPj(fcJ) + fcA;
      fcS := fcS + fcA^2;
    END;
    agFPi(fcI) := (fcS - fcM)/(fcM*(fcM - 1));
    agFPb := agFPb + agFPi(fcI);
  END;
  agFPb := agFPb/fcN;
  FOR fcJ FROM 1 TO fcQ DO
    agFPj(fcJ) := agFPj(fcJ)/(fcN*fcM);
    agFPe := agFPe + agFPj(fcJ)^2;
  END;
  agFSEj := SQRT(2/(fcN*fcM*(fcM - 1)));
  IF agFPe > 0.999999999 THEN agFOK := 2; RETURN 0; END;
  agFK := (agFPb - agFPe)/(1 - agFPe);
  fcSpq := 0; fcT := 0;
  FOR fcJ FROM 1 TO fcQ DO
    fcPq := agFPj(fcJ)*(1 - agFPj(fcJ));
    fcSpq := fcSpq + fcPq;
    fcT := fcT + fcPq*((1 - agFPj(fcJ)) - agFPj(fcJ));
    IF fcPq > 0 THEN
      fcV := 0;
      FOR fcI FROM 1 TO fcN DO
        fcA := agF((fcI - 1)*fcQ + fcJ);
        fcV := fcV + fcA*(fcM - fcA);
      END;
      agFKj(fcJ) := 1 - fcV/(fcN*fcM*(fcM - 1)*fcPq);
    END;
  END;
  agFSE0 := SQRT(MAX(0, 2/(fcN*fcM*(fcM - 1))*(fcSpq^2 - fcT)/fcSpq^2));
  IF agFSE0 > 0 THEN
    agFZ := agFK/agFSE0;
    agFP := NORMALD_CDF(0, 1, -agFZ);
  ELSE
    agFP := -1;
  END;
  agFOK := 1;
  RETURN 1;
END;

// The intraclass correlations (Shrout & Fleiss 1979; McGraw & Wong 1996)
// for agY: n targets (rows) x k raters (columns), from the two-way ANOVA:
//   MSR targets (rows), MSC raters (columns), MSE residual, MSW within
//   targets = (SSC + SSE)/(n(k - 1)).
//   ICC(1,1) = (MSR - MSW)/(MSR + (k-1) MSW)          one-way random
//   ICC(2,1) = (MSR - MSE)/(MSR + (k-1) MSE + k(MSC - MSE)/n)
//                                       two-way random, absolute agreement
//   ICC(3,1) = (MSR - MSE)/(MSR + (k-1) MSE)   two-way mixed, consistency
//   ICC(1,k) = (MSR - MSW)/MSR;  ICC(2,k) = (MSR - MSE)/(MSR + (MSC-MSE)/n);
//   ICC(3,k) = (MSR - MSE)/MSR   (the mean of the k ratings).
// 95% CIs: ICC(1): F0 = MSR/MSW on (n-1, n(k-1)); ICC(3): F0 = MSR/MSE on
//   (n-1, (n-1)(k-1)); FL = F0/F(.025; df1, df2), FU = F0 x F(.025; df2,
//   df1); single (FL-1)/(FL+k-1) to (FU-1)/(FU+k-1); average 1 - 1/FL to
//   1 - 1/FU.  ICC(2,1): Satterthwaite's df
//     a = k r/(n(1-r)), b = 1 + k r (n-1)/(n(1-r)), r = ICC(2,1),
//     v = (a MSC + b MSE)^2 / ((a MSC)^2/(k-1) + (b MSE)^2/((n-1)(k-1))),
//     F* = F(.025; n-1, v), F** = F(.025; v, n-1),
//     L = n(MSR - F* MSE)/(F*(k MSC + (kn - k - n) MSE) + n MSR),
//     U = n(F** MSR - MSE)/(k MSC + (kn - k - n) MSE + n F** MSR);
//   ICC(2,k): L k/(1 + (k-1) L) to U k/(1 + (k-1) U).
//   F(a; d1, d2) is the right-tail quantile.  The F test of each ICC = 0
//   uses MSR/MSW for ICC(1), MSR/MSE for ICC(2) and ICC(3) (McGraw & Wong
//   1996, Table 8).  agIDf(r) is 1 when form r's denominator is not zero
//   (else agIC(r) = -99 and "undefined (denominator 0)"); agICf(r) is 1 when
//   it has an interval, else agINo(r) says why (an infinite or zero F,
//   ICC(2,1) outside (-1, 1), Satterthwaite's v < 1).  AG_ICC holds -99 for
//   an undefined form.
//   ICC(2,k)'s transform L k/(1 + (k-1) L) has no finite value once
//   L <= -1/(k-1): that bound is then minus infinity, stored as -1E99.
// Sets agIOK: 1 computed, 2 undefined (the reason, as text, in agICI).
ag_ICompute()
BEGIN
  LOCAL icI, icJ, icN, icK, icG, icRm;
  LOCAL icCm, icSR, icSC, icST, icSE, icA;
  LOCAL icMR, icMC, icME, icMW, icF1, icF3;
  LOCAL icFL, icFU, icR, icAa, icBb, icV;
  LOCAL icFs, icFss, icL, icU, icD, icT;
  LOCAL icDn, icS;
  icN := agYN; icK := agYK;
  agIOK := 0; agICI := "";
  agIC := MAKELIST(0, icI, 1, 6); agIL := MAKELIST(-99, icI, 1, 6);
  agIU := MAKELIST(-99, icI, 1, 6); agIF := MAKELIST(0, icI, 1, 6);
  agID1 := MAKELIST(0, icI, 1, 6); agID2 := MAKELIST(0, icI, 1, 6);
  agIPv := MAKELIST(1, icI, 1, 6); agMS := {0, 0, 0, 0}; agIV := 0;
  agICf := {1, 1, 1, 1, 1, 1};
  icRm := MAKELIST(0, icI, 1, icN); icCm := MAKELIST(0, icI, 1, icK);
  icG := 0;
  FOR icI FROM 1 TO icN DO
    FOR icJ FROM 1 TO icK DO
      icA := agY((icI - 1)*icK + icJ);
      icRm(icI) := icRm(icI) + icA/icK;
      icCm(icJ) := icCm(icJ) + icA/icN;
      icG := icG + icA/(icN*icK);
    END;
  END;
  icSR := 0; icSC := 0; icST := 0;
  FOR icI FROM 1 TO icN DO icSR := icSR + icK*(icRm(icI) - icG)^2; END;
  FOR icJ FROM 1 TO icK DO icSC := icSC + icN*(icCm(icJ) - icG)^2; END;
  FOR icI FROM 1 TO icN DO
    FOR icJ FROM 1 TO icK DO
      icST := icST + (agY((icI - 1)*icK + icJ) - icG)^2;
    END;
  END;
  icSE := MAX(0, icST - icSR - icSC);
  // beta 1a (second review items 3, 4): a sum of squares below 1E-10 of
  // the total is zero to 12-digit arithmetic (a constant offset leaves
  // rounding dust in SSE); every point estimate is then computed from its
  // formula wherever its denominator is not zero
  icD := 1E-10*MAX(icST, 1E-300);
  IF icSR <= icD THEN icSR := 0; END;
  IF icSC <= icD THEN icSC := 0; END;
  IF icSE <= icD THEN icSE := 0; END;
  icMR := icSR/(icN - 1); icMC := icSC/(icK - 1);
  icME := icSE/((icN - 1)*(icK - 1)); icMW := (icSC + icSE)/(icN*(icK - 1));
  agMS := {icMR, icMC, icME, icMW};
  IF icST <= 0 THEN agIOK := 2; agICI := "every rating is the same: nothing to correlate"; RETURN 0; END;
  agIDf := {1, 1, 1, 1, 1, 1}; agICf := {0, 0, 0, 0, 0, 0}; agINo := {"", "", "", "", "", ""};
  // a denominator at or below 1E-10 of the largest mean square is zero
  icT := 1E-10*MAX(MAX(icMR, icMC), MAX(icME, icMW));
  icDn := icMR + (icK - 1)*icMW;
  IF icDn > icT THEN agIC(1) := (icMR - icMW)/icDn; ELSE agIDf(1) := 0; END;
  icDn := icMR + (icK - 1)*icME + icK*(icMC - icME)/icN;
  IF icDn > icT THEN agIC(2) := (icMR - icME)/icDn; ELSE agIDf(2) := 0; END;
  icDn := icMR + (icK - 1)*icME;
  IF icDn > icT THEN agIC(3) := (icMR - icME)/icDn; ELSE agIDf(3) := 0; END;
  icDn := icMR;
  IF icDn > icT THEN agIC(4) := (icMR - icMW)/icDn; agIC(6) := (icMR - icME)/icDn; ELSE agIDf(4) := 0; agIDf(6) := 0; END;
  icDn := icMR + (icMC - icME)/icN;
  IF icDn > icT THEN agIC(5) := (icMR - icME)/icDn; ELSE agIDf(5) := 0; END;
  FOR icI FROM 1 TO 6 DO
    IF agIDf(icI) == 0 THEN agIC(icI) := -99; agINo(icI) := "undefined (denominator 0)"; END;
  END;
  // the F statistics; -1 stands for infinite (a zero denominator MS)
  icF1 := -1; IF icMW > 0 THEN icF1 := icMR/icMW; END;
  icF3 := -1; IF icME > 0 THEN icF3 := icMR/icME; END;
  ag_Busy("ICC intervals (F quantiles)");
  // ICC(1,1) and ICC(1,k): an interval needs 0 < F < infinity
  IF icF1 > 0 THEN
    icFL := icF1/ag_FQ(0.025, icN - 1, icN*(icK - 1));
    icFU := icF1*ag_FQ(0.025, icN*(icK - 1), icN - 1);
    agIL(1) := (icFL - 1)/(icFL + icK - 1); agIU(1) := (icFU - 1)/(icFU + icK - 1);
    agIL(4) := 1 - 1/icFL; agIU(4) := 1 - 1/icFU;
    agICf(1) := agIDf(1); agICf(4) := agIDf(4);
  ELSE
    IF icF1 == -1 THEN icS := "no F interval (within MS = 0)"; ELSE icS := "no F interval (target MS = 0)"; END;
    IF agIDf(1) == 1 THEN agINo(1) := icS; END;
    IF agIDf(4) == 1 THEN agINo(4) := icS; END;
  END;
  // ICC(3,1) and ICC(3,k)
  IF icF3 > 0 THEN
    icFL := icF3/ag_FQ(0.025, icN - 1, (icN - 1)*(icK - 1));
    icFU := icF3*ag_FQ(0.025, (icN - 1)*(icK - 1), icN - 1);
    agIL(3) := (icFL - 1)/(icFL + icK - 1); agIU(3) := (icFU - 1)/(icFU + icK - 1);
    agIL(6) := 1 - 1/icFL; agIU(6) := 1 - 1/icFU;
    agICf(3) := agIDf(3); agICf(6) := agIDf(6);
  ELSE
    IF icF3 == -1 THEN icS := "no F interval (error MS = 0)"; ELSE icS := "no F interval (target MS = 0)"; END;
    IF agIDf(3) == 1 THEN agINo(3) := icS; END;
    IF agIDf(6) == 1 THEN agINo(6) := icS; END;
  END;
  // ICC(2,1) and ICC(2,k): Satterthwaite's df v
  icR := agIC(2);
  IF agIDf(2) == 1 AND icMR > 0 AND icR < 1 AND icR > -1 THEN
    icAa := icK*icR/(icN*(1 - icR));
    icBb := 1 + icK*icR*(icN - 1)/(icN*(1 - icR));
    icV := (icAa*icMC)^2/(icK - 1) + (icBb*icME)^2/((icN - 1)*(icK - 1));
    IF icV > 0 THEN icV := (icAa*icMC + icBb*icME)^2/icV; END;
    agIV := icV;
    IF icV >= 1 THEN
      icFs := ag_FQ(0.025, icN - 1, icV);
      icFss := ag_FQ(0.025, icV, icN - 1);
      icL := icN*(icMR - icFs*icME)/(icFs*(icK*icMC + (icK*icN - icK - icN)*icME) + icN*icMR);
      icU := icN*(icFss*icMR - icME)/(icK*icMC + (icK*icN - icK - icN)*icME + icN*icFss*icMR);
      agIL(2) := icL; agIU(2) := icU;
      agIL(5) := -1E99; agIU(5) := -1E99;
      IF 1 + icL*(icK - 1) > 0 THEN agIL(5) := icL*icK/(1 + icL*(icK - 1)); END;
      IF 1 + icU*(icK - 1) > 0 THEN agIU(5) := icU*icK/(1 + icU*(icK - 1)); END;
      agICf(2) := 1; agICf(5) := agIDf(5);
    END;
  END;
  IF agIDf(2) == 1 AND agICf(2) == 0 THEN agINo(2) := "no F interval"; END;
  IF agIDf(5) == 1 AND agICf(5) == 0 THEN agINo(5) := "no F interval"; END;
  // F tests of ICC = 0 (an infinite F: p = 0, shown as p < 0.0001)
  FOR icI FROM 1 TO 6 DO
    IF icI == 1 OR icI == 4 THEN
      agIF(icI) := icF1; agID1(icI) := icN - 1; agID2(icI) := icN*(icK - 1);
    ELSE
      agIF(icI) := icF3; agID1(icI) := icN - 1; agID2(icI) := (icN - 1)*(icK - 1);
    END;
    IF agIF(icI) == -1 THEN agIPv(icI) := 0; ELSE agIPv(icI) := ag_FUp(agIF(icI), agID1(icI), agID2(icI)); END;
  END;
  agIOK := 1;
  AG_ICC := agIC;
  RETURN 1;
END;

// Bland-Altman for raters (columns) agBA and agBB of agY.
//   d_i = A_i - B_i, m_i = (A_i + B_i)/2; bias = mean d, s = SD of d (n - 1);
//   limits of agreement bias -/+ 1.96 s (Bland & Altman 1986; their paper
//   rounds 1.96 to 2); t = t(.025, n - 1);
//   CI for the bias: bias -/+ t s/sqrt(n);
//   CI for a limit: limit -/+ t s sqrt(1/n + 1.96^2/(2(n - 1)))
//     (Bland & Altman 1999, section 2.2; about sqrt(3 s^2/n) in 1986);
//   proportional bias: least squares d = b0 + b1 m, t = b1/SE(b1) on
//   n - 2 df (Bland & Altman 1999, section 3.2).
// Sets agBOK: 1 computed, 2 not possible (reason in agBWhy).
ag_BCompute()
BEGIN
  LOCAL bcI, bcN, bcS, bcMx, bcSxx, bcSxy;
  LOCAL bcR, bcK;
  bcN := agYN; bcK := agYK;
  agBOK := 0; agBWhy := ""; agBN := bcN; agBCa := 0;
  agBD := MAKELIST(0, bcI, 1, bcN); agBM := MAKELIST(0, bcI, 1, bcN);
  FOR bcI FROM 1 TO bcN DO
    agBD(bcI) := agY((bcI - 1)*bcK + agBA) - agY((bcI - 1)*bcK + agBB);
    agBM(bcI) := (agY((bcI - 1)*bcK + agBA) + agY((bcI - 1)*bcK + agBB))/2;
  END;
  IF bcN < 3 THEN agBOK := 2; agBWhy := "need 3 or more targets"; RETURN 0; END;
  agBDb := 0; bcMx := 0;
  FOR bcI FROM 1 TO bcN DO agBDb := agBDb + agBD(bcI)/bcN; bcMx := bcMx + agBM(bcI)/bcN; END;
  bcS := 0; bcSxx := 0; bcSxy := 0;
  FOR bcI FROM 1 TO bcN DO
    bcS := bcS + (agBD(bcI) - agBDb)^2;
    bcSxx := bcSxx + (agBM(bcI) - bcMx)^2;
    bcSxy := bcSxy + (agBM(bcI) - bcMx)*(agBD(bcI) - agBDb);
  END;
  agBS := SQRT(bcS/(bcN - 1));
  agBLo := agBDb - 1.96*agBS; agBHi := agBDb + 1.96*agBS;
  agBT := ag_TQ(0.025, bcN - 1);
  agBSeb := agBS/SQRT(bcN);
  agBSL := agBS*SQRT(1/bcN + 1.96^2/(2*(bcN - 1)));
  agB1 := 0; agB0 := agBDb; agBSe1 := -1; agBTb := 0; agBPb := 1;
  IF bcSxx > 0 THEN
    agB1 := bcSxy/bcSxx;
    agB0 := agBDb - agB1*bcMx;
    bcR := 0;
    FOR bcI FROM 1 TO bcN DO bcR := bcR + (agBD(bcI) - agB0 - agB1*agBM(bcI))^2; END;
    agBSe1 := SQRT(bcR/(bcN - 2)/bcSxx);
    IF agBSe1 > 0 THEN
      agBTb := agB1/agBSe1;
      agBPb := MIN(1, 2*STUDENT_CDF(bcN - 2, -ABS(agBTb)));
    ELSE
      agBTb := 0; agBPb := -1;
    END;
  END;
  agBOK := 1;
  AG_LOA := {agBLo, agBDb, agBHi};
  RETURN 1;
END;

// Tile geometry: k x k tiles of agCZ pixels from (agGX0, agGY0); the
// drawing and the touch handler both read these.  21 px at k = 6.
ag_TGeom()
BEGIN
  agCZ := MIN(34, IP(126/agK));
  agGX0 := 22; agGY0 := 34;
END;

// The tile (as a flat index (i - 1) k + j) under the point (x, y), or 0.
ag_TCell(tcX, tcY)
BEGIN
  LOCAL tcI, tcJ;
  IF tcX < agGX0 OR tcY < agGY0 THEN RETURN 0; END;
  tcJ := IP((tcX - agGX0)/agCZ) + 1;
  tcI := IP((tcY - agGY0)/agCZ) + 1;
  IF tcI > agK OR tcJ > agK THEN RETURN 0; END;
  RETURN (tcI - 1)*agK + tcJ;
END;

// "A2, B1": a tile's name (row = rater A's category, column = rater B's).
ag_TNm(tnI, tnJ)
BEGIN
  RETURN "A" + ag_I(tnI) + ",B" + ag_I(tnJ);
END;

// The source line of the two-rater table: "Data: <source>", or, once a
// preset has been edited (beta 1a, second review item 2), "edited from

ag_TSrcL()

// The grid.  gdM = 1: counts, tiles shaded by count (diagonal green,
// off-diagonal red), with row and column totals and the selected tile
// outlined; gdM = 2: the agreement weights of agWt, shaded blue.
ag_TGrid(gdM)
BEGIN
  LOCAL gdI, gdJ, gdX, gdY, gdN, gdMx;
  LOCAL gdF, gdC, gdS, gdTc, gdE;
  ag_TGeom();
  gdMx := 1;
  FOR gdI FROM 1 TO agK*agK DO gdMx := MAX(gdMx, agT(gdI)); END;
  ag_Tx("A|B", 1, 22, agGRAY);
  FOR gdJ FROM 1 TO agK DO
    ag_Tx(ag_I(gdJ), agGX0 + (gdJ - 1)*agCZ + IP(agCZ/2) - 4, 22, agBLUE);
  END;
  FOR gdI FROM 1 TO agK DO
    ag_Tx(ag_I(gdI), 8, agGY0 + (gdI - 1)*agCZ + IP(agCZ/2) - 6, agBLUE);
    FOR gdJ FROM 1 TO agK DO
      gdX := agGX0 + (gdJ - 1)*agCZ; gdY := agGY0 + (gdI - 1)*agCZ;
      gdN := agT((gdI - 1)*agK + gdJ);
      IF gdM == 1 THEN
        gdF := 0;
        IF gdN > 0 THEN gdF := 0.15 + 0.85*gdN/gdMx; END;
        IF gdI == gdJ THEN
          gdC := ag_Mix(gdF, 0, 140, 60); gdE := agGRN;
        ELSE
          gdC := ag_Mix(gdF, 220, 40, 40); gdE := agLGRY;
        END;
        gdS := ag_I(gdN);
      ELSE
        gdF := ag_Wt(gdI, gdJ, agK, agWt);
        gdC := ag_Mix(0.85*gdF, 0, 60, 200); gdE := agLGRY;
        gdS := ag_W2(gdF);
      END;
      RECT_P(G1, gdX, gdY, gdX + agCZ - 2, gdY + agCZ - 2, gdE, gdC);
      gdTc := #000000h;
      IF gdF > 0.6 THEN gdTc := #FFFFFFh; END;
      TEXTOUT_P(gdS, G1, gdX + MAX(1, IP((agCZ - 2)/2 - 3.2*DIM(gdS))), gdY + IP(agCZ/2) - 6, 1, gdTc, agCZ - 2);
    END;
  END;
  IF gdM == 1 THEN
    // the selected tile, outlined twice in red
    gdX := agGX0 + (agSelJ - 1)*agCZ; gdY := agGY0 + (agSel - 1)*agCZ;
    ag_Box(gdX - 1, gdY - 1, gdX + agCZ - 1, gdY + agCZ - 1, agRED);
    ag_Box(gdX - 2, gdY - 2, gdX + agCZ, gdY + agCZ, agRED);
    // row totals (right) and column totals (below), in grey
    FOR gdI FROM 1 TO agK DO
      gdS := ag_I(agRw(gdI)*MAX(agTN, 0));
      IF agTN <= 0 THEN gdS := "0"; END;
      ag_Tx(gdS, agGX0 + agK*agCZ + 2, agGY0 + (gdI - 1)*agCZ + IP(agCZ/2) - 6, agGRAY);
      gdS := ag_I(agCl(gdI)*MAX(agTN, 0));
      IF agTN <= 0 THEN gdS := "0"; END;
      TEXTOUT_P(gdS, G1, agGX0 + (gdI - 1)*agCZ + 1, agGY0 + agK*agCZ + 1, 1, agGRAY, agCZ);
    END;
  END;
END;

// The bar under the tiles (and on the Fleiss page, with its own labels): 0 to 1, the chance agreement pe in magenta,
// what was achieved beyond it (pe to po) in green, or the shortfall
// (po below pe) in red.  kappa is the green length over (1 - pe).
ag_Bar(tbPo, tbPe, tbY, tbNo, tbNe)
BEGIN
  LOCAL tbX0, tbW, tbA, tbB, tbLa, tbLb;
  tbX0 := 10; tbW := 300;
  tbA := tbX0 + IP(tbW*MAX(0, MIN(1, tbPe)));
  tbB := tbX0 + IP(tbW*MAX(0, MIN(1, tbPo)));
  RECT_P(G1, tbX0, tbY, tbX0 + tbW, tbY + 9, agGRAY, #FFFFFFh);
  IF tbA > tbX0 THEN RECT_P(G1, tbX0 + 1, tbY + 1, tbA, tbY + 8, ag_Mix(0.3, 200, 0, 200)); END;
  IF tbB > tbA THEN RECT_P(G1, tbA, tbY + 1, tbB, tbY + 8, ag_Mix(0.55, 0, 140, 60)); END;
  IF tbB < tbA THEN RECT_P(G1, tbB, tbY + 1, tbA, tbY + 8, ag_Mix(0.45, 220, 40, 40)); END;
  LINE_P(G1, tbA, tbY - 2, tbA, tbY + 11, agMAG);
  LINE_P(G1, tbB, tbY - 2, tbB, tbY + 11, agGRN);
  tbLa := MAX(0, tbA - 6); tbLb := MAX(0, tbB - 6);
  IF ABS(tbLb - tbLa) < 16 THEN
    IF tbLb >= tbLa THEN tbLb := tbLa + 16; ELSE tbLa := tbLb + 16; END;
  END;
  ag_Tx(tbNe, MIN(296, tbLa), tbY + 11, agMAG);
  ag_Tx(tbNo, MIN(296, tbLb), tbY + 11, agGRN);
  ag_Tx("0", 2, tbY - 1, agGRAY);
  ag_Tx("1", 309, tbY - 1, agGRAY);
END;

// A density-curve frame shared by the null-curve pages: a nice x axis
// over [agXlo, agXhi] at the bottom of the plot box.
ag_XAxis()
BEGIN
  LOCAL xaS, xaV, xaP, xaD;
  LINE_P(G1, agPL, agPB, agPR, agPB, #000000h);
  xaS := ag_Nice((agXhi - agXlo)/5);
  xaD := MAX(0, -FLOOR(LOG(xaS) + 0.000000001));
  xaV := CEILING(agXlo/xaS)*xaS;
  WHILE xaV <= agXhi + xaS*0.000001 DO
    xaP := agPL + IP((xaV - agXlo)/(agXhi - agXlo)*(agPR - agPL));
    LINE_P(G1, xaP, agPB, xaP, agPB + 3, agGRAY);
    ag_Tx(ag_Fx(xaV, xaD), xaP - 8, agPB + 4, agGRAY);
    xaV := xaV + xaS;
  END;
END;

// The pixel column of a value on the x axis.

// The null curve: kappa's distribution if the raters agreed only by
// chance, N(0, SE0) (the large-sample normal of Fleiss, Cohen & Everitt
// 1969), the observed kappa in red with its upper tail shaded, the 95%
// CI (green) and kappa max (grey).  ncK, ncS0, ncLo, ncHi, ncMx: the
// numbers to draw; ncMx < -9 means "no kappa max to show".
ag_NullPlot(ncK, ncS0, ncLo, ncHi, ncMx)
BEGIN
  LOCAL npC, npX, npD, npY, npPk, npPy;
  LOCAL npXk, npW;
  agPL := 34; agPR := 310; agPT := 46; agPB := 160;
  agXlo := MIN(-4*ncS0, MIN(ncLo, ncK)); agXhi := MAX(4*ncS0, MAX(ncHi, ncK));
  IF ncMx > -9 THEN agXhi := MAX(agXhi, ncMx); END;
  npW := agXhi - agXlo;
  agXlo := agXlo - 0.05*npW; agXhi := agXhi + 0.05*npW;
  npPk := NORMALD(0, ncS0, 0);
  // the curve and the shaded tail, one pass over the pixel columns
  npXk := ag_PX(ncK);
  npPy := -1;
  FOR npC FROM agPL TO agPR DO
    npX := agXlo + (npC - agPL)/(agPR - agPL)*(agXhi - agXlo);
    npD := NORMALD(0, ncS0, npX);
    npY := agPB - IP(npD/npPk*(agPB - agPT));
    IF npC >= npXk AND npY < agPB THEN LINE_P(G1, npC, agPB, npC, npY, agSHD); END;
    IF npPy >= 0 THEN LINE_P(G1, npC - 1, npPy, npC, npY, agBLUE); END;
    npPy := npY;
  END;
  ag_XAxis();
  // kappa max, dotted grey
  IF ncMx > -9 THEN ag_DotV(ag_PX(ncMx), agPT, agPB, agGRAY); END;
  // the 95% CI for kappa: a green bar near the bottom of the plot
  LINE_P(G1, ag_PX(ncLo), agPB - 10, ag_PX(ncHi), agPB - 10, agGRN);
  LINE_P(G1, ag_PX(ncLo), agPB - 13, ag_PX(ncLo), agPB - 7, agGRN);
  LINE_P(G1, ag_PX(ncHi), agPB - 13, ag_PX(ncHi), agPB - 7, agGRN);
  // the observed kappa, red
  LINE_P(G1, npXk, agPT - 4, npXk, agPB, agRED);
  ag_Mark(npXk, agPB - 10, agRED);
END;

        dwLo := agWK(dwI) - 1.95996398454*agWSE(dwI); dwHi := agWK(dwI) + 1.95996398454*agWSE(dwI);
        IF agTInf == 0 OR agWSE(dwI) <= 0 THEN
          ag_Tx("no SE, CI or p", dwX, 62 + 42*(dwI - 1) + 12, dwC);
        ELSE
          IF dwLo < -1 OR dwHi > 1 THEN
            ag_Tx("95% CI (clipped):", dwX, 62 + 42*(dwI - 1) + 12, dwC);
          ELSE
            ag_Tx("SE " + ag_K3(agWSE(dwI)) + ", 95% CI", dwX, 62 + 42*(dwI - 1) + 12, dwC);
          END;
          ag_Tx(ag_K3(MAX(-1, dwLo)) + " to " + ag_K3(MIN(1, dwHi)), dwX, 62 + 42*(dwI - 1) + 24, dwC);
        END;
      END;
      IF agTOK == 1 THEN ag_Tx("unweighted = " + ag_K3(agKa), dwX, 146, agGRAY); END;
      IF agWP(dwW) >= 0 AND agTInf == 1 THEN
        ag_Tx("1-sided z = " + ag_Fx(agWZ(dwW), 2), dwX, 158, agBLUE);
        ag_Tx(ag_PE(agWP(dwW)), dwX, 170, agBLUE);
      END;
    END;
  END;
  IF agTOrd == 0 THEN
    ag_Tx("Unordered categories: use unweighted kappa.", 4, 184, agRED);
  ELSE
    ag_Tx("Near misses earn partial credit (the weights).", 4, 184, agGRAY);
  END;
  ag_Tx("Weights assume the codes 1.." + ag_I(agK) + " are in order.", 4, 195, agGRAY);
  ag_Foot("+/- = linear/quadratic, Ent = menu, Esc = back");
END;

// Move one count from tile mvA to tile mvB (flat indices) and say what it
// did to kappa.  N stays the same.
ag_TMove(mvA, mvB)
BEGIN
  LOCAL mvO, mvI1, mvJ1, mvI2, mvJ2, mvS;
  mvI1 := IP((mvA - 1)/agK) + 1; mvJ1 := mvA - (mvI1 - 1)*agK;
  mvI2 := IP((mvB - 1)/agK) + 1; mvJ2 := mvB - (mvI2 - 1)*agK;
  IF agT(mvA) <= 0 THEN
    agInfo := "Tile " + ag_TNm(mvI1, mvJ1) + " is empty: nothing to move.";
    RETURN 0;
  END;
  IF agT(mvB) >= 99999 THEN agInfo := "A tile holds at most 99999."; RETURN 0; END;
  mvO := agTOK; mvS := agKa;
  agT(mvA) := agT(mvA) - 1; agT(mvB) := agT(mvB) + 1;
  ag_TEdit();
  ag_TCompute();
  agSel := mvI2; agSelJ := mvJ2;
  agInfo := "1 moved " + ag_TNm(mvI1, mvJ1) + " to " + ag_TNm(mvI2, mvJ2);
  IF mvO == 1 AND agTOK == 1 THEN
    agInfo := agInfo + ": kappa " + ag_K3(mvS) + " " + CHAR(8594) + " " + ag_K3(agKa);
  ELSE
    IF agTOK == 1 THEN agInfo := agInfo + ": kappa now " + ag_K3(agKa); END;
  END;
  RETURN 1;
END;

// Add dlD (+1 or -1) to the selected tile.
ag_TAdd(dlD)
BEGIN
  LOCAL dlI, dlO;
  dlI := (agSel - 1)*agK + agSelJ;
  IF agT(dlI) + dlD < 0 THEN agInfo := "A count cannot go below 0."; RETURN 0; END;
  IF agT(dlI) + dlD > 99999 THEN agInfo := "A tile holds at most 99999."; RETURN 0; END;
  dlO := agKa;
  agT(dlI) := agT(dlI) + dlD;
  ag_TEdit();
  ag_TCompute();
  agInfo := "tile " + ag_TNm(agSel, agSelJ) + " now " + ag_I(agT(dlI)) + ", N = " + ag_I(agTN);
  IF agTOK == 1 THEN agInfo := agInfo + ", kappa " + ag_K3(agKa); END;
  RETURN 1;
END;

// Type a count into the selected tile.
ag_TType()
BEGIN
  LOCAL ttV, ttL, ttI;
  ttI := (agSel - 1)*agK + agSelJ;
  ttV := agT(ttI);
  ttL := ag_TNm(agSel, agSelJ) + ":";
  IF INPUT({ttV}, "Count in the selected tile", {ttL}, {"A whole number, 0 to 99999"}) == 1 THEN
    IF ttV < 0 OR ttV > 99999 OR FP(ttV) <> 0 THEN
      ag_Msg("A count must be a whole number from 0 to 99999.");
    ELSE
      IF ttV <> agT(ttI) THEN ag_TEdit(); END;
      agT(ttI) := ttV;
      ag_TCompute();
      agInfo := "tile " + ag_TNm(agSel, agSelJ) + " set to " + ag_I(ttV);
    END;
  END;
  ag_KDrain(); ag_Drain();
END;

// Move one count between the selected tile and the diagonal tile of its
// row (rater A's category stays the same; the row total is kept):
// tdW = 1 to the diagonal, 2 from it.
ag_TDiag(tdW)
BEGIN
  LOCAL tdS, tdD;
  IF agSel == agSelJ THEN
    agInfo := "Select an off-diagonal tile first.";
    RETURN 0;
  END;
  tdS := (agSel - 1)*agK + agSelJ; tdD := (agSel - 1)*agK + agSel;
  IF tdW == 1 THEN ag_TMove(tdS, tdD); ELSE ag_TMove(tdD, tdS); END;
  // keep the off-diagonal tile selected, so the button can be pressed again
  agSelJ := tdS - (agSel - 1)*agK;
  RETURN 1;
END;

// A tap on page 1: a tile (select it; tap it again to type a count; drag
// from one tile to another to move one count), a header number (its
// category's name), a number in the right-hand panel or the bar (what it
// means).
ag_TTap()
BEGIN
  LOCAL ttA, ttB, ttI, ttJ, ttS;
  ttA := ag_TCell(agTapX, agTapY);
  ttB := ag_TCell(agRelX, agRelY);
  IF ttA > 0 THEN
    // a move needs a different tile AND at least MAX(8, half a tile) px of
    // finger travel (beta 1a, second review item 1): a 1-px drift across an
    // edge is a tap
    IF ttB > 0 AND ttB <> ttA AND ABS(agRelX - agTapX) + ABS(agRelY - agTapY) >= MAX(8, agCZ/2) THEN
      ag_TMove(ttA, ttB);
      RETURN 0;
    END;
    ttI := IP((ttA - 1)/agK) + 1; ttJ := ttA - (ttI - 1)*agK;
    IF ttI == agSel AND ttJ == agSelJ THEN
      ag_TType();
    ELSE
      agSel := ttI; agSelJ := ttJ;
      agInfo := ag_TNm(ttI, ttJ) + ": A said " + agCN(ttI) + ", B said " + agCN(ttJ);
    END;
    RETURN 0;
  END;
  // header numbers: the category's name
  IF agTapY >= 20 AND agTapY < agGY0 AND agTapX >= agGX0 AND agTapX < agGX0 + agK*agCZ THEN
    ttJ := IP((agTapX - agGX0)/agCZ) + 1;
    agInfo := "column " + ag_I(ttJ) + " = B said " + agCN(ttJ);
    RETURN 0;
  END;
  IF agTapX < agGX0 AND agTapY >= agGY0 AND agTapY < agGY0 + agK*agCZ THEN
    ttI := IP((agTapY - agGY0)/agCZ) + 1;
    agInfo := "row " + ag_I(ttI) + " = A said " + agCN(ttI);
    RETURN 0;
  END;
  IF agTapX >= 196 AND agTOK == 1 THEN
    ttS := "";
    IF agTapY >= 32 AND agTapY < 46 THEN ttS := "Observed agreement po: the share of the N items on the diagonal, where both raters chose the same category."; END;
    IF agTapY >= 46 AND agTapY < 58 THEN ttS := "Chance agreement pe: the agreement expected if each rater chose at random at their own rates: sum of row share " + agTm + " column share."; END;
    IF agTapY >= 58 AND agTapY < 78 THEN ttS := "kappa = (po - pe)/(1 - pe): the share of the possible agreement beyond chance that was reached. 1 perfect, 0 chance level."; END;
    IF agTapY >= 78 AND agTapY < 102 THEN ttS := "95% CI: kappa " + agPM + " 1.96 SE, SE from Fleiss, Cohen & Everitt (1969). A large-sample interval: rough for small N."; END;
    IF agTapY >= 102 AND agTapY < 114 THEN ttS := "kappa max: the largest kappa these row and column totals allow (Cohen 1960). Raters who use the categories at different rates cannot reach 1."; END;
    IF agTapY >= 114 AND agTapY < 128 THEN ttS := "PABAK = (k po - 1)/(k - 1), Byrt, Bishop & Carlin (1993): po rescaled to run from -1 to 1, ignoring the margins."; END;
    IF agTapY >= 128 AND agTapY < 154 THEN ttS := "z = kappa/SE" + agS0 + ", SE" + agS0 + " being the SE if kappa were 0. p = P(z this big | kappa = 0): better than chance is a low bar."; END;
    IF DIM(ttS) > 0 THEN ag_Msg(ttS); END;
    RETURN 0;
  END;
  IF agTapY >= 170 AND agTapY < 192 AND agTOK >= 1 THEN
    ag_Msg("The bar runs from 0 to 1. Magenta: chance agreement pe. Green: agreement beyond it, up to po. kappa = green length / (1 - pe).");
  END;
  RETURN 0;
END;

// A tap on the weights page: what the tile's weight means.
ag_TTapW()
BEGIN
  LOCAL twA, twI, twJ, twW;
  twA := ag_TCell(agTapX, agTapY);
  IF twA > 0 THEN
    twI := IP((twA - 1)/agK) + 1; twJ := twA - (twI - 1)*agK;
    twW := ag_Wt(twI, twJ, agK, agWt);
    agInfo := ag_TNm(twI, twJ) + ": weight " + ag_Fx(twW, 3) + ", " + ag_I(ABS(twI - twJ)) + " categories apart";
    ag_Msg(agInfo + ". A disagreement this far apart earns " + ag_Fx(100*twW, 0) + "% of the credit of an exact match.");
  END;
  RETURN 0;
END;

// Step the Bland-Altman pair: +1 or -1 through every pair (a, b), a < b.
ag_CPair(cpD)
BEGIN
  LOCAL cpA, cpB, cpN, cpI, cpL, cpM;
  // list the pairs in order; find the current one; step
  cpL := {}; cpM := {};
  FOR cpA FROM 1 TO agYK - 1 DO
    FOR cpB FROM cpA + 1 TO agYK DO
      cpL := CONCAT(cpL, {cpA}); cpM := CONCAT(cpM, {cpB});
    END;
  END;
  cpN := SIZE(cpL); cpI := 1;
  FOR cpA FROM 1 TO cpN DO
    IF cpL(cpA) == agBA AND cpM(cpA) == agBB THEN cpI := cpA; END;
  END;
  cpI := cpI + cpD;
  IF cpI > cpN THEN cpI := 1; END;
  IF cpI < 1 THEN cpI := cpN; END;
  agBA := cpL(cpI); agBB := cpM(cpI); agBSel := 0;
  ag_BCompute();
  agInfo := "pair " + agYRN(agBA) + " vs " + agYRN(agBB);
END;

// A tap on the data or Bland-Altman plot: the nearest point within 10 px.

  LOCAL ctBi, ctBj;
  ctB := 101; ctBi := 0; ctBj := 0;
  IF agScr == 2 THEN
    FOR ctI FROM 1 TO agYN DO
      ctX := agPL + IP((ctI - 0.5)/agYN*(agPR - agPL));
      FOR ctJ FROM 1 TO agYK DO
        ctY := ag_PY(agY((ctI - 1)*agYK + ctJ));
        ctD := (ctX - agTapX)^2 + (ctY - agTapY)^2;
        IF ctD < ctB THEN ctB := ctD; ctBi := ctI; ctBj := ctJ; END;
      END;
    END;
    IF ctBi > 0 THEN
      agInfo := "target " + ag_I(ctBi) + ", " + agYRN(ctBj) + " = " + ag_Gn(agY((ctBi - 1)*agYK + ctBj)) + agYU;
    END;
  END;
  IF agScr == 3 AND agBOK == 1 THEN
    FOR ctI FROM 1 TO agBN DO
      ctD := (ag_PX(agBM(ctI)) - agTapX)^2 + (ag_PY(agBD(ctI)) - agTapY)^2;
      IF ctD < ctB THEN ctB := ctD; ctBi := ctI; END;
    END;
    IF ctBi > 0 THEN
      agBSel := ctBi;
      ctI := ctBi;
      agInfo := "#" + ag_I(ctI) + ": " + ag_Gn(agY((ctI - 1)*agYK + agBA)) + " and " + ag_Gn(agY((ctI - 1)*agYK + agBB)) + ", mean " + ag_Gn(agBM(ctI)) + ", d = " + ag_Gn(agBD(ctI));
    END;
  END;
  RETURN 0;
END;

// The continuous pages.  Soft keys: ICC, Data, B-A, Stats, Help, Back.
// Actions: 1-4 the pages, 5 help (ICC), 6 back, 7 next pair, 8 help
// (Bland-Altman), 9 the menu.
ag_CView(cvS)
BEGIN
  LOCAL cvK, cvA, cvGo, cvSl, cvR;
  IF agYHas == 0 THEN ag_Msg("No continuous ratings yet: load a preset or use your data."); RETURN 0; END;
  agScr := cvS; cvGo := 1; agInfo := "";
  IF agIOK == -1 THEN ag_ICompute(); END;
  ag_KDrain(); ag_Drain();
  WHILE cvGo == 1 DO
    ag_CDraw();
    ag_Show({"ICC", "Data", "B-A", "Stats", "Help", "Back"});
    cvK := ag_Poll();
    cvA := 0;
    IF cvK == -1 THEN
      IF agTapY >= 218 THEN
        cvSl := MIN(6, IP(agTapX/53.4) + 1);
        cvA := cvSl;
        IF cvSl == 5 AND agScr >= 3 THEN cvA := 8; END;
        IF cvSl == 6 THEN cvA := 6; END;
      ELSE
        IF agScr == 1 AND agIOK == 1 AND agTapY >= 36 AND agTapY < 150 THEN
          cvR := IP((agTapY - 36)/19) + 1;
          IF cvR >= 1 AND cvR <= 6 THEN agISel := cvR; agInfo := ag_ICIs(cvR); ag_Msg(ag_IWhat(cvR)); END;
        END;
        IF agScr == 2 OR agScr == 3 THEN ag_CTapPt(); END;
      END;
    ELSE
      IF cvK == 4 THEN cvA := 6; END;
      IF cvK == 30 THEN cvA := 9; END;
      IF cvK == 50 THEN cvA := 7; END;
      IF cvK == 45 THEN ag_CPair(-1); END;
      IF cvK == 7 THEN cvA := agScr - 1; IF cvA < 1 THEN cvA := 4; END; END;
      IF cvK == 8 THEN cvA := agScr + 1; IF cvA > 4 THEN cvA := 1; END; END;
      IF agScr == 1 AND (cvK == 2 OR cvK == 12) THEN
        IF cvK == 12 THEN agISel := agISel + 1; ELSE agISel := agISel - 1; END;
        IF agISel > 6 THEN agISel := 1; END;
        IF agISel < 1 THEN agISel := 6; END;
        IF agIOK == 1 THEN agInfo := ag_ICIs(agISel); END;
      END;
    END;
    IF cvA == 9 THEN cvA := ag_CMenu(); END;
    IF cvA >= 1 AND cvA <= 4 THEN
      IF cvA <> agScr THEN agInfo := ""; END;
      agScr := cvA;
    END;
    IF cvA == 5 THEN ag_Help(6); END;
    IF cvA == 8 THEN ag_Help(7); END;
    IF cvA == 7 THEN ag_CPair(1); END;
    IF cvA == 6 THEN cvGo := 0; END;
  END;
  IF agIOK == 1 THEN agLast := ag_ICIs(2) + "; ICC(3,1) " + ag_KV(3) + " (" + agYTtl + ")"; END;
  ag_KDrain();
  RETURN 0;
END;

// List Di, or {} when it is empty, not a list, or not there yet (the
// first read is inside IFERR: Di does not exist until the app has been
// opened once).
ag_ReadD(rdI)
BEGIN
  LOCAL rdL;
  rdL := {};
  IFERR
    IF rdI == 1 THEN rdL := D1; END;
    IF rdI == 2 THEN rdL := D2; END;
    IF rdI == 3 THEN rdL := D3; END;
    IF rdI == 4 THEN rdL := D4; END;
    IF rdI == 5 THEN rdL := D5; END;
    IF rdI == 6 THEN rdL := D6; END;
    IF rdI == 7 THEN rdL := D7; END;
    IF rdI == 8 THEN rdL := D8; END;
    IF rdI == 9 THEN rdL := D9; END;
  THEN
    rdL := {};
  END;
  IF TYPE(rdL) <> 6 THEN rdL := {}; END;
  RETURN rdL;
END;

// Two raters from D1 (rater A's codes) and D2 (rater B's): whole numbers
// 1 to 6, one pair per item.  k = the largest code used (at least 2).
ag_MyTwo()
BEGIN
  LOCAL mtA, mtB, mtN, mtI, mtK, mtV;
  mtA := ag_ReadD(1); mtB := ag_ReadD(2);
  mtN := SIZE(mtA);
  IF mtN < 1 OR SIZE(mtB) <> mtN THEN
    ag_Msg("Put rater A's codes in D1 and rater B's in D2 (Statistics 1Var): two lists of the same length.");
    RETURN 0;
  END;
  mtK := 2;
  FOR mtI FROM 1 TO mtN DO
    mtV := mtA(mtI);
    IF mtV < 1 OR mtV > 6 OR FP(mtV) <> 0 THEN ag_Msg("D1(" + ag_I(mtI) + ") is not a whole code 1 to 6."); RETURN 0; END;
    mtK := MAX(mtK, mtV);
    mtV := mtB(mtI);
    IF mtV < 1 OR mtV > 6 OR FP(mtV) <> 0 THEN ag_Msg("D2(" + ag_I(mtI) + ") is not a whole code 1 to 6."); RETURN 0; END;
    mtK := MAX(mtK, mtV);
  END;
  agK := mtK;
  agT := MAKELIST(0, mtI, 1, mtK*mtK);
  FOR mtI FROM 1 TO mtN DO
    mtV := (mtA(mtI) - 1)*mtK + mtB(mtI);
    agT(mtV) := agT(mtV) + 1;
  END;
  agCN := {};
  FOR mtI FROM 1 TO mtK DO agCN := CONCAT(agCN, {"code " + ag_I(mtI)}); END;
  agRA := "D1 (rater A)"; agRB := "D2 (rater B)";
  ag_TSet("Your data: D1 vs D2", "Statistics 1Var lists D1, D2", 0, 1, "D1, D2");
  RETURN 1;
END;

// Many raters from D1-D9: each non-empty list is a category, row i =
// subject i (its count in that category).  Every row must add to the
// same m (at least 2).
ag_MyMany()
BEGIN
  LOCAL mmI, mmJ, mmL, mmC, mmN, mmQ;
  LOCAL mmM, mmS, mmF, mmV;
  mmC := {}; mmN := -1;
  FOR mmI FROM 1 TO 9 DO
    mmL := ag_ReadD(mmI);
    IF SIZE(mmL) > 0 THEN
      IF mmN < 0 THEN mmN := SIZE(mmL); END;
      IF SIZE(mmL) <> mmN THEN
        ag_Msg("Every list must have one value per subject: D" + ag_I(mmI) + " has a different length.");
        RETURN 0;
      END;
      mmC := CONCAT(mmC, {mmI});
    END;
  END;
  mmQ := SIZE(mmC);
  IF mmQ < 2 OR mmN < 1 THEN
    ag_Msg("Put each category's counts in its own list (D1-D9), one row per subject: at least 2 lists.");
    RETURN 0;
  END;
  IF mmN > 200 THEN ag_Msg("More than 200 subjects: too many for this program."); RETURN 0; END;
  mmF := MAKELIST(0, mmI, 1, mmN*mmQ);
  FOR mmJ FROM 1 TO mmQ DO
    mmL := ag_ReadD(mmC(mmJ));
    FOR mmI FROM 1 TO mmN DO
      mmV := mmL(mmI);
      IF mmV < 0 OR FP(mmV) <> 0 THEN
        ag_Msg("D" + ag_I(mmC(mmJ)) + "(" + ag_I(mmI) + ") is not a whole count of 0 or more.");
        RETURN 0;
      END;
      mmF((mmI - 1)*mmQ + mmJ) := mmV;
    END;
  END;
  mmM := -1;
  FOR mmI FROM 1 TO mmN DO
    mmS := 0;
    FOR mmJ FROM 1 TO mmQ DO mmS := mmS + mmF((mmI - 1)*mmQ + mmJ); END;
    IF mmM < 0 THEN mmM := mmS; END;
    IF mmS <> mmM THEN
      ag_Msg("Row " + ag_I(mmI) + " adds to " + ag_I(mmS) + ", row 1 to " + ag_I(mmM) + ": every subject needs the same number of ratings.");
      RETURN 0;
    END;
  END;
  IF mmM < 2 THEN ag_Msg("Each subject needs 2 or more ratings."); RETURN 0; END;
  agF := mmF; agFN := mmN; agFQ := mmQ; agFM := mmM;
  agFCN := {};
  FOR mmI FROM 1 TO mmQ DO agFCN := CONCAT(agFCN, {"D" + ag_I(mmC(mmI))}); END;
  agFTtl := "Your data (D1-D9)"; agFSrc := "Statistics 1Var lists";
  agFIll := 0; agFHas := 1; agFSel := 0; agFSub := 0; agInfo := "";
  ag_FCompute();
  RETURN 1;
END;

// Continuous ratings from D1-D9: each non-empty list is a rater, row i =
// target i.  At least 2 raters and 3 targets, at most 1500 values in all
// (the ICC passes over every value three times; a G1 is slow).
ag_MyCont()
BEGIN
  LOCAL mcI, mcJ, mcL, mcC, mcN, mcK;
  LOCAL mcY;
  mcC := {}; mcN := -1;
  FOR mcI FROM 1 TO 9 DO
    mcL := ag_ReadD(mcI);
    IF SIZE(mcL) > 0 THEN
      IF mcN < 0 THEN mcN := SIZE(mcL); END;
      IF SIZE(mcL) <> mcN THEN
        ag_Msg("Every rater's list needs one value per target: D" + ag_I(mcI) + " has a different length.");
        RETURN 0;
      END;
      mcC := CONCAT(mcC, {mcI});
    END;
  END;
  mcK := SIZE(mcC);
  IF mcK < 2 OR mcN < 3 THEN
    ag_Msg("Put each rater's values in its own list (D1-D9), one row per target: 2 or more lists of 3 or more.");
    RETURN 0;
  END;
  IF mcN*mcK > 1500 THEN ag_Msg("More than 1500 values (targets " + agTm + " raters): too many for this program."); RETURN 0; END;
  mcY := MAKELIST(0, mcI, 1, mcN*mcK);
  FOR mcJ FROM 1 TO mcK DO
    mcL := ag_ReadD(mcC(mcJ));
    FOR mcI FROM 1 TO mcN DO mcY((mcI - 1)*mcK + mcJ) := mcL(mcI); END;
  END;
  agY := mcY; agYN := mcN; agYK := mcK;
  agYRN := {};
  FOR mcI FROM 1 TO mcK DO agYRN := CONCAT(agYRN, {"D" + ag_I(mcC(mcI))}); END;
  agYTtl := "Your data (D1-D9)"; agYSrc := "Statistics 1Var lists"; agYU := ""; agYIll := 0;
  ag_YSet();
  RETURN 1;
END;
