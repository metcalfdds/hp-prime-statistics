// ============================================================
// GALTONBOARD - beta 1
// Galton's quincunx (bean machine) for the HP Prime
// ============================================================
// Author: Roger Metcalf, 2026.  Written with AI assistance
//   (Anthropic Claude); concept, direction and all hardware
//   testing by the author.
//
// Idea after Francis Galton's quincunx of 1873 and the small
//   DOS QUINCUNX demo shipped with CDC Epi Info 6.  A ball
//   falls through R rows of pegs.  At every peg it bounces
//   right with probability p, otherwise left.  The bin it lands
//   in is the NUMBER OF RIGHT BOUNCES, so the bin counts build a
//   Binomial(R, p) histogram, and with many rows the bell of the
//   normal distribution appears (de Moivre-Laplace).
//
// FREE FOR EDUCATIONAL AND PERSONAL USE.  Keep this header with
//   the file and credit the author if you pass it on.  Not for
//   sale or commercial redistribution.  Provided as is, with no
//   warranty: check any result you intend to rely on.
//
// Firmware: written for the 2026-09-09 firmware (2.4.2).  This
//   version has NOT yet been run on hardware.
//
// Run GaltonBoard() from the Program Catalog.
// ============================================================
//
// WHAT YOU SEE
//   Top: the triangle of pegs.  A dropped ball is drawn
//        bouncing down through it.
//   Bottom: one bar per bin (blue, the counts so far).
//        Magenta squares: the counts the binomial predicts.
//        Green curve: the normal approximation N(Rp, Rp(1-p)).
//        Red dotted line: the observed mean position.
//        Magenta dotted line: the expected mean Rp.
//   Corners: n, the observed mean and SD of the bin positions,
//        and the theoretical mean Rp and SD sqrt(Rp(1-p)).
//
// KEYS AND TOUCH
//   tap the pegs ..... drop one ball and watch it fall
//   tap a bar ........ that bin: observed share vs binomial P
//   -  ............... drop one ball (watch)
//   +  ............... drop 100 balls (fast, no animation)
//   left / right ..... p down / up by 0.05   (clears the counts)
//   up / down ........ more / fewer rows      (clears the counts)
//   Enter ............ menu: 10 or 1000 balls, R, p, overlays,
//                      speed, send bins to D1/D2, reset, help
//   Esc .............. leave
//
// HOME EXPORTS (readable after the program ends)
//   GB_CNT  list of bin counts, bins 0 .. R
//   GB_N    balls dropped      GB_R  rows      GB_P  p
// ============================================================


// ------------------------------------------------------------
// RESULTS VISIBLE FROM HOME
// ------------------------------------------------------------
EXPORT GB_CNT, GB_N, GB_R, GB_P;

// ------------------------------------------------------------
// SETTINGS SHARED BY EVERY ROUTINE (private to this program)
// All of them are set in gbInit at the start of every run.
// ------------------------------------------------------------
LOCAL gbR, gbP, gbN, gbCnt, gbOvl, gbSpd;
//  gbR   rows of pegs (3..12)       gbP   chance of a right bounce
//  gbN   balls dropped so far       gbCnt counts, bins 0..R (R+1 entries)
//  gbOvl overlays: 0 none, 1 binomial, 2 binomial and normal
//  gbSpd animation frames per peg row (1 fast .. 4 slow)
LOCAL gbBw, gbX0, gbDy, gbSum, gbSq, gbYmx;
//  gbBw  bin width in pixels        gbX0  left edge of bin 0
//  gbDy  vertical spacing of peg rows
//  gbSum, gbSq  running sum and sum of squares of bin positions
//  gbYmx the count that reaches the top of the histogram box

// Screen layout.  Touch handling and drawing both use these, so a
// tap always lands where the picture is.
LOCAL gbTop, gbPegBot, gbHT, gbHB, gbMidX;
//  gbTop    y where a ball starts; peg row r is at gbTop + r*gbDy
//  gbPegBot lowest peg row is at or above this y
//  gbHT     tallest possible bar top      gbHB  histogram baseline
//  gbMidX   the board is centred here

// Colours (house palette)
LOCAL gbBLUE, gbFILL, gbMAG, gbGRN, gbRED, gbGRAY;
LOCAL gbFOOT, gbORNG;


// ------------------------------------------------------------
// SMALL HELPERS
// ------------------------------------------------------------

// A number as text with a FIXED number of decimals.  Mode 2 keeps
// trailing zeros; mode 1 would ignore the digits argument.
gbF(fv, fd)
BEGIN
  RETURN STRING(fv, 2, fd);
END;

// Pixel x of bin position v (v may be fractional).  After R rows a
// ball with k right bounces sits at bin k, whose centre is
// gbX0 + (k + 0.5) * gbBw.
gbPx(pxV)
BEGIN
  RETURN gbX0 + IP((pxV + 0.5) * gbBw + 0.5);
END;

// A small filled disc: three concentric circles and the centre
// pixel.  ARC_P with one colour draws an outline only.
gbDisc(dcG, dcX, dcY, dcCol)
BEGIN
  ARC_P(dcG, dcX, dcY, 3, dcCol);
  ARC_P(dcG, dcX, dcY, 2, dcCol);
  ARC_P(dcG, dcX, dcY, 1, dcCol);
  PIXON_P(dcG, dcX, dcY, dcCol);
END;

// Work out the geometry for the current number of rows.
// Bins are never wider than 40 px, and never wider than fits.
gbLayout()
BEGIN
  gbBw := MIN(40, FLOOR(300 / (gbR + 1)));
  gbX0 := gbMidX - IP((gbR + 1) * gbBw / 2);
  gbDy := FLOOR((gbPegBot - gbTop) / gbR);
END;

// Forget every ball dropped so far.
gbClear()
BEGIN
  gbN := 0;
  gbSum := 0;
  gbSq := 0;
  gbCnt := MAKELIST(0, K, 1, gbR + 1, 1);
END;

// One ball's final bin is k right bounces: record it.
gbRecord(rcK)
BEGIN
  gbN := gbN + 1;
  gbCnt(rcK + 1) := gbCnt(rcK + 1) + 1;
  gbSum := gbSum + rcK;
  gbSq := gbSq + rcK * rcK;
END;

// Observed mean of the bin positions.
gbMean()
BEGIN
  IF gbN == 0 THEN RETURN 0; END;
  RETURN gbSum / gbN;
END;

// Observed standard deviation of the bin positions, with the n-1
// divisor: the balls are a sample from the Binomial(R, p) process.
// MAX(0, ...) guards against rounding driving the variance a hair
// below zero when every ball landed in the same bin.
gbSD()
BEGIN
  LOCAL sdM;
  IF gbN < 2 THEN RETURN 0; END;
  sdM := gbSum / gbN;
  RETURN (MAX(0, (gbSq - gbN * sdM * sdM) / (gbN - 1)))^0.5;
END;

// Wait for any key or any tap, then let go of it.  Drain first: a
// key or finger left over from a menu would otherwise skip the
// screen that was just drawn.
gbWaitAny()
BEGIN
  LOCAL waK, waM, waM1;
  REPEAT waK := GETKEY; UNTIL waK == -1;
  REPEAT UNTIL MOUSE(1) == -1;
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  REPEAT UNTIL MOUSE(1) == -1;
  REPEAT waK := GETKEY; UNTIL waK == -1;
END;


// ------------------------------------------------------------
// THE STATIC SCENE, drawn into G2
// Pegs, bars, overlays and readouts.  The animation then only
// has to copy G2 and add one ball per frame, which keeps it
// fast even on a G1.
// ------------------------------------------------------------
gbScene()
BEGIN
  LOCAL scR, scK, scX, scY, scStr, scE;
  LOCAL scScale, scH, scTop, scPrev, scMu, scSg;
  LOCAL scV, scPY, scPX, scXb, scDec;

  DIMGROB_P(G2, 320, 240, #FFFFFFh);
  TEXTOUT_P("Galton Board", G2, 6, 1, 2, gbBLUE);

  // ---- the pegs: row r has r pegs, centred on the board ----
  FOR scR FROM 1 TO gbR DO
    scY := gbTop + scR * gbDy;
    FOR scK FROM 0 TO scR - 1 DO
      scX := gbMidX + IP((scK - (scR - 1) / 2) * gbBw + 0.5);
      ARC_P(G2, scX, scY, 2, gbGRAY);
      ARC_P(G2, scX, scY, 1, gbGRAY);
      PIXON_P(G2, scX, scY, gbGRAY);
    END;
  END;
  // the funnel the balls start from
  LINE_P(G2, gbMidX - 10, gbTop - 10, gbMidX - 3, gbTop - 1, gbGRAY);
  LINE_P(G2, gbMidX + 10, gbTop - 10, gbMidX + 3, gbTop - 1, gbGRAY);

  // ---- theory: binomial probabilities, mean and SD ----
  // With no balls yet, the overlays show the SHAPE (probabilities
  // scaled to fill the plot); with balls, they show expected counts.
  scE := MAKELIST(BINOMIAL(gbR, K - 1, gbP), K, 1, gbR + 1, 1);
  scMu := gbR * gbP;
  scSg := (gbR * gbP * (1 - gbP))^0.5;
  IF gbN > 0 THEN scScale := gbN; ELSE scScale := 1; END;

  // ---- y scale: tallest bar, or tallest expectation if shown ----
  gbYmx := 0;
  IF gbN > 0 THEN gbYmx := MAX(gbCnt); END;
  IF gbOvl > 0 THEN gbYmx := MAX(gbYmx, scScale * MAX(scE)); END;
  IF gbYmx <= 0 THEN gbYmx := 1; END;
  gbYmx := gbYmx * 1.08;
  scH := gbHB - gbHT;                 // pixels available for the tallest bar

  // ---- the bars (observed counts) ----
  LINE_P(G2, gbX0 - 4, gbHB, gbX0 + (gbR + 1) * gbBw + 4, gbHB, #000000h);
  FOR scK FROM 0 TO gbR DO
    scX := gbX0 + scK * gbBw;
    IF gbCnt(scK + 1) > 0 THEN
      scTop := gbHB - IP(gbCnt(scK + 1) / gbYmx * scH + 0.5);
      RECT_P(G2, scX + 1, scTop, scX + gbBw - 1, gbHB - 1, gbBLUE, gbFILL);
      // the count above its bar, when there is room for the digits
      IF gbBw >= 28 THEN
        scStr := STRING(gbCnt(scK + 1), 2, 0);
        TEXTOUT_P(scStr, G2, scX + IP(gbBw / 2) - 3 * SIZE(scStr), scTop - 11, 1, gbBLUE);
      END;
    END;
    // bin label: the number of right bounces
    scStr := STRING(scK, 2, 0);
    TEXTOUT_P(scStr, G2, scX + IP(gbBw / 2) - 3 * SIZE(scStr), gbHB + 3, 1, gbGRAY);
  END;

  // ---- overlay 1: binomial expected counts, magenta squares ----
  IF gbOvl >= 1 THEN
    scPrev := -1;
    FOR scK FROM 0 TO gbR DO
      scX := gbPx(scK);
      scY := gbHB - IP(scScale * scE(scK + 1) / gbYmx * scH + 0.5);
      RECT_P(G2, scX - 2, scY - 2, scX + 2, scY + 2, gbMAG, gbMAG);
      IF scPrev >= 0 THEN LINE_P(G2, gbPx(scK - 1), scPrev, scX, scY, gbMAG); END;
      scPrev := scY;
    END;
  END;

  // ---- overlay 2: the normal approximation, green curve ----
  // Drawn one pixel column at a time so it has no gaps.  Height is
  // the normal density at the (fractional) bin position, scaled by
  // n, because each bin is one unit wide in position.
  IF gbOvl >= 2 AND scSg > 0 THEN
    scPrev := -1;
    FOR scPX FROM gbX0 TO gbX0 + (gbR + 1) * gbBw DO
      scV := (scPX - gbX0) / gbBw - 0.5;          // bin position at this column
      scPY := gbHB - IP(scScale * NORMALD(scMu, scSg, scV) / gbYmx * scH + 0.5);
      IF scPY < gbHT THEN scPY := gbHT; END;
      IF scPrev >= 0 THEN LINE_P(G2, scPX - 1, scPrev, scPX, scPY, gbGRN); END;
      scPrev := scPY;
    END;
  END;

  // ---- expected mean Rp: magenta dotted line ----
  scX := gbPx(scMu);
  FOR scY FROM gbHT TO gbHB - 2 STEP 3 DO PIXON_P(G2, scX, scY, gbMAG); END;

  // ---- observed mean: red dotted line ----
  IF gbN > 0 THEN
    scXb := gbPx(gbMean());
    FOR scY FROM gbHT + 1 TO gbHB - 1 STEP 3 DO PIXON_P(G2, scXb, scY, gbRED); END;
  END;

  // ---- readouts: observed (left) and theory (right) ----
  // Up to 17 characters per line keeps clear of the pegs.
  scDec := 2;
  TEXTOUT_P("n = " + STRING(gbN, 2, 0), G2, 4, 20, 1, gbBLUE);
  IF gbN > 0 THEN
    TEXTOUT_P(CHAR(57344) + " = " + gbF(gbMean(), scDec), G2, 4, 33, 1, gbRED);
  ELSE
    TEXTOUT_P(CHAR(57344) + " = (no balls)", G2, 4, 33, 1, gbRED);
  END;
  IF gbN > 1 THEN
    TEXTOUT_P("s = " + gbF(gbSD(), scDec), G2, 4, 46, 1, gbBLUE);
  END;
  TEXTOUT_P("R=" + STRING(gbR, 2, 0) + "  p=" + gbF(gbP, 2), G2, 212, 20, 1, #000000h);
  TEXTOUT_P("Rp = " + gbF(scMu, scDec), G2, 212, 33, 1, gbMAG);
  TEXTOUT_P(CHAR(8730) + "Rp(1-p) = " + gbF(scSg, scDec), G2, 212, 46, 1, gbMAG);
  CASE
    IF gbOvl == 0 THEN scStr := "overlays off"; END;
    IF gbOvl == 1 THEN scStr := "binomial"; END;
    DEFAULT scStr := "binomial+normal";
  END;
  TEXTOUT_P(scStr, G2, 212, 59, 1, gbGRAY);

  // ---- key help ----
  TEXTOUT_P("+100  -1  " + CHAR(8592) + CHAR(8594) + " p  " + CHAR(8593) + CHAR(8595) + " rows  tap  Ent menu  Esc", G2, 4, 226, 1, gbFOOT);
END;

// Show the scene as it stands.
gbShow()
BEGIN
  BLIT_P(G0, 0, 0, G2);
END;


// ------------------------------------------------------------
// DROPPING BALLS
// ------------------------------------------------------------

// Drop one ball with animation.  Each frame copies the static scene
// from G2 into G1, draws the ball, and shows G1.  Returns the bin.
gbDropAnim()
BEGIN
  LOCAL daK, daR, daX0, daY0, daX1, daY1;
  LOCAL daF, daBX, daBY, daTop, daGo;
  DIMGROB_P(G1, 320, 240, #FFFFFFh);
  daK := 0;
  // out of the funnel, down to just above the first peg
  daX0 := gbMidX; daY0 := gbTop - 6;
  daY1 := gbTop + gbDy - 5;
  FOR daF FROM 1 TO 2 DO
    daBY := daY0 + IP((daY1 - daY0) * daF / 2 + 0.5);
    BLIT_P(G1, 0, 0, G2);
    gbDisc(G1, daX0, daBY, gbRED);
    BLIT_P(G0, 0, 0, G1);
    WAIT(0.02);
  END;
  daY0 := daY1;
  FOR daR FROM 1 TO gbR DO
    // the ball sits just above peg row daR and decides: right with chance p
    daGo := 0;
    IF RANDOM < gbP THEN daGo := 1; END;
    // where it will be just above the NEXT row (or the landing level)
    daX1 := gbMidX + IP((daK + daGo - daR / 2) * gbBw + 0.5);
    daY1 := gbTop + (daR + 1) * gbDy - 5;
    FOR daF FROM 1 TO gbSpd DO
      daBX := daX0 + IP((daX1 - daX0) * daF / gbSpd + 0.5);
      daBY := daY0 + IP((daY1 - daY0) * daF / gbSpd + 0.5);
      BLIT_P(G1, 0, 0, G2);
      gbDisc(G1, daBX, daBY, gbRED);
      BLIT_P(G0, 0, 0, G1);
      WAIT(0.02);
    END;
    daK := daK + daGo;
    daX0 := daX1; daY0 := daY1;
  END;
  // fall into the bin, onto the top of its bar as currently drawn
  daTop := gbHB - 4;
  IF gbCnt(daK + 1) > 0 THEN
    daTop := gbHB - IP(gbCnt(daK + 1) / gbYmx * (gbHB - gbHT) + 0.5) - 4;
  END;
  FOR daF FROM 1 TO 4 DO
    daBY := daY0 + IP((daTop - daY0) * daF / 4 + 0.5);
    BLIT_P(G1, 0, 0, G2);
    gbDisc(G1, daX0, daBY, gbRED);
    BLIT_P(G0, 0, 0, G1);
    WAIT(0.02);
  END;
  gbRecord(daK);
  RETURN daK;
END;

// Drop dfN balls with no animation: R random bounces each.
gbDropFast(dfN)
BEGIN
  LOCAL dfI, dfR, dfK;
  // say what is happening; 1000 balls take a few seconds on a G1
  RECT_P(G0, 60, 100, 260, 140, gbORNG, #FFF0DCh);
  TEXTOUT_P("Dropping " + STRING(dfN, 2, 0) + " balls ...", G0, 76, 112, 2, gbORNG);
  FOR dfI FROM 1 TO dfN DO
    dfK := 0;
    FOR dfR FROM 1 TO gbR DO
      IF RANDOM < gbP THEN dfK := dfK + 1; END;
    END;
    gbRecord(dfK);
  END;
END;


// ------------------------------------------------------------
// ASKING THE USER
// ------------------------------------------------------------

// Change the number of rows.  Clears the counts: bins from a
// different board mean something else.
gbSetRows()
BEGIN
  LOCAL srR;
  srR := gbR;
  IF INPUT({srR}, "Rows of pegs", {"R:"},
    {"3 to 12 rows; the bins are 0 .. R right bounces"}) == 1 THEN
    srR := IP(srR);
    IF srR < 3 OR srR > 12 THEN
      MSGBOX("R must be a whole number from 3 to 12.");
      RETURN 0;
    END;
    gbR := srR;
    gbLayout();
    gbClear();
    RETURN 1;
  END;
  RETURN 0;
END;

// Change p, the chance of bouncing right at a peg.  Try 0.3 or
// 0.7 to see a skewed binomial; 0.5 gives the symmetric board.
gbSetP()
BEGIN
  LOCAL spP;
  spP := gbP;
  IF INPUT({spP}, "Chance of a right bounce", {"p:"},
    {"Between 0.05 and 0.95; 0.5 is Galton's board"}) == 1 THEN
    IF spP < 0.05 OR spP > 0.95 THEN
      MSGBOX("p must be between 0.05 and 0.95 so that every bin is possible.");
      RETURN 0;
    END;
    gbP := spP;
    gbClear();
    RETURN 1;
  END;
  RETURN 0;
END;

// Animation speed: frames per peg row.
gbSetSpeed()
BEGIN
  LOCAL ssI;
  ssI := 0;
  CHOOSE(ssI, "Animation speed", {"Fast (1 frame per row)", "Normal (2)", "Slow (3)", "Very slow (4)"});
  IF ssI > 0 THEN gbSpd := ssI; END;
END;

// Put the bins where the Statistics 1Var app can use them:
// D1 = bin position 0..R, D2 = its count, so H1 can be D1 with
// frequency D2.  D1 and D2 are global, so this works from here;
// the app's own plot settings are not reachable, so the student
// sets the histogram up in the app (one step, and a useful one).
gbToStats()
BEGIN
  IFERR
    D1 := MAKELIST(K - 1, K, 1, gbR + 1, 1);
    D2 := gbCnt;
    MSGBOX("D1 = bin position (0 to " + STRING(gbR, 2, 0) + "), D2 = count.  In Statistics 1Var, Symbolic view: H1 data D1, frequency D2.");
  THEN
    MSGBOX("Could not write D1 and D2.  Open the Statistics 1Var app once, then try again.");
  END;
END;

// Tap on a bar: what did that bin get, and what did theory say?
gbBinInfo(biK)
BEGIN
  LOCAL biP, biObs;
  biP := BINOMIAL(gbR, biK, gbP);
  IF gbN > 0 THEN
    biObs := gbCnt(biK + 1) / gbN;
    MSGBOX("Bin " + STRING(biK, 2, 0) + " (" + STRING(biK, 2, 0) + " right bounces)" + CHAR(10)
      + "observed: " + STRING(gbCnt(biK + 1), 2, 0) + " of " + STRING(gbN, 2, 0) + " = " + gbF(biObs, 3) + CHAR(10)
      + "binomial P = " + gbF(biP, 3) + CHAR(10)
      + "expected count = " + gbF(gbN * biP, 1));
  ELSE
    MSGBOX("Bin " + STRING(biK, 2, 0) + ": binomial P = " + gbF(biP, 4) + CHAR(10)
      + "= C(" + STRING(gbR, 2, 0) + "," + STRING(biK, 2, 0) + ") p^" + STRING(biK, 2, 0)
      + " (1-p)^" + STRING(gbR - biK, 2, 0));
  END;
END;

  TEXTOUT_P("P(k) = C(R,k) p^k (1-p)^(R-k),  mean Rp,", G1, 12, 69, 1, gbMAG);
  TEXTOUT_P("SD = " + CHAR(8730) + "(Rp(1-p)).  Magenta squares show P(k).", G1, 12, 82, 1, gbMAG);

  TEXTOUT_P("and s settle toward Rp and " + CHAR(8730) + "(Rp(1-p)) as n", G1, 12, 142, 1, gbBLUE);
  TEXTOUT_P("grows (law of large numbers); p = 0.3 skews.", G1, 12, 155, 1, gbBLUE);

// The Enter-key menu.  Returns 1 to keep going, 0 to leave.
gbMenu()
BEGIN
  LOCAL mnI, mnJ;
  mnI := 0;
  CHOOSE(mnI, "Galton Board", {
    "Drop 1 ball (watch it fall)",
    "Drop 10 balls (watch)",
    "Drop 100 balls (fast)",
    "Drop 1000 balls (fast)",
    "Rows of pegs R",
    "Chance of a right bounce p",
    "Overlays: none / binomial / + normal",
    "Animation speed",
    "Send bins to D1, D2 (Statistics 1Var)",
    "Reset the counts",
    "Help",
    "Leave"});
  IF mnI == 0 THEN RETURN 1; END;
  IF mnI == 1 THEN gbDropAnim(); END;
  IF mnI == 2 THEN FOR mnJ FROM 1 TO 10 DO gbDropAnim(); gbScene(); END; END;
  IF mnI == 3 THEN gbDropFast(100); END;
  IF mnI == 4 THEN gbDropFast(1000); END;
  IF mnI == 5 THEN gbSetRows(); END;
  IF mnI == 6 THEN gbSetP(); END;
  IF mnI == 7 THEN gbOvl := (gbOvl + 1) MOD 3; END;
  IF mnI == 8 THEN gbSetSpeed(); END;
  IF mnI == 9 THEN gbToStats(); END;
  IF mnI == 10 THEN gbClear(); END;
  IF mnI == 11 THEN gbHelp(); END;
  IF mnI == 12 THEN RETURN 0; END;
  RETURN 1;
END;


// ------------------------------------------------------------
// START-UP
// ------------------------------------------------------------
gbInit()
BEGIN
  // layout
  gbTop := 22; gbPegBot := 124; gbHT := 134; gbHB := 200; gbMidX := 160;
  // colours
  gbBLUE := #003CC8h; gbFILL := #AACDFFh; gbMAG := #C800C8h;
  gbGRN := #008C3Ch; gbRED := #FF0000h; gbGRAY := #5A5A5Ah;
  gbFOOT := #1E90FFh; gbORNG := #FF4500h;
  // the board
  gbR := 8;
  gbP := 0.5;
  gbOvl := 1;
  gbSpd := 2;
  gbYmx := 1;
  gbLayout();
  gbClear();
END;

// ------------------------------------------------------------
// MAIN: the live board
// ------------------------------------------------------------
EXPORT GaltonBoard()
BEGIN
  LOCAL mlK, mlM, mlM1, mlGo, mlBin;
  gbInit();
  gbWelcome();
  mlGo := 1;
  WHILE mlGo == 1 DO
    gbScene();
    gbShow();
    REPEAT mlK := GETKEY; UNTIL mlK == -1;      // drain any leftover key
    REPEAT
      mlK := GETKEY;
      mlM := MOUSE;
      mlM1 := mlM(1);                            // the first finger, if any
    UNTIL mlK > -1 OR SIZE(mlM1) > 0;
    IF SIZE(mlM1) > 0 THEN
      // a tap: element 1 is x, element 2 is y
      IF mlM1(2) < gbHT - 4 THEN
        REPEAT UNTIL MOUSE(1) == -1;             // let go before the animation
        gbDropAnim();
      ELSE
        IF mlM1(2) <= gbHB + 12 AND mlM1(1) >= gbX0 AND mlM1(1) < gbX0 + (gbR + 1) * gbBw THEN
          mlBin := IP((mlM1(1) - gbX0) / gbBw);
          REPEAT UNTIL MOUSE(1) == -1;
          gbBinInfo(mlBin);
        END;
      END;
      REPEAT UNTIL MOUSE(1) == -1;
    ELSE
      IF mlK == 4 THEN mlGo := 0; END;                                      // Esc
      IF mlK == 50 THEN gbDropFast(100); END;                               // +
      IF mlK == 45 THEN gbDropAnim(); END;                                  // -
      IF mlK == 7 THEN gbP := MAX(0.05, gbP - 0.05); gbClear(); END;        // Left
      IF mlK == 8 THEN gbP := MIN(0.95, gbP + 0.05); gbClear(); END;        // Right
      IF mlK == 2 THEN gbR := MIN(12, gbR + 1); gbLayout(); gbClear(); END; // Up
      IF mlK == 12 THEN gbR := MAX(3, gbR - 1); gbLayout(); gbClear(); END; // Down
      IF mlK == 30 THEN mlGo := gbMenu(); END;                              // Enter
    END;
  END;
  gbExport();
  PRINT();
  PRINT("Galton Board: " + STRING(gbN, 2, 0) + " balls, R = " + STRING(gbR, 2, 0) + ", p = " + gbF(gbP, 2));
  PRINT("Bin counts (0 .. R) are in GB_CNT.");
END;
