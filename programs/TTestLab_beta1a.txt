//   * Exact values and the printed table's: t critical is worked out

//     column), step 3 shows both.  Where the lecture rounded along the
//     way, step 4 says so and gives the exact value.
//
// FROM THE LIBRARY (searched 31 Oct 2026: no Prime program in the
//   1,658-program corpus works a t test in steps).  Taken from the
//   fleet: the key and finger drains, the D1/D2 reader and the number
//   formats (PowerViz); the checked STUDENT_ICDF with a bisection
//   fallback (CICatcher's cqTCrit); the two-tailed p capped at 1
//   (AnovaLab); the pairs message (RankLab); the Inference app as a
//   check (TestChooser).
//
// HOW THE NUMBERS ARE WORKED OUT (group 1 minus group 2, as the
//   lecture and SPSS do):
//   pooled:    sp = sqrt(((N1-1)s1^2 + (N2-1)s2^2)/(N1+N2-2))
//              SE = sp * sqrt((N1+N2)/(N1*N2)),  df = N1 + N2 - 2
//   separate:  SE = sqrt(s1^2/N1 + s2^2/N2)  (Welch; SPSS's "Equal
//              variances not assumed" row)
//              df = (a+b)^2 / (a^2/(N1-1) + b^2/(N2-1)), a = s1^2/N1,
//              b = s2^2/N2  (Welch-Satterthwaite; fractional)
//   dependent: xD = x1 - x2 for each pair, sD their SD (N-1),
//              SE = sD/sqrt(Npairs),  df = Npairs - 1
//   t obtained = (mean1 - mean2)/SE, or mean(xD)/SE
//   t critical = -STUDENT_ICDF(df, alpha) one-tailed, alpha/2 two-
//              tailed (a lower tail, mirrored: never ICDF(1 - alpha))
//   p: two-tailed 2*STUDENT_CDF(df, -|t|); H1 mu1 < mu2: STUDENT_CDF(
//      df, t); H1 mu1 > mu2: STUDENT_CDF(df, -t).  Lower tails only.
//   Cohen's d: (mean1 - mean2)/sp (independent; SPSS's Cohen's d);
//      mean(xD)/sD (dependent; SPSS's paired Cohen's d).
//   Squares are taken after dividing by the larger SD (Lessons Learned
//   s11, "scale before squaring"), so no SD over- or underflows.
//
// KEYS AND TOUCH (a GETKEY + MOUSE polling loop, Lessons Learned s6):
//   soft keys 1-5 = the five steps, "Menu" = the menu
//   Left / Right = step back / on,  Enter = menu,  Esc = quit
//   the flowchart: tap Yes or No (or the soft keys), OK = go on
// After every form, menu and box the program drains keys AND waits for
// the finger to lift (tt_Drain, tt_TDrain): a form's OK sits where a
// soft key is, and a finger still on it was once read as a tap
// (PowerViz v2.5h on Roger's calculator, 31 Oct 2026).

// ---------------- shared state ----------------
LOCAL ttTY:=1;           // the test: 1 pooled, 2 separate, 3 dependent
LOCAL ttSRC:=1;          // the data: 1 summary numbers, 2 raw scores
LOCAL ttTL:=0;           // tails: 0 two-tailed, -1 H1 mu1 < mu2, 1 mu1 > mu2
LOCAL ttAL:=0.05;        // alpha
LOCAL ttEX:=0;           // the example on screen: 0 none, 1-3 the examples
LOCAL ttOK:=0;           // 1 = the numbers are usable
// independent samples: each group's mean, SD and N
LOCAL ttM1:=0, ttS1:=1, ttN1:=10, ttM2:=0, ttS2:=1, ttN2:=10;
// dependent samples: mean and SD of the differences, number of pairs,
// and the pairs themselves when they came as raw scores
LOCAL ttMD:=0, ttSDv:=1, ttNP:=5, ttSSq:=0;
LOCAL ttX1:={0}, ttX2:={0}, ttXD:={0};
// results
LOCAL ttSE:=1, ttT:=0, ttDF:=1, ttTC:=1, ttTB:=1, ttDFT:=1;
LOCAL ttP:=1, ttD:=0, ttSP:=1, ttREJ:=0, ttREJB:=0;
// words: the two groups, the outcome, and why the numbers were refused
LOCAL ttG1:="Group 1", ttG2:="Group 2", ttDV:="", ttMSG:="";
// palette (Lessons Learned s13), set in tt_Init
LOCAL ttBLUE, ttMAG, ttRED, ttGRN, ttORNG, ttGRAY;
LOCAL ttFOOT, ttGRID, ttLRED, ttLBLU, ttLORN, ttBLK;
LOCAL ttWHT, ttLMAG, ttLGRN;
// the picture's box (pixels)
LOCAL ttPL:=12, ttPR:=308;

// ---------------- waiting ----------------
// Empty the key buffer: the Enter that closed a menu or form must not
// be read as the next keypress.
LOCAL tt_Drain()
BEGIN
 LOCAL dk;
 REPEAT dk := GETKEY; UNTIL dk == -1;
END;

// Wait for the finger to lift, never more than a second (a stuck touch
// reading must not freeze the program).  MOUSE(1) is -1 when nothing
// touches the screen.
LOCAL tt_TDrain()
BEGIN
 LOCAL drT;
 drT := TICKS;
 REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// After a form, a menu or a box: keys AND the finger.
LOCAL tt_Settle()
BEGIN
 tt_Drain();
 tt_TDrain();
END;

// Wait for any key or tap (drain, wait, drain).
LOCAL tt_WaitAny()
BEGIN
 LOCAL waK, waM, waM1;
 tt_Settle();
 REPEAT
  waK := GETKEY;
  waM := MOUSE;
  waM1 := waM(1);
 UNTIL waK > -1 OR SIZE(waM1) > 0;
 tt_Settle();
END;

// ---------------- numbers as text ----------------
// Fixed decimals, rounded first: STRING(x, 2, d) prints scientific
// form below 10^-d, and a rounded value is 0 or at least 10^-d.
LOCAL tt_Fmt(fv, fd)
BEGIN
 RETURN STRING(ROUND(fv,fd),2,fd);
END;

// A probability: "< 0.001" below the floor, "> 0.999" above it.
LOCAL tt_FmtP(pv, pd)
BEGIN
 LOCAL thr;
 thr := 10^(-pd);
 IF pv <> 0 AND ABS(pv) < thr THEN RETURN "< "+STRING(thr,2,pd); END;
 IF pv <= 1 AND ROUND(pv,pd) >= 1 THEN RETURN "> "+STRING(1-thr,2,pd); END;
 RETURN STRING(pv,2,pd);
END;

// " = 0.031", or " < 0.001" when the text already carries its relation
LOCAL tt_Rel(rs)
BEGIN
 IF DIM(rs) > 0 THEN
  IF LEFT(rs,1) == "<" OR LEFT(rs,1) == ">" THEN RETURN " "+rs; END;
 END;
 RETURN " = "+rs;
END;

// A number as it was typed: no trailing zeros, at most 5 decimals
// (1.45 stays 1.45, 44 stays 44, SPSS's 3.28798 stays 3.28798).
LOCAL tt_Num(nv)
BEGIN
 LOCAL nd, nk, ns;
 nk := 5;
 FOR nd FROM 4 DOWNTO 0 DO
  // the fewest decimals that hold the number exactly
  IF ABS(nv*10^nd - ROUND(nv*10^nd,0)) < 0.000001 THEN nk := nd; END;
 END;
 IF nk == 0 THEN RETURN STRING(ROUND(nv,0),1,0); END;
 ns := STRING(ROUND(nv,nk),2,nk);
 IF nk == 5 THEN
  // more than 5 decimals typed: round to 5, then drop end zeros
  WHILE MID(ns,DIM(ns),1) == "0" DO ns := LEFT(ns,DIM(ns)-1); END;
  IF MID(ns,DIM(ns),1) == "." THEN ns := LEFT(ns,DIM(ns)-1); END;
 END;
 RETURN ns;
END;

// A worked value to 4 significant digits, in fixed form (0.05499,
// 1.745, 0.0001058): intermediate steps show enough digits to see where
// a rounded hand calculation would part company.
LOCAL tt_Sig(sv)
BEGIN
 LOCAL sd;
 IF sv == 0 THEN RETURN "0"; END;
 sd := MIN(10, MAX(0, 3 - FLOOR(LOG(ABS(sv)))));
 RETURN STRING(ROUND(sv,sd),2,sd);
END;

// ---------------- the arithmetic ----------------
// The row of the printed t table the lecture would read.  The table's
// rows are 1-30, 40, 60 and 120 (the lecture's appendix 1; no
// infinity row).  The rule is the lecture's, "the value that is
// closest to it" (df 73 -> row 60).  A tie goes to the smaller row,
// which is the safer one (its t critical is larger): 35 -> 30, 4.5 -> 4.
// Past 120, row 120.
LOCAL tt_TabDF(tdf)
BEGIN
 LOCAL tdr;
 IF tdf <= 30 THEN RETURN MAX(1,CEILING(tdf-0.5)); END;
 IF tdf >= 120 THEN RETURN 120; END;
 tdr := 30;
 IF tdf - 30 > 5 THEN tdr := 40; END;      // nearer 40 than 30
 IF tdf - 40 > 10 THEN tdr := 60; END;     // nearer 60 than 40
 IF tdf - 60 > 30 THEN tdr := 120; END;    // nearer 120 than 60
 RETURN tdr;
END;

// t critical (its size) for df, at alpha, for the tails on screen: the
// point with a = alpha (one-tailed) or alpha/2 (two-tailed) beyond it.
// A lower-tail quantile, mirrored: -STUDENT_ICDF(df, a), never
// ICDF(1 - a) (Lessons Learned s3).  STUDENT_ICDF matched SciPy to 12
// digits on the emulator but is not yet on the device-verified list,
// so, as CICatcher's cqTCrit does: try it inside IFERR, check the
// answer with STUDENT_CDF (device-verified), and if either fails,
// bisect on STUDENT_CDF.
LOCAL tt_Crit(cdf)
BEGIN
 LOCAL ca, cv, cok, clo, chi, cmid;
 LOCAL cj;
 ca := IFTE(ttTL == 0,ttAL/2,ttAL);
 cok := 0; cv := 0;
 IFERR
  cv := -STUDENT_ICDF(cdf,ca);
  IF cv > 0 THEN
   IF ABS(STUDENT_CDF(cdf,-cv)/ca - 1) < 0.000001 THEN cok := 1; END;
  END;
 THEN
  cok := 0;
 END;
 IF cok THEN RETURN cv; END;
 // the fallback: bracket (double until the tail is below a), then 60
 // halvings, which leave less than 1E-12 of the bracket
 clo := 0; chi := 4;
 WHILE STUDENT_CDF(cdf,-chi) > ca AND chi < 1000000 DO
  clo := chi; chi := 2*chi;
 END;
 FOR cj FROM 1 TO 60 DO
  cmid := (clo+chi)/2;
  IF STUDENT_CDF(cdf,-cmid) > ca THEN clo := cmid; ELSE chi := cmid; END;
 END;
 RETURN (clo+chi)/2;
END;

// Is a t beyond the critical value cv?  (two-tailed: either side;
// H1 mu1 < mu2: below -cv; H1 mu1 > mu2: above cv)
LOCAL tt_Beyond(bt, bc)
BEGIN
 IF ttTL == 0 THEN RETURN IFTE(ABS(bt) > bc,1,0); END;
 IF ttTL < 0 THEN RETURN IFTE(bt < -bc,1,0); END;
 RETURN IFTE(bt > bc,1,0);
END;

// The p value for t on df, for the tails on screen (lower tails only).
LOCAL tt_PVal(pt, pdf)
BEGIN
 // two-tailed: capped at 1, as AnovaLab's (t = 0 gives 2 x 0.5)
 IF ttTL == 0 THEN RETURN MIN(1,2*STUDENT_CDF(pdf,-ABS(pt))); END;
 IF ttTL < 0 THEN RETURN STUDENT_CDF(pdf,pt); END;
 RETURN STUDENT_CDF(pdf,-pt);
END;

// Pooled (pk = 1) or separate (pk = 2) t for the independent samples on
// screen: returns {SE, t, df, pooled SD}.  The SDs are divided by the
// larger first (cm), so nothing squares out of range.
LOCAL tt_Ind(pk)
BEGIN
 LOCAL cm, ca, cb, cv, csp, cse;
 LOCAL cdf;
 cm := MAX(ttS1,ttS2);
 ca := (ttS1/cm)^2;
 cb := (ttS2/cm)^2;
 csp := cm*(((ttN1-1)*ca + (ttN2-1)*cb)/(ttN1+ttN2-2))^.5;
 IF pk == 1 THEN
  cse := csp*((ttN1+ttN2)/(ttN1*ttN2))^.5;
  cdf := ttN1+ttN2-2;
 ELSE
  cv := ca/ttN1 + cb/ttN2;
  cse := cm*cv^.5;
  cdf := cv^2/((ca/ttN1)^2/(ttN1-1) + (cb/ttN2)^2/(ttN2-1));
 END;
 RETURN {cse, (ttM1-ttM2)/cse, cdf, csp};
END;

// Mean, SD and sum of squared deviations of the pairs' differences,
// from ttX1 and ttX2 (two passes: the mean first).
LOCAL tt_Pairs()
BEGIN
 LOCAL pj, ps;
 ttNP := SIZE(ttX1);
 ttXD := ttX1;
 ps := 0;
 FOR pj FROM 1 TO ttNP DO
  ttXD(pj) := ttX1(pj) - ttX2(pj);
  ps := ps + ttXD(pj);
 END;
 ttMD := ps/ttNP;
 ttSSq := 0;
 FOR pj FROM 1 TO ttNP DO ttSSq := ttSSq + (ttXD(pj)-ttMD)^2; END;
 ttSDv := (ttSSq/(ttNP-1))^.5;
END;

// The lecture's printed t table (its appendix 1) differs from the exact
// values, rounded to 3 decimals, in five cells, all in the last column
// (one-tailed .0005, two-tailed .001): rows 2, 3, 5, 7 and 23 print
// 31.598, 12.941, 6.859, 5.405 and 3.767 where the exact values are
// 31.599, 12.924, 6.869, 5.408 and 3.768 (SciPy; every other cell
// agrees, verify A6).  Returns the printed value when the row and
// column on screen are one of those five, else 0.  TTestLab shows the
// printed value as the table's (a student reads it there) and step 3
// gives the exact one beside it.
LOCAL tt_BookCell()
BEGIN
 LOCAL bk, bkr, bkv;
 IF ttAL <> IFTE(ttTL == 0,0.001,0.0005) THEN RETURN 0; END;
 bkr := {2,3,5,7,23};
 bkv := {31.598,12.941,6.859,5.405,3.767};
 FOR bk FROM 1 TO 5 DO
  IF ttDFT == bkr(bk) THEN RETURN bkv(bk); END;
 END;
 RETURN 0;
END;

// Everything the five steps show, from the numbers on screen.  Sets
// ttOK = 0 and ttMSG when the numbers cannot give a t.
LOCAL tt_Calc()
BEGIN
 LOCAL cr, cbk;
 ttOK := 1; ttMSG := "";
 IF ttTY == 3 THEN
  IF ttSRC == 2 THEN tt_Pairs(); END;
  IF ttNP < 2 THEN ttOK := 0; ttMSG := "A dependent-samples t test needs at least 2 pairs."; RETURN 0; END;
  IF ttSDv <= 0 THEN ttOK := 0; ttMSG := "Every pair differs by the same amount (sD = 0): there is no t."; RETURN 0; END;
  ttSE := ttSDv/ttNP^.5;
  ttDF := ttNP-1;
  ttT := ttMD/ttSE;
  ttD := ttMD/ttSDv;
  ttSP := ttSDv;
 ELSE
  IF ttN1 < 2 OR ttN2 < 2 THEN ttOK := 0; ttMSG := "Each group needs N of at least 2."; RETURN 0; END;
  IF ttS1 < 0 OR ttS2 < 0 THEN ttOK := 0; ttMSG := "An SD cannot be negative."; RETURN 0; END;
  IF ttS1 == 0 AND ttS2 == 0 THEN ttOK := 0; ttMSG := "Both SDs are 0: every score is the same, so there is no t."; RETURN 0; END;
  cr := tt_Ind(ttTY);
  ttSE := cr(1); ttT := cr(2); ttDF := cr(3); ttSP := cr(4);
  ttD := (ttM1-ttM2)/ttSP;
 END;
 ttTC := tt_Crit(ttDF);
 ttDFT := tt_TabDF(ttDF);
 // the printed table's value, as a student would read it: the exact
 // value to 3 decimals, or the lecture's own figure in its five odd
 // cells (tt_BookCell)
 cbk := tt_BookCell();
 ttTB := IFTE(cbk > 0,cbk,ROUND(tt_Crit(ttDFT),3));
 ttP := tt_PVal(ttT,ttDF);
 ttREJ := tt_Beyond(ttT,ttTC);
 ttREJB := tt_Beyond(ttT,ttTB);
 RETURN 1;
END;

BEGIN

BEGIN

// H1 in symbols
LOCAL tt_H1S()
BEGIN
 IF ttTL == 0 THEN RETURN "H"+CHAR(8321)+": μ"+CHAR(8321)+" ≠ μ"+CHAR(8322); END;
 IF ttTL < 0 THEN RETURN "H"+CHAR(8321)+": μ"+CHAR(8321)+" < μ"+CHAR(8322); END;
 RETURN "H"+CHAR(8321)+": μ"+CHAR(8321)+" > μ"+CHAR(8322);
END;

// "the alpha on screen", as ".05"-style text with house decimals
LOCAL tt_AlS()
BEGIN
 RETURN "α"+tt_Rel(tt_FmtP(ttAL,IFTE(ttAL < 0.001,4,3)));
END;

// t critical with its sign(s), as the decision rule needs it
LOCAL tt_CritS(cv)
BEGIN
 IF ttTL == 0 THEN RETURN "±"+tt_Fmt(cv,3); END;
 IF ttTL < 0 THEN RETURN tt_Fmt(-cv,3); END;
 RETURN tt_Fmt(cv,3);
END;

// "t(73)" or "t(72.44)"
LOCAL tt_TDf()
BEGIN
 IF ttTY == 2 THEN RETURN "t("+tt_Fmt(ttDF,2)+")"; END;
 RETURN "t("+STRING(ttDF,1,0)+")";
END;

// The t curve on df from y0 to y1: the rejection region red, the rest
// pale blue; t critical dashed red; with shT, t obtained in magenta and
// with shP the p value's area shaded magenta.  One pixel column at a
// time (Lessons Learned s6: shade by columns, never by samples).
LOCAL tt_Curve(cy0, cy1, shT, shP)
BEGIN
 LOCAL cxr, cpk, cH, cpx, cx, cf;
 LOCAL cyp, cyq, cin, cpin, ctk, ctx;
 LOCAL cdec, cstp, cs;
 // the window: -4..4, wider when t critical or t obtained lies past 3.2
 cxr := MAX(4, ttTC + 0.8);
 IF shT THEN cxr := MAX(cxr, ABS(ttT) + 0.8); END;
 cxr := MIN(cxr, 12);
 cpk := STUDENT(ttDF,0);
 cH := cy1 - cy0 - 4;
 cyq := cy1;
 FOR cpx FROM ttPL TO ttPR DO
  cx := -cxr + (cpx-ttPL)/(ttPR-ttPL)*2*cxr;
  cf := STUDENT(ttDF,cx);
  cyp := cy1 - IP(cf/cpk*cH);
  cin := tt_Beyond(cx,ttTC);
  // p value's area: beyond t obtained, in the direction(s) of H1
  cpin := 0;
  IF shP THEN
   IF ttTL == 0 THEN cpin := IFTE(ABS(cx) >= ABS(ttT),1,0); END;
   IF ttTL < 0 THEN cpin := IFTE(cx <= ttT,1,0); END;
   IF ttTL > 0 THEN cpin := IFTE(cx >= ttT,1,0); END;
  END;
  IF cyp < cy1 THEN
   IF cpin THEN
    LINE_P(G1,cpx,cyp,cpx,cy1,ttLMAG);
   ELSE
    LINE_P(G1,cpx,cyp,cpx,cy1,IFTE(cin,ttLRED,ttLBLU));
   END;
  END;
  // the curve itself, joined to the column before
  IF cpx > ttPL THEN LINE_P(G1,cpx-1,cyq,cpx,cyp,ttBLUE); END;
  cyq := cyp;
 END;
 // the axis, with whole-number ticks (every 2 when the window is wide)
 LINE_P(G1,ttPL,cy1,ttPR,cy1,ttBLK);
 cstp := IFTE(cxr > 6,2,1);
 FOR ctk FROM -FLOOR(cxr/cstp)*cstp TO FLOOR(cxr/cstp)*cstp STEP cstp DO
  ctx := ttPL + IP((ctk+cxr)/(2*cxr)*(ttPR-ttPL));
  LINE_P(G1,ctx,cy1,ctx,cy1+3,ttBLK);
  cs := STRING(ctk,1,0);
  TEXTOUT_P(cs,G1,ctx-3*DIM(cs),cy1+4,1,ttGRAY);
 END;
 // t critical: dashed red line(s)
 FOR cs FROM -1 TO 1 STEP 2 DO
  IF (ttTL == 0) OR (ttTL == cs) THEN
   ctx := ttPL + IP((cs*ttTC+cxr)/(2*cxr)*(ttPR-ttPL));
   FOR cpx FROM cy0 TO cy1 STEP 4 DO LINE_P(G1,ctx,cpx,ctx,MIN(cy1,cpx+1),ttRED); END;
  END;
 END;
 // t obtained: a magenta line, its value above it
 IF shT THEN
  ctx := ttPL + IP((MAX(-cxr,MIN(cxr,ttT))+cxr)/(2*cxr)*(ttPR-ttPL));
  LINE_P(G1,ctx,cy0,ctx,cy1,ttMAG);
  LINE_P(G1,ctx+1,cy0,ctx+1,cy1,ttMAG);
  cs := "t obt = "+tt_Fmt(ttT,2);
  cdec := MIN(ttPR-IP(5.5*DIM(cs)), MAX(ttPL, ctx-IP(2.75*DIM(cs))));
  TEXTOUT_P(cs,G1,cdec,cy0-11,1,ttMAG);
 END;
END;

// ---------------- step 1: the hypotheses and the data ----------------
LOCAL tt_Step1()
BEGIN
 LOCAL sy, sj, sn, sxd;
 tt_Frame("Step 1. The hypotheses");

// ---------------- pooled or separate? ----------------
LOCAL tt_Compare()
BEGIN
 LOCAL cp, cs, cpp, csp, ccp, ccs;
 LOCAL cw, ctl;
 IF ttTY == 3 THEN
  MSGBOX("Pooled or separate variances is a question for INDEPENDENT samples. Dependent samples have one SD, of the differences.");
  tt_Settle();
  RETURN 0;
 END;
 cp := tt_Ind(1); cs := tt_Ind(2);
 cpp := tt_PVal(cp(2),cp(3)); csp := tt_PVal(cs(2),cs(3));
 ccp := tt_Crit(cp(3)); ccs := tt_Crit(cs(3));
 tt_Frame("Pooled or separate variances?");
 TEXTOUT_P("Pooled",G1,120,30,1,ttBLUE);
 TEXTOUT_P("Separate",G1,215,30,1,ttMAG);
 TEXTOUT_P("SE",G1,10,44,1,ttGRAY);
 TEXTOUT_P(tt_Sig(cp(1)),G1,120,44,1,ttBLK);
 TEXTOUT_P(tt_Sig(cs(1)),G1,215,44,1,ttBLK);
 TEXTOUT_P("t obtained",G1,10,57,1,ttGRAY);
 TEXTOUT_P(tt_Fmt(cp(2),3),G1,120,57,1,ttBLK);
 TEXTOUT_P(tt_Fmt(cs(2),3),G1,215,57,1,ttBLK);
 TEXTOUT_P("df",G1,10,70,1,ttGRAY);
 TEXTOUT_P(STRING(cp(3),1,0),G1,120,70,1,ttBLK);
 TEXTOUT_P(tt_Fmt(cs(3),2),G1,215,70,1,ttBLK);
 TEXTOUT_P("t crit",G1,10,83,1,ttGRAY);
 TEXTOUT_P(tt_CritS(ccp),G1,120,83,1,ttBLK);
 TEXTOUT_P(tt_CritS(ccs),G1,215,83,1,ttBLK);
 TEXTOUT_P("p ("+tt_TailS()+")",G1,10,96,1,ttGRAY);
 TEXTOUT_P(tt_FmtP(cpp,3),G1,120,96,1,ttBLK);
 TEXTOUT_P(tt_FmtP(csp,3),G1,215,96,1,ttBLK);
 TEXTOUT_P("decision",G1,10,109,1,ttGRAY);
 TEXTOUT_P(IFTE(tt_Beyond(cp(2),ccp),"reject H"+CHAR(8320),"retain H"+CHAR(8320)),G1,120,109,1,ttBLK);
 TEXTOUT_P(IFTE(tt_Beyond(cs(2),ccs),"reject H"+CHAR(8320),"retain H"+CHAR(8320)),G1,215,109,1,ttBLK);
 // why they agree or differ: any mix of equal or unequal N and SD
 ctl := 128;
 IF ttN1 == ttN2 THEN
  TEXTOUT_P("N"+CHAR(8321)+" = N"+CHAR(8322)+": the same t either way; only df differs.",G1,8,ctl,1,ttBLK);
 ELSE
  IF ttS1 == ttS2 THEN
   TEXTOUT_P("s"+CHAR(8321)+" = s"+CHAR(8322)+": the same t either way; only df differs.",G1,8,ctl,1,ttBLK);
  ELSE
   cw := IFTE((ttN1 < ttN2 AND ttS1 > ttS2) OR (ttN2 < ttN1 AND ttS2 > ttS1),1,0);
   IF cw THEN
    // Welch 1938; Ruxton 2006; Delacre et al. 2017
    TEXTOUT_P("The SMALLER group has the LARGER SD: the pooled",G1,8,ctl,1,ttRED);
    TEXTOUT_P("test rejects a true H"+CHAR(8320)+" more often than α says.",G1,8,ctl+13,1,ttRED);
   ELSE
    TEXTOUT_P("The LARGER group has the LARGER SD: the pooled",G1,8,ctl,1,ttBLUE);
    TEXTOUT_P("test is too cautious: it misses real effects.",G1,8,ctl+13,1,ttBLUE);
   END;
  END;
 END;
 TEXTOUT_P("The lecture: pooled if the population variances",G1,8,160,1,ttGRAY);
 TEXTOUT_P("are equal, separate if not (Figure 11.2).  Many",G1,8,173,1,ttGRAY);
 TEXTOUT_P("now use separate always (Delacre et al. 2017).",G1,8,186,1,ttGRAY);
 TEXTOUT_P("any key or tap = back",G1,8,222,1,ttFOOT);
 BLIT_P(G0,0,0,G1);
 tt_WaitAny();
END;

// Returns the test chosen (1 pooled, 2 separate, 3 dependent), or 0 if
// left with Esc.
LOCAL tt_Flow()
BEGIN
 LOCAL fq1, fq2, fk, fm, fm1, fx;
 LOCAL fy, fa, fdone;
 fq1 := -1; fq2 := -1; fdone := 0;
 tt_Settle();
 REPEAT
  tt_DrawFlow(fq1,fq2);
  REPEAT
   fk := GETKEY;
   fm := MOUSE;
   fm1 := fm(1);
  UNTIL fk > -1 OR SIZE(fm1) > 0;
  fa := 0;                         // 1 yes, 2 no, 3 back, 6 OK, 9 quit
  IF SIZE(fm1) > 0 THEN
   fx := fm1(1); fy := fm1(2);
   tt_TDrain();
   IF fy >= 218 THEN
    fa := MIN(6,IP(fx/53.4)+1);
   ELSE
    // the drawn Yes and No boxes, for the question being asked
    IF fq1 == -1 AND fy >= 62 AND fy <= 80 THEN
     IF fx >= 60 AND fx <= 120 THEN fa := 1; END;
     IF fx >= 200 AND fx <= 260 THEN fa := 2; END;
    END;
    IF fq1 == 1 AND fq2 == -1 AND fy >= 124 AND fy <= 142 THEN
     IF fx >= 10 AND fx <= 70 THEN fa := 1; END;
     IF fx >= 110 AND fx <= 170 THEN fa := 2; END;
    END;
   END;
  END;
  IF fk == 4 THEN fa := 9; END;
  IF fk == 30 THEN fa := 6; END;
  IF fa == 1 OR fa == 2 THEN
   IF fq1 == -1 THEN
    fq1 := IFTE(fa == 1,1,0);
   ELSE
    IF fq1 == 1 AND fq2 == -1 THEN fq2 := IFTE(fa == 1,1,0); END;
   END;
  END;
  IF fa == 3 THEN
   IF fq2 <> -1 THEN fq2 := -1; ELSE fq1 := -1; END;
  END;
  IF fa == 6 THEN
   IF fq1 == 0 OR fq2 <> -1 THEN fdone := 1; END;
  END;
  IF fa == 9 THEN fdone := 2; END;
  tt_Drain();
 UNTIL fdone > 0;
 tt_Settle();
 IF fdone == 2 THEN RETURN 0; END;
 IF fq1 == 0 THEN RETURN 3; END;
 RETURN IFTE(fq2 == 1,1,2);
END;

// ---------------- data in ----------------
// A list from the Statistics 1Var app: D1 or D2, {} if it is not there
// (the app never opened) or not a list.
LOCAL tt_ReadD(rdi)
BEGIN
 LOCAL rdl;
 rdl := {};
 IFERR
  IF rdi == 1 THEN rdl := D1; END;
  IF rdi == 2 THEN rdl := D2; END;
 THEN
  rdl := {};
 END;
 IF TYPE(rdl) <> 6 THEN rdl := {}; END;
 RETURN rdl;
END;

// Mean and SD (N - 1) of a list: {mean, SD, N}
LOCAL tt_MSN(ml)
BEGIN
 LOCAL mi, ms, mn, mm;
 mn := SIZE(ml);
 ms := 0;
 FOR mi FROM 1 TO mn DO ms := ms + ml(mi); END;
 mm := ms/mn;
 ms := 0;
 FOR mi FROM 1 TO mn DO ms := ms + (ml(mi)-mm)^2; END;
 RETURN {mm, IFTE(mn > 1,(ms/(mn-1))^.5,0), mn};
END;

// Tails and alpha: two menus.  The alphas are the printed t table's
// columns.  Returns 0 if a menu was cancelled.
LOCAL tt_AskTails()
BEGIN
 LOCAL ak, aa, al;
 ak := 0;
 CHOOSE(ak,"Tails",{"Two-tailed: μ₁ ≠ μ₂","One-tailed: μ₁ < μ₂","One-tailed: μ₁ > μ₂"});
 tt_Settle();
 IF ak == 0 THEN RETURN 0; END;
 ttTL := IFTE(ak == 1,0,IFTE(ak == 2,-1,1));
 aa := 0;
 IF ttTL == 0 THEN
  CHOOSE(aa,"α (two-tailed)",{"0.20","0.10","0.05","0.02","0.01","0.001"});
  al := {0.20,0.10,0.05,0.02,0.01,0.001};
  IF aa > 0 THEN ttAL := al(aa); END;
 ELSE
  CHOOSE(aa,"α (one-tailed)",{"0.10","0.05","0.025","0.01","0.005","0.0005"});
  al := {0.10,0.05,0.025,0.01,0.005,0.0005};
  IF aa > 0 THEN ttAL := al(aa); END;
 END;
 tt_Settle();
 RETURN IFTE(aa > 0,1,0);
END;

// The numbers for the test chosen.  Returns 1 when they are in and
// usable, 0 if cancelled.
LOCAL tt_AskData()
BEGIN
 LOCAL ds, dok, dl1, dl2, dr1, dr2;
 LOCAL dm, dsd, dn;
 ds := 0;
 IF ttTY <> 3 THEN
  CHOOSE(ds,"The data",{"Type each group's mean, SD, N","Raw scores in D1 and D2"});
 ELSE
  CHOOSE(ds,"The pairs",{"Pairs in D1 (x₁) and D2 (x₂)","Type mean, SD of differences"});
 END;
 tt_Settle();
 IF ds == 0 THEN RETURN 0; END;
 ttEX := 0; ttG1 := "Group 1"; ttG2 := "Group 2"; ttDV := "";
 IF ttTY <> 3 THEN
  IF ds == 1 THEN
   ttSRC := 1;
   dok := INPUT({ttM1,ttS1,ttN1,ttM2,ttS2,ttN2},"Two independent samples",
    {"Mean 1:","SD 1:","N 1:","Mean 2:","SD 2:","N 2:"},
    {"Sample 1 mean","Sample 1 SD (s, with N - 1)","Sample 1 size","Sample 2 mean","Sample 2 SD (s, with N - 1)","Sample 2 size"});
   tt_Settle();
   IF dok == 0 THEN RETURN 0; END;
  ELSE
   ttSRC := 2;
   dl1 := tt_ReadD(1); dl2 := tt_ReadD(2);
   IF SIZE(dl1) < 2 OR SIZE(dl2) < 2 THEN
    MSGBOX("Put sample 1 in D1 and sample 2 in D2 of the Statistics 1Var app: at least 2 scores in each.");
    tt_Settle();
    RETURN 0;
   END;
   dr1 := tt_MSN(dl1); dr2 := tt_MSN(dl2);
   ttM1 := dr1(1); ttS1 := dr1(2); ttN1 := dr1(3);
   ttM2 := dr2(1); ttS2 := dr2(2); ttN2 := dr2(3);
   ttG1 := "D1"; ttG2 := "D2";
  END;
 ELSE
  IF ds == 1 THEN
   ttSRC := 2;
   dl1 := tt_ReadD(1); dl2 := tt_ReadD(2);
   IF SIZE(dl1) < 2 OR SIZE(dl1) <> SIZE(dl2) THEN
    MSGBOX("Put the pairs in D1 and D2 of the Statistics 1Var app, a pair per row: the same number of values in each, at least 2.");
    tt_Settle();
    RETURN 0;
   END;
   ttX1 := dl1; ttX2 := dl2;
   ttG1 := "D1"; ttG2 := "D2";
  ELSE
   ttSRC := 1;
   dm := ttMD; dsd := ttSDv; dn := ttNP;
   dok := INPUT({dm,dsd,dn},"Dependent samples",{"Mean xD:","SD sD:","N pairs:"},
    {"Mean of the differences x1 - x2","SD of the differences (N - 1)","Number of pairs"});
   tt_Settle();
   IF dok == 0 THEN RETURN 0; END;
   ttMD := dm; ttSDv := dsd; ttNP := IP(dn);
  END;
 END;
 RETURN 1;
END;

// ---------------- the menu ----------------
// Returns 1 to show the steps again, 0 to quit.
LOCAL tt_Menu()
BEGIN
 LOCAL ms, mt, mo;
 ms := 0;
 CHOOSE(ms,"TTestLab",{"Edit the numbers","Tails and α","Pooled or separate?","Choose the test","Examples","Help","Quit TTestLab"});
 tt_Settle();
 IF ms == 1 THEN
  IF tt_AskData() THEN tt_Calc(); END;
 END;
 IF ms == 2 THEN
  IF tt_AskTails() THEN tt_Calc(); END;
 END;
 IF ms == 3 THEN tt_Compare(); END;
 IF ms == 4 THEN
  mt := tt_Flow();
  IF mt > 0 THEN
   // The independent numbers serve both independent tests, so pooled
   // <-> separate keeps them; a change to or from dependent samples
   // needs new numbers, and a cancelled form keeps the old test.
   mo := ttTY;
   ttTY := mt;
   IF (mt == 3) <> (mo == 3) THEN
    IF tt_AskData() == 0 THEN ttTY := mo; END;
   END;
   IF ttTY == mt THEN tt_AskTails(); END;
   tt_Calc();
  END;
 END;
 IF ms == 5 THEN
  mt := 0;
  CHOOSE(mt,"Examples",{"Age at arrest (pooled)","Matched jails (dependent)","Drug use by sex (separate)"});
  tt_Settle();
  IF mt > 0 THEN tt_Example(mt); tt_Calc(); END;
 END;
 IF ms == 6 THEN tt_Help(); END;
 IF ms == 7 THEN RETURN 0; END;
 RETURN 1;
END;

// ---------------- the steps ----------------
LOCAL tt_Page(pst)
BEGIN
 IF pst == 1 THEN tt_Step1(); END;
 IF pst == 2 THEN tt_Step2(); END;
 IF pst == 3 THEN tt_Step3(); END;
 IF pst == 4 THEN tt_Step4(); END;
 IF pst == 5 THEN tt_Step5(); END;
END;

// The five steps: soft keys 1-5 or Left/Right move, Menu or Enter opens
// the menu, Esc quits.  Returns when the user quits.
LOCAL tt_Steps()
BEGIN
 LOCAL sst, sk, sm, sm1, sgo, sslt;
 sst := 1; sgo := 1;
 WHILE sgo DO
  IF ttOK == 0 THEN
   // say once why the numbers give no t, then the menu
   IF DIM(ttMSG) > 0 THEN MSGBOX(ttMSG); tt_Settle(); ttMSG := ""; END;
   sgo := tt_Menu();
  ELSE
   tt_Page(sst);
   tt_Settle();
   REPEAT
    sk := GETKEY;
    sm := MOUSE;
    sm1 := sm(1);
   UNTIL sk > -1 OR SIZE(sm1) > 0;
   sslt := 0;
   IF SIZE(sm1) > 0 THEN
    IF sm1(2) >= 218 THEN sslt := MIN(6,IP(sm1(1)/53.4)+1); END;
    tt_TDrain();
   END;
   IF sslt >= 1 AND sslt <= 5 THEN sst := sslt; END;
   IF sk == 7 THEN sst := MAX(1,sst-1); END;
   IF sk == 8 THEN sst := MIN(5,sst+1); END;
   IF sk == 4 THEN sgo := 0; END;
   IF sk == 30 OR sslt == 6 THEN
    tt_Settle();
    sgo := tt_Menu();
   END;
  END;
 END;
END;

// ---------------- main (zero-argument signature) ----------------
EXPORT TTESTLAB()
BEGIN
 LOCAL ms, mt;
 tt_Init();
 ms := 0;
 CHOOSE(ms,"TTestLab: t tests in 5 steps",{"Choose the test (Figure 11.2)","Example: age at arrest","Example: matched jails","Example: drug use by sex","Help"});
 tt_Settle();
 IF ms == 0 THEN RETURN "TTestLab: cancelled"; END;
 IF ms == 1 THEN
  mt := tt_Flow();
  IF mt == 0 THEN RETURN "TTestLab: cancelled"; END;
  ttTY := mt;
  IF tt_AskData() == 0 THEN RETURN "TTestLab: cancelled"; END;
  IF tt_AskTails() == 0 THEN RETURN "TTestLab: cancelled"; END;
 END;
 IF ms >= 2 AND ms <= 4 THEN tt_Example(ms-1); END;
 IF ms == 5 THEN
  tt_Help();
  tt_Example(1);
 END;
 tt_Calc();
 tt_Steps();
 IF ttOK == 0 THEN RETURN "TTestLab"; END;
 RETURN "TTestLab: "+tt_TDf()+" = "+tt_Fmt(ttT,2)+", p"+tt_Rel(tt_FmtP(ttP,3))+", "+tt_TailS();
END;
