//
// THE LESSON
//   An examiner takes n = 30 proficiency tests and makes 0 errors.
//   That is not a zero error rate: an examiner whose true rate is
//   5 % would also score 0 of 30 about one time in five
//   (0.95^30 = 21.5 %).  The honest summary is the UPPER 95 %
//   CONFIDENCE BOUND: the largest error rate that still produces
//   a record this good at least 5 % of the time.  For 0 errors in
//   n tests it is 1 - 0.05^(1/n) (9.50 % for n = 30); the "rule
//   of three", 3/n, is its large-n LIMIT.
//
//   PCAST (2016), Forensic Science in Criminal Courts, section 4.2,
//   p. 51 and its footnote 116: "When reporting a false positive
//   rate to a jury, it is scientifically important to state the
//   'upper 95 percent one-sided confidence bound'".  Appendix A,
//   pp. 152-153: "Upper 95 percent one-sided confidence bounds
//   should thus be used ... (In this report, we used the Clopper-
//   Pearson/Exact Binomial method.)"  That is the bound computed
//   here.  Section 4.6, pp. 57-58, is the caveat: proficiency
//   tests are not blind casework.
//
// WHAT YOU SEE (main screen)
//   Blue curve ....... P(k or fewer errors in n | true rate p):
//                      for k = 0 it is (1 - p)^n
//   Orange dots ...... the alpha line (0.05, one-sided)
//   Red marker ....... where the curve crosses alpha: the upper
//                      bound, with its value
//   Green dots ....... the rule-of-three line 3/n (for k > 0 or
//                      another alpha: its Poisson version)
//   Green / red bands  rates the record allows / rules out
//   Magenta marker ... the true rate under your finger, and the
//                      chance of the record at that rate
//   Right edge ....... the n slider (5 to 1000, log scale)
//
// KEYS AND TOUCH (main screen)
//   drag along the plot ..... the true rate (live)
//   drag the right edge ..... n (live; the window refits on lift)
//   Left / Right ............ the true rate down / up a step
//   Up / Down ............... n + 1 / n - 1
//   + / - ................... n up / down a step (10, 15, 20, 25,
//                             30, 40, 50, 75, 100, ...)
//   soft keys, page 1 ....... n, k | Rate | Why 3 | Bayes | Sim | More
//   soft keys, page 2 ....... Trawl | Log y | Help | Eng | Quit | More
//   Enter = menu, Esc = quit
//
// THE MATHS (every line checked against SciPy, verify_ErrorRate.py)
//   Bound, k = 0:  (1 - p)^n = alpha   so   p = 1 - alpha^(1/n)
//   Bound, k > 0:  BINOMIAL_CDF(n, p, k) = alpha, solved by
//                  bisection (45 halvings); this IS the one-sided
//                  Clopper-Pearson bound, beta.ppf(1-alpha, k+1, n-k)
//   Rule:          n * bound -> -ln(alpha) = ln 20 = 2.996 (k = 0);
//                  for k > 0 the Poisson bound mu with
//                  P(Poisson(mu) <= k) = alpha, which is
//                  CHISQUARE_ICDF(2k + 2, 1 - alpha) / 2
//   Bayes:         prior Beta(a, b), posterior Beta(a+k, b+n-k);
//                  mean (a+k)/(a+b+n); the upper credible bound by
//                  bisection on the Beta CDF, which is
//                  FISHER_CDF(2a, 2b, b x / (a (1 - x)))
//   Simulation:    each programme jumps from one error to the next;
//                  the run of clean tests before an error is
//                  geometric, floor(ln(U) / ln(1 - p))
//   Trawl:         P(at least one chance match) = 1 - (1 - RMP)^N
//
// THREE ENGINES, EACH WITH ONE HONEST JOB
//   PPL ......... draws every screen, reads every touch, and
//                 computes every number that is drawn
//   CAS ......... exact TEXT: fractions, closed forms, limits,
//                 series.  Called on events only (a page opening,
//                 n changing), never in a drag; through ONE routine
//                 (erlCas); every CAS line has a decimal fallback
//                 and a badge, "exact (CAS)" or "decimal"
//   MicroPython . 10,000 simulated proficiency programmes; if it
//                 fails, PPL simulates 1,000 and the badge says so
//

//   ER_UPPER  the upper bound        ER_N, ER_K, ER_ALPHA  the record

// A number as text with a FIXED number of decimals.  Mode 2 keeps
// trailing zeros; mode 1 would ignore the digits argument.
erlFix(fxV, fxD)
BEGIN
  RETURN STRING(fxV, 2, fxD);
END;

// A whole number as plain digits: no digit-grouping marks, no
// decimal point, no exponent (the CAS kit's xxIStr).  Home's
// STRING() follows the Home number format, which the CAS cannot
// read, so every piece of CAS text is built from this routine.
erlIStr(isN)
BEGIN
  LOCAL dS, dV, dD;
  dV := ROUND(ABS(isN), 0);
  IF dV == 0 THEN RETURN "0"; END;
  dS := "";
  WHILE dV > 0 DO
    dD := dV MOD 10;
    dS := CHAR(48 + dD) + dS;
    dV := (dV - dD) / 10;
  END;
  IF isN < 0 THEN dS := "-" + dS; END;
  RETURN dS;
END;

// A whole number with thousands commas, for the screen only:
// 10000 -> "10,000".
erlComma(cmV)
BEGIN
  LOCAL cmS, cmO, cmL, cmI;
  cmS := erlIStr(cmV);
  cmO := "";
  cmL := DIM(cmS);
  FOR cmI FROM 1 TO cmL DO
    cmO := cmO + MID(cmS, cmI, 1);
    IF (cmL - cmI) MOD 3 == 0 AND cmI < cmL THEN cmO := cmO + ","; END;
  END;
  RETURN cmO;
END;

// The smallest "nice" number (1, 2 or 5 times a power of ten) at
// least as big as nnV -- Heckbert's rule, for ticks and windows.
// The tiny tolerances keep 0.2 / 0.1 = 2 from reading as 2.0000001.
erlNice(nnV)
BEGIN
  LOCAL nnE, nnF;
  IF nnV <= 0 THEN RETURN 1; END;
  nnE := FLOOR(LOG(nnV));
  nnF := nnV / 10^nnE;
  IF nnF <= 1.000001 THEN RETURN 10^nnE; END;
  IF nnF <= 2.000001 THEN RETURN 2 * 10^nnE; END;
  IF nnF <= 5.000001 THEN RETURN 5 * 10^nnE; END;
  RETURN 10^(nnE + 1);
END;

// A probability as a percentage with sensible decimals:
// 0.2146 -> "21.5 %", 0.09503 -> "9.50 %", 0.00299 -> "0.299 %".
erlPctS(pcV)
BEGIN
  LOCAL pcP, pcD;
  pcP := 100 * pcV;
  IF pcP <= 0 THEN RETURN "0 %"; END;
  IF pcP < 0.0001 THEN RETURN "< 0.0001 %"; END;
  // just under 100 %: enough decimals that it never reads as 100
  // (1 - e^-10 = 99.9955 %, not "100.0 %")
  IF pcP > 99.9 AND pcP < 100 THEN
    pcD := 1 - FLOOR(LOG(100 - pcP));
    IF pcD > 5 THEN RETURN "> 99.9999 %"; END;
    RETURN STRING(pcP, 2, pcD) + " %";
  END;
  pcD := 1;
  IF pcP < 10 THEN pcD := 2; END;
  IF pcP < 1 THEN pcD := 3; END;
  IF pcP < 0.1 THEN pcD := 4; END;
  RETURN STRING(pcP, 2, pcD) + " %";
END;

// A true rate as a percentage, with the decimals of the step a
// finger snaps to (window 0-20 %: "5.0 %").
erlRateS(rsV)
BEGIN
  RETURN STRING(100 * rsV, 2, erlRDec) + " %";
END;

// alpha as typed: 0.05, 0.01, 0.025.
erlAlFmt()
BEGIN
  IF erlAm MOD 10 == 0 THEN RETURN STRING(erlAl, 2, 2); END;
  RETURN STRING(erlAl, 2, 3);
END;

// The confidence level that goes with alpha: "95", "99", "97.5".
erlLvS()
BEGIN
  IF erlAm MOD 10 == 0 THEN RETURN STRING(100 - erlAm / 10, 2, 0); END;
  RETURN STRING(100 - erlAm / 10, 2, 1);
END;

// A Beta parameter kept doubled (2a) as text: 2 -> "1", 1 -> "0.5".
erlHalfS(hfV)
BEGIN
  IF hfV MOD 2 == 0 THEN RETURN erlIStr(hfV / 2); END;
  RETURN erlIStr((hfV - 1) / 2) + ".5";
END;

// Wait for the finger to lift -- but never for more than a second.
// MOUSE(1) is -1 while nothing touches the screen.  Without the
// time limit a stuck or phantom touch would hold the program here
// for ever, and the only way out would be a reset.
erlDrain()
BEGIN
  LOCAL drT;
  drT := TICKS;                                  // milliseconds
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Drain, wait for a key OR a tap, drain.  Returns the key code,
// or -1 for a tap, whose x and y are left in erlTX, erlTY.  A GETKEY
// drain alone does not drain touch, which is why the MOUSE drain is
// here too (Lessons Learned v7f section 7).
erlWaitAny()
BEGIN
  LOCAL wk, wm, wm1;
  REPEAT wk := GETKEY; UNTIL wk == -1;
  erlDrain();
  REPEAT
    wk := GETKEY;
    wm := MOUSE;
    wm1 := wm(1);
  UNTIL wk > -1 OR SIZE(wm1) > 0;
  erlTX := -1; erlTY := -1;
  IF SIZE(wm1) > 0 THEN erlTX := wm1(1); erlTY := wm1(2); wk := -1; END;
  erlDrain();
  REPEAT UNTIL GETKEY == -1;
  RETURN wk;
END;

// Every rpO in rpS replaced by rpN.  Built from INSTRING, LEFT, MID
// and DIM; the guards keep LEFT and MID away from empty pieces.
erlRepl(rpS, rpO, rpN)
BEGIN
  LOCAL rpT, rpOut, rpP, rpLn;
  rpT := rpS;
  rpOut := "";
  rpLn := DIM(rpO);
  rpP := 0;
  IF DIM(rpT) > 0 THEN rpP := INSTRING(rpT, rpO); END;
  WHILE rpP > 0 DO
    IF rpP > 1 THEN rpOut := rpOut + LEFT(rpT, rpP - 1); END;
    rpOut := rpOut + rpN;
    IF rpP + rpLn > DIM(rpT) THEN
      rpT := "";
    ELSE
      rpT := MID(rpT, rpP + rpLn);
    END;
    rpP := 0;
    IF DIM(rpT) > 0 THEN rpP := INSTRING(rpT, rpO); END;
  END;
  RETURN rpOut + rpT;
END;

// A CAS answer made readable: the tagged CAS names become the
// letters the page uses (erlq -> p or N, erll -> mu), exp( becomes
// e^( (desktop Giac only: the Prime already writes e^(, CASProbe
// beta 1b T17, so on the calculator that step finds nothing), and *
// becomes the multiplication sign (the fence, rule 7).
// Giac writes (1 - p)^m as (p-1)^m (m even) or -(p-1)^m (m odd);
// both ARE (1-p)^m, so that leading factor is turned round.
erlShow(shS, shV)
BEGIN
  LOCAL shT, shP;
  shP := "" + shV;                         // the letter, as text
  shT := erlRepl(shS, "erlq", shP);
  shT := erlRepl(shT, "erll", erlMuS);
  shT := erlRepl(shT, "exp(", "e^(");
  shT := erlRepl(shT, "*", erlTms);
  IF DIM(shT) > 7 THEN
    IF LEFT(shT, 6) == "(" + shP + "-1)^" THEN
      shT := "(1-" + shP + ")^" + MID(shT, 7);
    ELSE
      IF LEFT(shT, 7) == "-(" + shP + "-1)^" THEN
        shT := "(1-" + shP + ")^" + MID(shT, 8);
      END;
    END;
  END;
  RETURN shT;
END;

// Where to break a long formula: the last "+", "=" or times sign
// in characters 2..bkW, else the last "-", else 0 (cut at bkW).
erlBreak(bkS, bkW)
BEGIN
  LOCAL bkI, bkC, bkA, bkM;
  bkA := 0; bkM := 0;
  FOR bkI FROM 2 TO MIN(bkW, DIM(bkS)) DO
    bkC := MID(bkS, bkI, 1);
    IF bkC == "+" OR bkC == "=" OR bkC == erlTms THEN bkA := bkI - 1; END;
    IF bkC == "-" THEN bkM := bkI - 1; END;
  END;
  IF bkA >= 2 THEN RETURN bkA; END;
  IF bkM >= 2 THEN RETURN bkM; END;
  RETURN bkW;
END;


// ------------------------------------------------------------
// THE CAS (the ONE place this program calls it)
// ------------------------------------------------------------
// It returns the CAS answer as text, or "" if the CAS failed, gave
// undef, an error or an infinity.
// The call form is the one CASProbe beta 1b found on the HP Prime Virtual
// Calculator (build 2026-09-09, 17 Oct 2026): CAS(s) with s a variable hands
// the text back UNEVALUATED, so the text is built into a literal CAS("...")
// that EXPR reads at run time, and the CAS is asked for string(...) of the
// answer, so a big integer crosses into PPL whole, not as a 12-digit real.
// The rules this program keeps (BRIEF_CAS, the fence):
//   - only whole numbers go into CAS text, built with erlIStr
//     (alpha 0.05 is sent as 50/1000), never STRING(real), never
//     a decimal point;
//   - the CAS answers events (a page opening, n changing), never a
//     frame or a drag; every answer is cached;
//   - CAS answers are TEXT for the screen: every number that is
//     drawn comes from PPL;
//   - the symbolic names are tagged (erlq, erll, never x, n or p).
erlCas(ecQ)
BEGIN
  LOCAL cR;
  cR := "";
  erlCasN := erlCasN + 1;
  IFERR
    cR := STRING(EXPR("CAS(" + CHAR(34) + "string(" + ecQ + ")" + CHAR(34) + ")"));
  THEN
    cR := "";
  END;
  // The answer is a PPL string, and STRING() of it adds one pair of quotes
  // (CASProbe beta 1b T12, emulator): drop that pair.
  IF DIM(cR) >= 2 THEN
    IF LEFT(cR, 1) == CHAR(34) AND MID(cR, DIM(cR), 1) == CHAR(34) THEN
      cR := MID(cR, 2, DIM(cR) - 2);
    END;
  END;
  // undef, an error, or an infinity (CHAR(8734) is the infinity sign; it
  // also catches the Prime's plus-minus infinity, 1/0): no usable answer
  IF INSTRING(cR, "undef") > 0 OR INSTRING(cR, "rror") > 0 OR INSTRING(cR, CHAR(8734)) > 0 THEN cR := ""; END;
  RETURN cR;
END;

// log-gamma, Lanczos (Numerical Recipes' gammln; Burr3 v2.4's
// b3_lngamma under this program's tag).  LNGAMMA does not exist
// outside the CAS (Lessons Learned section 3).  Valid for lgV > 0.
erlLnGam(lgV)
BEGIN
  LOCAL lgY, lgT, lgS, lgJ, lgC;
  lgC := {76.18009172947146, -86.50532032941677, 24.01409824083091, -1.231739572450155, 0.001208650973866179, -0.000005395239384953};
  lgY := lgV;
  lgT := lgV + 5.5;
  lgT := lgT - (lgV + 0.5) * LN(lgT);
  lgS := 1.000000000190015;
  FOR lgJ FROM 1 TO 6 DO
    lgY := lgY + 1;
    lgS := lgS + lgC(lgJ) / lgY;
  END;
  RETURN -lgT + LN(2.5066282746310005 * lgS / lgV);
END;

// The chance of tlK or fewer errors in tlN tests when the true
// error rate is tlP: the binomial CDF.  k = 0 is written out as
// (1 - p)^n, the formula on the "Why 3" page.  BINOMIAL_CDF takes
// (n, p, k) -- Lessons Learned section 3 and the harness notes.
erlTail(tlN, tlP, tlK)
BEGIN
  IF tlP <= 0 OR tlK >= tlN THEN RETURN 1; END;
  IF tlP >= 1 THEN RETURN 0; END;
  IF tlK == 0 THEN RETURN (1 - tlP)^tlN; END;
  RETURN BINOMIAL_CDF(tlN, tlP, tlK);
END;

// THE UPPER BOUND: the true rate at which a record of bdK or fewer
// errors in bdN tests has chance exactly bdA.  This is the one-
// sided Clopper-Pearson ("exact binomial") bound PCAST used.
//   k = 0: (1 - p)^n = alpha gives p = 1 - alpha^(1/n) directly.
//   k > 0: no closed form, so halve the interval [0, 1] 45 times,
//          keeping the half where the curve crosses alpha (the
//          curve falls as p rises).  2^-45 is 3E-14: far finer
//          than any digit shown.
//   k = n: every test was an error, and no rate is ruled out.
erlBound(bdN, bdK, bdA)
BEGIN
  LOCAL bdLo, bdHi, bdM, bdI;
  IF bdK >= bdN THEN RETURN 1; END;
  IF bdK == 0 THEN RETURN 1 - bdA^(1 / bdN); END;
  bdLo := 0;
  bdHi := 1;
  FOR bdI FROM 1 TO 45 DO
    bdM := (bdLo + bdHi) / 2;
    IF BINOMIAL_CDF(bdN, bdM, bdK) > bdA THEN
      bdLo := bdM;              // still too likely: the bound is higher
    ELSE
      bdHi := bdM;
    END;
  END;
  RETURN (bdLo + bdHi) / 2;
END;

// The large-n rule: n x bound tends to the Poisson bound mu, where
// P(Poisson(mu) <= k) = alpha.  For k = 0 that is -ln(alpha)
// (ln 20 = 2.996: "the rule of three"); in general it is half the
// chi-square quantile with 2k + 2 df (CHISQUARE_ICDF is verified).
erlRuleC(rcK, rcA)
BEGIN
  IF rcK == 0 THEN RETURN -LN(rcA); END;
  RETURN CHISQUARE_ICDF(2 * rcK + 2, 1 - rcA) / 2;
END;

// P(Beta(a, b) <= x) with a, b given doubled (bcA2 = 2a, bcB2 = 2b).
// If X is Beta(a, b), then (X / a) / ((1 - X) / b) has the F
// distribution with 2a and 2b df, so the Beta CDF is a FISHER_CDF
// (verified) with WHOLE-number df -- PPL has no Beta functions.
erlBetaCdf(bcA2, bcB2, bcX)
BEGIN
  IF bcX <= 0 THEN RETURN 0; END;
  IF bcX >= 1 THEN RETURN 1; END;
  RETURN FISHER_CDF(bcA2, bcB2, bcB2 * bcX / (bcA2 * (1 - bcX)));
END;

// The Beta quantile: the x with P(Beta <= x) = bqQ, by 50 halvings.
erlBetaQ(bqA2, bqB2, bqQ)
BEGIN
  LOCAL bqLo, bqHi, bqM, bqI;
  bqLo := 0;
  bqHi := 1;
  FOR bqI FROM 1 TO 50 DO
    bqM := (bqLo + bqHi) / 2;
    IF erlBetaCdf(bqA2, bqB2, bqM) < bqQ THEN bqLo := bqM; ELSE bqHi := bqM; END;
  END;
  RETURN (bqLo + bqHi) / 2;
END;

erlRuleS()

  RETURN erlFix(erlRuleM, 2) + "/n";
END;

// Copy the main results into the Home exports.
erlExport()
BEGIN
  ER_UPPER := erlPU; ER_N := erlN; ER_K := erlK; ER_ALPHA := erlAl;
  ER_RATE := erlP0; ER_PASS := erlPass; ER_RULE := erlRuleM / erlN;
END;

// A new true rate: the chance of the record at that rate.
erlSetP0(spV)
BEGIN
  erlP0 := spV;
  erlPass := erlTail(erlN, erlP0, erlK);
  ER_RATE := erlP0; ER_PASS := erlPass;
END;

// Everything that follows from n, k and alpha.
erlRecalc()
BEGIN
  erlAl := erlAm / 1000;
  erlLgMin := FLOOR(LOG(erlAl) + 0.000001) - 1;
  erlPU := erlBound(erlN, erlK, erlAl);
  erlRul := erlRuleC(erlK, erlAl);
  erlRuleM := erlRul;
  IF erlK == 0 AND erlAm == 50 THEN erlRuleM := 3; END;
  erlPass := erlTail(erlN, erlP0, erlK);
  erlExport();
END;

// Fit the window: twice the larger of the bound and the rule line,
// rounded up to 1, 2 or 5 times a power of ten (n = 30: 0 to 20 %).
// ftKeep = 1 widens it to keep the true rate in view (a typed
// rate); ftKeep = 0 moves the rate into the window instead.
erlFit(ftKeep)
BEGIN
  LOCAL ftW;
  ftW := 2 * MAX(erlPU, erlRuleM / erlN);
  IF ftKeep == 1 AND erlP0 * 1.15 > ftW THEN ftW := erlP0 * 1.15; END;
  erlPmax := MIN(1, erlNice(ftW));
  erlStp := erlNice(erlPmax / 5);
  erlTDec := MAX(0, -FLOOR(LOG(100 * erlStp) + 0.000001));
  erlRStp := erlNice(erlPmax / 200);
  erlRDec := MAX(0, -FLOOR(LOG(100 * erlRStp) + 0.000001));
  IF erlP0 > erlPmax THEN
    erlSetP0(erlRStp * ROUND(erlPmax / 4 / erlRStp, 0));
  END;
END;

// Refit only when the bound has drifted far from the middle of the
// window (arrow keys), so a few steps of n are seen on ONE window:
// the curve steepens and the bound slides, which is the lesson.
erlSoftFit()
BEGIN
  LOCAL sfM;
  sfM := MAX(erlPU, erlRuleM / erlN);
  IF sfM > 0.8 * erlPmax OR sfM < 0.2 * erlPmax THEN erlFit(0); END;
END;

// Pixel column of a rate on the main window.
erlPX(pxV)
BEGIN
  RETURN IP(erlPL + pxV / erlPmax * erlPW + 0.5);
END;

// Pixel row of a probability: linear 0..1, or log from 1 down to
// 10^erlLgMin (where the k = 0 curve is almost a straight line of
// slope -n: the rule of three, drawn).
erlYpx(ypV)
BEGIN
  IF erlLog == 0 THEN RETURN IP(erlPB - ypV * (erlPB - erlPT) + 0.5); END;
  IF ypV <= 10^erlLgMin THEN RETURN erlPB; END;
  RETURN IP(erlPT + LOG(ypV) / erlLgMin * (erlPB - erlPT) + 0.5);
END;

// The curve cache.  The probabilities are recomputed only when n,
// k or the window changed (or a finer stride is wanted); the pixel
// rows only when the scale changed too.  cvS = 1 computes every
// column; cvS = 3 (a live drag of n with k > 0) every third, with
// straight lines between, so a G1 keeps up with the finger.
erlCurve(cvS)
BEGIN
  LOCAL cvI, cvA, cvB, cvF;
  IF erlCN <> erlN OR erlCK <> erlK OR erlCM <> erlPmax OR erlCS > cvS THEN
    cvI := 0;
    WHILE cvI <= erlPW DO
      erlCP(cvI + 1) := erlTail(erlN, erlPmax * cvI / erlPW, erlK);
      IF cvI == erlPW THEN
        cvI := erlPW + 1;
      ELSE
        cvI := MIN(erlPW, cvI + cvS);
      END;
    END;
    IF cvS > 1 THEN
      FOR cvI FROM 1 TO erlPW - 1 DO
        IF cvI MOD cvS <> 0 THEN
          cvA := cvI - (cvI MOD cvS);
          cvB := MIN(erlPW, cvA + cvS);
          cvF := (cvI - cvA) / (cvB - cvA);
          erlCP(cvI + 1) := erlCP(cvA + 1) + cvF * (erlCP(cvB + 1) - erlCP(cvA + 1));
        END;
      END;
    END;
    erlCN := erlN; erlCK := erlK; erlCM := erlPmax; erlCS := cvS;
    erlCL := -99;                             // the rows must follow
  END;
  IF erlCL <> erlLog * 100 - erlLgMin THEN
    FOR cvI FROM 1 TO erlPW + 1 DO erlCY(cvI) := erlYpx(erlCP(cvI)); END;
    erlCL := erlLog * 100 - erlLgMin;
  END;
END;

erlYofN(ynN)
BEGIN
  LOCAL ynF;
  ynF := (LOG(MAX(5, MIN(1000, ynN))) - LOG(5)) / (LOG(1000) - LOG(5));
  RETURN IP(erlKB - ynF * (erlKB - erlKT) + 0.5);
END;

// The n under a finger at height nyY, snapped to values a student
// would type: whole numbers to 50, then steps of 5, 10 and 50.
erlNofY(nyY)
BEGIN
  LOCAL nyF, nyV;
  nyF := MAX(0, MIN(1, (erlKB - nyY) / (erlKB - erlKT)));
  nyV := 10^(LOG(5) + nyF * (LOG(1000) - LOG(5)));
  IF nyV < 50 THEN
    nyV := ROUND(nyV, 0);
  ELSE
    IF nyV < 200 THEN
      nyV := 5 * ROUND(nyV / 5, 0);
    ELSE
      IF nyV < 500 THEN nyV := 10 * ROUND(nyV / 10, 0); ELSE nyV := 50 * ROUND(nyV / 50, 0); END;
    END;
  END;
  RETURN MAX(MAX(5, erlK), MIN(1000, nyV));
END;

// The next value of n up (ldD > 0) or down on a ladder of round
// numbers, for the + and - keys.
erlLadder(ldN, ldD)
BEGIN
  LOCAL ldL, ldI, ldR;
  ldL := {1, 2, 3, 5, 10, 15, 20, 25, 30, 40, 50, 75, 100, 150, 200, 300, 500, 750, 1000, 1500, 2000, 3000, 5000, 10000};
  IF ldD > 0 THEN
    ldR := 10000;
    FOR ldI FROM 1 TO SIZE(ldL) DO
      IF ldL(ldI) > ldN AND ldL(ldI) < ldR THEN ldR := ldL(ldI); END;
    END;
  ELSE
    ldR := 1;
    FOR ldI FROM 1 TO SIZE(ldL) DO
      IF ldL(ldI) < ldN AND ldL(ldI) > ldR THEN ldR := ldL(ldI); END;
    END;
  END;
  RETURN ldR;
END;

// n up or down by one (snL = 0) or a ladder step (snL = 1).  n never
// falls below k: k errors need at least k tests.
erlStepN(snD, snL)
BEGIN
  LOCAL snN;
  IF snL == 0 THEN snN := erlN + snD; ELSE snN := erlLadder(erlN, snD); END;
  snN := MAX(MAX(1, erlK), MIN(10000, snN));
  IF snN <> erlN THEN
    erlN := snN;
    erlRecalc();
    erlSoftFit();
  END;
END;

// The rate down or up one arrow step (a 1/2/5 step, about 1/40 of
// the window), snapped onto that step's grid.
erlStepRate(stD)
BEGIN
  LOCAL stS;
  stS := erlNice(erlPmax / 40);
  erlSetP0(MAX(0, MIN(erlPmax, (ROUND(erlP0 / stS, 0) + stD) * stS)));
END;

// The rate under a finger at column rxX, snapped to erlRStp.
erlRateFromX(rxX)
BEGIN
  LOCAL rxV;
  rxV := (rxX - erlPL) / erlPW * erlPmax;
  rxV := ROUND(rxV / erlRStp, 0) * erlRStp;
  RETURN MAX(0, MIN(erlPmax, rxV));
END;

// THE SHOWCASE: drag the true rate along the axis.  While the finger
// stays down, every MOUSE sample moves the magenta marker and its
// chance is recomputed (one CDF); the curve comes from the cache.
// A key, a lift or a reading off the screen ends the drag.
erlDragRate(dgX)
BEGIN
  LOCAL dgK, dgM, dgM1, dgGo, dgV;
  erlSetP0(erlRateFromX(dgX));
  erlDrawMain(1);
  dgGo := 1;
  WHILE dgGo == 1 DO
    dgK := GETKEY;
    dgM := MOUSE;
    dgM1 := dgM(1);
    IF dgK > -1 OR SIZE(dgM1) == 0 THEN
      dgGo := 0;
    ELSE
      IF dgM1(1) < 0 OR dgM1(1) > 319 OR dgM1(2) < 0 OR dgM1(2) > 239 THEN
        dgGo := 0;
      ELSE
        dgV := erlRateFromX(dgM1(1));
        IF dgV <> erlP0 THEN
          erlSetP0(dgV);
          erlDrawMain(1);
        END;
      END;
    END;
  END;
  erlDrain();
  REPEAT dgK := GETKEY; UNTIL dgK == -1;
END;

// Drag n on the right edge.  The window stays FROZEN while the finger
// is down, so the curve visibly steepens and the red bound slides
// toward the axis as n grows (and n x bound creeps up toward 3).
// With k > 0 each new n costs a 45-step bisection and a curve of
// BINOMIAL_CDFs, so the drag draws every third column.  On lift the
// window is refit.
erlDragN(dnY)
BEGIN
  LOCAL dnK, dnM, dnM1, dnGo, dnV, dnS;
  dnS := 1;
  IF erlK > 0 THEN dnS := 3; END;
  dnV := erlNofY(dnY);
  IF dnV <> erlN THEN erlN := dnV; erlRecalc(); END;
  erlDrawMain(dnS);
  dnGo := 1;
  WHILE dnGo == 1 DO
    dnK := GETKEY;
    dnM := MOUSE;
    dnM1 := dnM(1);
    IF dnK > -1 OR SIZE(dnM1) == 0 THEN
      dnGo := 0;
    ELSE
      IF dnM1(1) < 0 OR dnM1(1) > 319 OR dnM1(2) < 0 OR dnM1(2) > 239 THEN
        dnGo := 0;
      ELSE

END;

// A CAS formula after a prefix, wrapped onto up to three lines at
// a "+", "=" or times sign.  The fence's rule 7: never cut an exact
// answer silently -- one that will not fit says how long it is.
erlAddW(awP, awS, awC)
BEGIN
  LOCAL awW, awT, awB, awI, awPad;
  awW := 44 - DIM(awP);
  IF DIM(awS) > 3 * awW THEN
    erlAdd(awP + "(the exact answer has " + erlIStr(DIM(awS)) + " characters:", awC);
    erlAdd("    too long to show; decimals below)", awC);
    RETURN;
  END;
  awPad := "";
  FOR awI FROM 1 TO DIM(awP) DO awPad := awPad + " "; END;
  awT := awS;
  awI := 0;
  WHILE DIM(awT) > awW DO
    awB := erlBreak(awT, awW);
    IF awI == 0 THEN erlAdd(awP + LEFT(awT, awB), awC); ELSE erlAdd(awPad + LEFT(awT, awB), awC); END;
    awT := MID(awT, awB + 1);
    awI := awI + 1;
  END;
  IF awI == 0 THEN erlAdd(awP + awT, awC); ELSE erlAdd(awPad + awT, awC); END;
END;

// "WHY 3": THE ALGEBRA, BY THE CAS

// The polynomial part of P(k or fewer errors).  Every term of the
// binomial sum has the factor (1-p)^(n-k), so
//   P(X <= k) = (1-p)^(n-k) * [ sum of C(n,j) p^j (1-p)^(k-j), j = 0..k ]
// and the CAS is asked to multiply out only the bracket (normal):
// a polynomial of degree k with EXACT coefficients (C(1000,5) has
// 13 digits, past PPL's 12), at the same cost for any n.  (Asking
// it to factor the whole sum expands (1-p)^n first: for n = 10000 that
// took desktop Giac over a minute, and would freeze a calculator.)
// Numeric j only: a sum over a symbolic index sends Giac solving
// Gamma equations (BRIEF_CAS).  For n = 30, k = 2 the text is
// comb(30,0)*erlq^0*(1-erlq)^2+comb(30,1)*erlq^1*(1-erlq)^1+...
erlBinTerms()
BEGIN
  LOCAL btS, btJ, btN;
  btN := "" + erlIStr(erlN);
  btS := "";
  FOR btJ FROM 0 TO erlK DO
    IF btJ > 0 THEN btS := btS + "+"; END;
    btS := btS + "comb(" + btN + "," + erlIStr(btJ) + ")*erlq^" + erlIStr(btJ) + "*(1-erlq)^" + erlIStr(erlK - btJ);
  END;
  RETURN btS;
END;

// The same chance with p = erll/erlq (erll = mu, erlq = n), the
// binomial coefficient written as a falling factorial over j!, so
// the CAS can let n grow: its limit is the Poisson chance.
erlPoiTerms()
BEGIN
  LOCAL ptS, ptJ, ptI, ptF, ptT;
  ptS := "(1-erll/erlq)^erlq";
  ptF := 1;
  FOR ptJ FROM 1 TO erlK DO
    ptF := ptF * ptJ;
    ptT := "erlq";
    FOR ptI FROM 1 TO ptJ - 1 DO ptT := ptT + "*(erlq-" + erlIStr(ptI) + ")"; END;
    ptS := ptS + "+" + ptT + "/" + erlIStr(ptF) + "*(erll/erlq)^" + erlIStr(ptJ) + "*(1-erll/erlq)^(erlq-" + erlIStr(ptJ) + ")";
  END;
  RETURN ptS;
END;

// Ask the CAS for this record's three lines -- once per record (n, k,
// alpha); the answers are cached.  An answer that failed is retried
// next time the page opens.
erlCasFill()
BEGIN
  LOCAL cfKey;
  cfKey := erlIStr(erlN) + "," + erlIStr(erlK) + "," + erlIStr(erlAm);
  IF cfKey == erlCasKey THEN RETURN; END;
  erlCa1 := ""; erlCa2 := ""; erlCa3 := "";
  IF erlK < erlN THEN
    erlCa1 := erlCas(erlIStr(erlAm) + "/1000");
    IF erlK == 0 THEN
      erlCa2 := erlCas("1-(" + erlIStr(erlAm) + "/1000)^(1/" + erlIStr(erlN) + ")");
      erlCa3 := erlCas("limit(erlq*(1-(" + erlIStr(erlAm) + "/1000)^(1/erlq)),erlq," + erlInf + ")");
    ELSE
      IF erlK <= 6 THEN
        erlCa2 := erlCas("normal(" + erlBinTerms() + ")");    // normal, not expand: expand left (-p)^2 beside p^2
        erlCa3 := erlCas("factor(limit(" + erlPoiTerms() + ",erlq," + erlInf + "))");
      END;
    END;
  END;
  erlCasOK := 1;
  IF erlCa1 == "" OR erlCa2 == "" OR erlCa3 == "" THEN erlCasOK := 0; END;
  IF erlK > 6 OR erlK >= erlN THEN erlCasOK := -1; END;     // no CAS lines on this page
  IF erlCasOK == 0 THEN erlCasKey := ""; ELSE erlCasKey := cfKey; END;
END;

// The page.  Magenta lines are the CAS's exact text (or, if it failed,
// the same formula written by PPL, and the badge says "decimal");
// red lines are PPL's decimals.

  ELSE

      erlAdd("The upper bound is the rate p that makes", erlBLK);
      erlAdd("a clean record only " + cdF + " likely:", erlBLK);
      erlAdd("   (1-p)^" + cdN + " = " + cdF, erlBLUE);
      IF erlCa2 <> "" THEN cdS := erlShow(erlCa2, "p"); ELSE cdS := "1-" + erlAlFmt() + "^(1/" + cdN + ")"; END;

      erlAdd("As n grows, the CAS finds the limit:", erlBLK);
      IF erlCa3 <> "" THEN cdS := erlShow(erlCa3, "p"); ELSE cdS := "-ln(" + erlAlFmt() + ")"; END;

// The posterior Beta(a + k, b + n - k) (all kept doubled), its mean
// and upper credible bound, the window, the densities per column,
// and the mean as an exact fraction from the CAS (cached).
erlBayesCalc()
BEGIN
  LOCAL bcA2, bcB2, bcKey;
  bcA2 := erlBa2 + 2 * erlK;
  bcB2 := erlBb2 + 2 * (erlN - erlK);
  erlBMean := bcA2 / (bcA2 + bcB2);
  erlBUp := erlBetaQ(bcA2, bcB2, 1 - erlAl);
  ER_BMEAN := erlBMean; ER_BUPPER := erlBUp;
  erlBMax := MIN(1, erlNice(1.25 * MAX(MAX(erlBUp, MIN(1, erlPU)), erlP0)));
  erlBStp := erlNice(erlBMax / 5);
  erlBDec := MAX(0, -FLOOR(LOG(100 * erlBStp) + 0.000001));
  erlBayesDens(bcA2, bcB2);
  // (a + k)/(a + b + n) = (2a + 2k)/(2a + 2b + 2n): whole numbers only
  bcKey := erlIStr(bcA2) + "/" + erlIStr(bcA2 + bcB2);
  IF bcKey <> erlCaBKey THEN
    erlCaB := erlCas("(" + erlIStr(bcA2) + ")/(" + erlIStr(bcA2 + bcB2) + ")");
    IF erlCaB <> "" THEN erlCaBKey := bcKey; ELSE erlCaBKey := ""; END;
  END;
  erlBOK := 0;
  IF erlCaB <> "" THEN erlBOK := 1; END;
END;

// The posterior and prior densities at the centre of every column:
// log f(x) = (a - 1) ln x + (b - 1) ln(1 - x) - ln B(a, b), with
// ln B from the Lanczos log-gamma.  The y-scale skips the first two
// columns, so a prior with a < 1 (Jeffreys, k = 0) climbs off the
// top near 0 instead of flattening everything else.
erlBayesDens(bdA2, bdB2)
BEGIN
  LOCAL bdA, bdB, bdLB, bdI, bdX, bdF;
  LOCAL bdPa, bdPb, bdPLB, bdM;
  bdA := bdA2 / 2; bdB := bdB2 / 2;
  bdLB := erlLnGam(bdA) + erlLnGam(bdB) - erlLnGam(bdA + bdB);
  bdPa := erlBa2 / 2; bdPb := erlBb2 / 2;
  bdPLB := erlLnGam(bdPa) + erlLnGam(bdPb) - erlLnGam(bdPa + bdPb);
  erlBD := MAKELIST(0, K, 1, erlPW + 1, 1);
  erlBPr := MAKELIST(0, K, 1, erlPW + 1, 1);
  bdM := 0;
  FOR bdI FROM 0 TO erlPW DO
    bdX := erlBMax * (bdI + 0.5) / (erlPW + 1);
    bdF := EXP((bdA - 1) * LN(bdX) + (bdB - 1) * LN(1 - bdX) - bdLB);
    erlBD(bdI + 1) := bdF;
    IF bdI >= 2 AND bdF > bdM THEN bdM := bdF; END;
    erlBPr(bdI + 1) := EXP((bdPa - 1) * LN(bdX) + (bdPb - 1) * LN(1 - bdX) - bdPLB);
  END;
  IF bdM <= 0 THEN bdM := 1; END;
  erlBYm := 1.1 * bdM;
END;

// Pixel column of a rate on the Bayes window, and the row of a
// density (clipped at the top).
erlBX(bxV)
BEGIN
  RETURN IP(erlPL + bxV / erlBMax * erlPW + 0.5);
END;

erlBYpx(byV)
BEGIN
  RETURN IP(erlPB - MIN(1, byV / erlBYm) * (erlPB - erlPT) + 0.5);
END;

// A label beside a vertical line at column lbX, row lbY: to the
// right, or to the left near the right edge.
erlLabel(lbS, lbX, lbY, lbC)
BEGIN
  LOCAL lbW;
  lbW := 6 * DIM(lbS) + 2;
  IF lbX + 3 + lbW > erlPR THEN
    TEXTOUT_P(lbS, G1, lbX - 3 - lbW, lbY, 1, lbC, lbW);
  ELSE
    TEXTOUT_P(lbS, G1, lbX + 3, lbY, 1, lbC, lbW);
  END;
END;

// The Bayes frame.  Under the posterior: light blue up to the true
// rate, pink beyond it (that pink area IS P(true rate > p0)).
erlBayesDraw()
BEGIN
  LOCAL bwI, bwX, bwY, bwV, bwS, bwK1;
  LOCAL bwA2, bwB2, bwL;
  bwA2 := erlBa2 + 2 * erlK;
  bwB2 := erlBb2 + 2 * (erlN - erlK);
  DIMGROB_P(G1, 320, 240, #FFFFFFh);
  // header: prior, badge, posterior
  TEXTOUT_P("Bayes: prior Beta(" + erlHalfS(erlBa2) + ", " + erlHalfS(erlBb2) + "), " + erlBNm, G1, 2, 1, 1, erlBLK, 228);
  IF erlBOK == 1 THEN bwS := "exact (CAS)"; ELSE bwS := "decimal"; END;
  TEXTOUT_P(bwS, G1, 240, 1, 1, erlFOOT, 78);
  TEXTOUT_P("posterior Beta(" + erlHalfS(bwA2) + ", " + erlHalfS(bwB2) + ") after " + erlIStr(erlK) + " of " + erlIStr(erlN) + " wrong", G1, 2, 13, 1, erlMAG, 316);
  // the area under the posterior, column by column
  FOR bwI FROM 0 TO erlPW DO
    bwY := erlBYpx(erlBD(bwI + 1));
    bwV := erlBMax * (bwI + 0.5) / (erlPW + 1);
    IF bwY < erlPB THEN
      IF bwV <= erlP0 THEN
        LINE_P(G1, erlPL + bwI, bwY, erlPL + bwI, erlPB - 1, erlSHAD);
      ELSE
        LINE_P(G1, erlPL + bwI, bwY, erlPL + bwI, erlPB - 1, erlTAIL);
      END;
    END;
  END;
  // grid and rate ticks (percent)
  bwK1 := FLOOR(erlBMax / erlBStp + 0.000001);
  FOR bwI FROM 0 TO bwK1 DO
    bwX := erlBX(bwI * erlBStp);
    IF bwI > 0 THEN LINE_P(G1, bwX, erlPT, bwX, erlPB, erlGRID); END;
    LINE_P(G1, bwX, erlPB, bwX, erlPB + 3, erlGRAY);
    bwS := STRING(100 * bwI * erlBStp, 2, erlBDec);
    IF bwI == bwK1 THEN bwS := bwS + " %"; END;
    TEXTOUT_P(bwS, G1, MAX(0, bwX - 3 * DIM(bwS)), erlPB + 4, 1, erlGRAY, 40);
  END;
  LINE_P(G1, erlPL, erlPB, erlPR, erlPB, erlBLK);
  LINE_P(G1, erlPL, erlPT, erlPL, erlPB, erlBLK);
  TEXTOUT_P("density", G1, erlPL + 3, erlPT + 2, 1, erlGRAY, 50);
  // the prior: grey dots
  FOR bwI FROM 0 TO erlPW STEP 2 DO
    bwY := erlBYpx(erlBPr(bwI + 1));
    IF bwY < erlPB THEN PIXON_P(G1, erlPL + bwI, bwY, erlGRAY); END;
  END;
  // the posterior: magenta, two pixels thick
  FOR bwI FROM 1 TO erlPW DO
    LINE_P(G1, erlPL + bwI - 1, erlBYpx(erlBD(bwI)), erlPL + bwI, erlBYpx(erlBD(bwI + 1)), erlMAG);
    LINE_P(G1, erlPL + bwI - 1, erlBYpx(erlBD(bwI)) - 1, erlPL + bwI, erlBYpx(erlBD(bwI + 1)) - 1, erlMAG);
  END;
  // the upper credible bound (magenta), Clopper-Pearson (red dots),
  // the posterior mean (green dots) and the true rate (orange)
  bwX := erlBX(erlBUp);
  LINE_P(G1, bwX, erlPT, bwX, erlPB, erlMAG);
  erlLabel(erlLvS() + " %", bwX, erlPT + 2, erlMAG);
  IF erlPU <= erlBMax THEN
    bwX := erlBX(erlPU);
    FOR bwY FROM erlPT TO erlPB STEP 3 DO PIXON_P(G1, bwX, bwY, erlRED); PIXON_P(G1, bwX + 1, bwY, erlRED); END;
    erlLabel("CP", bwX, erlPT + 14, erlRED);
  END;
  bwX := erlBX(erlBMean);
  FOR bwY FROM erlPT TO erlPB STEP 3 DO PIXON_P(G1, bwX, bwY, erlGRN); END;
  erlLabel("mean", bwX, erlPT + 26, erlGRN);
  bwX := erlBX(erlP0);
  LINE_P(G1, bwX, erlPT + 38, bwX, erlPB, erlORNG);
  // the panel
  IF erlBOK == 1 THEN
    bwL := erlCaB + " = " + erlPctS(erlBMean);
  ELSE
    bwL := erlPctS(erlBMean);
  END;
  TEXTOUT_P("mean (a+k)/(a+b+n) = " + bwL, G1, 2, 181, 1, erlGRN, 316);
  TEXTOUT_P(erlLvS() + " % credible bound " + erlPctS(erlBUp) + " (CP " + erlPctS(erlPU) + ")", G1, 2, 193, 1, erlMAG, 316);
  bwV := 1 - erlBetaCdf(bwA2, bwB2, erlP0);
  TEXTOUT_P("P(true rate > " + erlRateS(erlP0) + ") = " + erlPctS(bwV) + "  (drag)", G1, 2, 205, 1, erlORNG, 316);
  BLIT_P(G0, 0, 0, G1);
  DRAWMENU({"Unif", "Jeff", "a, b", "", "Back", ""});
END;

// Your own prior Beta(a, b): a and b in halves from 0.5 to 100.
erlAskPrior()
BEGIN
  LOCAL apA, apB, apOK, apGo, apMsg;
  apA := erlBa2 / 2; apB := erlBb2 / 2;
  apGo := 1;
  WHILE apGo == 1 DO
    apOK := INPUT({apA, apB}, "Prior Beta(a, b)", {"a:", "b:"}, {"0.5 to 100, in steps of 0.5 (prior errors)", "0.5 to 100, in steps of 0.5 (prior passes)"});
    IF apOK <> 1 THEN RETURN 0; END;
    apMsg := "";
    IF FP(2 * apA) <> 0 OR FP(2 * apB) <> 0 THEN apMsg := "a and b must be multiples of 0.5."; END;
    IF apA < 0.5 OR apA > 100 OR apB < 0.5 OR apB > 100 THEN apMsg := "a and b must be from 0.5 to 100."; END;
    IF apMsg == "" THEN apGo := 0; ELSE MSGBOX(apMsg); END;
  END;
  erlBa2 := 2 * apA; erlBb2 := 2 * apB;
  erlBNm := "your prior";
  IF erlBa2 == 2 AND erlBb2 == 2 THEN erlBNm := "uniform"; END;
  IF erlBa2 == 1 AND erlBb2 == 1 THEN erlBNm := "Jeffreys"; END;
  RETURN 1;
END;

// The Bayes page: soft keys pick the prior; drag along the plot (or
// Left / Right) moves the true rate, and the pink area follows.
erlBayesPage()
BEGIN
  LOCAL bpK, bpM, bpM1, bpGo, bpX, bpY;
  LOCAL bpA, bpS, bpV, bpG2;
  erlBayesCalc();
  bpGo := 1;
  REPEAT bpK := GETKEY; UNTIL bpK == -1;
  erlDrain();
  WHILE bpGo == 1 DO
    erlBayesDraw();
    REPEAT
      bpK := GETKEY;
      bpM := MOUSE;
      bpM1 := bpM(1);
    UNTIL bpK > -1 OR SIZE(bpM1) > 0;
    bpA := 0;
    bpS := erlNice(erlBMax / 200);
    IF SIZE(bpM1) > 0 THEN
      bpX := bpM1(1); bpY := bpM1(2);
      IF bpY >= 218 THEN
        erlDrain();
        bpA := MIN(6, IP(bpX / 53.4) + 1);
      ELSE
        IF bpY >= erlPT AND bpY <= erlPB + 12 THEN
          // a live drag of the true rate (densities come from the cache)
          bpG2 := 1;
          WHILE bpG2 == 1 DO
            bpV := MAX(0, MIN(erlBMax, ROUND((bpX - erlPL) / erlPW * erlBMax / bpS, 0) * bpS));
            IF bpV <> erlP0 THEN erlSetP0(bpV); erlBayesDraw(); END;
            bpK := GETKEY;
            bpM := MOUSE;
            bpM1 := bpM(1);
            IF bpK > -1 OR SIZE(bpM1) == 0 THEN
              bpG2 := 0;
            ELSE
              bpX := bpM1(1);
              IF bpX < 0 OR bpX > 319 THEN bpG2 := 0; END;
            END;
          END;
        END;
        erlDrain();
      END;
    ELSE
      IF bpK == 4 OR bpK == 30 THEN bpGo := 0; END;
      IF bpK == 7 THEN erlSetP0(MAX(0, (ROUND(erlP0 / (5 * bpS), 0) - 1) * 5 * bpS)); END;
      IF bpK == 8 THEN erlSetP0(MIN(erlBMax, (ROUND(erlP0 / (5 * bpS), 0) + 1) * 5 * bpS)); END;
    END;
    IF bpA == 1 THEN erlBa2 := 2; erlBb2 := 2; erlBNm := "uniform"; erlBayesCalc(); END;
    IF bpA == 2 THEN erlBa2 := 1; erlBb2 := 1; erlBNm := "Jeffreys"; erlBayesCalc(); END;
    IF bpA == 3 THEN
      IF erlAskPrior() == 1 THEN erlBayesCalc(); END;
      REPEAT bpK := GETKEY; UNTIL bpK == -1;
      erlDrain();
    END;
    IF bpA == 5 THEN bpGo := 0; END;
  END;
  // the true rate may now be outside the main window
  IF erlP0 > erlPmax THEN erlFit(1); END;
END;


// ------------------------------------------------------------
// SIMULATION -- "PPL draws, Python crunches"
// NOT YET RUN ON HARDWARE.  Test with the Terminal open: a Python
// error prints there, and PPL carries on with its fallback.
// Route A (Lessons Learned section 8): the Python lives in this file
// and is run by PYTHON(erlPySim) -- there is no .py file.
// The bridge (section 9): PPL writes ER_PYN, ER_PYP, ER_PYM, ER_PYK
// and ER_PYW; Python reads them with hpprime.eval (never argv) and
// writes ER_PYO back ONE VALUE AT A TIME -- 41 bin counts, then the
// zero count, the pass count and the total, the total last -- so
// the payload is the same 44 numbers whether 100 or 10,000
// programmes were run.
// ------------------------------------------------------------
#PYTHON erlPySim
# ErrorRate beta 1: proficiency programmes in MicroPython.
# NOT YET RUN ON HARDWARE.  MicroPython 1.9.4 has no 'random' module:
# urandom is the one to import (Lessons Learned v7f section 8).
from hpprime import eval as ppl
from urandom import random
import math

def mark(s):
    # how far the script got, readable from Home as ER_PYZ
    ppl("ER_PYZ:=" + str(s))

def main():
    mark(1)
    n = int(float(ppl("ER_PYN")))
    p = float(ppl("ER_PYP"))
    m = int(float(ppl("ER_PYM")))
    k = int(float(ppl("ER_PYK")))
    w = int(float(ppl("ER_PYW")))
    bins = [0] * 41
    zero = 0
    ok = 0
    tot = 0
    L = 0.0
    if 0.0 < p < 1.0:
        L = math.log(1.0 - p)
    for _ in range(m):
        e = 0
        if p >= 1.0:
            e = n
        elif L < -1e-15:
            # Jump from one error to the next instead of tossing a coin
            # for every test: the run of clean tests before an error is
            # geometric, so its length is floor(ln(U) / ln(1 - p)) with
            # U uniform on (0, 1].  The next error is test pos + run + 1;
            # if that is past test n, this programme is finished.
            pos = 0
            while True:
                g = math.log(1.0 - random()) / L
                if g >= n - pos:
                    break
                pos += int(g) + 1
                e += 1
        b = e // w
        if b > 40:
            b = 40
        bins[b] += 1
        if e == 0:
            zero += 1
        if e <= k:
            ok += 1
        tot += e
    mark(2)
    # one value per crossing, built by concatenation (section 9)
    for i in range(41):
        ppl("ER_PYO(" + str(i + 1) + "):=" + str(bins[i]))
    ppl("ER_PYO(42):=" + str(zero))
    ppl("ER_PYO(43):=" + str(ok))
    ppl("ER_PYO(44):=" + str(tot))
    mark(3)

main()
#END

// The PPL fallback: the same jumps from error to error, spM
// programmes.  RANDOM is in [0, 1), so 1 - RANDOM is in (0, 1] and
// its LN is never undefined.
erlSimPPL(spM)
BEGIN
  LOCAL spI, spE, spPos, spG, spL, spB;
  LOCAL spGo;
  erlSmB := MAKELIST(0, K, 1, 41, 1);
  erlSm0 := 0; erlSmOK := 0; erlSmTot := 0; erlSmM := spM;
  spL := 0;
  IF erlP0 > 0 AND erlP0 < 1 THEN spL := LN(1 - erlP0); END;
  FOR spI FROM 1 TO spM DO
    spE := 0;
    IF erlP0 >= 1 THEN spE := erlN; END;
    IF spL < 0 THEN
      spPos := 0;
      spGo := 1;
      WHILE spGo == 1 DO
        spG := LN(1 - RANDOM) / spL;          // clean tests before the next error
        IF spG >= erlN - spPos THEN
          spGo := 0;                         // the next error is past test n
        ELSE
          spPos := spPos + IP(spG) + 1;
          spE := spE + 1;
        END;
      END;
    END;
    spB := MIN(40, IP(spE / erlSmW)) + 1;
    erlSmB(spB) := erlSmB(spB) + 1;
    IF spE == 0 THEN erlSm0 := erlSm0 + 1; END;
    IF spE <= erlK THEN erlSmOK := erlSmOK + 1; END;
    erlSmTot := erlSmTot + spE;
  END;
END;

// Run the simulation at the current n, k and true rate (Lessons
// Learned section 9, all three guards): clear the output first,
// validate by READING the last element inside IFERR (and check the
// bins add up), and always keep the PPL fallback -- 1,000 programmes.
// The badge then names the engine that really ran and the TICKS.
erlRunSim()
BEGIN
  LOCAL rsT, rsOK, rsChk, rsI, rsS, rsMax;
  // bars wide enough for the bulk of the error counts (at most 40)
  rsMax := CEILING(erlN * erlP0 + 4 * (erlN * erlP0 * (1 - erlP0))^0.5 + 1);
  rsMax := MIN(erlN, MAX(MAX(erlK + 2, 5), rsMax));
  erlSmW := MAX(1, CEILING((rsMax + 1) / 40));
  erlSmNb := MIN(41, CEILING((rsMax + 1) / erlSmW));
  erlSmN := erlN; erlSmP := erlP0; erlSmK := erlK;
  // 1. the inputs Python reads, and the output cleared to -1
  ER_PYN := erlN; ER_PYP := erlP0; ER_PYM := 10000;
  ER_PYK := erlK; ER_PYW := erlSmW; ER_PYZ := 0;
  ER_PYO := MAKELIST(-1, K, 1, 44, 1);
  rsT := TICKS;
  IFERR
    PYTHON(erlPySim);
  THEN
    rsOK := 0;          // (an error INSIDE Python never reaches here)
  END;
  rsT := TICKS - rsT;
  // 2. validate by USING the result: the last element, inside IFERR
  rsOK := 0;
  IFERR
    rsChk := ER_PYO(44) + 0;
    IF rsChk >= 0 THEN rsOK := 1; END;
  THEN
    rsOK := 0;
  END;
  // and the bins must add up to the programmes asked for
  IF rsOK == 1 THEN
    rsS := 0;
    FOR rsI FROM 1 TO 41 DO rsS := rsS + ER_PYO(rsI); END;
    IF rsS <> 10000 OR ER_PYO(42) > ER_PYO(43) OR ER_PYO(43) > 10000 THEN rsOK := 0; END;
  END;
  IF rsOK == 1 THEN
    erlSmB := MAKELIST(0, K, 1, 41, 1);
    FOR rsI FROM 1 TO 41 DO erlSmB(rsI) := ER_PYO(rsI); END;
    erlSm0 := ER_PYO(42); erlSmOK := ER_PYO(43); erlSmTot := ER_PYO(44);
    erlSmM := 10000;
    erlSmEng := 1;
    erlSmMs := rsT;
  ELSE
    // 3. the PPL fallback
    rsT := TICKS;
    erlSimPPL(1000);
    erlSmMs := TICKS - rsT;
    erlSmEng := 2;
  END;
  ER_SIM := erlSmOK / erlSmM;
END;

// P(at least one chance match) among tpN profiles, RMP = 10^-erlTm:
// 1 - (1 - RMP)^N = 1 - e^(N ln(1 - RMP)).  ln(1 - RMP) is taken
// from its series (RMP <= 0.001, four terms are exact to 12 digits),
// and for a tiny exponent so is 1 - e^-x, so no digits are lost.
erlTrP(tpN)
BEGIN
  LOCAL tpR, tpL, tpX;
  tpR := 10^(-erlTm);
  tpL := tpR + tpR * tpR / 2 + tpR^3 / 3 + tpR^4 / 4;   // -ln(1 - RMP)
  tpX := tpN * tpL;
  IF tpX < 0.0001 THEN RETURN tpX - tpX * tpX / 2 + tpX^3 / 6; END;
  RETURN 1 - EXP(-tpX);
END;

// Pixel column of a database size (log scale, 1 to 100 million).
erlTrX(txN)
BEGIN
  RETURN IP(erlPL + LOG(MAX(1, txN)) / 8 * erlPW + 0.5);
END;

// The curve and the N x RMP line, once per RMP.
erlTrCurve()
BEGIN
  LOCAL tcI, tcN, tcP;

  erlTMY := MAKELIST(0, K, 1, erlPW + 1, 1);
  FOR tcI FROM 0 TO erlPW DO

    erlTMY(tcI + 1) := IP(erlPB - MIN(1, tcN * 10^(-erlTm)) * (erlPB - erlPT) + 0.5);
  END;
  erlTKey := erlTm;
END;

// A database size snapped to two significant digits, 1 to 1E8.
erlTrSnap(tsV)
BEGIN
  LOCAL tsE;
  IF tsV < 10 THEN RETURN MAX(1, ROUND(tsV, 0)); END;
  tsE := FLOOR(LOG(tsV) + 0.000001) - 1;
  RETURN MIN(100000000, ROUND(tsV / 10^tsE, 0) * 10^tsE);
END;

// The next 1-2-5 database size up or down (Left / Right).
erlTrStep(tsD)
BEGIN
  LOCAL tsE, tsM, tsV;
  tsE := FLOOR(LOG(erlTN) + 0.000001);
  tsM := erlTN / 10^tsE;
  IF tsD > 0 THEN
    tsV := 10^(tsE + 1);
    IF tsM < 4.999999 THEN tsV := 5 * 10^tsE; END;
    IF tsM < 1.999999 THEN tsV := 2 * 10^tsE; END;
  ELSE
    tsV := 5 * 10^(tsE - 1);
    IF tsM > 1.000001 THEN tsV := 10^tsE; END;
    IF tsM > 2.000001 THEN tsV := 2 * 10^tsE; END;
    IF tsM > 5.000001 THEN tsV := 5 * 10^tsE; END;
  END;
  erlTN := MAX(1, MIN(100000000, ROUND(tsV, 0)));
END;

// The CAS lines for the trawl, cached per RMP.  The exact chance is
// never evaluated by the CAS: (999999/1000000)^1000000 as an exact
// fraction would have millions of digits.  It shows the expression;
// PPL computes the number.
erlTrCas()
BEGIN
  IF erlCaTKey == erlIStr(erlTm) THEN RETURN; END;
  erlCaT1 := erlCas("1-(1-1/" + erlIStr(10^erlTm) + ")^erlq");
  erlCaT2 := erlCas("limit(1-(1-erll/erlq)^erlq,erlq," + erlInf + ")");
  erlCaT3 := erlCas("convert(series(1-exp(-erll),erll=0,3),polynom)");
  erlCaTKey := "";
  IF erlCaT1 <> "" AND erlCaT2 <> "" AND erlCaT3 <> "" THEN erlCaTKey := erlIStr(erlTm); END;
END;

// The "Exact" page of the trawl: the exact expression, its limit and
// its series, from the CAS, with PPL's numbers at this N.
erlTrExact()
BEGIN
  LOCAL teS, teR, teM, teB, teOK;
  erlTrCas();
  teR := 10^(-erlTm);
  teM := erlTN * teR;
  teOK := 1;
  erlPgReset();
  erlAdd("Each of N profiles is compared with the", erlBLK);
  erlAdd("crime-scene profile.  If the source is not", erlBLK);
  erlAdd("in the database, each one matches by chance", erlBLK);

// The trawl page: drag along the plot (or Left / Right) for N.
erlTrawlPage()
BEGIN
  LOCAL tgK, tgM, tgM1, tgGo, tgX, tgY;
  LOCAL tgA, tgG2, tgV;
  tgGo := 1;
  REPEAT tgK := GETKEY; UNTIL tgK == -1;
  erlDrain();
  WHILE tgGo == 1 DO
    erlTrDraw();
    REPEAT
      tgK := GETKEY;
      tgM := MOUSE;
      tgM1 := tgM(1);
    UNTIL tgK > -1 OR SIZE(tgM1) > 0;
    tgA := 0;
    IF SIZE(tgM1) > 0 THEN
      tgX := tgM1(1); tgY := tgM1(2);
      IF tgY >= 218 THEN
        erlDrain();
        tgA := MIN(6, IP(tgX / 53.4) + 1);
      ELSE
        IF tgY >= erlPT AND tgY <= erlPB + 12 THEN
          tgG2 := 1;
          WHILE tgG2 == 1 DO
            tgV := erlTrSnap(10^(8 * MAX(0, MIN(erlPW, tgX - erlPL)) / erlPW));
            IF tgV <> erlTN THEN erlTN := tgV; erlTrDraw(); END;
            tgK := GETKEY;
            tgM := MOUSE;
            tgM1 := tgM(1);
            IF tgK > -1 OR SIZE(tgM1) == 0 THEN
              tgG2 := 0;
            ELSE
              tgX := tgM1(1);
              IF tgX < 0 OR tgX > 319 THEN tgG2 := 0; END;
            END;
          END;
        END;
        erlDrain();
      END;
    ELSE
      IF tgK == 4 OR tgK == 30 THEN tgGo := 0; END;
      IF tgK == 7 THEN erlTrStep(-1); END;
      IF tgK == 8 THEN erlTrStep(1); END;
    END;
    IF tgA == 1 THEN erlAskRMP(); END;
    IF tgA == 2 THEN erlAskTN(); END;
    IF tgA == 3 THEN erlTrExact(); END;
    IF tgA == 5 THEN tgGo := 0; END;
    IF tgA >= 1 THEN
      REPEAT tgK := GETKEY; UNTIL tgK == -1;
      erlDrain();
    END;
  END;
END;

// n, k and alpha.  A bad value gets a message saying why, and the
// form comes back.  alpha may be typed as a percent (5 = 0.05).
erlAskNK()
BEGIN
  LOCAL akN, akK, akA, akOK, akMsg, akGo;
  LOCAL akL3;
  akN := erlN; akK := erlK; akA := erlAl;
  akL3 := erlAlS + ":";
  akGo := 1;
  WHILE akGo == 1 DO
    akOK := INPUT({akN, akK, akA}, "The test record", {"n:", "k:", akL3}, {"Tests taken: a whole number, 1 to 10000", "Errors made: a whole number, 0 to n", "One-sided, 0.001 to 0.5 (PCAST: 0.05)"});
    IF akOK <> 1 THEN RETURN 0; END;
    akMsg := "";
    IF akA >= 1 THEN akA := akA / 100; END;          // typed as a percent
    IF akA < 0.001 OR akA > 0.5 THEN
      akMsg := erlAlS + " must be from 0.001 to 0.5 (0.05 for a 95 % bound).";
    END;
    IF FP(akK) <> 0 OR akK < 0 OR akK > akN THEN
      akMsg := "k must be a whole number from 0 to n: you cannot make more errors than tests.";
    END;
    IF FP(akN) <> 0 OR akN < 1 OR akN > 10000 THEN
      akMsg := "n must be a whole number from 1 to 10000.";
    END;
    IF akMsg == "" THEN akGo := 0; ELSE MSGBOX(akMsg); END;
  END;
  erlN := akN; erlK := akK;
  erlAm := MAX(1, ROUND(akA * 1000, 0));
  erlRecalc();
  erlFit(0);
  RETURN 1;
END;

  END;
END;

  ER_SIM := 0; ER_TRAWL := 0; ER_BMEAN := 0; ER_BUPPER := 0;
  erlRecalc();
  erlFit(0);
END;

// The live screen: draw, wait for a key or a touch, act.
erlLoop()
BEGIN
  LOCAL lpK, lpM, lpM1, lpGo, lpX, lpY;
  LOCAL lpA;
  lpGo := 1;
  REPEAT lpK := GETKEY; UNTIL lpK == -1;
  erlDrain();
  WHILE lpGo == 1 DO
    erlDrawMain(1);
    REPEAT
      lpK := GETKEY;
      lpM := MOUSE;
      lpM1 := lpM(1);
    UNTIL lpK > -1 OR SIZE(lpM1) > 0;
    // lpA: 1-5 page-1 soft keys, 11-15 page-2 soft keys, 20 menu
    lpA := 0;
    IF SIZE(lpM1) > 0 THEN
      lpX := lpM1(1);
      lpY := lpM1(2);
      IF lpY >= 218 THEN
        // the soft-key strip: six 53-px slots; slot 6 turns the page
        erlDrain();
        lpA := MIN(6, IP(lpX / 53.4) + 1);
        IF lpA == 6 THEN
          erlSKP := erlSKP + 1;
          IF erlSKP > 2 THEN erlSKP := 1; END;
          lpA := 0;
        ELSE
          IF erlSKP == 2 THEN lpA := lpA + 10; END;
        END;
      ELSE
        IF lpX >= erlSLX AND lpY >= erlKT - 10 AND lpY <= erlKB + 10 THEN
          erlDragN(lpY);                      // the n slider
        ELSE
          IF lpX < erlSLX AND lpY >= erlPT AND lpY <= erlPB + 12 THEN
            erlDragRate(lpX);                 // the true rate
          ELSE
            erlDrain();
          END;
        END;
      END;
    ELSE
      // the keys remain as shortcuts for everything the touch does
      IF lpK == 4 THEN lpGo := 0; END;                 // Esc   = quit
      IF lpK == 30 THEN lpA := 20; END;                // Enter = menu
      IF lpK == 7 THEN erlStepRate(-1); END;           // Left  = rate down
      IF lpK == 8 THEN erlStepRate(1); END;            // Right = rate up
      IF lpK == 2 THEN erlStepN(1, 0); END;            // Up    = n + 1
      IF lpK == 12 THEN erlStepN(-1, 0); END;          // Down  = n - 1
      IF lpK == 50 THEN erlStepN(1, 1); END;           // +     = n up a step
      IF lpK == 45 THEN erlStepN(-1, 1); END;          // -     = n down a step
    END;
    IF lpA == 1 THEN erlAskNK(); END;
    IF lpA == 2 THEN erlAskRate(); END;
    IF lpA == 3 THEN erlCasPage(); END;
    IF lpA == 4 THEN erlBayesPage(); END;
    IF lpA == 5 THEN erlSimPage(); END;
    IF lpA == 11 THEN erlTrawlPage(); END;
    IF lpA == 12 THEN erlLog := 1 - erlLog; END;
    IF lpA == 13 THEN erlHelp(); END;
    IF lpA == 14 THEN erlEngPage(); END;
    IF lpA == 15 THEN lpGo := 0; END;
    IF lpA == 20 THEN lpGo := erlMenu(); END;
    // a dialog's closing tap or key can arrive again on the screen
    // underneath, so drain both before the next frame
    IF lpA > 0 THEN
      REPEAT lpK := GETKEY; UNTIL lpK == -1;
      erlDrain();
    END;
  END;
END;

EXPORT ErrorRate()
BEGIN
  erlInit();
  erlWelcome();
  erlLoop();
  erlExport();
  // Hand the summary back as the program's value: Home shows it in
  // the history and no Terminal screen opens.
  RETURN "ErrorRate: " + erlRecS() + " -> " + erlLvS() + " % upper bound " + erlPctS(erlPU) + " (" + erlRuleS() + " = " + erlPctS(erlRuleM / erlN) + "); see ER_UPPER";
END;
