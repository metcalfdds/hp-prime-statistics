// ------------------------------------------------------------
// THE KNOWLEDGE: names, table cells, and five lines per test

// Test id for a scale (1 interval, 2 nominal, 3 ordinal, 4 survival)
// and a design (1 one group vs a value, 2 two independent groups,
// 3 three or more groups, 4 before/after same people, 5 several
// conditions same people, 6 association of two variables).
tcIdFor(idS, idD)
BEGIN
  LOCAL idMap;
  idMap := {21, 1, 2, 3, 4, 5,
            22, 6, 7, 8, 9, 10,
            23, 11, 12, 13, 14, 15,
            24, 16, 17, 18, 19, 20};
  RETURN idMap((idS - 1) * 6 + idD);
END;

// The scale, in one word (fits a 96-pixel box).
tcScaleShort(ssS)
BEGIN
  IF ssS == 1 THEN RETURN "interval"; END;
  IF ssS == 2 THEN RETURN "nominal"; END;
  IF ssS == 3 THEN RETURN "ordinal"; END;
  RETURN "survival";
END;

// The design, short enough for a subtitle (at most 26 characters).
tcDesignShort(dsD)
BEGIN
  IF dsD == 1 THEN RETURN "1 group vs a value"; END;
  IF dsD == 2 THEN RETURN "2 independent groups"; END;
  IF dsD == 3 THEN RETURN "3+ independent groups"; END;
  IF dsD == 4 THEN RETURN "before/after, same people"; END;
  IF dsD == 5 THEN RETURN "3+ conditions, same people"; END;
  RETURN "association of 2 variables";
END;

    IF fnI == 9 THEN RETURN "Cochran's Q test"; END;
    IF fnI == 10 THEN RETURN "Relative risk, odds ratio, chi-square"; END;
    IF fnI == 11 THEN RETURN "Mann-Whitney U (Wilcoxon rank-sum)"; END;
    IF fnI == 12 THEN RETURN "Kruskal-Wallis test"; END;
    IF fnI == 13 THEN RETURN "Wilcoxon signed-rank test"; END;
    IF fnI == 14 THEN RETURN "Friedman test"; END;
    IF fnI == 15 THEN RETURN "Spearman rank correlation"; END;

    IF fnI == 17 THEN RETURN "Log-rank test (several groups)"; END;

// Wait for the finger to lift -- but never for more than a second.
// MOUSE(1) is -1 while nothing touches the screen.  Without the
// time limit a stuck or phantom touch would hold the program here
// for ever, and the only way out would be a reset.
tcDrain()
BEGIN
  LOCAL drT;
  drT := TICKS;                                  // milliseconds
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Wait for any key or tap (drain, wait, drain).
tcWaitAny()
BEGIN
  LOCAL waK, waM, waM1;
  REPEAT waK := GETKEY; UNTIL waK == -1;
  tcDrain();
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  tcDrain();
  REPEAT waK := GETKEY; UNTIL waK == -1;
END;

  IF coSc == 1 THEN RETURN tcBLUE; END;
  IF coSc == 2 THEN RETURN tcORNG; END;

  RETURN tcGRN;

tcLightOf(loS)

  IF loS == 1 THEN RETURN tcLBLU; END;
  IF loS == 2 THEN RETURN tcLORN; END;
  IF loS == 3 THEN RETURN tcLMAG; END;
  RETURN tcLGRN;
END;


// ------------------------------------------------------------
// A SCREEN OF BIG BUTTONS: tap one, or move with the arrows
// and press Enter.  Returns the index chosen, 0 for Esc.
// pkCols / pkLts: the border and fill colour of each button.
// ------------------------------------------------------------
tcPick(pkTtl, pkItems, pkCols, pkLts, pkCur)
BEGIN
  LOCAL pkN, pkI, pkH, pkY0, pkK, pkM;
  LOCAL pkM1, pkSel, pkDone, pkY, pkTY, pkHit;
  pkN := SIZE(pkItems);
  pkY0 := 30;
  pkH := IP((198 - pkY0) / pkN);              // 42 px for 4 items, 28 for 6
  pkSel := pkCur;
  IF pkSel < 1 OR pkSel > pkN THEN pkSel := 1; END;
  pkDone := 0;
  REPEAT pkK := GETKEY; UNTIL pkK == -1;
  tcDrain();
  REPEAT
    DIMGROB_P(G1, 320, 240, #FFFFFFh);
    TEXTOUT_P(pkTtl, G1, 10, 6, 2, #000000h);
    FOR pkI FROM 1 TO pkN DO
      pkY := pkY0 + (pkI - 1) * pkH;
      pkTY := pkY + IP((pkH - 4 - 14) / 2);
      IF pkI == pkSel THEN
        RECT_P(G1, 10, pkY, 310, pkY + pkH - 4, pkCols(pkI), pkCols(pkI));
        TEXTOUT_P(pkItems(pkI), G1, 22, pkTY, 2, #FFFFFFh);
      ELSE
        RECT_P(G1, 10, pkY, 310, pkY + pkH - 4, pkCols(pkI), pkLts(pkI));
        TEXTOUT_P(pkItems(pkI), G1, 22, pkTY, 2, pkCols(pkI));
      END;
    END;
    TEXTOUT_P("tap an answer, or " + CHAR(8593) + CHAR(8595) + " and Enter.  Esc: back", G1, 10, 222, 1, tcFOOT);
    BLIT_P(G0, 0, 0, G1);
    REPEAT
      pkK := GETKEY;
      pkM := MOUSE;
      pkM1 := pkM(1);
    UNTIL pkK > -1 OR SIZE(pkM1) > 0;
    IF SIZE(pkM1) > 0 THEN
      // a tap: which button?  (taps above the first button are ignored)
      pkHit := IP((pkM1(2) - pkY0) / pkH) + 1;
      IF pkM1(2) >= pkY0 AND pkHit >= 1 AND pkHit <= pkN THEN
        pkSel := pkHit;
        pkDone := 1;
      END;
      tcDrain();
    ELSE
      IF pkK == 2 THEN pkSel := MAX(1, pkSel - 1); END;
      IF pkK == 12 THEN pkSel := MIN(pkN, pkSel + 1); END;
      IF pkK == 30 THEN pkDone := 1; END;
      IF pkK == 4 THEN pkDone := 1; pkSel := 0; END;
    END;
  UNTIL pkDone == 1;
  REPEAT pkK := GETKEY; UNTIL pkK == -1;
  RETURN pkSel;
END;


// ------------------------------------------------------------
// THE DETAILS PAGE for one test
// dtS / dtD: the scale and design that led here (0 = not known).

BEGIN

  dtL := tcDetail(dtI);
  dtN := tcFull(dtI);
  TC_TEST := dtN;

  IF dtS > 0 THEN dtCol := tcColOf(dtS); dtLt := tcLightOf(dtS); END;

  RECT_P(G1, 4, 4, 315, 235, dtCol, dtLt);
  TEXTOUT_P(dtN, G1, 12, 10, 2, dtCol, 298);
  IF dtS > 0 THEN
    TEXTOUT_P(tcScaleShort(dtS) + "  |  " + tcDesignShort(dtD), G1, 12, 32, 1, tcGRAY, 298);
  END;
  LINE_P(G1, 12, 46, 307, 46, tcGRAY);
  TEXTOUT_P("What:", G1, 12, 54, 1, tcGRAY);
  TEXTOUT_P(dtL(1), G1, 14, 66, 1, #000000h, 296);
  TEXTOUT_P("Assumes:", G1, 12, 84, 1, tcGRAY);
  TEXTOUT_P(dtL(2), G1, 14, 96, 1, #000000h, 296);
  TEXTOUT_P("On the Prime:", G1, 12, 114, 1, tcGRAY);
  TEXTOUT_P(dtL(3), G1, 14, 126, 1, tcBLUE, 296);

tcGuide()
BEGIN
  LOCAL gdS, gdD, gdI, gdK, gdStage, gdM;
  LOCAL gdM1;
  gdS := 0; gdD := 0; gdStage := 1;
  WHILE gdStage > 0 DO
    IF gdStage == 1 THEN
      gdS := tcPick("What kind of OUTCOME variable?",
        {"Interval or ratio, roughly normal", "Nominal: categories, yes/no", "Ordinal, or skewed / small n", "Survival time, with censoring"},
        {tcBLUE, tcORNG, tcMAG, tcGRN}, {tcLBLU, tcLORN, tcLMAG, tcLGRN}, gdS);
      IF gdS == 0 THEN gdStage := 0; ELSE gdStage := 2; END;
    END;
    IF gdStage == 2 THEN
      gdD := tcPick("What is being compared?",
        {"One group vs a known value", "Two groups, different people", "Three or more groups", "Before & after, same people", "3+ conditions, same people", "Association of two variables"},
        {tcBLUE, tcORNG, tcMAG, tcGRN, tcDODG, tcGRAY}, {tcLBLU, tcLORN, tcLMAG, tcLGRN, tcLDOD, tcLGRY}, gdD);
      IF gdD == 0 THEN gdStage := 1; ELSE gdStage := 3; END;
    END;
    IF gdStage == 3 THEN
      gdI := tcIdFor(gdS, gdD);
      tcShowPath(gdS, gdD, gdI);
      REPEAT gdK := GETKEY; UNTIL gdK == -1;
      tcDrain();
      REPEAT
        gdK := GETKEY;
        gdM := MOUSE;
        gdM1 := gdM(1);
      UNTIL gdK > -1 OR SIZE(gdM1) > 0;
      IF SIZE(gdM1) > 0 THEN
        tcDrain();
        gdK := 30;                                  // a tap opens the details
      END;
      CASE
        IF gdK == 30 THEN tcDetails(gdI, gdS, gdD); END;
        IF gdK == 7 THEN gdStage := 2; END;
        IF gdK == 4 THEN gdStage := 0; END;
      END;
    END;
  END;
END;

// THE TABLE: 4 scales x 5 designs, Glantz's chapter table
// Cells are 52 px wide from x = 54 (columns) and 38 px tall
// from y = 46 (rows).  The touch handler uses the same numbers.
// ------------------------------------------------------------
tcDrawTable()
BEGIN
  LOCAL tbR, tbC, tbX, tbY, tbI, tbCell;
  LOCAL tbCol, tbLt, tbName, tbS;
  DIMGROB_P(G1, 320, 240, #FFFFFFh);
  TEXTOUT_P("Which test?  scale x design", G1, 54, 4, 1, #000000h);
  // column headings, two lines each
  tbS := {"2 groups", "3+ grps", "before/", "repeat", "assoc."};
  FOR tbC FROM 1 TO 5 DO
    tbX := 54 + (tbC - 1) * 53;
    TEXTOUT_P(tbS(tbC), G1, tbX + 3, 20, 1, tcGRAY, 50);
  END;
  tbS := {"indep.", "indep.", "after", "same ppl", "2 vars"};
  FOR tbC FROM 1 TO 5 DO
    tbX := 54 + (tbC - 1) * 53;
    TEXTOUT_P(tbS(tbC), G1, tbX + 3, 32, 1, tcGRAY, 50);
  END;
  tbS := {"Interval", "Nominal", "Ordinal", "Survival"};
  FOR tbR FROM 1 TO 4 DO
    tbY := 46 + (tbR - 1) * 38;
    tbCol := tcColOf(tbR); tbLt := tcLightOf(tbR);
    TEXTOUT_P(tbS(tbR), G1, 2, tbY + 13, 1, tbCol, 51);
    FOR tbC FROM 1 TO 5 DO
      tbX := 54 + (tbC - 1) * 53;
      tbI := tcIdFor(tbR, tbC + 1);
      tbCell := tcCell(tbI);
      IF tbR == tcRow AND tbC == tcCol THEN
        RECT_P(G1, tbX, tbY, tbX + 52, tbY + 37, tbCol, tbCol);
        TEXTOUT_P(tbCell(1), G1, tbX + 3, tbY + 7, 1, #FFFFFFh, 49);
        TEXTOUT_P(tbCell(2), G1, tbX + 3, tbY + 20, 1, #FFFFFFh, 49);
      ELSE
        RECT_P(G1, tbX, tbY, tbX + 52, tbY + 37, tcGRAY, tbLt);
        TEXTOUT_P(tbCell(1), G1, tbX + 3, tbY + 7, 1, tbCol, 49);
        TEXTOUT_P(tbCell(2), G1, tbX + 3, tbY + 20, 1, tbCol, 49);
      END;
    END;
  END;
  // the selected test, in full
  tbName := tcFull(tcIdFor(tcRow, tcCol + 1));
  TEXTOUT_P(tbName, G1, 6, 204, 1, tcColOf(tcRow), 308);
  TEXTOUT_P("arrows or tap, Enter: details, Esc: menu", G1, 6, 224, 1, tcFOOT);
  BLIT_P(G0, 0, 0, G1);
END;

tcTable()
BEGIN
  LOCAL tbK, tbM, tbM1, tbGo, tbR, tbC;
  tbGo := 1;
  WHILE tbGo == 1 DO
    tcDrawTable();
    REPEAT tbK := GETKEY; UNTIL tbK == -1;
    REPEAT
      tbK := GETKEY;
      tbM := MOUSE;
      tbM1 := tbM(1);
    UNTIL tbK > -1 OR SIZE(tbM1) > 0;
    IF SIZE(tbM1) > 0 THEN
      tcDrain();
      IF tbM1(1) >= 54 AND tbM1(2) >= 46 AND tbM1(2) < 198 THEN
        tbC := IP((tbM1(1) - 54) / 53) + 1;
        tbR := IP((tbM1(2) - 46) / 38) + 1;
        IF tbC >= 1 AND tbC <= 5 AND tbR >= 1 AND tbR <= 4 THEN
          tcRow := tbR; tcCol := tbC;
          tcDrawTable();
          tcDetails(tcIdFor(tcRow, tcCol + 1), tcRow, tcCol + 1);
        END;
      END;
    ELSE
      IF tbK == 7 THEN tcCol := MAX(1, tcCol - 1); END;
      IF tbK == 8 THEN tcCol := MIN(5, tcCol + 1); END;
      IF tbK == 2 THEN tcRow := MAX(1, tcRow - 1); END;
      IF tbK == 12 THEN tcRow := MIN(4, tcRow + 1); END;
      IF tbK == 30 THEN tcDetails(tcIdFor(tcRow, tcCol + 1), tcRow, tcCol + 1); END;
      IF tbK == 4 THEN tbGo := 0; END;
    END;
  END;
END;

EXPORT TestChooser()
BEGIN
  LOCAL mnI, mnRun;
  tcInit();
  mnRun := 1;
  WHILE mnRun == 1 DO
    mnI := 0;
    CHOOSE(mnI, "Which statistical test?", {
      "1  Guide me: two questions",
      "2  Browse the table (Glantz)",
      "3  Help",
      "Quit"});
    CASE
      IF mnI == 0 OR mnI == 4 THEN mnRun := 0; END;
      IF mnI == 1 THEN tcGuide(); END;
      IF mnI == 2 THEN tcTable(); END;
      IF mnI == 3 THEN tcHelp(); END;
    END;
  END;
END;
