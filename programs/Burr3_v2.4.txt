// ---- log-gamma (Lanczos): pure numeric, no CAS ----
// b3V is the argument (it was x in v2.2: bare x is the CAS variable).
b3_lngamma(b3V)
BEGIN
  LOCAL b3Y, tmp, ser, b3J, cof;
  cof := {76.18009172947146,-86.50532032941677,24.01409824083091,-1.231739572450155,0.001208650973866179,-0.000005395239384953};
  b3Y := b3V;
  tmp := b3V + 5.5;
  tmp := tmp - (b3V + 0.5)*LN(tmp);
  ser := 1.000000000190015;
  FOR b3J FROM 1 TO 6 DO
    b3Y := b3Y + 1;
    ser := ser + cof(b3J)/b3Y;
  END;
  RETURN -tmp + LN(2.5066282746310005*ser/b3V);
END;

// ---- pdf helper: Burr XII density at b3V for shape C, K and scale S ----
b3_bpdf(b3V,C,K,S)
BEGIN
  LOCAL b3Z;
  IF b3V <= 0 THEN RETURN 0; END;
  b3Z := (b3V/S)^C;
  RETURN (K*C*b3Z)/(b3V*(1+b3Z)^(K+1));
END;

// ---- fixed-decimal formatter with a floor ----
// Mode 2 keeps trailing zeros. The floor matters in the tail: a pdf of
// 3e-6 printed "0.0000", which reads as zero for a density that is
// merely small. Params pv/dg, not p/d — short parameter names collide
// with builtins. Non-EXPORT so it stays out of the catalog.
b3_fmtp(pv, dg)
BEGIN
  LOCAL thr;
  thr := 10^(-dg);
  IF pv != 0 AND ABS(pv) < thr THEN
    IF pv > 0 THEN RETURN "< " + STRING(thr,2,dg); END;
    RETURN "> -" + STRING(thr,2,dg);
  END;
  RETURN STRING(pv,2,dg);
END;

// ---- "nice" number for axis ticks ----
// b3V is the raw step; roundf = 1 rounds to the nearest 1/2/5, 0 rounds up.
b3_NiceNum(b3V, roundf)
BEGIN
  LOCAL expv, b3F, nf;
  IF b3V <= 0 THEN RETURN 1; END;
  expv := FLOOR(LOG(b3V));
  b3F := b3V / 10^expv;
  IF roundf THEN
    CASE
      IF b3F < 1.5 THEN nf := 1; END;
      IF b3F < 3 THEN nf := 2; END;
      IF b3F < 7 THEN nf := 5; END;
      DEFAULT nf := 10;
    END;
  ELSE
    CASE
      IF b3F <= 1 THEN nf := 1; END;
      IF b3F <= 2 THEN nf := 2; END;
      IF b3F <= 5 THEN nf := 5; END;
      DEFAULT nf := 10;
    END;
  END;
  RETURN nf * 10^expv;
END;

// ---- wait for the finger to lift, but never for more than a second ----
// MOUSE(1) is -1 while nothing touches the screen.  Without the time limit
// a stuck or phantom touch would hold the program here for ever, and the
// only way out would be a reset (Lessons Learned section 7; ChiTable's
// csDrain, device-verified).  TICKS is the millisecond clock.
b3_Drain()
BEGIN
  LOCAL drT;
  drT := TICKS;
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// ---- hold a page until any key OR a tap: drain, wait, drain ----
// The house page-hold (ChiTable's csWaitAny).  A GETKEY drain does not
// drain touch, so both are drained before and after the wait.
b3_WaitAny()
BEGIN
  LOCAL waK, waM, waM1;
  REPEAT waK := GETKEY; UNTIL waK == -1;
  b3_Drain();
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  b3_Drain();
  REPEAT waK := GETKEY; UNTIL waK == -1;
END;

// ---- a small label centred on a marker, kept inside the plot ----
// lbS the text, lbX the marker's pixel column, lbY the row, lbC colour.
// Font 1 is about 6 px a character.
b3_Label(lbS, lbX, lbY, lbC)
BEGIN
  LOCAL lbL;
  lbL := lbX - 3*SIZE(lbS);
  lbL := MAX(26, MIN(300 - 6*SIZE(lbS), lbL));
  TEXTOUT_P(lbS, G1, lbL, lbY, 1, lbC);
END;

EXPORT Burr3(C,K,S,x₀)
BEGIN
  LOCAL LFT,RGT,TOPy,BOT;
  LOCAL W,H,N;
  LOCAL xmax,dx,ymax,dX;
  LOCAL b3I,b3X,b3Y,b3J;
  LOCAL pxx,pyy,ppx,ppy;
  LOCAL x0px,pdf0,cdf0,b3X0;
  LOCAL cSh,cCv,cAx,cX0,cLR;
  LOCAL cTx,cGrid,cMn,cGy,cOrg;
  LOCAL cLO,cGh,cNm;
  LOCAL mmean,mvar,msd,sMu,sSD;
  LOCAL kk,showMode,lastcy,cy;
  LOCAL stp,ystp,nfrac,ynfrac;
  LOCAL tickx,ticky,xv,yv;
  LOCAL b3Pdf,b3Cdf,b3Nrm,b3Gh;
  LOCAL b3Q,b3Cap,b3New,sKC,b3Wn;
  LOCAL b3M,b3M1,b3Ev,b3Act,b3Quit;
  LOCAL b3Drag,b3TX,b3TY,b3LX,b3LY;
  LOCAL b3PX,b3PY,b3P0,b3P,b3Msg;
  LOCAL b3Re,b3CdfOK,b3NrmOK,b3Frz,b3Snap;
  LOCAL b3Sel,b3GhOn,b3GhDx,b3C,b3Lbl;
  LOCAL b3Mode,b3Med,b3Pk,b3Tap,b3Hx;

  IF C  <= 0 THEN RETURN "C must be > 0";  END;
  IF K  <= 0 THEN RETURN "K must be > 0";  END;
  IF S  <= 0 THEN RETURN "S must be > 0";  END;
  IF x₀ <= 0 THEN RETURN "x₀ must be > 0"; END;
  // Below K×C = 0.5 the tail is so heavy that the 0.995 quantile runs to
  // thousands of S and beyond (6.5E36*S at C = K = 0.25): no window can
  // show it.  The keys and the drags keep K×C >= 0.5 too.
  IF K*C < 0.5 THEN RETURN "K×C must be ≥ 0.5 (a heavier tail has no usable window)"; END;
  // x₀ arrives as the fourth argument; the program works on an ASCII copy,
  // b3X0 (house rule: no new Unicode in executable code).
  b3X0 := x₀;

  // ---- constants (screen 320x240, origin top-left, y increases DOWN) ----
  // Rows ABOVE the plot: 1 = title (C, K, S, x₀; the chosen one boxed),
  // 12 = the pdf / P(X <= x₀) readout.  The plot starts at TOPy = 24.
  // Rows BELOW it: x tick labels at BOT+4 = 194, the status row (moments,
  // K×C) at BOT+15 = 205, ending by 215.  Nothing is drawn below 216:
  // the soft-key strip DRAWMENU paints is 218..239.
  // RGT is 302 (was 314) so the right-hand CDF axis labels fit.
  // THE TOUCH HANDLER USES THESE SAME CONSTANTS: a finger at pixel column
  // px means x = (px - LFT)/W*xmax, the drawing's own map.
  LFT:= 24; RGT:=302; TOPy:=24; BOT:=190;
  W:=RGT-LFT; H:=BOT-TOPy; N:=W;
  // palette (Lessons Learned section 13), set once, not inside loops
  cSh:=RGB(170,205,255); cCv:=RGB(0,60,200); cAx:=RGB(0,0,0);
  cX0:=RGB(255,0,0); cTx:=RGB(200,0,200); cGrid:=RGB(225,228,235);
  cMn:=RGB(0,140,60); cGy:=RGB(90,90,90); cOrg:=RGB(255,69,0);
  cLR:=RGB(255,235,235); cLO:=RGB(255,240,220);
  cGh:=RGB(200,200,200);            // the ghost: light grey
  cNm:=RGB(90,90,90);               // the fitted normal: grey
  // 0 = grid+CDF, 1 = grid, 2 = CDF, 3 = plain, 4 = grid + fitted normal
  showMode := 0;
  mmean := 0; mvar := 0; msd := 0;
  b3Cap := 0;

  // ---- the state of the touch loop ----
  // b3Sel: the quantity the arrows and a drag adjust -- 1 C, 2 K, 3 S,
  //   4 x₀ (the soft keys 1-4 choose it).  C to start, so Left/Right do
  //   what they did in v2.3.
  // b3Drag: what the finger holds -- 0 nothing, 1 x₀, 2 the S-curve
  //   (inverse), 4 a press with C/K/S chosen that has not moved 8 px yet
  //   (a tap so far), 5 a parameter drag.
  // b3Frz = 1: the x-window is held (a finger is down).
  // b3Re = 1: refill the pdf cache.  b3CdfOK, b3NrmOK: the CDF and the
  //   fitted-normal caches match it.
  // b3GhOn = 1: draw the ghost (the curve before the last step).
  // b3Msg: row 2 shows 0 the readout, 1 the inverse, 2 a refused step.
  b3Sel := 1; b3Drag := 0; b3Frz := 0; b3Msg := 0; b3Act := 0; b3Quit := 0;
  b3Re := 1; b3CdfOK := 0; b3NrmOK := 0; b3GhOn := 0; b3GhDx := 1;
  b3LX := -1; b3LY := -1; b3PX := 0; b3PY := 0; b3P0 := 1; b3P := 0.5;
  b3Tap := 0; b3Pk := 1; xmax := 0; ymax := 1; pdf0 := 0; cdf0 := 0;

  // ---- caches, sized once: b3Pdf(k), b3Cdf(k), b3Nrm(k) hold the pdf,
  //      the CDF and the fitted normal at pixel column k (x = k*dx),
  //      k = 1..N; b3Gh is the ghost (a copy of b3Pdf before a step).
  //      The MAKELIST counter is Home's X (HP's own example form,
  //      MAKELIST(X^2,X,1,5); StudRange's form on hardware): it must NOT
  //      be K here, because K is this program's shape parameter and
  //      MAKELIST writes its counter.  X in Home is overwritten; this
  //      program does not use it.  A LOCAL counter waits for GfxProbe
  //      T19. ----
  b3Pdf := MAKELIST(0, X, 1, N, 1);
  b3Cdf := MAKELIST(0, X, 1, N, 1);
  b3Nrm := MAKELIST(0, X, 1, N, 1);
  b3Gh  := MAKELIST(0, X, 1, N, 1);

  DIMGROB_P(G1,320,240);            // off-screen buffer, built once

  // ---- drain the key that closed the launch form, and any touch, so
  //      neither can satisfy the first wait below (section 7) ----
  REPEAT kk := GETKEY; UNTIL kk == -1;
  b3_Drain();

  REPEAT
    // ==== 1. the x-range: the 0.995 quantile, capped, x₀ kept on screen ====
    // The quantile S*((0.005)^(-1/K)-1)^(1/C) grows fast as K×C shrinks:
    // 199*S at C = K = 1, about 1.6E9*S at C = K = 0.5, about 6.5E36*S at
    // C = K = 0.25 -- v2.2 then printed 37-digit tick labels and a blank
    // plot.  The window is capped at 50*S; when the cap bites the plot
    // says so (section 6: say on screen what was suppressed).
    // While a finger is down (b3Frz = 1) the window is HELD, so the map
    // from the finger's pixel to x cannot move under it; it is refitted
    // on the lift (BRIEF_TOUCH: freeze during a drag, refit on lift).
    IF b3Frz == 0 THEN
      b3Q  := S*((0.005)^(-1/K)-1)^(1/C);
      b3Wn := MIN(b3Q, 50*S);
      b3Wn := MAX(b3Wn, b3X0*1.15);
      IF b3Wn < b3Q THEN b3Cap := 1; ELSE b3Cap := 0; END;
      IF b3Wn <> xmax THEN xmax := b3Wn; b3Re := 1; END;
    END;
    dx   := xmax/N;
    dX   := xmax/40;                  // x₀ slide step
    // the grid x₀ snaps to on lift: 1/100 of the window, 1-2-5
    b3Snap := b3_NiceNum(xmax/100, 1);

    // ==== 2. moments (guards: mean K·C>1, variance K·C>2) ====
    // The order matters: K·C>2 implies K·C>1, so mmean is always defined
    // before the variance line uses it.  "No mean" is a teaching feature
    // of this distribution, not a failure.  (Before the caches now: the
    // fitted normal is built from these.)
    IF K*C > 1 THEN
      mmean := S*EXP(b3_lngamma(1+1/C)+b3_lngamma(K-1/C)-b3_lngamma(K));
      sMu := "μ = " + STRING(mmean,2,3);
    ELSE
      mmean := 0;
      sMu := "no mean (K×C ≤ 1)";
    END;
    IF K*C > 2 THEN
      mvar := S^2*EXP(b3_lngamma(1+2/C)+b3_lngamma(K-2/C)-b3_lngamma(K)) - mmean^2;
      IF mvar < 0 THEN mvar := 0; END;
      msd := √(mvar);
      sSD := "σ = " + STRING(msd,2,3);
    ELSE
      mvar := 0; msd := 0;
      sSD := "no variance (K×C ≤ 2)";
    END;
    // mode and median, closed forms (verify_Burr3.py vs scipy.stats.burr12):
    // the mode is where the pdf turns, x^C = (C-1)/(K*C+1), which needs
    // C > 1 (for C <= 1 the pdf falls from x = 0, so the mode is 0); the
    // median solves F(x) = 1/2.
    IF C > 1 THEN b3Mode := S*((C-1)/(K*C+1))^(1/C); ELSE b3Mode := 0; END;
    b3Med := S*(2^(1/K)-1)^(1/C);

    // ==== 3. the caches: refilled only when C, K, S or the window moved ====
    // The scan stores each value in b3Pdf and keeps the largest (b3Pk),
    // the Burr's peak on screen, for the y-scale (section 6: cache the
    // density).  Moving x₀ needs no refill.
    IF b3Re == 1 THEN
      b3Pk := 0;
      FOR b3I FROM 1 TO N DO
        b3Y := b3_bpdf(b3I*dx,C,K,S);
        b3Pdf(b3I) := b3Y;
        IF b3Y > b3Pk THEN b3Pk := b3Y; END;
      END;
      b3Re := 0; b3CdfOK := 0; b3NrmOK := 0;
    END;
    // the CDF, filled only in the views that draw the S-curve
    IF (showMode == 0 OR showMode == 2) AND b3CdfOK == 0 THEN
      FOR b3I FROM 1 TO N DO
        b3Cdf(b3I) := 1-(1+(b3I*dx/S)^C)^(-K);
      END;
      b3CdfOK := 1;
    END;
    // the fitted normal (view 4), only when the sd exists (K×C > 2)
    IF showMode == 4 AND K*C > 2 AND b3NrmOK == 0 THEN
      FOR b3I FROM 1 TO N DO
        b3Nrm(b3I) := NORMALD(mmean, msd, b3I*dx);
      END;
      b3NrmOK := 1;
    END;
    // y-range: the Burr's peak on screen, or the fitted normal's peak
    // when that is taller (check BOTH curves before drawing a reference:
    // Lessons Learned section 6), plus 15 %.  Not frozen: it follows C,
    // K and S, also during a drag.
    ymax := b3Pk;
    IF showMode == 4 AND K*C > 2 THEN
      ymax := MAX(ymax, NORMALD(mmean, msd, mmean));
    END;
    ymax := ymax*1.15;
    IF ymax <= 0 THEN ymax := 1; END;

    // ==== 4. the values at x₀ ====
    pdf0 := b3_bpdf(b3X0,C,K,S);
    cdf0 := 1-(1+(b3X0/S)^C)^(-K);
    x0px := IP(LFT + b3X0/xmax*W);

    RECT_P(G1,0,0,319,239,RGB(255,255,255));      // clear the buffer

    // ---- nice-number tick positions ----
    stp := b3_NiceNum(xmax/6, 1);
    nfrac := MAX(0, -FLOOR(LOG(stp)));
    ystp := b3_NiceNum(ymax/4, 1);
    ynfrac := MAX(0, -FLOOR(LOG(ystp)));

    // ---- grid at nice values (behind everything) ----
    IF showMode == 0 OR showMode == 1 OR showMode == 4 THEN
      xv := stp;
      WHILE xv <= xmax + stp/1000 DO
        tickx := LFT + IP(xv/xmax*W);
        IF tickx > LFT AND tickx <= RGT THEN
          LINE_P(G1, tickx, TOPy, tickx, BOT-1, cGrid);
        END;
        xv := xv + stp;
      END;
      yv := ystp;
      WHILE yv < ymax DO
        ticky := BOT - IP(yv/ymax*H);
        LINE_P(G1, LFT+1, ticky, RGT, ticky, cGrid);
        yv := yv + ystp;
      END;
    END;

    // ---- heavy tail (K×C <= 1): hatch the right 20 % of the plot ----
    // A light-red band with a sparse diagonal dot pattern (PIXON_P: there
    // is no fill pattern on the Prime): the tail goes on past the window
    // so slowly that the mean is infinite.  b3Hx is the band's left edge.
    // Rows 4 px apart, dots 8 px apart, shifted 2 px a row: ~290 dots.
    IF K*C <= 1 THEN
      b3Hx := LFT + IP(0.8*W);
      RECT_P(G1, b3Hx, TOPy, RGT, BOT-1, cLR);
      FOR b3I FROM TOPy+2 TO BOT-2 STEP 4 DO
        b3J := ((b3I - TOPy)/2) MOD 8;
        FOR pxx FROM b3Hx + b3J TO RGT-1 STEP 8 DO
          PIXON_P(G1, pxx, b3I, cX0);
        END;
      END;
    END;

    // ---- shading: iterate PIXEL COLUMNS (no gaps), x<=x₀ ----
    // Column pxx is cache entry pxx - LFT (1..N): the same x as the curve.
    FOR pxx FROM LFT+1 TO RGT DO
      b3X := (pxx - LFT)/W*xmax;
      IF b3X <= b3X0 THEN
        b3Y := b3Pdf(pxx - LFT);
        pyy := IP(BOT - b3Y/ymax*H);
        IF pyy < TOPy THEN pyy := TOPy; END;
        LINE_P(G1, pxx, BOT, pxx, pyy, cSh);
      END;
    END;

    // ---- the ghost: the curve before the last C, K or S step, in light
    //      grey, drawn BEFORE the new curve.  Its columns are those of
    //      the window it was drawn in (b3GhDx), mapped into this one. ----
    IF b3GhOn == 1 THEN
      ppx := -1;
      FOR b3I FROM 1 TO N DO
        pxx := IP(LFT + b3I*b3GhDx/xmax*W);
        IF pxx <= RGT THEN
          pyy := IP(BOT - b3Gh(b3I)/ymax*H);
          IF pyy < TOPy THEN pyy := TOPy; END;
          IF ppx >= 0 THEN LINE_P(G1, ppx, ppy, pxx, pyy, cGh); END;
          ppx := pxx; ppy := pyy;
        END;
      END;
    END;

    // ---- pdf curve on top (from the cache) ----
    ppx := -1;
    FOR b3I FROM 1 TO N DO
      pxx := LFT + b3I;
      pyy := IP(BOT - b3Pdf(b3I)/ymax*H);
      IF pyy < TOPy THEN pyy := TOPy; END;
      IF ppx >= 0 THEN LINE_P(G1, ppx, ppy, pxx, pyy, cCv); END;
      ppx := pxx; ppy := pyy;
    END;

    // ---- view 4: the fitted normal, same mean and sd, in grey ----
    // Skewed against symmetric with the same μ and σ; the normal also
    // puts probability below 0, where the Burr has none -- said on screen.
    IF showMode == 4 THEN
      IF K*C > 2 THEN
        ppx := -1;
        FOR b3I FROM 1 TO N DO
          pxx := LFT + b3I;
          pyy := IP(BOT - b3Nrm(b3I)/ymax*H);
          IF pyy < TOPy THEN pyy := TOPy; END;
          IF ppx >= 0 THEN LINE_P(G1, ppx, ppy, pxx, pyy, cNm); END;
          ppx := pxx; ppy := pyy;
        END;
        // (rows TOPy+38 and +48: below the three marker labels)
        TEXTOUT_P("grey = normal, same μ and σ", G1, RGT-168, TOPy+38, 1, cNm);
        TEXTOUT_P("its P(X < 0) = "+b3_fmtp(NORMALD_CDF(mmean, msd, 0),4),
                  G1, RGT-168, TOPy+48, 1, cNm);
      ELSE
        TEXTOUT_P("normal fit needs K×C > 2 (now "+STRING(K*C,2,2)+")",
                  G1, RGT-246, TOPy+38, 1, cOrg);
      END;
    END;

    // ---- CDF S-curve (0..1 over the full plot height) and its axis ----
    IF showMode == 0 OR showMode == 2 THEN
      lastcy := -1;
      FOR b3I FROM 1 TO N DO
        cy := IP(BOT - b3Cdf(b3I)*H);
        pxx := LFT + b3I;
        IF lastcy >= 0 THEN LINE_P(G1, pxx-1, lastcy, pxx, cy, cTx); END;
        lastcy := cy;
      END;
      // the right-hand 0-1 axis for the S-curve: a magenta line at RGT
      // with ticks at 0, .5 and 1 (it replaces v2.3's lone "1")
      LINE_P(G1, RGT, TOPy, RGT, BOT, cTx);
      FOR b3I FROM 0 TO 2 DO
        ticky := BOT - IP(b3I*H/2);
        LINE_P(G1, RGT, ticky, RGT+3, ticky, cTx);
      END;
      TEXTOUT_P("1", G1, RGT+6, TOPy-2, 1, cTx);
      TEXTOUT_P(".5", G1, RGT+5, BOT-IP(H/2)-4, 1, cTx);
      TEXTOUT_P("0", G1, RGT+6, BOT-9, 1, cTx);
    END;

    // ---- three markers: mode (orange), median (magenta), mean (green) ----
    // Dotted verticals, each with its name and value on its own row at
    // the top so that close markers do not print over each other.  A
    // marker beyond the window is named at the right edge with an arrow.
    // As C falls (drag it) the three slide apart: the longer the right
    // tail, the further the mean is pulled past the median and the mode.
    tickx := LFT + IP(b3Mode/xmax*W);
    IF tickx <= RGT THEN
      FOR b3I FROM TOPy+11 TO BOT STEP 4 DO PIXON_P(G1, MAX(LFT+1, tickx), b3I, cOrg); END;
      b3_Label("mode "+STRING(b3Mode,2,2), tickx, TOPy+1, cOrg);
    ELSE
      b3_Label("mode "+STRING(b3Mode,2,2)+" "+CHAR(8594), 400, TOPy+1, cOrg);
    END;
    tickx := LFT + IP(b3Med/xmax*W);
    IF tickx <= RGT THEN
      FOR b3I FROM TOPy+22 TO BOT STEP 4 DO PIXON_P(G1, tickx, b3I, cTx); END;
      b3_Label("median "+STRING(b3Med,2,2), tickx, TOPy+12, cTx);
    ELSE
      b3_Label("median "+STRING(b3Med,2,2)+" "+CHAR(8594), 400, TOPy+12, cTx);
    END;
    IF K*C > 1 THEN
      tickx := LFT + IP(mmean/xmax*W);
      IF tickx <= RGT THEN
        FOR b3I FROM TOPy+33 TO BOT STEP 4 DO PIXON_P(G1, tickx, b3I, cMn); END;
        b3_Label("mean "+STRING(mmean,2,2), tickx, TOPy+23, cMn);
      ELSE
        b3_Label("mean "+STRING(mmean,2,2)+" "+CHAR(8594), 400, TOPy+23, cMn);
      END;
    END;

    // ---- axes + labeled nice ticks ----
    LINE_P(G1, LFT, BOT, RGT, BOT, cAx);
    LINE_P(G1, LFT, BOT, LFT, TOPy, cAx);
    // "0" at the origin: the x-axis starts at 0 (the Burr has no mass
    // below it) and the first labeled tick is one step to the right.
    TEXTOUT_P("0", G1, LFT-3, BOT+4, 1, cGy);
    xv := stp;
    WHILE xv <= xmax + stp/1000 DO
      tickx := LFT + IP(xv/xmax*W);
      IF tickx <= RGT THEN
        LINE_P(G1, tickx, BOT, tickx, BOT+3, cAx);
        // BELOW the axis, centred under the tick (3*SIZE is half the
        // label's width in font 1, as Norm3 v2.3 does).
        b3Lbl := STRING(xv,2,nfrac);
        TEXTOUT_P(b3Lbl, G1, tickx - 3*SIZE(b3Lbl), BOT+4, 1, cGy);
      END;
      xv := xv + stp;
    END;
    yv := ystp;
    WHILE yv < ymax DO
      ticky := BOT - IP(yv/ymax*H);
      LINE_P(G1, LFT-3, ticky, LFT, ticky, cAx);
      TEXTOUT_P(STRING(yv,2,ynfrac), G1, 1, ticky-4, 1, cGy);
      yv := yv + ystp;
    END;

    // ---- x₀ marker (vertical) + dotted horizontal to the y-axis ----
    pyy  := IP(BOT - pdf0/ymax*H);
    IF pyy < TOPy THEN pyy := TOPy; END;
    LINE_P(G1, x0px, BOT, x0px, pyy, cX0);
    FOR b3I FROM LFT+1 TO x0px STEP 4 DO
      PIXON_P(G1, b3I, pyy, cX0);
    END;
    // ---- the magenta point riding the S-curve at x₀, with a dotted
    //      line across to the right-hand axis ----
    IF showMode == 0 OR showMode == 2 THEN
      cy := IP(BOT - cdf0*H);
      FOR b3I FROM x0px+4 TO RGT STEP 4 DO
        PIXON_P(G1, b3I, cy, cTx);
      END;
      RECT_P(G1, x0px-2, cy-2, x0px+2, cy+2, cTx);
    END;

    // ---- said on the plot: no finite mean, and a capped window ----
    IF K*C <= 1 THEN
      RECT_P(G1, RGT-164, BOT-45, RGT-2, BOT-32, cX0, RGB(255,255,255));
      TEXTOUT_P("heavy tail: no finite mean", G1, RGT-160, BOT-43, 1, cX0);
    END;
    IF b3Cap == 1 THEN
      RECT_P(G1, RGT-134, BOT-29, RGT-2, BOT-16, cOrg, RGB(255,255,255));
      TEXTOUT_P("window capped at 50×S", G1, RGT-130, BOT-27, 1, cOrg);
    END;

    // ---- row 1: the title, the four quantities, the chosen one boxed ----
    TEXTOUT_P("Burr", G1, 2, 1, 1, cAx);
    b3I := 34 + (b3Sel - 1)*68;
    RECT_P(G1, b3I-3, 0, b3I+65, 11, cOrg, cLO);
    TEXTOUT_P("C = "+STRING(C,2,2), G1, 34, 1, 1, cAx);
    TEXTOUT_P("K = "+STRING(K,2,2), G1, 102, 1, 1, cAx);
    TEXTOUT_P("S = "+STRING(S,2,2), G1, 170, 1, 1, cAx);
    TEXTOUT_P("x₀ = "+STRING(b3X0,2,2), G1, 238, 1, 1, cAx);
    // ---- row 2: the readout, the inverse, or a refused step ----
    // pdf is the curve's HEIGHT at x₀ (not a probability); P(X <= x₀)
    // is the shaded AREA to the left of x₀.
    IF b3Msg == 1 THEN
      TEXTOUT_P("p = "+STRING(b3P,2,3)+" "+CHAR(8594)+" x₀ = "+STRING(b3X0,2,2),
                G1, 4, 12, 1, cTx);
    ELSE
      IF b3Msg == 2 THEN
        TEXTOUT_P("step refused: K×C must stay ≥ 0.5", G1, 4, 12, 1, cOrg);
      ELSE
        TEXTOUT_P("pdf = "+b3_fmtp(pdf0,4)+" (a height), P(X ≤ x₀) = "+
                  b3_fmtp(cdf0,4), G1, 4, 12, 1, cTx);
      END;
    END;
    // ---- the status row: the moments (they may not exist) and K×C,
    //      orange when it sits on the 0.5 floor ----
    IF K*C > 2 THEN
      TEXTOUT_P(sMu+", "+sSD, G1, 4, BOT+15, 1, cAx);
    ELSE
      IF K*C > 1 THEN
        TEXTOUT_P(sMu+", "+sSD, G1, 4, BOT+15, 1, cAx);
      ELSE
        TEXTOUT_P(sMu, G1, 4, BOT+15, 1, cX0);
      END;
    END;
    sKC := "K×C = " + STRING(K*C,2,2);
    IF ROUND(K*C, 6) <= 0.5 THEN
      TEXTOUT_P(sKC+" (floor)", G1, 208, BOT+15, 1, cOrg);
    ELSE
      TEXTOUT_P(sKC, G1, 248, BOT+15, 1, cGy);
    END;

    BLIT_P(G0, 0, 0, G1);             // one flicker-free page flip
    // the soft keys, painted on the screen after the blit, every frame
    DRAWMENU({"C", "K", "S", "x₀", "View", "Quit"});

    // ==== 5. input: a key, a press, a drag sample, or a lift ====
    // b3Ev: 1 a press or a moved finger, 2 the finger lifted, 3 a key.
    // While something is held the loop waits until the finger MOVES or
    // lifts (a resting finger redraws nothing).  A key ends a hold.
    b3Ev := 0; kk := -1;
    REPEAT
      kk := GETKEY;
      b3M := MOUSE;
      b3M1 := b3M(1);
      IF b3Drag > 0 THEN
        IF SIZE(b3M1) == 0 THEN
          b3Ev := 2;
        ELSE
          IF b3M1(1) <> b3LX OR b3M1(2) <> b3LY THEN b3Ev := 1; END;
        END;
      ELSE
        IF SIZE(b3M1) > 0 THEN b3Ev := 1; END;
      END;
      IF kk > -1 THEN b3Ev := 3; END;
    UNTIL b3Ev > 0;
    // The ghost stays on screen until the next action; it is kept
    // through a parameter drag (it is the curve the drag started from).
    IF b3Drag < 4 THEN b3GhOn := 0; END;

    // ---- the finger lifted (or a key ended the hold) ----
    b3Tap := 0;
    IF b3Ev == 2 OR (b3Ev == 3 AND b3Drag > 0) THEN
      IF b3Drag == 4 THEN
        // C, K or S was chosen but the finger never moved 8 px: a TAP.
        // It sets x₀ (or, on the S-curve, the inverse) at the press
        // point, exactly as when x₀ is chosen -- handled just below.
        b3Tap := 1; b3TX := b3PX; b3TY := b3PY;
      END;
      IF b3Drag == 1 THEN
        // x₀ snaps to the grid, staying inside the window and above 0
        b3X0 := MAX(b3Snap, ROUND(b3X0/b3Snap, 0)*b3Snap);
        IF b3X0 > xmax THEN b3X0 := b3X0 - b3Snap; END;
      END;
      b3Drag := 0; b3Frz := 0;
      b3_Drain();
    END;

    // ---- a press, a drag sample, or the tap just completed ----
    IF b3Ev == 1 OR b3Tap == 1 THEN
      IF b3Ev == 1 THEN b3TX := b3M1(1); b3TY := b3M1(2); END;
      IF b3Drag == 0 THEN
        // A NEW PRESS: decide what it holds, once, from the first sample.
        IF b3TY >= 218 THEN
          // the soft-key strip: six 53-px slots; act once, on the lift
          b3_Drain();
          b3Act := MIN(6, IP(b3TX/53.4) + 1);
        ELSE
          IF b3TX >= LFT AND b3TX <= RGT AND b3TY >= TOPy AND b3TY <= BOT THEN
            b3PX := b3TX; b3PY := b3TY;
            IF b3Sel == 4 OR b3Tap == 1 THEN
              // x₀ follows the finger -- unless the press is ON the
              // S-curve (within 16 px, and nearer it than the pdf)
              b3Drag := 1;
              IF showMode == 0 OR showMode == 2 THEN
                b3J := MAX(1, MIN(N, b3TX - LFT));
                b3Y := ABS(b3TY - (BOT - b3Cdf(b3J)*H));
                IF b3Y <= 16 AND b3Y < ABS(b3TY - (BOT - b3Pdf(b3J)/ymax*H)) THEN
                  b3Drag := 2;
                END;
              END;
            ELSE
              // C, K or S chosen: a tap or a drag?  Wait and see (4);
              // keep the curve as it is now, for the ghost
              b3Drag := 4;
              IF b3Sel == 1 THEN b3P0 := C; END;
              IF b3Sel == 2 THEN b3P0 := K; END;
              IF b3Sel == 3 THEN b3P0 := S; END;
              b3Gh := b3Pdf; b3GhDx := dx;
            END;
          ELSE
            // on the right-hand 0-1 axis: the inverse as well
            IF (showMode == 0 OR showMode == 2) AND b3TX > RGT
               AND b3TY >= TOPy AND b3TY <= BOT THEN
              b3Drag := 2;
            ELSE
              b3_Drain();             // outside the plot: nothing to do
            END;
          END;
        END;
      END;
      // APPLY THIS SAMPLE to what is held (the first sample too); the
      // finger is clamped to the plot box.
      IF b3Drag > 0 THEN
        IF b3Tap == 0 THEN b3LX := b3TX; b3LY := b3TY; END;
        b3TX := MAX(LFT+1, MIN(RGT, b3TX));
        b3TY := MAX(TOPy, MIN(BOT, b3TY));
        b3Frz := 1;                   // hold the window while the finger is down
        IF b3Drag == 1 THEN
          // x₀ from the pixel column: the drawing's own map
          b3X0 := (b3TX - LFT)/W*xmax;
          b3Msg := 0;
        END;
        IF b3Drag == 2 THEN
          // p from the height on the right-hand 0-1 axis, rounded to 0.01
          // (one pixel is 1/H = 0.006 of p); 0.025 and 0.975 are caught
          // within 0.003.  p is kept inside the part of the S-curve on
          // screen, then x₀ = S*((1-p)^(-1/K) - 1)^(1/C), the Burr XII
          // inverse CDF (verify_Burr3.py).
          b3P := (BOT - b3TY)/H;
          IF ABS(b3P - 0.975) < 0.003 THEN
            b3P := 0.975;
          ELSE
            IF ABS(b3P - 0.025) < 0.003 THEN
              b3P := 0.025;
            ELSE
              b3P := ROUND(b3P*100, 0)/100;
            END;
          END;
          b3P := MAX(b3P, MAX(0.005, b3Cdf(2)));
          b3P := MIN(b3P, MIN(0.995, b3Cdf(N-1)));
          IF b3P > 0 AND b3P < 1 THEN
            b3X0 := S*((1-b3P)^(-1/K)-1)^(1/C);
            b3Msg := 1;
          END;
        END;
        IF b3Drag == 4 THEN
          // more than 8 px sideways: it is a drag of the chosen parameter
          IF ABS(b3TX - b3PX) > 8 THEN b3Drag := 5; b3GhOn := 1; END;
        END;
        IF b3Drag == 5 THEN
          // C and K: 0.01 a pixel; S: x1.02 a pixel.  The K×C >= 0.5
          // floor holds on every sample: C or K stops at the smallest
          // hundredth that keeps the product at or above 0.5.
          IF b3Sel == 1 THEN
            C := ROUND(b3P0 + 0.01*(b3TX - b3PX), 2);
            IF C*K < 0.5 THEN C := CEILING(50/K)/100; END;
          END;
          IF b3Sel == 2 THEN
            K := ROUND(b3P0 + 0.01*(b3TX - b3PX), 2);
            IF K*C < 0.5 THEN K := CEILING(50/C)/100; END;
          END;
          IF b3Sel == 3 THEN
            S := MAX(0.01, b3P0*1.02^(b3TX - b3PX));
          END;
          b3Re := 1; b3Msg := 0;
        END;
      END;
      IF b3Tap == 1 THEN
        // the tap is over: snap as for a lifted x₀ drag, release the window
        IF b3Drag == 1 THEN
          b3X0 := MAX(b3Snap, ROUND(b3X0/b3Snap, 0)*b3Snap);
          IF b3X0 > xmax THEN b3X0 := b3X0 - b3Snap; END;
        END;
        b3Drag := 0; b3Frz := 0;
      END;
    END;

    // ---- keys: the hardware keys stay as shortcuts for the strip ----
    // b3Act: 1-4 choose C, K, S, x₀; 5 view; 6 quit (the six soft keys);
    //        7 the chosen quantity down, 8 up; 9 x₀-, 10 x₀+;
    //        11 choose the next name; 12 menu
    IF b3Ev == 3 THEN
      IF kk == 7 OR kk == 12 THEN b3Act := 7;  END;  // Left / Down : down
      IF kk == 8 OR kk == 2  THEN b3Act := 8;  END;  // Right / Up  : up
      IF kk == 4  THEN b3Act := 6;  END;             // Esc   : quit
      IF kk == 45 THEN b3Act := 9;  END;             // (-)   : x₀ down
      IF kk == 50 THEN b3Act := 10; END;             // (+)   : x₀ up
      IF kk == 42 THEN b3Act := 11; END;             // 0     : next name
      IF kk == 30 THEN b3Act := 12; END;             // Enter : menu
    END;

    // ---- the menu (Enter): Help, Next view, Quit ----
    IF b3Act == 12 THEN
      b3Act := 0;
      b3C := 0;
      CHOOSE(b3C, "Burr3", {"Help: touch and keys", "Next view", "Quit"});
      // the press that closed the menu can arrive again underneath
      REPEAT kk := GETKEY; UNTIL kk == -1;
      b3_Drain();
      IF b3C == 1 THEN b3_Help(); END;
      IF b3C == 2 THEN b3Act := 5; END;
      IF b3C == 3 THEN b3Act := 6; END;
    END;

    // ---- do it ----
    IF b3Act >= 1 AND b3Act <= 4 THEN b3Sel := b3Act; END;
    IF b3Act == 11 THEN b3Sel := (b3Sel MOD 4) + 1; END;
    IF b3Act == 5 THEN showMode := (showMode+1) MOD 5; END;
    IF b3Act == 6 THEN b3Quit := 1; END;
    IF b3Act == 9 THEN b3X0 := MAX(0.05, b3X0-dX); b3Msg := 0; END;
    IF b3Act == 10 THEN b3X0 := b3X0+dX; b3Msg := 0; END;
    IF b3Act == 7 OR b3Act == 8 THEN
      IF b3Sel == 4 THEN
        // x₀ chosen: the arrows slide it, as +/- do
        IF b3Act == 7 THEN b3X0 := MAX(0.05, b3X0-dX); ELSE b3X0 := b3X0+dX; END;
        b3Msg := 0;
      ELSE
        // C, K or S: keep the curve as it is now for the ghost.  C and K
        // step by 0.25; a step DOWN that would take K×C below 0.5 is
        // refused and row 2 says so (below that the 0.995 quantile runs
        // to thousands of S and the plot has nothing useful to show).
        // The floor is on the PRODUCT, so C can go down to 0.25 when K
        // is large enough, and vice versa.  S scales by x1.25.
        b3Gh := b3Pdf; b3GhDx := dx; b3GhOn := 1; b3Msg := 0;
        IF b3Sel == 1 THEN
          IF b3Act == 8 THEN
            C := C+0.25;
          ELSE
            b3New := C - 0.25;
            IF b3New*K >= 0.5 THEN C := b3New; ELSE b3GhOn := 0; b3Msg := 2; END;
          END;
        END;
        IF b3Sel == 2 THEN
          IF b3Act == 8 THEN
            K := K+0.25;
          ELSE
            b3New := K - 0.25;
            IF b3New*C >= 0.5 THEN K := b3New; ELSE b3GhOn := 0; b3Msg := 2; END;
          END;
        END;
        IF b3Sel == 3 THEN
          IF b3Act == 8 THEN S := S*1.25; ELSE S := MAX(0.01, S/1.25); END;
        END;
        b3Re := 1;
      END;
    END;
    b3Act := 0;
  UNTIL b3Quit == 1;

  // ---- export the final state so it can be read in Home ----
  EP_BURR_MEAN := mmean;
  EP_BURR_SD   := msd;
  EP_BURR_PDF  := b3_bpdf(b3X0,C,K,S);
  EP_BURR_CDF  := 1-(1+(b3X0/S)^C)^(-K);
  EP_BURR_X0   := b3X0;

  RETURN "Burr3: C = " + STRING(C,2,2) + ", K = " + STRING(K,2,2) + ", S = " +
         STRING(S,2,2) + ", P(X ≤ " + STRING(b3X0,2,2) + ") = " +
         b3_fmtp(EP_BURR_CDF,4) + "; values in EP_BURR_*";
END;
