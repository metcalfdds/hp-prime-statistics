// Normal distribution — interactive pdf/CDF explorer — Roger D Metcalf
//   μ = mean   σ = std dev   x₀ = evaluation point
// Launch:  Norm3(0, 1, 1)   (parallel to Burr3)
// TOUCH:     drag in the plot = x₀;  press on the S-curve (or on the
//            right-hand 0-1 axis) = the inverse, p -> x₀;  press on the
//            green μ line = drag the mean;  soft keys μ− μ+ σ− σ+ View Quit
// KEYS:      Left/Right = μ    Up/Down = σ (animated)    (-)/(+) = x₀
//            0 = re-centre the window    Enter = menu (Help, Next view,
//            Re-centre, Quit)    Esc = quit
// Live readouts: μ and σ (title row), x₀, z, the pdf height, and both
// areas P(X ≤ x₀) and P(X > x₀).
// Window frozen at μ ± 4σ on entry (x₀ can push it; 0 re-centres);
// nice-number labelled ticks on both axes; a z-ruler under the x-axis;
// CDF S-curve with its own 0-1 axis on the right; dotted mean marker;
// solid vertical x₀ marker + dotted horizontal line to the y-axis.
//
// v2 CHANGES (28 Jul 2026) — presentation and controls only, no maths
//   * Flicker-free: draw to G1, BLIT once per frame. v1 cleared G0 and
//     rebuilt it in view, so every keypress flashed.
//   * μ/σ panel: v1 padded the strings with 6 leading spaces inside a
//     206..296 box with text at x=212. That fits "μ = 0" but a negative
//     mean overflows — "      μ = -12.5" runs to x≈302 and clips. Since a
//     normal mean is routinely negative (unlike Burr3's x, which is always
//     positive) this bites in ordinary use. Padding dropped, box widened
//     to 194..306 with text at 200: "μ = -123.456" now ends near x=272.
//     (v2.4: the panel is gone; μ and σ are in the title row.)
//   * ADAPTIVE STEPS. v1 moved μ and σ by a fixed 0.25 regardless of scale.
//     At σ=15 that is 0.017σ per keypress; at σ=100 it is invisible on a
//     290 px plot. Now μ steps by σ/4 and σ scales by ×1.25 / ÷1.25, so the
//     controls feel identical at every scale.
//   * All STRING(ROUND(x,d)) -> fixed-decimal STRING(x,2,d); mode 1 drops
//     trailing zeros regardless of the digit count, so σ = 1.500 printed
//     "1.5" and the tick labels came out ragged.
//   * pdf/CDF readouts through n3_fmtp(), which floors at "< 0.0001" instead
//     of printing "0.0000" — four sigma out the density is small, not zero,
//     and showing that is the point of the explorer.
//   * x-axis tick labels moved BELOW the axis (v1 drew them at BOT-11, i.e.
//     inside the plot, on top of the shading and curve). BOT 208 -> 200 to
//     make room.
//   * Mean marker dotted, to read differently from the solid red x₀ line.
//   * Exports EP_NORM_* so results are readable in Home after Esc.
//
// VERIFIED (verify_Norm3.py, all checks pass):
//   NORMALD(μ,σ,μ) is the closed-form peak 1/(σ√(2π)); the μ±4σ window is
//   symmetric and covers 99.9937%; edge/peak density ratio is exp(-8) at
//   every scale; CDF(x₀) == Φ(z) for the z readout; and the tick ladder
//   CEILING(xmin/stp)*stp places every tick inside a NEGATIVE range (the
//   one structural difference from Burr3, whose x starts at 0).
//   v2.4 (verify_Norm3.py, rewritten): the pixel -> x0 map, the p -> x0 inverse,

// ---- fixed-decimal formatter with a floor ----
// Mode 2 keeps trailing zeros. The floor matters in the tail: at z = -4 the
// density is ~1.3e-4 and beyond that it prints "0.0000", which reads as zero
// for a value that is merely small. Params pv/dg, not p/d — short parameter
// names collide with builtins. Non-EXPORT so it stays out of the catalog.
n3_fmtp(pv, dg)
BEGIN
  LOCAL thr;
  thr := 10^(-dg);
  IF pv != 0 AND ABS(pv) < thr THEN
    IF pv > 0 THEN RETURN "< " + STRING(thr,2,dg); END;
    RETURN "> -" + STRING(thr,2,dg);
  END;
  RETURN STRING(pv,2,dg);
END;

// ---- "nice" number for axis ticks ----
// n3V is the raw step; roundf = 1 rounds to the nearest 1/2/5, 0 rounds up.
// (The parameter was x in v2.2: bare x is the CAS variable, section 5.)
n3_NiceNum(n3V, roundf)
BEGIN
  LOCAL expv, n3F, nf;
  IF n3V <= 0 THEN RETURN 1; END;
  expv := FLOOR(LOG(n3V));
  n3F := n3V / 10^expv;
  IF roundf THEN
    CASE
      IF n3F < 1.5 THEN nf := 1; END;
      IF n3F < 3 THEN nf := 2; END;
      IF n3F < 7 THEN nf := 5; END;
      DEFAULT nf := 10;
    END;
  ELSE
    CASE
      IF n3F <= 1 THEN nf := 1; END;
      IF n3F <= 2 THEN nf := 2; END;
      IF n3F <= 5 THEN nf := 5; END;
      DEFAULT nf := 10;
    END;
  END;
  RETURN nf * 10^expv;
END;

// ---- wait for the finger to lift, but never for more than a second ----
// MOUSE(1) is -1 while nothing touches the screen.  Without the time limit
// a stuck or phantom touch would hold the program here for ever, and the
// only way out would be a reset (Lessons Learned section 7; ChiTable's
// csDrain, device-verified).  TICKS is the millisecond clock.
n3_Drain()
BEGIN
  LOCAL drT;
  drT := TICKS;
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// ---- hold a page until any key OR a tap: drain, wait, drain ----
// The house page-hold (ChiTable's csWaitAny).  A GETKEY drain does not
// drain touch, so both are drained before and after the wait; GETKEY and
// MOUSE are polled together so a tap closes the page as well as a key.
n3_WaitAny()
BEGIN
  LOCAL waK, waM, waM1;
  REPEAT waK := GETKEY; UNTIL waK == -1;
  n3_Drain();
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  n3_Drain();
  REPEAT waK := GETKEY; UNTIL waK == -1;
END;

EXPORT Norm3(mu,sd,x₀)
BEGIN
  LOCAL LFT,RGT,TOPy,BOT;
  LOCAL W,H,N;
  LOCAL xmin,xmax,dx,ymax,dX;
  LOCAL n3I,n3J,n3X,n3Y;
  LOCAL pxx,pyy,ppx,ppy;
  LOCAL x0px,pdf0,cdf0,zz,n3Lo,n3Hi;
  LOCAL cSh,cCp,cCv,cAx,cX0;
  LOCAL cTx,cGrid,cMn,cGy,cOrg;
  LOCAL kk,showMode,lastcy,cy;
  LOCAL stp,ystp,nfrac,ynfrac;
  LOCAL tickx,ticky,xv,yv;
  LOCAL n3Pdf,n3Cdf,n3Lbl,n3X0;
  LOCAL n3M,n3M1,n3Ev,n3Act,n3Quit;
  LOCAL n3Drag,n3TX,n3TY,n3LX,n3LY;
  LOCAL n3PX,n3Mu0,n3Sh,n3P,n3Msg;
  LOCAL n3Re,n3CdfOK,n3Fit,n3YFit,n3Snap;
  LOCAL n3Anim,n3SdA,n3SdB,n3C,n3Off;
  LOCAL n3Zp,n3Za,n3Zs,n3K,n3Mx;

  IF sd <= 0 THEN RETURN "σ must be > 0"; END;
  // x₀ arrives as the third argument; the program works on an ASCII copy,
  // n3X0, so that no new non-ASCII name appears in the code (house rule:
  // no new Unicode in executable code; x₀ stays as the Home signature).
  n3X0 := x₀;

  // ---- constants (screen 320x240, origin top-left, y increases DOWN) ----
  // Rows ABOVE the plot: 1 = title (μ, σ), 11 = x₀ / z / pdf readout,
  // 22-34 = the two area keys.  So the plot starts at TOPy = 37.
  // Rows BELOW it: x tick labels at BOT+4 = 194, the z-ruler at
  // BOT+14 = 204 (ends by 214).  Nothing is drawn below 216: the soft-key
  // strip DRAWMENU paints is 218..239.
  // RGT is 302 (was 314) so that the right-hand CDF axis labels ("1",
  // ".5", "0") fit between it and the screen edge.
  // THE TOUCH HANDLER USES THESE SAME CONSTANTS: a finger at pixel column
  // px means x = xmin + (px - LFT)/W*(xmax - xmin), the drawing's own map.
  LFT:= 24; RGT:=302; TOPy:=37; BOT:=190;
  W:=RGT-LFT; H:=BOT-TOPy; N:=W;
  // palette (Lessons Learned section 13), set once, not inside loops
  cSh:=RGB(170,205,255); cCp:=RGB(255,235,235); cCv:=RGB(0,60,200);
  cAx:=RGB(0,0,0); cX0:=RGB(255,0,0); cTx:=RGB(200,0,200);
  cGrid:=RGB(225,228,235); cMn:=RGB(0,140,60); cGy:=RGB(90,90,90);
  cOrg:=RGB(255,69,0);
  showMode := 0;    // 0 = grid+CDF, 1 = grid, 2 = CDF, 3 = plain

  // ---- the state of the touch loop ----
  // n3Drag: what the finger holds -- 0 nothing, 1 x₀, 2 the S-curve
  //   (inverse), 3 the mean.  n3Sh: while the mean is held, how many
  //   pixel columns it has moved (the cache is read shifted by that).
  // n3Re = 1: the pdf cache must be refilled (μ, σ or the window moved).
  // n3CdfOK = 1: the CDF cache matches the pdf cache.
  // n3Fit = 1: re-centre the window on μ ± 4σ.  n3YFit = 1: refit the
  //   y-scale.  n3Anim: frames of the σ animation still to draw.
  // n3Msg = 1: row 2 shows the inverse "p = ... -> x₀ = ...".
  n3Drag := 0; n3Sh := 0; n3Msg := 0; n3Act := 0; n3Quit := 0;
  n3Anim := 0; n3SdA := sd; n3SdB := sd; n3P := 0.5;
  n3Fit := 1; n3YFit := 1; n3Re := 1; n3CdfOK := 0;
  n3LX := -1; n3LY := -1; n3PX := 0; n3Mu0 := mu;
  xmin := mu - 4*sd; xmax := mu + 4*sd; ymax := 1;
  pdf0 := 0; cdf0 := 0;

  // ---- caches, sized once: n3Pdf(k) and n3Cdf(k) hold the pdf and CDF
  //      at pixel column k (x = xmin + k*dx), k = 1..N.  Home's K is the
  //      MAKELIST counter, the device-verified form (ChiTable,
  //      NormExplorer); it overwrites K in Home, which this program does
  //      not use.  A LOCAL counter waits for GfxProbe T19. ----
  n3Pdf := MAKELIST(0, K, 1, N, 1);
  n3Cdf := MAKELIST(0, K, 1, N, 1);

  DIMGROB_P(G1,320,240);            // off-screen buffer, built once

  // ---- drain the key that closed the launch form, and any touch, so
  //      neither can satisfy the first wait below (section 7) ----
  REPEAT kk := GETKEY; UNTIL kk == -1;
  n3_Drain();

  REPEAT
    // ==== 1. the σ animation: one step per pass, no waiting ====
    // Frames 4, 3, 2, 1 to go move σ a quarter of the way each, with the
    // window AND the y-scale held, so the peak visibly falls and spreads
    // (or rises and narrows) while the area under it stays 1.  The last
    // frame (n3Anim = 1) refits the y-scale to the new peak.
    IF n3Anim > 0 THEN
      IF n3Anim > 1 THEN
        sd := n3SdB + (n3SdA - n3SdB)*(n3Anim - 2)/4;
        n3Re := 1;
      ELSE
        n3YFit := 1;
      END;
      n3Anim := n3Anim - 1;
    END;

    // ==== 2. the window: frozen, re-centred only on request ====
    // Lessons Learned section 6: a window that tracks μ keeps the bell
    // centred for ever and moving the mean looks like nothing happening.
    // So the window is set to μ ± 4σ on entry and on 0 (re-centre), and
    // otherwise held.  x₀ may still PUSH it: past an edge, the window
    // stretches to x₀ ± 0.5σ, as in v2.3.
    IF n3Fit == 1 THEN
      xmin := mu - 4*sd;
      xmax := mu + 4*sd;
      n3Fit := 0; n3YFit := 1; n3Re := 1;
    END;
    IF n3X0 < xmin THEN xmin := n3X0 - 0.5*sd; n3Re := 1; END;
    IF n3X0 > xmax THEN xmax := n3X0 + 0.5*sd; n3Re := 1; END;
    // y-range: the peak is known in closed form.  Headroom 1.45 (v2.3:
    // 1.15) so that one σ step DOWN (peak x 1.25) still fits while the
    // animation holds the scale.  Refitted after σ changes only -- μ does
    // not change a normal's peak height.
    IF n3YFit == 1 THEN
      ymax := NORMALD(mu, sd, mu)*1.45;
      n3YFit := 0;
    END;
    dx := (xmax - xmin)/N;
    dX := (xmax - xmin)/40;           // x₀ slide step for the +/- keys
    // the grid x₀ and μ snap to on lift: 1/100 of the window, 1-2-5
    n3Snap := n3_NiceNum((xmax - xmin)/100, 1);

    // ==== 3. the cache: refilled only when μ, σ or the window changed ====
    // Moving x₀ (keys, finger, S-curve) needs no refill at all: the pdf
    // and CDF at every column are the same; only the shading boundary
    // and the marker move.  The CDF is filled only in the views that
    // draw it.
    IF n3Re == 1 THEN
      FOR n3I FROM 1 TO N DO
        n3Pdf(n3I) := NORMALD(mu, sd, xmin + n3I*dx);
      END;
      n3Re := 0; n3CdfOK := 0;
    END;
    IF (showMode == 0 OR showMode == 2) AND n3CdfOK == 0 THEN
      FOR n3I FROM 1 TO N DO
        n3Cdf(n3I) := NORMALD_CDF(mu, sd, xmin + n3I*dx);
      END;
      n3CdfOK := 1;
    END;

    // ==== 4. the values at x₀ (two calls a frame) ====
    pdf0 := NORMALD(mu, sd, n3X0);
    cdf0 := NORMALD_CDF(mu, sd, n3X0);
    zz   := (n3X0 - mu)/sd;
    x0px := IP(LFT + (n3X0 - xmin)/(xmax - xmin)*W);

    RECT_P(G1,0,0,319,239,RGB(255,255,255));      // clear the buffer

    // ---- nice-number tick positions ----
    stp := n3_NiceNum((xmax - xmin)/6, 1);
    nfrac := MAX(0, -FLOOR(LOG(stp)));
    ystp := n3_NiceNum(ymax/4, 1);
    ynfrac := MAX(0, -FLOOR(LOG(ystp)));

    // ---- grid at nice values (behind everything) ----
    IF showMode == 0 OR showMode == 1 THEN
      xv := CEILING(xmin/stp)*stp;
      WHILE xv <= xmax + stp/1000 DO
        tickx := LFT + IP((xv - xmin)/(xmax - xmin)*W);
        IF tickx > LFT AND tickx <= RGT THEN
          LINE_P(G1, tickx, TOPy, tickx, BOT-1, cGrid);
        END;
        xv := xv + stp;
      END;
      yv := ystp;
      WHILE yv < ymax DO
        ticky := BOT - IP(yv/ymax*H);
        LINE_P(G1, LFT+1, ticky, RGT, ticky, cGrid);
        yv := yv + ystp;
      END;
    END;

    // ---- ONE pass over the pixel columns: shading, then the curve ----
    // Column k (screen x = LFT + k) is x = xmin + k*dx.  Left of x₀ the
    // column is shaded blue (the area P(X <= x₀)), right of it light red
    // (the complement P(X > x₀)); then the curve segment into this column
    // is drawn on top.  Each column's height comes from the cache.  While
    // the mean is held (n3Sh <> 0) the bell has slid by n3Sh whole
    // columns, so column k reads cache entry k - n3Sh; the few columns
    // the cache does not cover are computed directly.
    ppx := -1;
    FOR n3I FROM 1 TO N DO
      pxx := LFT + n3I;
      n3X := xmin + n3I*dx;
      n3J := n3I - n3Sh;
      IF n3J >= 1 AND n3J <= N THEN
        n3Y := n3Pdf(n3J);
      ELSE
        n3Y := NORMALD(mu, sd, n3X);
      END;
      pyy := IP(BOT - n3Y/ymax*H);
      IF pyy < TOPy THEN pyy := TOPy; END;
      IF n3X <= n3X0 THEN
        LINE_P(G1, pxx, BOT, pxx, pyy, cSh);
      ELSE
        LINE_P(G1, pxx, BOT, pxx, pyy, cCp);
      END;
      IF ppx >= 0 THEN LINE_P(G1, ppx, ppy, pxx, pyy, cCv); END;
      ppx := pxx; ppy := pyy;
    END;

    // ---- CDF S-curve (0..1 over the full plot height) and its axis ----
    IF showMode == 0 OR showMode == 2 THEN
      lastcy := -1;
      FOR n3I FROM 1 TO N DO
        n3J := n3I - n3Sh;
        IF n3J >= 1 AND n3J <= N THEN
          n3Y := n3Cdf(n3J);
        ELSE
          n3Y := NORMALD_CDF(mu, sd, xmin + n3I*dx);
        END;
        cy := IP(BOT - n3Y*H);
        pxx := LFT + n3I;
        IF lastcy >= 0 THEN LINE_P(G1, pxx-1, lastcy, pxx, cy, cTx); END;
        lastcy := cy;
      END;
      // the right-hand 0-1 axis for the S-curve: a magenta line at RGT
      // with ticks at 0, .5 and 1 (it replaces v2.3's lone "1")
      LINE_P(G1, RGT, TOPy, RGT, BOT, cTx);
      FOR n3I FROM 0 TO 2 DO
        ticky := BOT - IP(n3I*H/2);
        LINE_P(G1, RGT, ticky, RGT+3, ticky, cTx);
      END;
      TEXTOUT_P("1", G1, RGT+6, TOPy-4, 1, cTx);
      TEXTOUT_P(".5", G1, RGT+5, BOT-IP(H/2)-4, 1, cTx);
      TEXTOUT_P("0", G1, RGT+6, BOT-9, 1, cTx);
    END;

    // ---- mean marker: DOTTED vertical at μ (3 px wide while held) ----
    n3Mx := LFT + IP((mu - xmin)/(xmax - xmin)*W);
    IF n3Mx > LFT AND n3Mx < RGT THEN
      FOR n3I FROM TOPy TO BOT STEP 4 DO
        PIXON_P(G1, n3Mx, n3I, cMn);
        IF n3Drag == 3 THEN
          PIXON_P(G1, n3Mx-1, n3I, cMn);
          PIXON_P(G1, n3Mx+1, n3I, cMn);
          PIXON_P(G1, n3Mx, n3I+1, cMn);
        END;
      END;
    END;

    // ---- axes + labelled nice ticks ----
    LINE_P(G1, LFT, BOT, RGT, BOT, cAx);
    LINE_P(G1, LFT, BOT, LFT, TOPy, cAx);
    xv := CEILING(xmin/stp)*stp;
    WHILE xv <= xmax + stp/1000 DO
      tickx := LFT + IP((xv - xmin)/(xmax - xmin)*W);
      IF tickx >= LFT AND tickx <= RGT THEN
        LINE_P(G1, tickx, BOT, tickx, BOT+3, cAx);
        // BELOW the axis, centred under the tick: font 1 runs about 6 px a
        // character, so 3*SIZE(label) is half the label's width.
        n3Lbl := STRING(xv,2,nfrac);
        TEXTOUT_P(n3Lbl, G1, tickx - 3*SIZE(n3Lbl), BOT+4, 1, cGy);
      END;
      xv := xv + stp;
    END;
    yv := ystp;
    WHILE yv < ymax DO
      ticky := BOT - IP(yv/ymax*H);
      LINE_P(G1, LFT-3, ticky, LFT, ticky, cAx);
      TEXTOUT_P(STRING(yv,2,ynfrac), G1, 1, ticky-4, 1, cGy);
      yv := yv + ystp;
    END;

    // ---- z-ruler: z = -3..3 at μ + k*σ, under the x labels ----
    // The same column map as everything else, so a z label sits exactly
    // under the x value it standardises.  n3Zp is one σ in pixels: below
    // 14 px only z = -3, 0, 3 are labelled, below 5 px only z = 0.
    TEXTOUT_P("x", G1, 8, BOT+4, 1, cGy);
    TEXTOUT_P("z", G1, 8, BOT+14, 1, cGy);
    n3Zp := sd/(xmax - xmin)*W;
    n3Za := -3; n3Zs := 1;
    IF n3Zp < 14 THEN n3Zs := 3; END;
    IF n3Zp < 5 THEN n3Za := 0; END;
    FOR n3K FROM n3Za TO -n3Za STEP n3Zs DO
      tickx := LFT + IP((mu + n3K*sd - xmin)/(xmax - xmin)*W);
      IF tickx >= LFT AND tickx <= RGT THEN
        n3Lbl := STRING(n3K,2,0);
        TEXTOUT_P(n3Lbl, G1, tickx - 3*SIZE(n3Lbl), BOT+14, 1, cGy);
      END;
    END;

    // ---- x₀ marker (solid red vertical) + dotted line to the y-axis ----
    pyy  := IP(BOT - pdf0/ymax*H);
    IF pyy < TOPy THEN pyy := TOPy; END;
    LINE_P(G1, x0px, BOT, x0px, pyy, cX0);
    FOR n3I FROM LFT+1 TO x0px STEP 4 DO
      PIXON_P(G1, n3I, pyy, cX0);
    END;
    // ---- the magenta point riding the S-curve at x₀, with a dotted
    //      line across to the right-hand axis, where its height is read
    //      as P(X <= x₀) ----
    IF showMode == 0 OR showMode == 2 THEN
      cy := IP(BOT - cdf0*H);
      FOR n3I FROM x0px+4 TO RGT STEP 4 DO
        PIXON_P(G1, n3I, cy, cTx);
      END;
      RECT_P(G1, x0px-2, cy-2, x0px+2, cy+2, cTx);
    END;

    // ---- the rows above the plot ----
    // Row 1: the distribution; an orange hint when pressing 0 would move
    // the frozen window by more than a tenth of its width: μ or σ has
    // moved a long way, or x₀ pushed the window and has come back since.
    // (n3Lo..n3Hi is the window a re-centre would give: μ ± 4σ,
    // stretched for x₀ exactly as the fit in step 2 does.)
    TEXTOUT_P("Normal  μ = "+STRING(mu,2,3)+", σ = "+STRING(sd,2,3),
              G1, 4, 1, 1, cAx);
    n3Lo := mu - 4*sd; n3Hi := mu + 4*sd;
    IF n3X0 < n3Lo THEN n3Lo := n3X0 - 0.5*sd; END;
    IF n3X0 > n3Hi THEN n3Hi := n3X0 + 0.5*sd; END;
    n3Off := ABS(n3Lo - xmin) + ABS(n3Hi - xmax);
    IF n3Off > (xmax - xmin)/10 THEN
      TEXTOUT_P("0 = re-centre", G1, 236, 1, 1, cOrg);
    END;
    // Row 2: x₀, z and the pdf -- a HEIGHT at x₀, not a probability (it
    // passes 1 when σ is small).  After the S-curve, the inverse instead.
    IF n3Msg == 1 THEN
      TEXTOUT_P("p = "+STRING(n3P,2,3)+" "+CHAR(8594)+" x₀ = "+STRING(n3X0,2,2)+
                ", z = "+STRING(zz,2,2), G1, 4, 11, 1, cTx);
    ELSE
      TEXTOUT_P("x₀ = "+STRING(n3X0,2,2)+", z = "+STRING(zz,2,2)+
                ", pdf = "+n3_fmtp(pdf0,4)+" (a height)", G1, 4, 11, 1, cTx);
    END;
    // Row 3: the two areas, each on a key filled with its shading colour
    RECT_P(G1, 3, 22, 129, 34, cCv, cSh);
    TEXTOUT_P("P(X ≤ x₀) = "+n3_fmtp(cdf0,4), G1, 6, 24, 1, cAx);
    RECT_P(G1, 135, 22, 261, 34, cX0, cCp);
    TEXTOUT_P("P(X > x₀) = "+n3_fmtp(1-cdf0,4), G1, 138, 24, 1, cAx);

    BLIT_P(G0, 0, 0, G1);             // one flicker-free page flip
    // the soft keys, painted on the screen after the blit, every frame
    DRAWMENU({"μ−", "μ+", "σ−", "σ+", "View", "Quit"});

    // ==== 5. input: a key, a press, a drag sample, or a lift ====
    // n3Ev: 1 a press or a moved finger, 2 the finger lifted, 3 a key.
    // While something is held (n3Drag > 0) the loop waits until the
    // finger MOVES or lifts: MOUSE keeps reporting a finger that rests,
    // and redrawing for it would only burn time (NormExplorer beta 41).
    // A key during a hold ends the hold (the way out of a stuck touch).
    n3Ev := 0; kk := -1;
    IF n3Anim > 0 THEN
      n3Ev := 9;                      // the animation draws its next frame
    ELSE
      REPEAT
        kk := GETKEY;
        n3M := MOUSE;
        n3M1 := n3M(1);
        IF n3Drag > 0 THEN
          IF SIZE(n3M1) == 0 THEN
            n3Ev := 2;
          ELSE
            IF n3M1(1) <> n3LX OR n3M1(2) <> n3LY THEN n3Ev := 1; END;
          END;
        ELSE
          IF SIZE(n3M1) > 0 THEN n3Ev := 1; END;
        END;
        IF kk > -1 THEN n3Ev := 3; END;
      UNTIL n3Ev > 0;
    END;

    // ---- a press or a drag sample ----
    IF n3Ev == 1 THEN
      n3TX := n3M1(1); n3TY := n3M1(2);
      IF n3Drag == 0 THEN
        // A NEW PRESS: decide what it holds, once, from the first sample.
        IF n3TY >= 218 THEN
          // the soft-key strip: six 53-px slots; act once, on the lift
          n3_Drain();
          n3Act := MIN(6, IP(n3TX/53.4) + 1);
        ELSE
          IF n3TX >= LFT AND n3TX <= RGT AND n3TY >= TOPy AND n3TY <= BOT THEN
            n3PX := n3TX;
            // the mean line, if the press is within 6 px of it and
            // nearer it than the x₀ line (a tie goes to x₀)
            IF n3Mx > LFT AND n3Mx < RGT AND ABS(n3TX - n3Mx) <= 6
               AND ABS(n3TX - n3Mx) < ABS(n3TX - x0px) THEN
              n3Drag := 3; n3Mu0 := mu; n3Sh := 0;
            ELSE
              n3Drag := 1;
              // the S-curve, if it is drawn and the press is ON it (within
              // 16 px, vertically) and nearer it than the bell in that
              // column (the caches hold both).  Without the 16 px a press
              // low in the middle of the plot, where the S-curve crosses
              // half height, would be read as the inverse and x₀ would
              // jump to a quantile instead of to the finger.
              IF showMode == 0 OR showMode == 2 THEN
                n3J := MAX(1, MIN(N, n3TX - LFT));
                n3Y := ABS(n3TY - (BOT - n3Cdf(n3J)*H));
                IF n3Y <= 16 AND n3Y < ABS(n3TY - (BOT - n3Pdf(n3J)/ymax*H)) THEN
                  n3Drag := 2;
                END;
              END;
            END;
          ELSE
            // on the right-hand 0-1 axis: the inverse as well
            IF (showMode == 0 OR showMode == 2) AND n3TX > RGT
               AND n3TY >= TOPy AND n3TY <= BOT THEN
              n3Drag := 2;
            ELSE
              n3_Drain();             // outside the plot: nothing to do
            END;
          END;
        END;
      END;
      // APPLY THIS SAMPLE to what is held (the first sample too).  The
      // finger is clamped to the plot box, so a drag that wanders off the
      // plot holds x₀ at the edge instead of throwing it off screen.
      IF n3Drag > 0 THEN
        n3LX := n3TX; n3LY := n3TY;
        n3TX := MAX(LFT+1, MIN(RGT, n3TX));
        n3TY := MAX(TOPy, MIN(BOT, n3TY));
        IF n3Drag == 1 THEN
          // x₀ from the pixel column: the drawing's own map
          n3X0 := xmin + (n3TX - LFT)/W*(xmax - xmin);
          n3Msg := 0;
        END;
        IF n3Drag == 2 THEN
          // p from the height on the right-hand 0-1 axis.  One pixel is
          // 1/H = 0.0065 of p, so p is rounded to 0.01 (every 0.01 can be
          // reached; a 0.005 grid would skip 0.500), and 0.025 / 0.975,
          // the 95% central limits (z = -/+1.96), are caught within
          // 0.0035.  p is kept inside the part of the S-curve on screen,
          // so x₀ stays in the frozen window; then the inverse normal.
          n3P := (BOT - n3TY)/H;
          IF ABS(n3P - 0.975) < 0.0035 THEN
            n3P := 0.975;
          ELSE
            IF ABS(n3P - 0.025) < 0.0035 THEN
              n3P := 0.025;
            ELSE
              n3P := ROUND(n3P*100, 0)/100;
            END;
          END;
          n3P := MAX(n3P, MAX(0.005, n3Cdf(2)));
          n3P := MIN(n3P, MIN(0.995, n3Cdf(N-1)));
          IF n3P > 0 AND n3P < 1 THEN
            n3X0 := NORMALD_ICDF(mu, sd, n3P);
            n3Msg := 1;
          END;
        END;
        IF n3Drag == 3 THEN
          // the mean slides by whole pixel columns: the bell is the
          // cached one, n3Sh columns over
          n3Sh := n3TX - n3PX;
          mu := n3Mu0 + n3Sh*dx;
          n3Msg := 0;
        END;
      END;
    END;

    // ---- the finger lifted (or a key ended the hold) ----
    IF n3Ev == 2 OR (n3Ev == 3 AND n3Drag > 0) THEN
      IF n3Drag == 1 THEN
        // x₀ snaps to the grid, staying inside the window
        n3X0 := ROUND(n3X0/n3Snap, 0)*n3Snap;
        IF n3X0 > xmax THEN n3X0 := n3X0 - n3Snap; END;
        IF n3X0 < xmin THEN n3X0 := n3X0 + n3Snap; END;
      END;
      IF n3Drag == 3 THEN
        // μ snaps to the grid; the cache is refilled for the true μ
        mu := ROUND(mu/n3Snap, 0)*n3Snap;
        n3Sh := 0; n3Re := 1;
      END;
      n3Drag := 0;
      n3_Drain();
    END;

    // ---- keys: the hardware keys stay as shortcuts for the strip ----
    // n3Act: 1 μ-, 2 μ+, 3 σ-, 4 σ+, 5 view, 6 quit (the six soft keys),
    //        7 x₀-, 8 x₀+, 9 re-centre, 10 menu
    IF n3Ev == 3 THEN
      IF kk == 7  THEN n3Act := 1;  END;  // Left  : μ down
      IF kk == 8  THEN n3Act := 2;  END;  // Right : μ up
      IF kk == 12 THEN n3Act := 3;  END;  // Down  : σ down
      IF kk == 2  THEN n3Act := 4;  END;  // Up    : σ up
      IF kk == 4  THEN n3Act := 6;  END;  // Esc   : quit
      IF kk == 45 THEN n3Act := 7;  END;  // (-)   : x₀ down
      IF kk == 50 THEN n3Act := 8;  END;  // (+)   : x₀ up
      IF kk == 42 THEN n3Act := 9;  END;  // 0     : re-centre
      IF kk == 30 THEN n3Act := 10; END;  // Enter : menu
    END;

    // ---- the menu (Enter): Help, Next view, Re-centre, Quit ----
    IF n3Act == 10 THEN
      n3Act := 0;
      n3C := 0;
      CHOOSE(n3C, "Norm3", {"Help: touch and keys", "Next view",
                            "Re-centre the window", "Quit"});
      // the press that closed the menu can arrive again underneath
      REPEAT kk := GETKEY; UNTIL kk == -1;
      n3_Drain();
      IF n3C == 1 THEN n3_Help(); END;
      IF n3C == 2 THEN n3Act := 5; END;
      IF n3C == 3 THEN n3Act := 9; END;
      IF n3C == 4 THEN n3Act := 6; END;
    END;

    // ---- do it ----
    // ADAPTIVE STEPS: μ moves by σ/4 and σ scales by x1.25, so the controls
    // behave the same whether σ is 0.25 or 100.
    IF n3Act == 1 THEN mu := mu - sd/4; n3Re := 1; n3Msg := 0; END;
    IF n3Act == 2 THEN mu := mu + sd/4; n3Re := 1; n3Msg := 0; END;
    IF n3Act == 3 OR n3Act == 4 THEN
      // start the σ animation: from σ to the new σ over four frames
      n3SdA := sd;
      IF n3Act == 3 THEN n3SdB := MAX(0.0001, sd/1.25); ELSE n3SdB := sd*1.25; END;
      n3Anim := 5;
      n3Msg := 0;
    END;
    IF n3Act == 5 THEN showMode := (showMode+1) MOD 4; END;
    IF n3Act == 6 THEN n3Quit := 1; END;
    IF n3Act == 7 THEN n3X0 := n3X0 - dX; n3Msg := 0; END;
    IF n3Act == 8 THEN n3X0 := n3X0 + dX; n3Msg := 0; END;
    IF n3Act == 9 THEN n3Fit := 1; END;
    n3Act := 0;
  UNTIL n3Quit == 1;

  // ---- export the final state so it can be read in Home ----
  EP_NORM_MU  := mu;
  EP_NORM_SD  := sd;
  EP_NORM_X0  := n3X0;
  EP_NORM_Z   := (n3X0 - mu)/sd;
  EP_NORM_PDF := NORMALD(mu, sd, n3X0);
  EP_NORM_CDF := NORMALD_CDF(mu, sd, n3X0);

  RETURN "Norm3: N(" + STRING(mu,2,3) + ", " + STRING(sd,2,3) + "), P(X ≤ " +
         STRING(n3X0,2,2) + ") = " + n3_fmtp(EP_NORM_CDF,4) + "; values in EP_NORM_*";
END;
