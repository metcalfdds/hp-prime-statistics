// WHAT IT IS FOR (Roger, 31 Oct 2026, with his lecture note "Hypothesis
//   Testing: t-Test (with two population means)"): the t-tests a
//   criminal-justice statistics course teaches, worked the way the
//   lecture works them, in five steps, with a picture.
//   * Which t-test?  Figure 11.2's two questions as a tappable
//     flowchart: are the samples independent?  are the population
//     variances equal?  -> pooled-variances, separate-variances or
//     dependent-samples t.
//   * Independent samples from each group's N, mean and SD (the way a
//     problem states them), or from raw scores in D1 and D2 (Apps,
//     Statistics 1Var).  Any mix of equal or unequal variances and
//     equal or unequal N; the "Pooled or separate?" page sets the two
//     tests side by side and says when and why they differ.
//   * Dependent samples (matched pairs, before and after) from the
//     pairs in D1 and D2, or from the mean and SD of the differences.
//   * (beta 1k) One sample and proportions, in the same five steps: the
//     one-sample t-test (a sample's mean against mu0, the value H0
//     claims; from the mean, SD and N, or from scores in D1), the
//     one-proportion z-test and the two-proportion z-test (from counts).
//     The z-tests use the normal curve; step 2 checks that every count
//     is at least 5, the book's rule for the normal approximation.
//   * Two-tailed, or one-tailed either way; alpha from the t table's
//     columns.
//   * The five steps, a screen each: 1 the hypotheses (and the data),
//     2 the distribution and df, 3 the critical value and the decision
//     rule (a t curve with the rejection region shaded), 4 the standard
//     error and t obtained with the numbers put in, 5 the decision and
//     the conclusion (t obtained and the p value on the curve).
//   * Three examples: the lecture's two (age at arrest, pooled; matched
//     jails, dependent) and Roger's SPSS output (drug use by sex,
//     separate variances: t = -4.351 on 265.924 df, as SPSS).
//     Three more since beta 1k, each worked in the book: Student's sleep
//     data (Cushny and Peebles 1905; Student 1908: one-sample t(9) =
//     4.06), the grand juries of Castaneda v. Partida, 430 U.S. 482
//     (1977) (339 of 870 against p0 = 0.791: z = -29.1) and the Rossi
//     experiment (Rossi, Berk and Lenihan 1980: re-arrested, 48 of 216
//     with aid against 66 of 216 without: z = -1.965).
//   * Exact values and the printed table's: t critical is worked out
//     for the exact df (STUDENT_ICDF) and also read from the row the
//     printed table would use (df 73 -> row 60).  In the five cells
//     where the lecture's table differs from the exact value (its .001
//     column), step 3 shows both.  Where the lecture rounded along the
//     way, step 4 says so and gives the exact value.
//
// FROM THE LIBRARY (searched 31 Oct 2026: no Prime program in the
//   1,658-program corpus works a t-test in steps).  Taken from the
//   fleet: the key and finger drains, the D1/D2 reader and the number
//   formats (PowerViz); the checked STUDENT_ICDF with a bisection
//   fallback (CICatcher's cqTCrit); the two-tailed p capped at 1
//   (AnovaLab); the pairs message (RankLab); the Inference app as a
//   check (TestChooser).
//
// HOW THE NUMBERS ARE WORKED OUT (group 1 minus group 2, as the
//   lecture and SPSS do):
//   pooled:    sp = sqrt(((N1-1)s1^2 + (N2-1)s2^2)/(N1+N2-2))
//              SE = sp * sqrt((N1+N2)/(N1*N2)),  df = N1 + N2 - 2
//   separate:  SE = sqrt(s1^2/N1 + s2^2/N2)  (Welch; SPSS's "Equal
//              variances not assumed" row)
//              df = (a+b)^2 / (a^2/(N1-1) + b^2/(N2-1)), a = s1^2/N1,
//              b = s2^2/N2  (Welch-Satterthwaite; fractional)
//   dependent: xD = x1 - x2 for each pair, sD their SD (N-1),
//              SE = sD/sqrt(Npairs),  df = Npairs - 1
//   t obtained = (mean1 - mean2)/SE, or mean(xD)/SE
//   t critical = -STUDENT_ICDF(df, alpha) one-tailed, alpha/2 two-
//              tailed (a lower tail, mirrored: never ICDF(1 - alpha))
//   p: two-tailed 2*STUDENT_CDF(df, -|t|); H1 mu1 < mu2: STUDENT_CDF(
//      df, t); H1 mu1 > mu2: STUDENT_CDF(df, -t).  Lower tails only.
//   Cohen's d: (mean1 - mean2)/sp (independent; SPSS's Cohen's d);
//      mean(xD)/sD (dependent; SPSS's paired Cohen's d).
//   one sample (beta 1k): SE = s/sqrt(N), t obtained = (mean - mu0)/SE,
//      df = N - 1; Cohen's d = (mean - mu0)/s
//   one proportion: p-hat = x/n, SE = sqrt(p0(1 - p0)/n) (from p0, not
//      p-hat: the test assumes H0), z obtained = (p-hat - p0)/SE
//   two proportions: pooled p = (x1 + x2)/(n1 + n2),
//      SE = sqrt(p(1 - p)(1/n1 + 1/n2)), z obtained = (p-hat1 - p-hat2)/SE
//   z critical = -NORMALD_ICDF(0, 1, alpha) one-tailed, alpha/2 two-
//      tailed; p from NORMALD_CDF(0, 1, .), lower tails only, as for t
//   Squares are taken after dividing by the larger SD (Lessons Learned
//   s11, "scale before squaring"), so no SD over- or underflows.
//

// ---------------- palette ----------------
// ---- tt_Cm: commas in big numbers (round 5, 2 Nov 2026, night) ----
// Roger, 2 Nov 2026: "always put comma delimiters in big numbers".  STRING never
// groups digits, whatever Home's setting, so this inserts a comma every three
// digits in each run of four or more digits that does not follow a decimal
// point: "1234567.25" -> "1,234,567.25", "-98765" -> "-98,765"; "0.00012345",
// "1.8E10" and "1,000" stay as they are.  Only text for the screen comes here,
// never text the CAS or EXPR reads ("1,000" there is two numbers).
// Run on the Virtual Calculator: CommaProbe beta 1, 12 of 12 (2 Nov 2026).
tt_Cm(cmS)
BEGIN
  LOCAL cmO, cmI, cmJ, cmN, cmR, cmK;
  cmN := DIM(cmS);
  IF cmN < 4 THEN RETURN cmS; END;
  cmO := ""; cmI := 1;
  WHILE cmI <= cmN DO
    IF INSTRING("0123456789", MID(cmS, cmI, 1)) > 0 THEN
      // a run of digits from cmI to cmJ
      cmJ := cmI;
      WHILE cmJ < cmN AND INSTRING("0123456789", MID(cmS, cmJ + 1, 1)) > 0 DO cmJ := cmJ + 1; END;
      cmR := MID(cmS, cmI, cmJ - cmI + 1);
      IF DIM(cmR) >= 4 AND (cmI == 1 OR MID(cmS, MAX(1, cmI - 1), 1) <> ".") THEN
        // the first group takes 1 to 3 digits, then a comma before every three
        cmK := DIM(cmR) MOD 3;
        IF cmK == 0 THEN cmK := 3; END;
        cmO := cmO + LEFT(cmR, cmK);
        WHILE cmK < DIM(cmR) DO
          cmO := cmO + "," + MID(cmR, cmK + 1, 3);
          cmK := cmK + 3;
        END;
      ELSE
        cmO := cmO + cmR;
      END;
      cmI := cmJ + 1;
    ELSE
      cmO := cmO + MID(cmS, cmI, 1);
      cmI := cmI + 1;
    END;
  END;
  RETURN cmO;
END;

// ---------------- waiting ----------------
// Empty the key buffer: the Enter that closed a menu or form must not
// be read as the next keypress.
LOCAL tt_Drain()
BEGIN
 LOCAL dk;
 REPEAT dk := GETKEY; UNTIL dk == -1;
END;

// Wait for the finger to lift, never more than a second (a stuck touch
// reading must not freeze the program).  MOUSE(1) is -1 when nothing
// touches the screen.
LOCAL tt_TDrain()
BEGIN
 LOCAL drT;
 drT := TICKS;
 REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// After a form, a menu or a box: keys AND the finger.
LOCAL tt_Settle()
BEGIN
 tt_Drain();
 tt_TDrain();
END;

// Wait for any key or tap (drain, wait, drain).
LOCAL tt_WaitAny()
BEGIN
 LOCAL waK, waM, waM1;
 tt_Settle();
 REPEAT
  waK := GETKEY;
  waM := MOUSE;
  waM1 := waM(1);
 UNTIL waK > -1 OR SIZE(waM1) > 0;
 tt_Settle();
END;

// ---------------- numbers as text ----------------
// Fixed decimals, rounded first: STRING(x, 2, d) prints scientific
// form below 10^-d, and a rounded value is 0 or at least 10^-d.
LOCAL tt_Fmt(fv, fd)
BEGIN
 RETURN STRING(ROUND(fv,fd),2,fd);
END;

// A probability: "< 0.001" below the floor (a 0, which only an
// underflow of STUDENT_CDF can give, is below it too: a true p is never
// 0, Lessons Learned s3), "> 0.999" above it, "1.000" for exactly 1.
LOCAL tt_FmtP(pv, pd)
BEGIN
 LOCAL thr;
 thr := 10^(-pd);
 IF pv < thr THEN RETURN "< "+STRING(thr,2,pd); END;
 IF pv >= 1 THEN RETURN STRING(1,2,pd); END;
 IF ROUND(pv,pd) >= 1 THEN RETURN "> "+STRING(1-thr,2,pd); END;
 RETURN STRING(pv,2,pd);
END;

// " = 0.031", or " < 0.001" when the text already carries its relation
LOCAL tt_Rel(rs)
BEGIN
 IF DIM(rs) > 0 THEN
  IF LEFT(rs,1) == "<" OR LEFT(rs,1) == ">" THEN RETURN " "+rs; END;
 END;
 RETURN " = "+rs;
END;

// The fewest decimals (0-5) that hold a typed number exactly
LOCAL tt_NDec(nv)
BEGIN
 LOCAL nd, nk;
 nk := 5;
 FOR nd FROM 4 DOWNTO 0 DO
  IF ABS(nv*10^nd - ROUND(nv*10^nd,0)) < 0.000001 THEN nk := nd; END;
 END;
 RETURN nk;
END;

// The decimals a column of typed values needs: the most any of them
// needs, so the pairs print as the lecture's Table 11.3 does, 1.30 and
// 1.00 beside 1.25 (review 11)
LOCAL tt_ColDec(cl)
BEGIN
 LOCAL cq, cm;
 cm := 0;
 FOR cq FROM 1 TO SIZE(cl) DO cm := MAX(cm, tt_NDec(cl(cq))); END;
 RETURN cm;
END;

// A value with a given number of decimals (0: no point)
LOCAL tt_FixD(fv, fk)
BEGIN
 IF fk == 0 THEN RETURN STRING(ROUND(fv,0),1,0); END;
 RETURN STRING(ROUND(fv,fk),2,fk);
END;

// A number as it was typed: no trailing zeros, at most 5 decimals
// (1.45 stays 1.45, 44 stays 44, SPSS's 3.28798 stays 3.28798).
LOCAL tt_Num(nv)
BEGIN
 LOCAL nk, ns;
 nk := tt_NDec(nv);
 IF nk == 0 THEN RETURN STRING(ROUND(nv,0),1,0); END;
 ns := STRING(ROUND(nv,nk),2,nk);
 IF nk == 5 THEN
  // more than 5 decimals typed: round to 5, then drop end zeros
  WHILE MID(ns,DIM(ns),1) == "0" DO ns := LEFT(ns,DIM(ns)-1); END;
  IF MID(ns,DIM(ns),1) == "." THEN ns := LEFT(ns,DIM(ns)-1); END;
 END;
 RETURN ns;
END;

// A worked value to 4 significant digits, in fixed form (0.05499,
// 1.745, 0.0001058): intermediate steps show enough digits to see where
// a rounded hand calculation would part company.
LOCAL tt_Sig(sv)
BEGIN
 LOCAL sd;
 IF sv == 0 THEN RETURN "0"; END;
 sd := MIN(10, MAX(0, 3 - FLOOR(LOG(ABS(sv)))));
 RETURN STRING(ROUND(sv,sd),2,sd);
END;

// ---------------- the arithmetic ----------------
// The row of the printed t table the lecture would read.  The table's
// rows are 1-30, 40, 60 and 120 (the lecture's appendix 1; no
// infinity row).  The rule is the lecture's, "the value that is
// closest to it" (df 73 -> row 60).  A tie goes to the smaller row,
// which is the safer one (its t critical is larger): 35 -> 30, 4.5 -> 4.
// Past 120, row 120.
LOCAL tt_TabDF(tdf)
BEGIN
 LOCAL tdr;
 IF tdf <= 30 THEN RETURN MAX(1,CEILING(tdf-0.5)); END;
 IF tdf >= 120 THEN RETURN 120; END;
 tdr := 30;
 IF tdf - 30 > 5 THEN tdr := 40; END;      // nearer 40 than 30
 IF tdf - 40 > 10 THEN tdr := 60; END;     // nearer 60 than 40
 IF tdf - 60 > 30 THEN tdr := 120; END;    // nearer 120 than 60
 RETURN tdr;
END;

// t critical (its size) for df, at alpha, for the tails on screen: the
// point with a = alpha (one-tailed) or alpha/2 (two-tailed) beyond it.
// A lower-tail quantile, mirrored: -STUDENT_ICDF(df, a), never
// ICDF(1 - a) (Lessons Learned s3).  STUDENT_ICDF matched SciPy to 12
// digits on the emulator but is not yet on the device-verified list,
// so, as CICatcher's cqTCrit does: try it inside IFERR, check the
// answer with STUDENT_CDF (device-verified), and if either fails,
// bisect on STUDENT_CDF.
LOCAL tt_Crit(cdf)
BEGIN
 LOCAL ca, cv, cok, clo, chi, cmid;
 LOCAL cj;
 ca := IFTE(ttTL == 0,ttAL/2,ttAL);
 cok := 0; cv := 0;
 IFERR
  cv := -STUDENT_ICDF(cdf,ca);
  IF cv > 0 THEN
   IF ABS(STUDENT_CDF(cdf,-cv)/ca - 1) < 0.000001 THEN cok := 1; END;
  END;
 THEN
  cok := 0;
 END;
 IF cok THEN RETURN cv; END;
 // the fallback: bracket (double until the tail is below a), then 60
 // halvings, which leave less than 1E-12 of the bracket
 clo := 0; chi := 4;
 WHILE STUDENT_CDF(cdf,-chi) > ca AND chi < 1000000 DO
  clo := chi; chi := 2*chi;
 END;
 FOR cj FROM 1 TO 60 DO
  cmid := (clo+chi)/2;
  IF STUDENT_CDF(cdf,-cmid) > ca THEN clo := cmid; ELSE chi := cmid; END;
 END;
 RETURN (clo+chi)/2;
END;

// Is a t beyond the critical value cv?  (two-tailed: either side;
// H1 mu1 < mu2: below -cv; H1 mu1 > mu2: above cv)
LOCAL tt_Beyond(bt, bc)
BEGIN
 IF ttTL == 0 THEN RETURN IFTE(ABS(bt) > bc,1,0); END;
 IF ttTL < 0 THEN RETURN IFTE(bt < -bc,1,0); END;
 RETURN IFTE(bt > bc,1,0);
END;

// The p value for t on df, for the tails on screen (lower tails only).
LOCAL tt_PVal(pt, pdf)
BEGIN
 // two-tailed: capped at 1, as AnovaLab's (t = 0 gives 2 x 0.5)
 IF ttTL == 0 THEN RETURN MIN(1,2*STUDENT_CDF(pdf,-ABS(pt))); END;
 IF ttTL < 0 THEN RETURN STUDENT_CDF(pdf,pt); END;
 RETURN STUDENT_CDF(pdf,-pt);
END;

// ---------------- z-tests and the tests' symbols (beta 1k) ----------------
// TTestLab's first three tests are t-tests for two means.  Since beta 1k
// it also works a one-sample t-test (ttTY = 4) and two z-tests for
// proportions (5: one proportion, 6: two).  A z-test uses the standard
// normal curve, which has no df.

// The statistic's letter for the test on screen: "z" for a proportion
// test, "t" for the rest
LOCAL tt_SL()
BEGIN
 IF ttTY >= 5 THEN RETURN "z"; END;
 RETURN "t";
END;

// The height at dvx of the curve the test uses: the standard normal
// curve for a z-test, Student's t on ttDF for a t-test
LOCAL tt_Dens(dvx)
BEGIN
 IF ttTY >= 5 THEN RETURN NORMALD(0,1,dvx); END;
 RETURN STUDENT(ttDF,dvx);
END;

// z critical (its size) at alpha for the tails on screen: the point of
// the standard normal curve with a = alpha (one-tailed) or alpha/2
// (two-tailed) beyond it.  A lower-tail quantile, mirrored, as tt_Crit's
// (Lessons Learned s3: never ICDF(1 - a)).
LOCAL tt_ZCrit()
BEGIN
 RETURN -NORMALD_ICDF(0,1,IFTE(ttTL == 0,ttAL/2,ttAL));
END;

// The p value for a z, for the tails on screen (lower tails only; the
// two-tailed p is capped at 1, as tt_PVal's)
LOCAL tt_ZP(zpv)
BEGIN
 IF ttTL == 0 THEN RETURN MIN(1,2*NORMALD_CDF(0,1,-ABS(zpv))); END;
 IF ttTL < 0 THEN RETURN NORMALD_CDF(0,1,zpv); END;
 RETURN NORMALD_CDF(0,1,-zpv);
END;

// The two sides of the hypotheses in symbols, for the test on screen:
// mu1 and mu2 (two means), mu and mu0 (one mean), p and p0 (one
// proportion), p1 and p2 (two proportions)
LOCAL tt_HS()
BEGIN
 IF ttTY == 4 THEN RETURN {"μ","μ"+CHAR(8320)}; END;
 IF ttTY == 5 THEN RETURN {"p","p"+CHAR(8320)}; END;
 IF ttTY == 6 THEN RETURN {"p"+CHAR(8321),"p"+CHAR(8322)}; END;
 RETURN {"μ"+CHAR(8321),"μ"+CHAR(8322)};
END;

// Cohen's d as the pages print it after "Cohen's d": " = 1.28", or, past
// 999.99 in size, " > 999.99" or " < -999.99".  A d that large comes from
// an SD typed far too small, and its digits would run into the label
// beside it (the label monkey, 14 Nov 2026: "Cohen's d = -34,200,918,..."
// over "magenta area = p", on the older tests' step 5 as well).
LOCAL tt_DS()
BEGIN
 IF ttD >= 1000 THEN RETURN " > 999.99"; END;
 IF ttD <= -1000 THEN RETURN " < "+CHAR(8722)+"999.99"; END;
 RETURN " = "+tt_Cm(tt_Fmt(ttD,2));
END;

// Pooled (pk = 1) or separate (pk = 2) t for the independent samples on
// screen: returns {SE, t, df, pooled SD}.  The SDs are divided by the
// larger first (cm), so nothing squares out of range.
LOCAL tt_Ind(pk)
BEGIN
 LOCAL cm, ca, cb, cv, csp, cse;
 LOCAL cdf;
 cm := MAX(ttS1,ttS2);
 ca := (ttS1/cm)^2;
 cb := (ttS2/cm)^2;
 csp := cm*(((ttN1-1)*ca + (ttN2-1)*cb)/(ttN1+ttN2-2))^.5;
 IF pk == 1 THEN
  cse := csp*((ttN1+ttN2)/(ttN1*ttN2))^.5;
  cdf := ttN1+ttN2-2;
 ELSE
  cv := ca/ttN1 + cb/ttN2;
  cse := cm*cv^.5;
  cdf := cv^2/((ca/ttN1)^2/(ttN1-1) + (cb/ttN2)^2/(ttN2-1));
 END;
 RETURN {cse, (ttM1-ttM2)/cse, cdf, csp};
END;

// Mean, SD and sum of squared deviations of the pairs' differences,
// from ttX1 and ttX2 (two passes: the mean first).
LOCAL tt_Pairs()
BEGIN
 LOCAL pj, ps;
 ttNP := SIZE(ttX1);
 ttXD := ttX1;
 ps := 0;
 FOR pj FROM 1 TO ttNP DO
  ttXD(pj) := ttX1(pj) - ttX2(pj);
  ps := ps + ttXD(pj);
 END;
 ttMD := ps/ttNP;
 ttSSq := 0;
 FOR pj FROM 1 TO ttNP DO ttSSq := ttSSq + (ttXD(pj)-ttMD)^2; END;
 ttSDv := (ttSSq/(ttNP-1))^.5;
END;

// The lecture's printed t table (its appendix 1) differs from the exact
// values, rounded to 3 decimals, in five cells, all in the last column
// (one-tailed .0005, two-tailed .001): rows 2, 3, 5, 7 and 23 print
// 31.598, 12.941, 6.859, 5.405 and 3.767 where the exact values are
// 31.599, 12.924, 6.869, 5.408 and 3.768 (SciPy; every other cell
// agrees, verify A6).  Returns the printed value when the row and
// column on screen are one of those five, else 0.  TTestLab shows the
// printed value as the table's (a student reads it there) and step 3
// gives the exact one beside it.
LOCAL tt_BookCell()
BEGIN
 LOCAL bk, bkr, bkv;
 IF ttAL <> IFTE(ttTL == 0,0.001,0.0005) THEN RETURN 0; END;
 bkr := {2,3,5,7,23};
 bkv := {31.598,12.941,6.859,5.405,3.767};
 FOR bk FROM 1 TO 5 DO
  IF ttDFT == bkr(bk) THEN RETURN bkv(bk); END;
 END;
 RETURN 0;
END;

// Too large to work with?  The program squares the numbers it is given
// (an SD, a difference from the mean) and step 4 squares s^2/N again
// for Welch's df.  On the Prime a square past 1E499 is infinity, with
// no error (Lessons Learned s3), and the screens would fill with it.
// So every number taken in, typed or read from D1/D2, must be smaller
// than 1E50 in size: 50 digits before the point, far beyond any data a
// course meets, and its fourth power still fits (beta 1c, from the
// monkey test).  An infinity is refused too.
LOCAL tt_Big(bv)
BEGIN
 RETURN IFTE(ABS(bv) >= 1E50,1,0);
END;

// The position of the first value in a list that is too large
// (tt_Big), or 0 when every value is fine.
LOCAL tt_BigL(bl)
BEGIN
 LOCAL bi;
 FOR bi FROM 1 TO SIZE(bl) DO
  IF tt_Big(bl(bi)) THEN RETURN bi; END;
 END;
 RETURN 0;
END;

// The sentence for a list read from the Statistics 1Var app that holds
// a value too large: which list, and which value in it.
LOCAL tt_BigLMsg(gl1, gl2)
BEGIN
 LOCAL gk, gw;
 gk := tt_BigL(gl1); gw := "D1";
 IF gk == 0 THEN gk := tt_BigL(gl2); gw := "D2"; END;
 RETURN "Value "+tt_Cm(STRING(gk,1,0))+" in "+gw+" has more than 50 digits: too large to work with. Check "+gw+" in the Statistics 1Var app.";
END;

// The sentence for a typed N over 1,000,000.  The Prime's STUDENT_CDF
// loses its digits at a huge df (Virtual Calculator 2.4.2, 1 Nov 2026,
// at t = -1.96: df 1E6 gave 0.024998033593, right to 8 digits; df 1E10
// 0.02498, where the value is 0.024998; df 1E13 0.014; df 1E15 and past
// 0.5).  N of at most 1,000,000 keeps every df below 2,000,000, where
// it is right, and t and z there agree to 5 digits anyway.  Only a
// typed N is checked: an N from D1/D2 counts the values in a list.
LOCAL tt_NMsg()
BEGIN
 RETURN "N can be at most 10,000. Choose Edit the numbers to change it.";
END;

// ---------------- one sample and proportions: the arithmetic (beta 1k) ----------------
// Mean and SD (N - 1) of the one sample's scores in ttXO: two passes, the
// mean first, then the squared distances from it (Lessons Learned s11:
// never from the sum of squares, which loses its digits)
LOCAL tt_One()
BEGIN
 LOCAL oj, oq;
 ttNO := SIZE(ttXO);
 oq := 0;
 FOR oj FROM 1 TO ttNO DO oq := oq + ttXO(oj); END;
 ttMO := oq/ttNO;
 oq := 0;
 FOR oj FROM 1 TO ttNO DO oq := oq + (ttXO(oj)-ttMO)^2; END;
 ttSO := (oq/(ttNO-1))^.5;
END;

// Is wv a whole number from wlo to whi?  (a count, or a sample size)
LOCAL tt_Whole(wv, wlo, whi)
BEGIN
 IF wv < wlo OR wv > whi THEN RETURN 0; END;
 IF FP(wv) <> 0 THEN RETURN 0; END;
 RETURN 1;
END;

// The one-sample t-test and the two z-tests: check the numbers on screen
// and work out what the five steps show.  Returns 1, or 0 with ttOK = 0
// and the reason in ttMSG when the numbers give no test.  The z-tests
// are finished here; the one-sample t goes back to tt_Calc for its t
// critical and p, as the other t-tests.
LOCAL tt_CalcN()
BEGIN
 IF ttTY == 4 THEN
  // ---- one sample: t = (mean - mu0)/(s/sqrt(N)) on N - 1 df ----
  // mu0 is always typed.  Typed numbers keep to the fleet's ranges;
  // scores from D1 were checked when the list was read.
  IF tt_Big(ttMU) THEN
   ttOK := 0; ttMSG := "A number you typed has more than 50 digits: too large to work with. Choose Edit the numbers to change it."; RETURN 0;
  END;
  IF ttSRC == 1 THEN
   IF tt_Big(ttMO) OR tt_Big(ttSO) OR tt_Big(ttNO) THEN
    ttOK := 0; ttMSG := "A number you typed has more than 50 digits: too large to work with. Choose Edit the numbers to change it."; RETURN 0;
   END;
   IF ttNO > 10000 THEN ttOK := 0; ttMSG := tt_NMsg(); RETURN 0; END;
   IF MAX(ABS(ttMO), ABS(ttMU)) > 10000 OR ttSO > 1000 THEN ttOK := 0; ttMSG := "The mean and μ"+CHAR(8320)+" must be from -10,000 to 10,000 and the SD at most 1,000. Choose Edit the numbers to change it."; RETURN 0; END;
  ELSE
   ttNO := SIZE(ttXO);
  END;
  IF ttNO < 2 THEN ttOK := 0; ttMSG := "A one-sample t-test needs N of at least 2."; RETURN 0; END;
  IF ttSRC == 2 THEN tt_One(); END;
  IF ttSO < 0 THEN ttOK := 0; ttMSG := "An SD cannot be negative."; RETURN 0; END;
  IF ttSO == 0 THEN ttOK := 0; ttMSG := "The SD is 0: every score is the same, so there is no t."; RETURN 0; END;
  ttSE := ttSO/ttNO^.5;
  ttDF := ttNO-1;
  ttT := (ttMO-ttMU)/ttSE;
  ttD := (ttMO-ttMU)/ttSO;
  ttSP := ttSO;
  RETURN 1;
 END;
 IF ttTY == 5 THEN
  // ---- one proportion: z = (p-hat - p0)/sqrt(p0(1 - p0)/n) ----
  // the SE is built from p0, not from p-hat: the test works out what
  // would happen if H0 were true
  IF tt_Whole(ttNA,1,1000000) == 0 THEN ttOK := 0; ttMSG := "n must be a whole number from 1 to 1,000,000. Choose Edit the numbers to change it."; RETURN 0; END;
  IF tt_Whole(ttKA,0,ttNA) == 0 THEN ttOK := 0; ttMSG := "x, the number of successes, must be a whole number from 0 to n. Choose Edit the numbers to change it."; RETURN 0; END;
  IF ttP0 < 0.00001 OR ttP0 > 0.99999 THEN ttOK := 0; ttMSG := "p"+CHAR(8320)+" must be from 0.00001 to 0.99999. Choose Edit the numbers to change it."; RETURN 0; END;
  ttPA := ttKA/ttNA;
  ttSE := (ttP0*(1-ttP0)/ttNA)^.5;
  ttT := (ttPA-ttP0)/ttSE;
  ttD := ttPA-ttP0;
 ELSE
  // ---- two proportions: z = (p-hat1 - p-hat2)/SE ----
  // under H0 the two samples share one proportion, so the SE is built
  // from the pooled p = (x1 + x2)/(n1 + n2)
  IF tt_Whole(ttNA,1,1000000) == 0 OR tt_Whole(ttNB,1,1000000) == 0 THEN ttOK := 0; ttMSG := "Each n must be a whole number from 1 to 1,000,000. Choose Edit the numbers to change it."; RETURN 0; END;
  IF tt_Whole(ttKA,0,ttNA) == 0 OR tt_Whole(ttKB,0,ttNB) == 0 THEN ttOK := 0; ttMSG := "Each x, a number of successes, must be a whole number from 0 to its n. Choose Edit the numbers to change it."; RETURN 0; END;
  IF ttKA+ttKB == 0 OR ttKA+ttKB == ttNA+ttNB THEN ttOK := 0; ttMSG := "Every case is a success, or none is: the two samples cannot differ, so there is no z."; RETURN 0; END;
  ttPA := ttKA/ttNA; ttPB := ttKB/ttNB;
  ttPP := (ttKA+ttKB)/(ttNA+ttNB);
  ttSE := (ttPP*(1-ttPP)*(1/ttNA+1/ttNB))^.5;
  ttT := (ttPA-ttPB)/ttSE;
  ttD := ttPA-ttPB;
 END;
 // a z-test: no df; z critical, p and the decision from the normal curve
 ttDF := 0;
 ttTC := tt_ZCrit();
 ttP := tt_ZP(ttT);
 ttREJ := tt_Beyond(ttT,ttTC);
 RETURN 1;
END;

// Everything the five steps show, from the numbers on screen.  Sets
// ttOK = 0 and ttMSG when the numbers cannot give a t.
LOCAL tt_Calc()
BEGIN
 LOCAL cr, cbk;
 ttOK := 1; ttMSG := "";
 // beta 1k: the one-sample t-test and the z-tests check and work out
 // their own numbers (tt_CalcN).  The z-tests are finished there; the
 // one-sample t goes on to t critical and p below.
 IF ttTY >= 4 THEN
  IF tt_CalcN() == 0 THEN RETURN 0; END;
  IF ttTY >= 5 THEN RETURN 1; END;
 END;
 IF ttTY <= 3 THEN
 // A typed number too large to work with (tt_Big).  Only typed ones:
 // numbers worked out from D1/D2 come from values already checked
 // when the lists were read, and an SD worked out from values below
 // 1E50 can lie a little above it.
 IF ttSRC == 1 THEN
  IF ttTY == 3 THEN
   IF tt_Big(ttMD) OR tt_Big(ttSDv) OR tt_Big(ttNP) THEN
    ttOK := 0; ttMSG := "A number you typed has more than 50 digits: too large to work with. Choose Edit the numbers to change it."; RETURN 0;
   END;
   IF ttNP > 10000 THEN ttOK := 0; ttMSG := tt_NMsg(); RETURN 0; END;
   // Round 10: the fleet's limits for a typed mean and SD
   IF ABS(ttMD) > 10000 OR ttSDv > 1000 THEN ttOK := 0; ttMSG := "The mean must be from -10,000 to 10,000 and the SD at most 1,000. Choose Edit the numbers to change it."; RETURN 0; END;
  ELSE
   IF tt_Big(ttM1) OR tt_Big(ttS1) OR tt_Big(ttN1) OR tt_Big(ttM2) OR tt_Big(ttS2) OR tt_Big(ttN2) THEN
    ttOK := 0; ttMSG := "A number you typed has more than 50 digits: too large to work with. Choose Edit the numbers to change it."; RETURN 0;
   END;
   IF ttN1 > 10000 OR ttN2 > 10000 THEN ttOK := 0; ttMSG := tt_NMsg(); RETURN 0; END;
   IF MAX(ABS(ttM1), ABS(ttM2)) > 10000 OR MAX(ttS1, ttS2) > 1000 THEN ttOK := 0; ttMSG := "A mean must be from -10,000 to 10,000 and an SD at most 1,000. Choose Edit the numbers to change it."; RETURN 0; END;
  END;
 END;
 IF ttTY == 3 THEN
  // the guard before the SD of the differences is formed (it divides
  // by N - 1): review 10
  IF ttSRC == 2 THEN ttNP := SIZE(ttX1); END;
  IF ttNP < 2 THEN ttOK := 0; ttMSG := "A dependent-samples t-test needs at least 2 pairs."; RETURN 0; END;
  IF ttSRC == 2 THEN tt_Pairs(); END;
  IF ttSDv < 0 THEN ttOK := 0; ttMSG := "An SD cannot be negative."; RETURN 0; END;
  IF ttSDv == 0 THEN ttOK := 0; ttMSG := "Every pair differs by the same amount (sD = 0): there is no t."; RETURN 0; END;
  ttSE := ttSDv/ttNP^.5;
  ttDF := ttNP-1;
  ttT := ttMD/ttSE;
  ttD := ttMD/ttSDv;
  ttSP := ttSDv;
 ELSE
  IF ttN1 < 2 OR ttN2 < 2 THEN ttOK := 0; ttMSG := "Each group needs N of at least 2."; RETURN 0; END;
  IF ttS1 < 0 OR ttS2 < 0 THEN ttOK := 0; ttMSG := "An SD cannot be negative."; RETURN 0; END;
  IF ttS1 == 0 AND ttS2 == 0 THEN ttOK := 0; ttMSG := "Both SDs are 0: every score is the same, so there is no t."; RETURN 0; END;
  cr := tt_Ind(ttTY);
  ttSE := cr(1); ttT := cr(2); ttDF := cr(3); ttSP := cr(4);
  ttD := (ttM1-ttM2)/ttSP;
 END;
 END;
 ttTC := tt_Crit(ttDF);
 ttDFT := tt_TabDF(ttDF);
 // the printed table's value, as a student would read it: the exact
 // value to 3 decimals, or the lecture's own figure in its five odd
 // cells (tt_BookCell)
 cbk := tt_BookCell();
 ttTB := IFTE(cbk > 0,cbk,ROUND(tt_Crit(ttDFT),3));
 ttP := tt_PVal(ttT,ttDF);
 ttREJ := tt_Beyond(ttT,ttTC);
 ttREJB := tt_Beyond(ttT,ttTB);
 RETURN 1;
END;

// The t curve on df from y0 to y1: the rejection region red, the rest
// pale blue; t critical dashed red; with shT, t obtained in magenta and
// with shP the p value's area shaded magenta.  One pixel column at a
// time (Lessons Learned s6: shade by columns, never by samples).
// beta 1k: for a z-test the curve is the standard normal curve (tt_Dens)
// and the labels say z (tt_SL).
LOCAL tt_Curve(cy0, cy1, shT, shP)
BEGIN
 LOCAL cxr, cpk, cH, cpx, cx, cf;
 LOCAL cyp, cyq, cin, cpin, ctk, ctx;
 LOCAL cdec, cstp, cs;
 // the window: -4..4, wider when t critical or t obtained lies past 3.2
 cxr := MAX(4, ttTC + 0.8);
 IF shT THEN cxr := MAX(cxr, ABS(ttT) + 0.8); END;
 cxr := MIN(cxr, 12);
 cpk := tt_Dens(0);
 cH := cy1 - cy0 - 4;
 cyq := cy1;
 FOR cpx FROM ttPL TO ttPR DO
  cx := -cxr + (cpx-ttPL)/(ttPR-ttPL)*2*cxr;
  cf := tt_Dens(cx);
  cyp := cy1 - IP(cf/cpk*cH);
  cin := tt_Beyond(cx,ttTC);
  // p value's area: beyond t obtained, in the direction(s) of H1
  cpin := 0;
  IF shP THEN
   IF ttTL == 0 THEN cpin := IFTE(ABS(cx) >= ABS(ttT),1,0); END;
   IF ttTL < 0 THEN cpin := IFTE(cx <= ttT,1,0); END;
   IF ttTL > 0 THEN cpin := IFTE(cx >= ttT,1,0); END;
  END;
  IF cyp < cy1 THEN
   IF cpin THEN
    LINE_P(G1,cpx,cyp,cpx,cy1,ttLMAG);
   ELSE
    LINE_P(G1,cpx,cyp,cpx,cy1,IFTE(cin,ttLRED,ttLBLU));
   END;
  END;
  // the curve itself, joined to the column before
  IF cpx > ttPL THEN LINE_P(G1,cpx-1,cyq,cpx,cyp,ttBLUE); END;
  cyq := cyp;
 END;
 // the axis, with whole-number ticks (every 2 when the window is wide)
 LINE_P(G1,ttPL,cy1,ttPR,cy1,ttBLK);
 cstp := IFTE(cxr > 6,2,1);
 FOR ctk FROM -FLOOR(cxr/cstp)*cstp TO FLOOR(cxr/cstp)*cstp STEP cstp DO
  ctx := ttPL + IP((ctk+cxr)/(2*cxr)*(ttPR-ttPL));
  LINE_P(G1,ctx,cy1,ctx,cy1+3,ttBLK);
  cs := tt_Cm(STRING(ctk,1,0));
  TEXTOUT_P(cs,G1,ctx-3*DIM(cs),cy1+4,1,ttGRAY);
 END;

LOCAL tt_Step1N()
BEGIN

 tt_Frame("Step 1. The hypotheses");

 ss := " ≠ ";
 IF ttTL < 0 THEN ss := " < "; END;
 IF ttTL > 0 THEN ss := " > "; END;
 ss := "H"+CHAR(8321)+": "+sh(1)+ss+sh(2);
 IF ttTL == 0 THEN
  IF ttTY == 6 THEN
   TEXTOUT_P(ss+"   (they differ, either way)",G1,8,60,1,ttMAG);

  TEXTOUT_P(ss+"   (a predicted direction)",G1,8,60,1,ttMAG);

 IF ttTY == 4 AND ttSRC == 1 THEN
  // one sample, from its typed mean, SD and N
  RECT_P(G1,4,76,315,134,ttORNG,ttLORN);
  TEXTOUT_P("The sample:",G1,10,80,1,ttGRAY);
  TEXTOUT_P(CHAR(57344)+" = "+tt_Cm(tt_Num(ttMO))+",   s = "+tt_Cm(tt_Num(ttSO))+",   N = "+tt_Cm(tt_Num(ttNO)),G1,10,95,1,ttBLK);

  IF DIM(ttDV) > 0 THEN TEXTOUT_P("Outcome: "+ttDV,G1,8,144,1,ttGRAY); END;

  // one sample, from the scores in D1: the first of them in two rows,
  // as many to a row as their widest text allows

  FOR sj FROM 1 TO MIN(ttNO,10) DO sk := MAX(sk, tt_NDec(ttXO(sj))); END;

// ---- step 2: the distribution; df, or the check of the counts ----
LOCAL tt_Step2N()
BEGIN
 LOCAL sa, sb, sok;
 IF ttTY == 4 THEN
  tt_Frame("Step 2. Distribution and df");
  TEXTOUT_P("The distribution is t: its shape depends on df.",G1,8,32,1,ttBLK);
  TEXTOUT_P("One sample:  df = (N - 1)",G1,8,54,1,ttBLUE);
  TEXTOUT_P("   = "+tt_Cm(tt_Num(ttNO))+" - 1 = "+tt_Cm(STRING(ttDF,1,0)),G1,8,69,1,ttBLUE);

  TEXTOUT_P("from it always add to zero.",G1,8,103,1,ttGRAY);
  TEXTOUT_P("The more df, the closer t comes to the normal",G1,8,160,1,ttGRAY);
  TEXTOUT_P("curve; few df give a wider, flatter t.",G1,8,173,1,ttGRAY);
 ELSE
  tt_Frame("Step 2. The distribution");
  TEXTOUT_P("The distribution is z, the standard normal curve.",G1,8,32,1,ttBLK);
  TEXTOUT_P("A z-test has no df: the curve never changes.",G1,8,47,1,ttBLUE);
  // the normal curve stands in for the binomial distribution, and fits
  // it well only when the counts are large enough: the book's rule
  TEXTOUT_P("The curve is an approximation.  It fits well when",G1,8,68,1,ttBLK);
  IF ttTY == 5 THEN
   sa := ttNA*ttP0; sb := ttNA*(1-ttP0);
   TEXTOUT_P("n p"+CHAR(8320)+" and n(1 - p"+CHAR(8320)+") are both at least 5:",G1,8,81,1,ttBLK);
   TEXTOUT_P("n p"+CHAR(8320)+" = "+tt_Cm(tt_Num(ttNA))+" × "+tt_Cm(tt_Num(ttP0))+" = "+tt_Cm(tt_Sig(sa)),G1,8,98,1,ttBLUE);
   TEXTOUT_P("n(1 - p"+CHAR(8320)+") = "+tt_Cm(tt_Num(ttNA))+" × "+tt_Cm(tt_Num(1-ttP0))+" = "+tt_Cm(tt_Sig(sb)),G1,8,111,1,ttBLUE);
   sok := 0;

   TEXTOUT_P("each sample has at least 5 successes and 5 failures:",G1,8,81,1,ttBLK);

// ---------------- pooled or separate? ----------------
LOCAL tt_Compare()
BEGIN
 LOCAL cp, cs, cpp, csp, ccp, ccs;
 LOCAL cw, ctl;
 // beta 1k: the one-sample t-test and the z-tests have no such choice
 IF ttTY >= 4 THEN
  MSGBOX("Pooled or separate variances is a question for a test of TWO independent means. The test on screen has no such choice.");
  tt_Settle();
  RETURN 0;
 END;
 IF ttTY == 3 THEN
  MSGBOX("Pooled or separate variances is a question for INDEPENDENT samples. Dependent samples have one SD, of the differences.");
  tt_Settle();
  RETURN 0;
 END;
 // numbers that gave no t (N = 1, both SDs 0) would divide by zero
 // here (review 3)
 IF ttOK == 0 THEN
  MSGBOX("The numbers on screen give no t: Edit the numbers first.");
  tt_Settle();
  RETURN 0;
 END;
 cp := tt_Ind(1); cs := tt_Ind(2);
 cpp := tt_PVal(cp(2),cp(3)); csp := tt_PVal(cs(2),cs(3));
 ccp := tt_Crit(cp(3)); ccs := tt_Crit(cs(3));
 tt_Frame("Pooled or separate variances?");
 TEXTOUT_P("Pooled",G1,120,30,1,ttBLUE);
 TEXTOUT_P("Separate",G1,215,30,1,ttMAG);
 TEXTOUT_P("SE",G1,10,44,1,ttGRAY);
 TEXTOUT_P(tt_Cm(tt_Sig(cp(1))),G1,120,44,1,ttBLK);
 TEXTOUT_P(tt_Cm(tt_Sig(cs(1))),G1,215,44,1,ttBLK);
 TEXTOUT_P("t obtained",G1,10,57,1,ttGRAY);
 TEXTOUT_P(tt_Cm(tt_Fmt(cp(2),3)),G1,120,57,1,ttBLK);
 TEXTOUT_P(tt_Cm(tt_Fmt(cs(2),3)),G1,215,57,1,ttBLK);
 TEXTOUT_P("df",G1,10,70,1,ttGRAY);
 TEXTOUT_P(tt_Cm(STRING(cp(3),1,0)),G1,120,70,1,ttBLK);
 TEXTOUT_P(tt_Cm(tt_Fmt(cs(3),2)),G1,215,70,1,ttBLK);
 TEXTOUT_P("t crit",G1,10,83,1,ttGRAY);
 TEXTOUT_P(tt_CritS(ccp),G1,120,83,1,ttBLK);
 TEXTOUT_P(tt_CritS(ccs),G1,215,83,1,ttBLK);
 TEXTOUT_P("p ("+tt_TailS()+")",G1,10,96,1,ttGRAY);
 TEXTOUT_P(tt_Cm(tt_FmtP(cpp,3)),G1,120,96,1,ttBLK);
 TEXTOUT_P(tt_Cm(tt_FmtP(csp,3)),G1,215,96,1,ttBLK);
 TEXTOUT_P("decision",G1,10,109,1,ttGRAY);
 TEXTOUT_P(tt_Cm(IFTE(tt_Beyond(cp(2),ccp),"reject H"+CHAR(8320),"retain H"+CHAR(8320))),G1,120,109,1,ttBLK);
 TEXTOUT_P(tt_Cm(IFTE(tt_Beyond(cs(2),ccs),"reject H"+CHAR(8320),"retain H"+CHAR(8320))),G1,215,109,1,ttBLK);
 // why they agree or differ: any mix of equal or unequal N and SD
 ctl := 128;
 IF ttN1 == ttN2 THEN
  TEXTOUT_P("N"+CHAR(8321)+" = N"+CHAR(8322)+": the same t either way; only df differs.",G1,8,ctl,1,ttBLK);
 ELSE
  IF ttS1 == ttS2 THEN
   TEXTOUT_P("s"+CHAR(8321)+" = s"+CHAR(8322)+": the same t either way; only df differs.",G1,8,ctl,1,ttBLK);
  ELSE
   cw := IFTE((ttN1 < ttN2 AND ttS1 > ttS2) OR (ttN2 < ttN1 AND ttS2 > ttS1),1,0);
   IF cw THEN
    // Welch 1938; Ruxton 2006; Delacre et al. 2017
    TEXTOUT_P("The SMALLER group has the LARGER SD: the pooled",G1,8,ctl,1,ttRED);
    TEXTOUT_P("test rejects a true H"+CHAR(8320)+" more often than α says.",G1,8,ctl+13,1,ttRED);
   ELSE
    TEXTOUT_P("The LARGER group has the LARGER SD: the pooled",G1,8,ctl,1,ttBLUE);
    TEXTOUT_P("test is too cautious: it misses real effects.",G1,8,ctl+13,1,ttBLUE);
   END;
  END;
 END;
 TEXTOUT_P("The lecture: pooled if the population variances",G1,8,160,1,ttGRAY);
 TEXTOUT_P("are equal, separate if not.  Many",G1,8,173,1,ttGRAY);
 TEXTOUT_P("now use separate always (Delacre et al. 2017).",G1,8,186,1,ttGRAY);
 TEXTOUT_P("any key or tap = back",G1,8,222,1,ttFOOT);
 BLIT_P(G0,0,0,G1);
 tt_WaitAny();
END;

// Returns the test chosen (1 pooled, 2 separate, 3 dependent), or 0 if
// left with Esc.
LOCAL tt_Flow()
BEGIN
 LOCAL fq1, fq2, fk, fm, fm1, fx;
 LOCAL fy, fa, fdone;
 fq1 := -1; fq2 := -1; fdone := 0;
 tt_Settle();
 REPEAT
  tt_DrawFlow(fq1,fq2);
  REPEAT
   fk := GETKEY;
   fm := MOUSE;
   fm1 := fm(1);
  UNTIL fk > -1 OR SIZE(fm1) > 0;
  fa := 0;                         // 1 yes, 2 no, 3 back, 6 Run, 9 quit
  IF SIZE(fm1) > 0 THEN
   fx := fm1(1); fy := fm1(2);
   tt_TDrain();
   IF fy >= 218 THEN
    fa := MIN(6,IP(fx/53.4)+1);
   ELSE
    // the drawn Yes and No boxes, for the question being asked
    IF fq1 == -1 AND fy >= 62 AND fy <= 80 THEN
     IF fx >= 60 AND fx <= 120 THEN fa := 1; END;
     IF fx >= 200 AND fx <= 260 THEN fa := 2; END;
    END;
    IF fq1 == 1 AND fq2 == -1 AND fy >= 124 AND fy <= 142 THEN
     IF fx >= 10 AND fx <= 70 THEN fa := 1; END;
     IF fx >= 110 AND fx <= 170 THEN fa := 2; END;
    END;
   END;
  END;
  IF fk == 4 THEN fa := 9; END;
  IF fk == 30 THEN fa := 6; END;
  IF fa == 1 OR fa == 2 THEN
   IF fq1 == -1 THEN
    fq1 := IFTE(fa == 1,1,0);
   ELSE
    IF fq1 == 1 AND fq2 == -1 THEN fq2 := IFTE(fa == 1,1,0); END;
   END;
  END;
  IF fa == 3 THEN
   IF fq2 <> -1 THEN fq2 := -1; ELSE fq1 := -1; END;
  END;
  IF fa == 6 THEN
   IF fq1 == 0 OR fq2 <> -1 THEN fdone := 1; END;
  END;
  IF fa == 9 THEN fdone := 2; END;
  tt_Drain();
 UNTIL fdone > 0;
 tt_Settle();
 IF fdone == 2 THEN RETURN 0; END;
 IF fq1 == 0 THEN RETURN 3; END;
 RETURN IFTE(fq2 == 1,1,2);
END;

// ---------------- which test? (beta 1k) ----------------
// Which numbers a test uses: the two independent t-tests share theirs
// (family 1); every other test has numbers of its own.  A change of
// family asks for new numbers.
LOCAL tt_Fam(fty)
BEGIN
 IF fty <= 2 THEN RETURN 1; END;
 RETURN fty;
END;

// The test to run: a short menu first, and for two means the flowchart.
// Returns 1-6 (1 pooled, 2 separate, 3 dependent, 4 one-sample t, 5 one-
// proportion z, 6 two-proportion z), or 0 if the menu or the flowchart
// was left with Esc.
LOCAL tt_Pick()
BEGIN

 IF pkc == 1 THEN RETURN tt_Flow(); END;
 RETURN pkc + 2;
END;

// ---------------- data in ----------------
// A list from the Statistics 1Var app: D1 or D2, {} if it is not there
// (the app never opened) or not a list.
LOCAL tt_ReadD(rdi)
BEGIN
 LOCAL rdl;
 rdl := {};
 IFERR
  IF rdi == 1 THEN rdl := D1; END;
  IF rdi == 2 THEN rdl := D2; END;
 THEN
  rdl := {};
 END;
 IF TYPE(rdl) <> 6 THEN rdl := {}; END;

END;

// Mean and SD (N - 1) of a list: {mean, SD, N}
LOCAL tt_MSN(ml)
BEGIN
 LOCAL mi, ms, mn, mm;
 mn := SIZE(ml);
 ms := 0;
 FOR mi FROM 1 TO mn DO ms := ms + ml(mi); END;
 mm := ms/mn;
 ms := 0;
 FOR mi FROM 1 TO mn DO ms := ms + (ml(mi)-mm)^2; END;
 RETURN {mm, IFTE(mn > 1,(ms/(mn-1))^.5,0), mn};
END;

// Tails and alpha: two menus.  The alphas are the printed t table's
// columns.  Returns 0 if a menu was cancelled.
LOCAL tt_AskTails()
BEGIN
 LOCAL ak, aa, al, at;
 LOCAL ah;
 ak := 0;
 // the two sides of the hypotheses, in the symbols of the test on
 // screen (beta 1k): mu1 and mu2, mu and mu0, p and p0, or p1 and p2
 ah := tt_HS();
 CHOOSE(ak,"Tails",{"Two-tailed: "+ah(1)+" ≠ "+ah(2),"One-tailed: "+ah(1)+" < "+ah(2),"One-tailed: "+ah(1)+" > "+ah(2)});
 tt_Settle();
 IF ak == 0 THEN RETURN 0; END;
 // the new tails wait here until an alpha is chosen too: a cancelled
 // alpha list leaves tails and alpha as they were (review 1)
 at := IFTE(ak == 1,0,IFTE(ak == 2,-1,1));
 aa := 0;
 IF at == 0 THEN
  CHOOSE(aa,"α (two-tailed)",{"0.20","0.10","0.05","0.02","0.01","0.001"});
  al := {0.20,0.10,0.05,0.02,0.01,0.001};
 ELSE
  CHOOSE(aa,"α (one-tailed)",{"0.10","0.05","0.025","0.01","0.005","0.0005"});
  al := {0.10,0.05,0.025,0.01,0.005,0.0005};
 END;
 tt_Settle();
 IF aa == 0 THEN RETURN 0; END;
 ttTL := at; ttAL := al(aa);
 RETURN 1;
END;

// The numbers for the test chosen.  Returns 1 when they are in and
// usable, 0 if cancelled.
LOCAL tt_AskData()
BEGIN
 LOCAL ds, dok, dl1, dl2, dr1, dr2;
 LOCAL dm, dsd, dn;
 // beta 1k: the one-sample t-test and the z-tests have forms of their own
 IF ttTY >= 4 THEN RETURN tt_AskDataN(); END;
 ds := 0;
 IF ttTY <> 3 THEN
  CHOOSE(ds,"The data",{"Type each group's mean, SD, N","Raw scores in D1 and D2"});
 ELSE
  CHOOSE(ds,"The pairs",{"Pairs in D1 (x₁) and D2 (x₂)","Type mean, SD of differences"});
 END;
 tt_Settle();
 IF ds == 0 THEN RETURN 0; END;
 // The source, the names and the example change only once new numbers
 // are in: a cancelled form or a refused D1/D2 read leaves everything
 // as it was (review 2 and 6).  The forms write into the variables
 // themselves only on OK.
 // Every field is {name,[0]}: [0] is the list of types the field
 // allows, and 0 is a real number, so the form refuses a list or a
 // text typed into it (beta 1c: the monkey test typed {1,2} as a
 // mean, and the list went on into the arithmetic).
 IF ttTY <> 3 THEN
  IF ds == 1 THEN
   dok := INPUT({{ttM1,[0]},{ttS1,[0]},{ttN1,[0]},{ttM2,[0]},{ttS2,[0]},{ttN2,[0]}},"Two independent samples",
    {"Mean 1:","SD 1:","N 1:","Mean 2:","SD 2:","N 2:"},
    {"Mean 1, -10,000 to 10,000","SD 1 (N - 1), at most 1,000","Size 1, 2 to 10,000","Mean 2, -10,000 to 10,000","SD 2 (N - 1), at most 1,000","Size 2, 2 to 10,000"});
   tt_Settle();
   IF dok == 0 THEN RETURN 0; END;
   // a whole N, as the pairs form takes it (review 7)
   ttN1 := IP(ttN1); ttN2 := IP(ttN2);
   ttSRC := 1;
   ttG1 := "Group 1"; ttG2 := "Group 2";
  ELSE
   dl1 := tt_ReadD(1); dl2 := tt_ReadD(2);
   IF SIZE(dl1) < 2 OR SIZE(dl2) < 2 THEN
    MSGBOX("Put sample 1 in D1 and sample 2 in D2 of the Statistics 1Var app: at least 2 scores in each.");
    tt_Settle();
    RETURN 0;
   END;
   // a value too large to square is refused before any arithmetic
   // (beta 1c, tt_Big), and nothing changes
   IF tt_BigL(dl1) > 0 OR tt_BigL(dl2) > 0 THEN
    MSGBOX(tt_BigLMsg(dl1,dl2));
    tt_Settle();
    RETURN 0;
   END;
   dr1 := tt_MSN(dl1); dr2 := tt_MSN(dl2);
   ttM1 := dr1(1); ttS1 := dr1(2); ttN1 := dr1(3);
   ttM2 := dr2(1); ttS2 := dr2(2); ttN2 := dr2(3);
   ttSRC := 2;
   ttG1 := "D1"; ttG2 := "D2";
  END;
 ELSE
  IF ds == 1 THEN
   dl1 := tt_ReadD(1); dl2 := tt_ReadD(2);
   IF SIZE(dl1) < 2 OR SIZE(dl1) <> SIZE(dl2) THEN
    MSGBOX("Put the pairs in D1 and D2 of the Statistics 1Var app, a pair per row: the same number of values in each, at least 2.");
    tt_Settle();
    RETURN 0;
   END;
   // a value too large to square is refused, and nothing changes
   // (beta 1c, tt_Big)
   IF tt_BigL(dl1) > 0 OR tt_BigL(dl2) > 0 THEN
    MSGBOX(tt_BigLMsg(dl1,dl2));
    tt_Settle();
    RETURN 0;
   END;
   ttX1 := dl1; ttX2 := dl2;
   ttSRC := 2;
   ttG1 := "D1"; ttG2 := "D2";
  ELSE
   dm := ttMD; dsd := ttSDv; dn := ttNP;
   dok := INPUT({{dm,[0]},{dsd,[0]},{dn,[0]}},"Dependent samples",{"Mean xD:","SD sD:","N pairs:"},
    {"Mean of x1 - x2, -10,000 to 10,000","SD of the differences, at most 1,000","Pairs, 2 to 10,000"});
   tt_Settle();
   IF dok == 0 THEN RETURN 0; END;
   ttMD := dm; ttSDv := dsd; ttNP := IP(dn);
   ttSRC := 1;
   ttG1 := "Group 1"; ttG2 := "Group 2";
  END;
 END;
 // new numbers: no longer one of the examples
 ttEX := 0; ttDV := "";
 RETURN 1;
END;

// ---------------- the menu ----------------
// Returns 1 to show the steps again, 0 to quit.
LOCAL tt_Menu()
BEGIN
 LOCAL ms, mt, mo;
 ms := 0;
 CHOOSE(ms,"TTestLab",{"Edit the numbers","Tails and α","Pooled or separate?","Choose the test","Examples","Help","Quit TTestLab"});
 tt_Settle();
 IF ms == 1 THEN
  IF tt_AskData() THEN tt_Calc(); END;
 END;
 IF ms == 2 THEN
  tt_AskTails();
  tt_Calc();
 END;
 IF ms == 3 THEN tt_Compare(); END;
 IF ms == 4 THEN
  mt := tt_Pick();
  IF mt > 0 THEN
   // The independent numbers serve both independent tests, so pooled
   // <-> separate keeps them; a change to or from dependent samples
   // needs new numbers, and a cancelled form keeps the old test.
   // beta 1k: so does a change to or from the one-sample t-test or a
   // z-test, each with numbers of its own (tt_Fam).
   mo := ttTY;
   ttTY := mt;
   IF tt_Fam(mt) <> tt_Fam(mo) THEN
    IF tt_AskData() == 0 THEN ttTY := mo; END;
   END;
   IF ttTY == mt THEN tt_AskTails(); END;
   tt_Calc();
  END;
 END;
 IF ms == 5 THEN
  mt := 0;
  CHOOSE(mt,"Examples",{"Age at arrest (pooled)","Matched jails (dependent)","Drug use by sex (separate)","Student's sleep (1 sample)","Grand juries (1 proportion)","Re-arrest (2 proportions)"});
  tt_Settle();
  IF mt > 0 THEN tt_Example(mt); tt_Calc(); END;
 END;
 IF ms == 6 THEN tt_Help(); END;
 IF ms == 7 THEN RETURN 0; END;
 RETURN 1;
END;

  IF pst == 2 THEN tt_Step2N(); END;
  IF pst == 3 THEN
   IF ttTY == 4 THEN tt_Step3(); ELSE tt_Step3Z(); END;
  END;
  IF pst == 4 THEN tt_Step4N(); END;
  IF pst == 5 THEN tt_Step5N(); END;
  RETURN 0;
 END;
 IF pst == 1 THEN tt_Step1(); END;
 IF pst == 2 THEN tt_Step2(); END;
 IF pst == 3 THEN tt_Step3(); END;
 IF pst == 4 THEN tt_Step4(); END;
 IF pst == 5 THEN tt_Step5(); END;
END;

// The five steps: soft keys 1-5 or Left/Right move, Menu or Enter opens
// the menu, Esc quits.  Returns when the user quits.
LOCAL tt_Steps()
BEGIN
 LOCAL sst, sk, sm, sm1, sgo, sslt;
 sst := 1; sgo := 1;
 WHILE sgo DO
  IF ttOK == 0 THEN
   // say once why the numbers give no t, then the menu
   IF DIM(ttMSG) > 0 THEN MSGBOX(ttMSG); tt_Settle(); ttMSG := ""; END;
   sgo := tt_Menu();
  ELSE
   tt_Page(sst);
   tt_Settle();
   REPEAT
    sk := GETKEY;
    sm := MOUSE;
    sm1 := sm(1);
   UNTIL sk > -1 OR SIZE(sm1) > 0;
   sslt := 0;
   IF SIZE(sm1) > 0 THEN
    IF sm1(2) >= 218 THEN sslt := MIN(6,IP(sm1(1)/53.4)+1); END;
    tt_TDrain();
   END;
   IF sslt >= 1 AND sslt <= 5 THEN sst := sslt; END;
   IF sk == 7 THEN sst := MAX(1,sst-1); END;
   IF sk == 8 THEN sst := MIN(5,sst+1); END;
   IF sk == 4 THEN sgo := 0; END;
   IF sk == 30 OR sslt == 6 THEN
    tt_Settle();
    sgo := tt_Menu();
   END;
  END;
 END;
END;
