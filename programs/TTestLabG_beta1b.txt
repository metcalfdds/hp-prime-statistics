//   steps, with a picture.
//   * Which t test?  Two questions as a tappable flowchart: are the

//     dependent-samples t.

//     the two tests side by side and says when and why they differ.
//   * Dependent samples (matched pairs, before and after) from the
//     pairs in D1 and D2, or from the mean and SD of the differences.
//   * Two-tailed, or one-tailed either way.
//   * The five steps, a screen each: 1 the hypotheses (and the data),
//     2 the distribution and df, 3 the critical value and the decision
//     rule (a t curve with the rejection region shaded), 4 the standard
//     error and t obtained with the numbers put in, 5 the decision and
//     the conclusion (t obtained and the p value on the curve).
//
// HOW THE NUMBERS ARE WORKED OUT (group 1 minus group 2, as SPSS and
//   R do):
//   pooled:    sp = sqrt(((N1-1)s1^2 + (N2-1)s2^2)/(N1+N2-2))
//              SE = sp * sqrt((N1+N2)/(N1*N2)),  df = N1 + N2 - 2
//   separate:  SE = sqrt(s1^2/N1 + s2^2/N2)  (Welch; SPSS's "Equal
//              variances not assumed" row; R's t.test default)
//              df = (a+b)^2 / (a^2/(N1-1) + b^2/(N2-1)), a = s1^2/N1,
//              b = s2^2/N2  (Welch-Satterthwaite; fractional)
//   dependent: xD = x1 - x2 for each pair, sD their SD (N-1),
//              SE = sD/sqrt(Npairs),  df = Npairs - 1
//   t obtained = (mean1 - mean2)/SE, or mean(xD)/SE
//   t critical = -STUDENT_ICDF(df, alpha) one-tailed, alpha/2 two-
//              tailed (a lower tail, mirrored: never ICDF(1 - alpha))
//   p: two-tailed 2*STUDENT_CDF(df, -|t|); H1 mu1 < mu2: STUDENT_CDF(
//      df, t); H1 mu1 > mu2: STUDENT_CDF(df, -t).  Lower tails only.
//   Cohen's d: (mean1 - mean2)/sp (independent; SPSS's Cohen's d);
//      mean(xD)/sD (dependent; SPSS's paired Cohen's d).
//   Squares are taken after dividing by the larger SD (Lessons Learned
//   s11, "scale before squaring"), so no SD over- or underflows.
//

// ---------------- shared state ----------------
LOCAL tzTY:=2;           // the test: 1 pooled, 2 separate (the default), 3 dependent
LOCAL tzSRC:=1;          // the data: 1 summary numbers, 2 raw scores
LOCAL tzTL:=0;           // tails: 0 two-tailed, -1 H1 mu1 < mu2, 1 mu1 > mu2
LOCAL tzAL:=0.05;        // alpha
LOCAL tzEX:=0;           // the example on screen: 0 none, 1-4 the examples
LOCAL tzOK:=0;           // 1 = the numbers are usable
// independent samples: each group's mean, SD and N
LOCAL tzM1:=0, tzS1:=1, tzN1:=10, tzM2:=0, tzS2:=1, tzN2:=10;
// dependent samples: mean and SD of the differences, number of pairs,
// and the pairs themselves when they came as raw scores
LOCAL tzMD:=0, tzSDv:=1, tzNP:=5, tzSSq:=0;
LOCAL tzX1:={0}, tzX2:={0}, tzXD:={0};
// results
LOCAL tzSE:=1, tzT:=0, tzDF:=1, tzTC:=1;
LOCAL tzP:=1, tzD:=0, tzSP:=1, tzREJ:=0;
// words: the two groups, the outcome, and why the numbers were refused
LOCAL tzG1:="Group 1", tzG2:="Group 2", tzDV:="", tzMSG:="";
// palette (Lessons Learned s13), set in tz_Init
LOCAL tzBLUE, tzMAG, tzRED, tzGRN, tzORNG, tzGRAY;
LOCAL tzFOOT, tzGRID, tzLRED, tzLBLU, tzLORN, tzBLK;
LOCAL tzWHT, tzLMAG, tzLGRN;
// the picture's box (pixels)
LOCAL tzPL:=12, tzPR:=308;

// ---------------- palette ----------------
// ---- tz_Cm: commas in big numbers (round 5, 2 Nov 2026, night) ----
// Roger, 2 Nov 2026: "always put comma delimiters in big numbers".  STRING never
// groups digits, whatever Home's setting, so this inserts a comma every three
// digits in each run of four or more digits that does not follow a decimal
// point: "1234567.25" -> "1,234,567.25", "-98765" -> "-98,765"; "0.00012345",
// "1.8E10" and "1,000" stay as they are.  Only text for the screen comes here,
// never text the CAS or EXPR reads ("1,000" there is two numbers).
// Run on the Virtual Calculator: CommaProbe beta 1, 12 of 12 (2 Nov 2026).
tz_Cm(cmS)
BEGIN
  LOCAL cmO, cmI, cmJ, cmN, cmR, cmK;
  cmN := DIM(cmS);
  IF cmN < 4 THEN RETURN cmS; END;
  cmO := ""; cmI := 1;
  WHILE cmI <= cmN DO
    IF INSTRING("0123456789", MID(cmS, cmI, 1)) > 0 THEN
      // a run of digits from cmI to cmJ
      cmJ := cmI;
      WHILE cmJ < cmN AND INSTRING("0123456789", MID(cmS, cmJ + 1, 1)) > 0 DO cmJ := cmJ + 1; END;
      cmR := MID(cmS, cmI, cmJ - cmI + 1);
      IF DIM(cmR) >= 4 AND (cmI == 1 OR MID(cmS, MAX(1, cmI - 1), 1) <> ".") THEN
        // the first group takes 1 to 3 digits, then a comma before every three
        cmK := DIM(cmR) MOD 3;
        IF cmK == 0 THEN cmK := 3; END;
        cmO := cmO + LEFT(cmR, cmK);
        WHILE cmK < DIM(cmR) DO
          cmO := cmO + "," + MID(cmR, cmK + 1, 3);
          cmK := cmK + 3;
        END;
      ELSE
        cmO := cmO + cmR;
      END;
      cmI := cmJ + 1;
    ELSE
      cmO := cmO + MID(cmS, cmI, 1);
      cmI := cmI + 1;
    END;
  END;
  RETURN cmO;
END;

// ---------------- waiting ----------------
// Empty the key buffer: the Enter that closed a menu or form must not
// be read as the next keypress.
LOCAL tz_Drain()
BEGIN
 LOCAL dk;
 REPEAT dk := GETKEY; UNTIL dk == -1;
END;

// Wait for the finger to lift, never more than a second (a stuck touch
// reading must not freeze the program).  MOUSE(1) is -1 when nothing
// touches the screen.
LOCAL tz_TDrain()
BEGIN
 LOCAL drT;
 drT := TICKS;
 REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// After a form, a menu or a box: keys AND the finger.
LOCAL tz_Settle()
BEGIN
 tz_Drain();
 tz_TDrain();
END;

// Wait for any key or tap (drain, wait, drain).
LOCAL tz_WaitAny()
BEGIN
 LOCAL waK, waM, waM1;
 tz_Settle();
 REPEAT
  waK := GETKEY;
  waM := MOUSE;
  waM1 := waM(1);
 UNTIL waK > -1 OR SIZE(waM1) > 0;
 tz_Settle();
END;

// ---------------- numbers as text ----------------
// Fixed decimals, rounded first: STRING(x, 2, d) prints scientific
// form below 10^-d, and a rounded value is 0 or at least 10^-d.
LOCAL tz_Fmt(fv, fd)
BEGIN
 RETURN STRING(ROUND(fv,fd),2,fd);
END;

// A probability: "< 0.001" below the floor (a 0, which only an
// underflow of STUDENT_CDF can give, is below it too: a true p is never
// 0, Lessons Learned s3), "> 0.999" above it, "1.000" for exactly 1.
LOCAL tz_FmtP(pv, pd)
BEGIN
 LOCAL thr;
 thr := 10^(-pd);
 IF pv < thr THEN RETURN "< "+STRING(thr,2,pd); END;
 IF pv >= 1 THEN RETURN STRING(1,2,pd); END;
 IF ROUND(pv,pd) >= 1 THEN RETURN "> "+STRING(1-thr,2,pd); END;
 RETURN STRING(pv,2,pd);
END;

// " = 0.031", or " < 0.001" when the text already carries its relation
LOCAL tz_Rel(rs)
BEGIN
 IF DIM(rs) > 0 THEN
  IF LEFT(rs,1) == "<" OR LEFT(rs,1) == ">" THEN RETURN " "+rs; END;
 END;
 RETURN " = "+rs;
END;

// The fewest decimals (0-5) that hold a typed number exactly
LOCAL tz_NDec(nv)
BEGIN
 LOCAL nd, nk;
 nk := 5;
 FOR nd FROM 4 DOWNTO 0 DO
  IF ABS(nv*10^nd - ROUND(nv*10^nd,0)) < 0.000001 THEN nk := nd; END;
 END;
 RETURN nk;
END;

// The decimals a column of typed values needs: the most any of them
// needs, so a column prints alike: 1.30 and 1.00 beside 1.25
LOCAL tz_ColDec(cl)
BEGIN
 LOCAL cq, cm;
 cm := 0;
 FOR cq FROM 1 TO SIZE(cl) DO cm := MAX(cm, tz_NDec(cl(cq))); END;
 RETURN cm;
END;

// A value with a given number of decimals (0: no point)
LOCAL tz_FixD(fv, fk)
BEGIN
 IF fk == 0 THEN RETURN STRING(ROUND(fv,0),1,0); END;
 RETURN STRING(ROUND(fv,fk),2,fk);
END;

// A number as it was typed: no trailing zeros, at most 5 decimals
// (1.45 stays 1.45, 44 stays 44, SPSS's 3.28798 stays 3.28798).
LOCAL tz_Num(nv)
BEGIN
 LOCAL nk, ns;
 nk := tz_NDec(nv);
 IF nk == 0 THEN RETURN STRING(ROUND(nv,0),1,0); END;
 ns := STRING(ROUND(nv,nk),2,nk);
 IF nk == 5 THEN
  // more than 5 decimals typed: round to 5, then drop end zeros
  WHILE MID(ns,DIM(ns),1) == "0" DO ns := LEFT(ns,DIM(ns)-1); END;
  IF MID(ns,DIM(ns),1) == "." THEN ns := LEFT(ns,DIM(ns)-1); END;
 END;
 RETURN ns;
END;

// A worked value to 4 significant digits, in fixed form (0.05499,
// 1.745, 0.0001058): intermediate steps show enough digits to see where
// a rounded hand calculation would part company.
LOCAL tz_Sig(sv)
BEGIN
 LOCAL sd;
 IF sv == 0 THEN RETURN "0"; END;
 sd := MIN(10, MAX(0, 3 - FLOOR(LOG(ABS(sv)))));
 RETURN STRING(ROUND(sv,sd),2,sd);
END;

// A mean or SD on the data screens: as it was typed (tz_Num) when it
// was typed; to 4 significant digits (tz_Sig) when the program worked
// it out from raw scores (D1/D2 or an example's data), so a draft
// lottery mean shows as 206.3, not 206.32418.
LOCAL tz_Val(vv)
BEGIN
 IF tzSRC == 2 THEN RETURN tz_Cm(tz_Sig(vv)); END;
 RETURN tz_Cm(tz_Num(vv));
END;

// ---------------- the arithmetic ----------------
// t critical (its size) for df, at alpha, for the tails on screen: the
// point with a = alpha (one-tailed) or alpha/2 (two-tailed) beyond it.
// A lower-tail quantile, mirrored: -STUDENT_ICDF(df, a), never
// ICDF(1 - a) (Lessons Learned s3).  STUDENT_ICDF matched SciPy to 12
// digits on the emulator but is not yet on the device-verified list,
// so, as CICatcher's cqTCrit does: try it inside IFERR, check the
// answer with STUDENT_CDF (device-verified), and if either fails,
// bisect on STUDENT_CDF.
LOCAL tz_Crit(cdf)
BEGIN
 LOCAL ca, cv, cok, clo, chi, cmid;
 LOCAL cj;
 ca := IFTE(tzTL == 0,tzAL/2,tzAL);
 cok := 0; cv := 0;
 IFERR
  cv := -STUDENT_ICDF(cdf,ca);
  IF cv > 0 THEN
   IF ABS(STUDENT_CDF(cdf,-cv)/ca - 1) < 0.000001 THEN cok := 1; END;
  END;
 THEN
  cok := 0;
 END;
 IF cok THEN RETURN cv; END;
 // the fallback: bracket (double until the tail is below a), then 60
 // halvings, which leave less than 1E-12 of the bracket
 clo := 0; chi := 4;
 WHILE STUDENT_CDF(cdf,-chi) > ca AND chi < 1000000 DO
  clo := chi; chi := 2*chi;
 END;
 FOR cj FROM 1 TO 60 DO
  cmid := (clo+chi)/2;
  IF STUDENT_CDF(cdf,-cmid) > ca THEN clo := cmid; ELSE chi := cmid; END;
 END;
 RETURN (clo+chi)/2;
END;

// Is a t beyond the critical value cv?  (two-tailed: either side;
// H1 mu1 < mu2: below -cv; H1 mu1 > mu2: above cv)
LOCAL tz_Beyond(bt, bc)
BEGIN
 IF tzTL == 0 THEN RETURN IFTE(ABS(bt) > bc,1,0); END;
 IF tzTL < 0 THEN RETURN IFTE(bt < -bc,1,0); END;
 RETURN IFTE(bt > bc,1,0);
END;

// The p value for t on df, for the tails on screen (lower tails only).
LOCAL tz_PVal(pt, pdf)
BEGIN
 // two-tailed: capped at 1, as AnovaLab's (t = 0 gives 2 x 0.5)
 IF tzTL == 0 THEN RETURN MIN(1,2*STUDENT_CDF(pdf,-ABS(pt))); END;
 IF tzTL < 0 THEN RETURN STUDENT_CDF(pdf,pt); END;
 RETURN STUDENT_CDF(pdf,-pt);
END;

// Pooled (pk = 1) or separate (pk = 2) t for the independent samples on
// screen: returns {SE, t, df, pooled SD}.  The SDs are divided by the
// larger first (cm), so nothing squares out of range.
LOCAL tz_Ind(pk)
BEGIN
 LOCAL cm, ca, cb, cv, csp, cse;
 LOCAL cdf;
 cm := MAX(tzS1,tzS2);
 ca := (tzS1/cm)^2;
 cb := (tzS2/cm)^2;
 csp := cm*(((tzN1-1)*ca + (tzN2-1)*cb)/(tzN1+tzN2-2))^.5;
 IF pk == 1 THEN
  cse := csp*((tzN1+tzN2)/(tzN1*tzN2))^.5;
  cdf := tzN1+tzN2-2;
 ELSE
  cv := ca/tzN1 + cb/tzN2;
  cse := cm*cv^.5;
  cdf := cv^2/((ca/tzN1)^2/(tzN1-1) + (cb/tzN2)^2/(tzN2-1));
 END;
 RETURN {cse, (tzM1-tzM2)/cse, cdf, csp};
END;

// Mean, SD and sum of squared deviations of the pairs' differences,
// from tzX1 and tzX2 (two passes: the mean first).
LOCAL tz_Pairs()
BEGIN
 LOCAL pj, ps;
 tzNP := SIZE(tzX1);
 tzXD := tzX1;
 ps := 0;
 FOR pj FROM 1 TO tzNP DO
  tzXD(pj) := tzX1(pj) - tzX2(pj);
  ps := ps + tzXD(pj);
 END;
 tzMD := ps/tzNP;
 tzSSq := 0;
 FOR pj FROM 1 TO tzNP DO tzSSq := tzSSq + (tzXD(pj)-tzMD)^2; END;
 tzSDv := (tzSSq/(tzNP-1))^.5;
END;

// Too large to work with?  The program squares the numbers it is given
// (an SD, a difference from the mean) and step 4 squares s^2/N again
// for Welch's df.  On the Prime a square past 1E499 is infinity, with
// no error (Lessons Learned s3), and the screens would fill with it.
// So every number taken in, typed or read from D1/D2, must be smaller
// than 1E50 in size: 50 digits before the point, far beyond any data a
// course meets, and its fourth power still fits (beta 1c, from the
// monkey test).  An infinity is refused too.
LOCAL tz_Big(bv)
BEGIN
 RETURN IFTE(ABS(bv) >= 1E50,1,0);
END;

// The position of the first value in a list that is too large
// (tz_Big), or 0 when every value is fine.
LOCAL tz_BigL(bl)
BEGIN
 LOCAL bi;
 FOR bi FROM 1 TO SIZE(bl) DO
  IF tz_Big(bl(bi)) THEN RETURN bi; END;
 END;
 RETURN 0;
END;

// The sentence for a list read from the Statistics 1Var app that holds
// a value too large: which list, and which value in it.
LOCAL tz_BigLMsg(gl1, gl2)
BEGIN
 LOCAL gk, gw;
 gk := tz_BigL(gl1); gw := "D1";
 IF gk == 0 THEN gk := tz_BigL(gl2); gw := "D2"; END;
 RETURN "Value "+tz_Cm(STRING(gk,1,0))+" in "+gw+" has more than 50 digits: too large to work with. Check "+gw+" in the Statistics 1Var app.";
END;

// The sentence for a typed N over 1,000,000.  The Prime's STUDENT_CDF
// loses its digits at a huge df (Virtual Calculator 2.4.2, 1 Nov 2026,
// at t = -1.96: df 1E6 gave 0.024998033593, right to 8 digits; df 1E10
// 0.02498, where the value is 0.024998; df 1E13 0.014; df 1E15 and past
// 0.5).  N of at most 1,000,000 keeps every df below 2,000,000, where
// it is right, and t and z there agree to 5 digits anyway.  Only a
// typed N is checked: an N from D1/D2 counts the values in a list.
LOCAL tz_NMsg()
BEGIN
 RETURN "N can be at most 1,000,000: the calculator's t distribution loses accuracy far beyond it (and there t equals z).";
END;

// Everything the five steps show, from the numbers on screen.  Sets
// tzOK = 0 and tzMSG when the numbers cannot give a t.
LOCAL tz_Calc()
BEGIN
 LOCAL cr;
 tzOK := 1; tzMSG := "";
 // A typed number too large to work with (tz_Big).  Only typed ones:
 // numbers worked out from D1/D2 come from values already checked
 // when the lists were read, and an SD worked out from values below
 // 1E50 can lie a little above it.
 IF tzSRC == 1 THEN
  IF tzTY == 3 THEN
   IF tz_Big(tzMD) OR tz_Big(tzSDv) OR tz_Big(tzNP) THEN
    tzOK := 0; tzMSG := "A number you typed has more than 50 digits: too large to work with. Choose Edit the numbers to change it."; RETURN 0;
   END;
   IF tzNP > 1000000 THEN tzOK := 0; tzMSG := tz_NMsg(); RETURN 0; END;
  ELSE
   IF tz_Big(tzM1) OR tz_Big(tzS1) OR tz_Big(tzN1) OR tz_Big(tzM2) OR tz_Big(tzS2) OR tz_Big(tzN2) THEN
    tzOK := 0; tzMSG := "A number you typed has more than 50 digits: too large to work with. Choose Edit the numbers to change it."; RETURN 0;
   END;
   IF tzN1 > 1000000 OR tzN2 > 1000000 THEN tzOK := 0; tzMSG := tz_NMsg(); RETURN 0; END;
  END;
 END;
 IF tzTY == 3 THEN
  // the guard before the SD of the differences is formed (it divides
  // by N - 1): review 10
  IF tzSRC == 2 THEN tzNP := SIZE(tzX1); END;
  IF tzNP < 2 THEN tzOK := 0; tzMSG := "A dependent-samples t test needs at least 2 pairs."; RETURN 0; END;
  IF tzSRC == 2 THEN tz_Pairs(); END;
  IF tzSDv < 0 THEN tzOK := 0; tzMSG := "An SD cannot be negative."; RETURN 0; END;
  IF tzSDv == 0 THEN tzOK := 0; tzMSG := "Every pair differs by the same amount (sD = 0): there is no t."; RETURN 0; END;
  tzSE := tzSDv/tzNP^.5;
  tzDF := tzNP-1;
  tzT := tzMD/tzSE;
  tzD := tzMD/tzSDv;
  tzSP := tzSDv;
 ELSE
  IF tzN1 < 2 OR tzN2 < 2 THEN tzOK := 0; tzMSG := "Each group needs N of at least 2."; RETURN 0; END;
  IF tzS1 < 0 OR tzS2 < 0 THEN tzOK := 0; tzMSG := "An SD cannot be negative."; RETURN 0; END;
  IF tzS1 == 0 AND tzS2 == 0 THEN tzOK := 0; tzMSG := "Both SDs are 0: every score is the same, so there is no t."; RETURN 0; END;
  cr := tz_Ind(tzTY);
  tzSE := cr(1); tzT := cr(2); tzDF := cr(3); tzSP := cr(4);
  tzD := (tzM1-tzM2)/tzSP;
 END;
 // t critical for the exact df (a printed table is only a note on
 // step 3: the books' tables differ past df 30)
 tzTC := tz_Crit(tzDF);
 tzP := tz_PVal(tzT,tzDF);
 tzREJ := tz_Beyond(tzT,tzTC);
 RETURN 1;
END;

// ---------------- words ----------------
LOCAL tz_TName()
BEGIN

 IF tzTY == 2 THEN RETURN "Separate-variances t test"; END;
 RETURN "Dependent-samples t test";
END;

LOCAL tz_TailS()
BEGIN

END;

// H1 in symbols
LOCAL tz_H1S()
BEGIN
 IF tzTL == 0 THEN RETURN "H"+CHAR(8321)+": μ"+CHAR(8321)+" ≠ μ"+CHAR(8322); END;
 IF tzTL < 0 THEN RETURN "H"+CHAR(8321)+": μ"+CHAR(8321)+" < μ"+CHAR(8322); END;
 RETURN "H"+CHAR(8321)+": μ"+CHAR(8321)+" > μ"+CHAR(8322);
END;

// "the alpha on screen", as ".05"-style text with house decimals
LOCAL tz_AlS()
BEGIN
 RETURN "α"+tz_Rel(tz_Cm(tz_FmtP(tzAL,IFTE(tzAL < 0.001,4,3))));
END;

// t critical with its sign(s), as the decision rule needs it
LOCAL tz_CritS(cv)
BEGIN

 IF tzTL < 0 THEN RETURN tz_Cm(tz_Fmt(-cv,3)); END;
 RETURN tz_Cm(tz_Fmt(cv,3));
END;

// The t curve on df from y0 to y1: the rejection region red, the rest
// pale blue; t critical dashed red; with shT, t obtained in magenta and
// with shP the p value's area shaded magenta.  One pixel column at a
// time (Lessons Learned s6: shade by columns, never by samples).
LOCAL tz_Curve(cy0, cy1, shT, shP)
BEGIN
 LOCAL cxr, cpk, cH, cpx, cx, cf;
 LOCAL cyp, cyq, cin, cpin, ctk, ctx;
 LOCAL cdec, cstp, cs;
 // the window: -4..4, wider when t critical or t obtained lies past 3.2
 cxr := MAX(4, tzTC + 0.8);
 IF shT THEN cxr := MAX(cxr, ABS(tzT) + 0.8); END;
 cxr := MIN(cxr, 12);
 cpk := STUDENT(tzDF,0);
 cH := cy1 - cy0 - 4;
 cyq := cy1;
 FOR cpx FROM tzPL TO tzPR DO
  cx := -cxr + (cpx-tzPL)/(tzPR-tzPL)*2*cxr;
  cf := STUDENT(tzDF,cx);
  cyp := cy1 - IP(cf/cpk*cH);
  cin := tz_Beyond(cx,tzTC);
  // p value's area: beyond t obtained, in the direction(s) of H1
  cpin := 0;
  IF shP THEN
   IF tzTL == 0 THEN cpin := IFTE(ABS(cx) >= ABS(tzT),1,0); END;
   IF tzTL < 0 THEN cpin := IFTE(cx <= tzT,1,0); END;
   IF tzTL > 0 THEN cpin := IFTE(cx >= tzT,1,0); END;
  END;
  IF cyp < cy1 THEN
   IF cpin THEN
    LINE_P(G1,cpx,cyp,cpx,cy1,tzLMAG);
   ELSE
    LINE_P(G1,cpx,cyp,cpx,cy1,IFTE(cin,tzLRED,tzLBLU));
   END;
  END;
  // the curve itself, joined to the column before
  IF cpx > tzPL THEN LINE_P(G1,cpx-1,cyq,cpx,cyp,tzBLUE); END;
  cyq := cyp;
 END;
 // the axis, with whole-number ticks (every 2 when the window is wide)
 LINE_P(G1,tzPL,cy1,tzPR,cy1,tzBLK);
 cstp := IFTE(cxr > 6,2,1);
 FOR ctk FROM -FLOOR(cxr/cstp)*cstp TO FLOOR(cxr/cstp)*cstp STEP cstp DO
  ctx := tzPL + IP((ctk+cxr)/(2*cxr)*(tzPR-tzPL));
  LINE_P(G1,ctx,cy1,ctx,cy1+3,tzBLK);
  cs := tz_Cm(STRING(ctk,1,0));
  TEXTOUT_P(cs,G1,ctx-3*DIM(cs),cy1+4,1,tzGRAY);
 END;
 // t critical: dashed red line(s).  Past the window (2 pairs at .05:
 // 12.706) the line sits at the window's edge, with its name, so the
 // page never shows a rule with nothing red (review 8).
 FOR cs FROM -1 TO 1 STEP 2 DO
  IF (tzTL == 0) OR (tzTL == cs) THEN
   ctx := tzPL + IP((MAX(-cxr,MIN(cxr,cs*tzTC))+cxr)/(2*cxr)*(tzPR-tzPL));
   FOR cpx FROM cy0 TO cy1 STEP 4 DO LINE_P(G1,ctx,cpx,ctx,MIN(cy1,cpx+1),tzRED); END;
   IF tzTC > cxr THEN
    TEXTOUT_P("t crit",G1,IFTE(cs < 0,tzPL+3,tzPR-38),cy0+2,1,tzRED);
   END;
  END;
 END;
 // t obtained: a magenta line, its value above it
 IF shT THEN
  ctx := tzPL + IP((MAX(-cxr,MIN(cxr,tzT))+cxr)/(2*cxr)*(tzPR-tzPL));
  LINE_P(G1,ctx,cy0,ctx,cy1,tzMAG);
  LINE_P(G1,ctx+1,cy0,ctx+1,cy1,tzMAG);
  cs := "t obt = "+tz_Cm(tz_Fmt(tzT,2));
  cdec := MIN(tzPR-IP(6.5*DIM(cs)), MAX(tzPL, ctx-IP(2.75*DIM(cs))));
  TEXTOUT_P(cs,G1,cdec,cy0-11,1,tzMAG);
 END;
END;

// ---------------- step 2: the distribution and df ----------------
LOCAL tz_Step2()
BEGIN
 LOCAL sa, sb, sden, sdh;
 tz_Frame("Step 2. Distribution and df");
 TEXTOUT_P("The distribution is t: its shape depends on df.",G1,8,32,1,tzBLK);
 IF tzTY == 1 THEN
  TEXTOUT_P("Pooled variances:  df = N"+CHAR(8321)+" + N"+CHAR(8322)+" - 2",G1,8,54,1,tzBLUE);
  TEXTOUT_P("   = "+tz_Cm(tz_Num(tzN1))+" + "+tz_Cm(tz_Num(tzN2))+" - 2 = "+tz_Cm(STRING(tzDF,1,0)),G1,8,69,1,tzBLUE);
 END;
 IF tzTY == 2 THEN
  sa := tzS1^2/tzN1; sb := tzS2^2/tzN2;
  sden := sa^2/(tzN1-1) + sb^2/(tzN2-1);
  TEXTOUT_P("Separate variances (Welch-Satterthwaite):",G1,8,54,1,tzBLUE);
  TEXTOUT_P("df = (a + b)² / [a²/(N"+CHAR(8321)+"-1) + b²/(N"+CHAR(8322)+"-1)]",G1,8,69,1,tzBLUE);
  TEXTOUT_P("a = s"+CHAR(8321)+"²/N"+CHAR(8321)+" = "+tz_Cm(tz_Sig(sa))+",  b = s"+CHAR(8322)+"²/N"+CHAR(8322)+" = "+tz_Cm(tz_Sig(sb)),G1,8,84,1,tzBLUE);
  TEXTOUT_P("df = "+tz_Cm(tz_Sig((sa+sb)^2))+" / "+tz_Cm(tz_Sig(sden))+" = "+tz_Cm(tz_Fmt(tzDF,2)),G1,8,99,1,tzBLUE);
  // the hand shortcut many books give for separate variances: the
  // smaller N - 1, which gives a larger t crit than Welch's df, so it
  // errs on the safe side (the companion guide's Conventions)
  sdh := MIN(tzN1,tzN2) - 1;
  TEXTOUT_P("A fraction is fine: SPSS and R print this df.",G1,8,116,1,tzGRAY);
  TEXTOUT_P("By hand: df = the smaller of N"+CHAR(8321)+"-1, N"+CHAR(8322)+"-1 = "+tz_Cm(STRING(sdh,1,0))+",",G1,8,131,1,tzGRAY);
  TEXTOUT_P("t crit = "+tz_CritS(tz_Crit(sdh))+": larger, so a safer test.",G1,8,144,1,tzGRAY);
 END;
 IF tzTY == 3 THEN
  TEXTOUT_P("Dependent samples:  df = N pairs - 1",G1,8,54,1,tzBLUE);
  TEXTOUT_P("   = "+tz_Cm(tz_Num(tzNP))+" - 1 = "+tz_Cm(STRING(tzDF,1,0)),G1,8,69,1,tzBLUE);
  TEXTOUT_P("N is the number of PAIRS, not of people or",G1,8,90,1,tzGRAY);
  TEXTOUT_P("places: each pair gives one difference.",G1,8,103,1,tzGRAY);
 END;
 TEXTOUT_P("The more df, the closer t comes to the normal",G1,8,160,1,tzGRAY);
 TEXTOUT_P("curve; few df give a wider, flatter t.",G1,8,173,1,tzGRAY);
 tz_ShowSteps();
END;

// ---------------- step 3: the critical value and the rule ----------------
LOCAL tz_Step3()
BEGIN
 LOCAL sdf;
 tz_Frame("Step 3. Critical value, decision rule");
 sdf := IFTE(tzTY == 2,tz_Cm(tz_Fmt(tzDF,2)),tz_Cm(STRING(tzDF,1,0)));
 TEXTOUT_P(tz_AlS()+", "+tz_TailS()+", df = "+sdf,G1,8,30,1,tzBLK);
 // exact, for this df: the calculator needs no printed table
 TEXTOUT_P("t crit = "+tz_CritS(tzTC)+"   (exact, for df = "+sdf+")",G1,8,44,1,tzRED);
 IF tzTL == 0 THEN
  TEXTOUT_P("Rule: if t obt > "+tz_Cm(tz_Fmt(tzTC,3))+" or t obt < "+tz_Cm(tz_Fmt(-tzTC,3))+",",G1,8,60,1,tzBLK);
 ELSE
  IF tzTL < 0 THEN
   TEXTOUT_P("Rule: if t obt < "+tz_Cm(tz_Fmt(-tzTC,3))+",",G1,8,60,1,tzBLK);
  ELSE
   TEXTOUT_P("Rule: if t obt > "+tz_Cm(tz_Fmt(tzTC,3))+",",G1,8,60,1,tzBLK);
  END;
 END;
 IF tzTC > 12 THEN
  TEXTOUT_P("  reject H"+CHAR(8320)+".  (t crit lies beyond the plot.)",G1,8,73,1,tzBLK);
 ELSE
  TEXTOUT_P("  reject H"+CHAR(8320)+".  Red = the rejection region (α).",G1,8,73,1,tzBLK);
 END;
 // A printed t table lists only some df, and the books' tables differ
 // past 30.  The companion guide's rule: a df the table lacks takes the
 // next SMALLER row it lists, whose t crit is larger, so the test never
 // claims more than the data show; a fractional df is rounded down
 // first.  Up to df 30 every table has the row: no note.
 IF tzTY == 2 THEN
  TEXTOUT_P("Printed table?  Round df down to "+tz_Cm(STRING(IP(tzDF),1,0))+"; if the",G1,8,87,1,tzGRAY);
  TEXTOUT_P("table lacks that row, take the next smaller.",G1,8,100,1,tzGRAY);
 ELSE
  IF tzDF > 30 THEN
   TEXTOUT_P("Printed table without row "+tz_Cm(STRING(tzDF,1,0))+"?  Take the next",G1,8,87,1,tzGRAY);
   TEXTOUT_P("smaller row it lists (its t crit is larger).",G1,8,100,1,tzGRAY);
  END;
 END;
 tz_Curve(114,196,0,0);
 tz_ShowSteps();
END;

// ---------------- step 4: SE and t obtained ----------------
LOCAL tz_Step4()
BEGIN
 LOCAL sa, sb, sr, sj, sn, sy;
 tz_Frame("Step 4. Standard error and t obtained");
 IF tzTY == 1 THEN
  sr := (tzN1+tzN2)/(tzN1*tzN2);
  TEXTOUT_P("SE = √[((N"+CHAR(8321)+"-1)s"+CHAR(8321)+"² + (N"+CHAR(8322)+"-1)s"+CHAR(8322)+"²)/(N"+CHAR(8321)+"+N"+CHAR(8322)+"-2)]",G1,8,30,1,tzBLUE);
  TEXTOUT_P("       × √[(N"+CHAR(8321)+"+N"+CHAR(8322)+")/(N"+CHAR(8321)+"N"+CHAR(8322)+")]",G1,8,43,1,tzBLUE);
  TEXTOUT_P("   = √[("+tz_Cm(tz_Num(tzN1-1))+" × "+tz_Val(tzS1)+"² + "+tz_Cm(tz_Num(tzN2-1))+" × "+tz_Val(tzS2)+"²)/"+tz_Cm(STRING(tzDF,1,0))+"]",G1,8,58,1,tzBLK);
  TEXTOUT_P("       × √["+tz_Cm(tz_Num(tzN1+tzN2))+"/"+tz_Cm(tz_Num(tzN1*tzN2))+"]",G1,8,71,1,tzBLK);
  TEXTOUT_P("   = √"+tz_Cm(tz_Sig(tzSP^2))+" × √"+tz_Cm(tz_Sig(sr))+" = "+tz_Cm(tz_Sig(tzSP))+" × "+tz_Cm(tz_Sig(sr^.5)),G1,8,84,1,tzBLK);
  TEXTOUT_P("   = "+tz_Cm(tz_Sig(tzSE)),G1,8,97,1,tzBLK);
 END;
 IF tzTY == 2 THEN
  sa := tzS1^2/tzN1; sb := tzS2^2/tzN2;
  TEXTOUT_P("SE = √(s"+CHAR(8321)+"²/N"+CHAR(8321)+" + s"+CHAR(8322)+"²/N"+CHAR(8322)+")",G1,8,30,1,tzBLUE);
  TEXTOUT_P("   = √("+tz_Val(tzS1)+"²/"+tz_Cm(tz_Num(tzN1))+" + "+tz_Val(tzS2)+"²/"+tz_Cm(tz_Num(tzN2))+")",G1,8,45,1,tzBLK);
  TEXTOUT_P("   = √("+tz_Cm(tz_Sig(sa))+" + "+tz_Cm(tz_Sig(sb))+") = √"+tz_Cm(tz_Sig(sa+sb)),G1,8,60,1,tzBLK);
  TEXTOUT_P("   = "+tz_Cm(tz_Sig(tzSE)),G1,8,75,1,tzBLK);
 END;
 IF tzTY <> 3 THEN
  TEXTOUT_P("t obt = ("+CHAR(57344)+CHAR(8321)+" - "+CHAR(57344)+CHAR(8322)+")/SE = ("+tz_Val(tzM1)+" - "+tz_Val(tzM2)+")/"+tz_Cm(tz_Sig(tzSE)),G1,8,116,1,tzMAG);
  TEXTOUT_P("      = "+tz_Cm(tz_Sig(tzM1-tzM2))+"/"+tz_Cm(tz_Sig(tzSE))+" = "+tz_Cm(tz_Fmt(tzT,2)),G1,8,129,1,tzMAG);
 ELSE
  IF tzSRC == 2 THEN
   // each difference, its deviation from the mean, and that squared
   TEXTOUT_P("x"+CHAR(8321)+"-x"+CHAR(8322)+" = xD",G1,8,28,1,tzGRAY);
   TEXTOUT_P("xD - "+CHAR(57344)+"D",G1,110,28,1,tzGRAY);
   TEXTOUT_P("(xD - "+CHAR(57344)+"D)²",G1,210,28,1,tzGRAY);
   sn := MIN(tzNP,5);
   FOR sj FROM 1 TO sn DO
    sy := 28 + 12*sj;
    TEXTOUT_P(tz_Cm(tz_Sig(tzXD(sj))),G1,12,sy,1,tzBLK);
    TEXTOUT_P(tz_Cm(tz_Sig(tzXD(sj)-tzMD)),G1,114,sy,1,tzBLK);
    TEXTOUT_P(tz_Cm(tz_Sig((tzXD(sj)-tzMD)^2)),G1,214,sy,1,tzBLK);
   END;
   IF tzNP > 5 THEN TEXTOUT_P("... "+tz_Cm(STRING(tzNP-5,1,0))+" more",G1,12,100,1,tzGRAY); END;
   TEXTOUT_P(CHAR(57344)+"D = "+tz_Cm(tz_Sig(tzMD)),G1,12,112,1,tzBLK);
   // the column's sum (the sigma sign is not on the verified glyph
   // list, Lessons Learned s5, so the word)
   TEXTOUT_P("sum = "+tz_Cm(tz_Sig(tzSSq)),G1,214,112,1,tzBLK);
   TEXTOUT_P("sD = √(sum/(N-1)) = √("+tz_Cm(tz_Sig(tzSSq))+"/"+tz_Cm(STRING(tzNP-1,1,0))+") = "+tz_Cm(tz_Sig(tzSDv)),G1,8,128,1,tzBLUE);
  ELSE
   TEXTOUT_P(CHAR(57344)+"D = "+tz_Cm(tz_Num(tzMD))+",  sD = "+tz_Cm(tz_Num(tzSDv))+",  N pairs = "+tz_Cm(tz_Num(tzNP)),G1,8,40,1,tzBLK);
  END;
  // under the pairs' table, or right under the typed summary
  sy := IFTE(tzSRC == 2,141,60);
  TEXTOUT_P("SE = sD/√N = "+tz_Cm(tz_Sig(tzSDv))+"/√"+tz_Cm(tz_Num(tzNP))+" = "+tz_Cm(tz_Sig(tzSE)),G1,8,sy,1,tzBLUE);
  TEXTOUT_P("t obt = "+CHAR(57344)+"D/SE = "+tz_Cm(tz_Sig(tzMD))+"/"+tz_Cm(tz_Sig(tzSE))+" = "+tz_Cm(tz_Fmt(tzT,2)),G1,8,sy+13,1,tzMAG);
 END;
 tz_ShowSteps();
END;

// ---------------- step 5: the decision ----------------
LOCAL tz_Step5()
BEGIN
 tz_Frame("Step 5. Decision and conclusion");
 IF tzREJ THEN
  TEXTOUT_P("t obt = "+tz_Cm(tz_Fmt(tzT,3))+" is beyond t crit = "+tz_CritS(tzTC)+":",G1,8,30,1,tzRED);
  TEXTOUT_P("REJECT H"+CHAR(8320)+".",G1,8,43,1,tzRED);
  IF tzTL == 0 THEN
   TEXTOUT_P("The population means differ significantly:",G1,8,58,1,tzBLK);
  ELSE
   TEXTOUT_P("The means differ the way H"+CHAR(8321)+" predicted:",G1,8,58,1,tzBLK);
  END;
  // which way, in the samples' own names (sample 1 minus sample 2)
  TEXTOUT_P("sample 1 ("+tzG1+") is "+tz_Cm(IFTE(tzT < 0,"lower","higher"))+" on average.",G1,8,71,1,tzBLK);
 ELSE
  TEXTOUT_P("t obt = "+tz_Cm(tz_Fmt(tzT,3))+" is not beyond t crit = "+tz_CritS(tzTC)+":",G1,8,30,1,tzGRN);
  TEXTOUT_P("RETAIN H"+CHAR(8320)+".",G1,8,43,1,tzGRN);
  TEXTOUT_P("No significant difference: no support for H"+CHAR(8321)+".",G1,8,58,1,tzBLK);
  // one-tailed, and the samples differ the opposite way to H1's
  IF tzTL <> 0 AND tzT*tzTL < 0 THEN
   TEXTOUT_P("The samples even differ the other way from H"+CHAR(8321)+".",G1,8,71,1,tzBLK);
  END;
 END;
 // A p that rounds to alpha at 3 decimals ("p = 0.050" beside a
 // rejection at alpha .05 reads as a contradiction) also gets its 4
 // significant digits (Darwin's plants: p = 0.04970).
 IF tzAL >= 0.001 AND ROUND(tzP,3) == ROUND(tzAL,3) AND tzP <> tzAL THEN
  TEXTOUT_P("(p = "+tz_Cm(tz_Sig(tzP))+": just "+tz_Cm(IFTE(tzP < tzAL,"below","above"))+" α = "+tz_Cm(tz_Fmt(tzAL,3))+".)",G1,8,84,1,tzGRAY);
 END;
 TEXTOUT_P("Report "+tz_TDf()+" = "+tz_Cm(tz_Fmt(tzT,2))+", p"+tz_Rel(tz_Cm(tz_FmtP(tzP,3)))+", "+tz_TailS(),G1,8,97,1,tzMAG);
 TEXTOUT_P("Cohen's d = "+tz_Cm(tz_Fmt(tzD,2))+tz_Cm(IFTE(tzTY == 3,"  ("+CHAR(57344)+"D / sD)","  (pooled s)")),G1,8,110,1,tzMAG);
 TEXTOUT_P("magenta area = p",G1,216,110,1,tzMAG);
 tz_Curve(134,196,1,1);
 tz_ShowSteps();
END;

// ---------------- pooled or separate? ----------------
LOCAL tz_Compare()
BEGIN
 LOCAL cp, cs, cpp, csp, ccp, ccs;
 LOCAL cw, ctl;
 IF tzTY == 3 THEN
  MSGBOX("Pooled or separate variances is a question for INDEPENDENT samples. Dependent samples have one SD, of the differences.");
  tz_Settle();
  RETURN 0;
 END;
 // numbers that gave no t (N = 1, both SDs 0) would divide by zero
 // here (review 3)
 IF tzOK == 0 THEN
  MSGBOX("The numbers on screen give no t: Edit the numbers first.");
  tz_Settle();
  RETURN 0;
 END;
 cp := tz_Ind(1); cs := tz_Ind(2);

// Returns the test chosen (1 pooled, 2 separate, 3 dependent), or 0 if
// left with Esc.
LOCAL tz_Flow()
BEGIN
 LOCAL fq1, fq2, fk, fm, fm1, fx;
 LOCAL fy, fa, fdone;
 fq1 := -1; fq2 := -1; fdone := 0;
 tz_Settle();
 REPEAT
  tz_DrawFlow(fq1,fq2);
  REPEAT
   fk := GETKEY;
   fm := MOUSE;
   fm1 := fm(1);
  UNTIL fk > -1 OR SIZE(fm1) > 0;
  fa := 0;                         // 1 yes, 2 no, 3 back, 6 OK, 9 quit
  IF SIZE(fm1) > 0 THEN
   fx := fm1(1); fy := fm1(2);
   tz_TDrain();
   IF fy >= 218 THEN
    fa := MIN(6,IP(fx/53.4)+1);
   ELSE
    // the drawn Yes and No boxes, for the question being asked
    IF fq1 == -1 AND fy >= 62 AND fy <= 80 THEN
     IF fx >= 60 AND fx <= 120 THEN fa := 1; END;
     IF fx >= 200 AND fx <= 260 THEN fa := 2; END;
    END;
    IF fq1 == 1 AND fq2 == -1 AND fy >= 124 AND fy <= 142 THEN
     IF fx >= 10 AND fx <= 70 THEN fa := 1; END;
     IF fx >= 110 AND fx <= 170 THEN fa := 2; END;
    END;
   END;
  END;
  IF fk == 4 THEN fa := 9; END;
  IF fk == 30 THEN fa := 6; END;
  IF fa == 1 OR fa == 2 THEN
   IF fq1 == -1 THEN
    fq1 := IFTE(fa == 1,1,0);
   ELSE
    IF fq1 == 1 AND fq2 == -1 THEN fq2 := IFTE(fa == 1,1,0); END;
   END;
  END;
  IF fa == 3 THEN
   IF fq2 <> -1 THEN fq2 := -1; ELSE fq1 := -1; END;
  END;
  IF fa == 6 THEN
   IF fq1 == 0 OR fq2 <> -1 THEN fdone := 1; END;
  END;
  IF fa == 9 THEN fdone := 2; END;
  tz_Drain();
 UNTIL fdone > 0;
 tz_Settle();
 IF fdone == 2 THEN RETURN 0; END;
 IF fq1 == 0 THEN RETURN 3; END;
 RETURN IFTE(fq2 == 1,1,2);
END;

// ---------------- data in ----------------
// A list from the Statistics 1Var app: D1 or D2, {} if it is not there
// (the app never opened) or not a list.
LOCAL tz_ReadD(rdi)
BEGIN
 LOCAL rdl;
 rdl := {};
 IFERR
  IF rdi == 1 THEN rdl := D1; END;
  IF rdi == 2 THEN rdl := D2; END;
 THEN
  rdl := {};
 END;
 IF TYPE(rdl) <> 6 THEN rdl := {}; END;
 RETURN rdl;
END;

// Mean and SD (N - 1) of a list: {mean, SD, N}
LOCAL tz_MSN(ml)
BEGIN
 LOCAL mi, ms, mn, mm;
 mn := SIZE(ml);
 ms := 0;
 FOR mi FROM 1 TO mn DO ms := ms + ml(mi); END;
 mm := ms/mn;
 ms := 0;
 FOR mi FROM 1 TO mn DO ms := ms + (ml(mi)-mm)^2; END;
 RETURN {mm, IFTE(mn > 1,(ms/(mn-1))^.5,0), mn};
END;

// Tails and alpha: two menus.  The alphas are the printed t table's
// columns.  Returns 0 if a menu was cancelled.
LOCAL tz_AskTails()
BEGIN
 LOCAL ak, aa, al, at;
 ak := 0;
 CHOOSE(ak,"Tails",{"Two-tailed: μ₁ ≠ μ₂","One-tailed: μ₁ < μ₂","One-tailed: μ₁ > μ₂"});
 tz_Settle();
 IF ak == 0 THEN RETURN 0; END;
 // the new tails wait here until an alpha is chosen too: a cancelled
 // alpha list leaves tails and alpha as they were (review 1)
 at := IFTE(ak == 1,0,IFTE(ak == 2,-1,1));
 aa := 0;
 IF at == 0 THEN
  CHOOSE(aa,"α (two-tailed)",{"0.20","0.10","0.05","0.02","0.01","0.001"});
  al := {0.20,0.10,0.05,0.02,0.01,0.001};
 ELSE
  CHOOSE(aa,"α (one-tailed)",{"0.10","0.05","0.025","0.01","0.005","0.0005"});
  al := {0.10,0.05,0.025,0.01,0.005,0.0005};
 END;
 tz_Settle();
 IF aa == 0 THEN RETURN 0; END;
 tzTL := at; tzAL := al(aa);
 RETURN 1;
END;

// The numbers for the test chosen.  Returns 1 when they are in and
// usable, 0 if cancelled.
LOCAL tz_AskData()
BEGIN
 LOCAL ds, dok, dl1, dl2, dr1, dr2;
 LOCAL dm, dsd, dn;
 ds := 0;
 IF tzTY <> 3 THEN
  CHOOSE(ds,"The data",{"Type each group's mean, SD, N","Raw scores in D1 and D2"});
 ELSE
  CHOOSE(ds,"The pairs",{"Pairs in D1 (x₁) and D2 (x₂)","Type mean, SD of differences"});
 END;
 tz_Settle();
 IF ds == 0 THEN RETURN 0; END;
 // The source, the names and the example change only once new numbers
 // are in: a cancelled form or a refused D1/D2 read leaves everything
 // as it was (review 2 and 6).  The forms write into the variables
 // themselves only on OK.
 // Every field is {name,[0]}: [0] is the list of types the field
 // allows, and 0 is a real number, so the form refuses a list or a
 // text typed into it (beta 1c: the monkey test typed {1,2} as a
 // mean, and the list went on into the arithmetic).
 IF tzTY <> 3 THEN
  IF ds == 1 THEN
   dok := INPUT({{tzM1,[0]},{tzS1,[0]},{tzN1,[0]},{tzM2,[0]},{tzS2,[0]},{tzN2,[0]}},"Two independent samples",
    {"Mean 1:","SD 1:","N 1:","Mean 2:","SD 2:","N 2:"},
    {"Sample 1 mean","Sample 1 SD (s, with N - 1)","Sample 1 size","Sample 2 mean","Sample 2 SD (s, with N - 1)","Sample 2 size"});
   tz_Settle();
   IF dok == 0 THEN RETURN 0; END;
   // a whole N, as the pairs form takes it (review 7)
   tzN1 := IP(tzN1); tzN2 := IP(tzN2);
   tzSRC := 1;
   tzG1 := "Group 1"; tzG2 := "Group 2";
  ELSE
   dl1 := tz_ReadD(1); dl2 := tz_ReadD(2);
   IF SIZE(dl1) < 2 OR SIZE(dl2) < 2 THEN
    MSGBOX("Put sample 1 in D1 and sample 2 in D2 of the Statistics 1Var app: at least 2 scores in each.");
    tz_Settle();
    RETURN 0;
   END;
   // a value too large to square is refused before any arithmetic
   // (beta 1c, tz_Big), and nothing changes
   IF tz_BigL(dl1) > 0 OR tz_BigL(dl2) > 0 THEN
    MSGBOX(tz_BigLMsg(dl1,dl2));
    tz_Settle();
    RETURN 0;
   END;
   dr1 := tz_MSN(dl1); dr2 := tz_MSN(dl2);
   tzM1 := dr1(1); tzS1 := dr1(2); tzN1 := dr1(3);
   tzM2 := dr2(1); tzS2 := dr2(2); tzN2 := dr2(3);
   tzSRC := 2;
   tzG1 := "D1"; tzG2 := "D2";
  END;
 ELSE
  IF ds == 1 THEN
   dl1 := tz_ReadD(1); dl2 := tz_ReadD(2);
   IF SIZE(dl1) < 2 OR SIZE(dl1) <> SIZE(dl2) THEN
    MSGBOX("Put the pairs in D1 and D2 of the Statistics 1Var app, a pair per row: the same number of values in each, at least 2.");
    tz_Settle();
    RETURN 0;
   END;
   // a value too large to square is refused, and nothing changes
   // (beta 1c, tz_Big)
   IF tz_BigL(dl1) > 0 OR tz_BigL(dl2) > 0 THEN
    MSGBOX(tz_BigLMsg(dl1,dl2));
    tz_Settle();
    RETURN 0;
   END;
   tzX1 := dl1; tzX2 := dl2;
   tzSRC := 2;
   tzG1 := "D1"; tzG2 := "D2";
  ELSE
   dm := tzMD; dsd := tzSDv; dn := tzNP;
   dok := INPUT({{dm,[0]},{dsd,[0]},{dn,[0]}},"Dependent samples",{"Mean xD:","SD sD:","N pairs:"},
    {"Mean of the differences x1 - x2","SD of the differences (N - 1)","Number of pairs"});
   tz_Settle();
   IF dok == 0 THEN RETURN 0; END;
   tzMD := dm; tzSDv := dsd; tzNP := IP(dn);
   tzSRC := 1;
   tzG1 := "Group 1"; tzG2 := "Group 2";
  END;
 END;
 // new numbers: no longer one of the examples
 tzEX := 0; tzDV := "";
 RETURN 1;
END;

 IF ms == 2 THEN
  tz_AskTails();
  tz_Calc();
 END;
 IF ms == 3 THEN tz_Compare(); END;
 IF ms == 4 THEN
  mt := tz_Flow();
  IF mt > 0 THEN
   // The independent numbers serve both independent tests, so pooled
   // <-> separate keeps them; a change to or from dependent samples
   // needs new numbers, and a cancelled form keeps the old test.
   mo := tzTY;
   tzTY := mt;
   IF (mt == 3) <> (mo == 3) THEN
    IF tz_AskData() == 0 THEN tzTY := mo; END;
   END;
   IF tzTY == mt THEN tz_AskTails(); END;
   tz_Calc();
  END;
 END;
 IF ms == 5 THEN
  mt := 0;
  CHOOSE(mt,"Examples",{"Darwin's plants (pairs)","Draft lottery (separate)","Age at arrest (pooled)","Matched jails (pairs)"});
  tz_Settle();
  IF mt > 0 THEN tz_Example(mt); tz_Calc(); END;
 END;
 IF ms == 6 THEN tz_Help(); END;
 IF ms == 7 THEN RETURN 0; END;
 RETURN 1;
END;

// ---------------- the steps ----------------
LOCAL tz_Page(pst)
BEGIN
 IF pst == 1 THEN tz_Step1(); END;
 IF pst == 2 THEN tz_Step2(); END;
 IF pst == 3 THEN tz_Step3(); END;
 IF pst == 4 THEN tz_Step4(); END;
 IF pst == 5 THEN tz_Step5(); END;
END;

// The five steps: soft keys 1-5 or Left/Right move, Menu or Enter opens
// the menu, Esc quits.  Returns when the user quits.
LOCAL tz_Steps()
BEGIN
 LOCAL sst, sk, sm, sm1, sgo, sslt;
 sst := 1; sgo := 1;
 WHILE sgo DO
  IF tzOK == 0 THEN
   // say once why the numbers give no t, then the menu
   IF DIM(tzMSG) > 0 THEN MSGBOX(tzMSG); tz_Settle(); tzMSG := ""; END;
   sgo := tz_Menu();
  ELSE
   tz_Page(sst);
   tz_Settle();
   REPEAT
    sk := GETKEY;
    sm := MOUSE;
    sm1 := sm(1);
   UNTIL sk > -1 OR SIZE(sm1) > 0;
   sslt := 0;
   IF SIZE(sm1) > 0 THEN
    IF sm1(2) >= 218 THEN sslt := MIN(6,IP(sm1(1)/53.4)+1); END;
    tz_TDrain();
   END;
   IF sslt >= 1 AND sslt <= 5 THEN sst := sslt; END;
   IF sk == 7 THEN sst := MAX(1,sst-1); END;
   IF sk == 8 THEN sst := MIN(5,sst+1); END;
   IF sk == 4 THEN sgo := 0; END;
   IF sk == 30 OR sslt == 6 THEN
    tz_Settle();
    sgo := tz_Menu();
   END;
  END;
 END;
END;

// ---------------- main (zero-argument signature) ----------------
EXPORT TTESTLABG()
BEGIN
 LOCAL ms, mt;
 tz_Init();
 // Help returns to this menu (it used to go on to example 1 unasked)
 REPEAT
  ms := 0;
  CHOOSE(ms,"TTestLabG: t tests in 5 steps",{"Choose the test (flowchart)","Example: Darwin's plants","Example: draft lottery","Example: age at arrest","Example: matched jails","Help"});
  tz_Settle();
  IF ms == 6 THEN tz_Help(); END;
 UNTIL ms <> 6;
 IF ms == 0 THEN RETURN "TTestLabG: cancelled"; END;
 IF ms == 1 THEN
  mt := tz_Flow();
  IF mt == 0 THEN RETURN "TTestLabG: cancelled"; END;
  tzTY := mt;
  IF tz_AskData() == 0 THEN RETURN "TTestLabG: cancelled"; END;
  IF tz_AskTails() == 0 THEN RETURN "TTestLabG: cancelled"; END;
 END;
 IF ms >= 2 AND ms <= 5 THEN tz_Example(ms-1); END;
 tz_Calc();
 tz_Steps();
 IF tzOK == 0 THEN RETURN "TTestLabG"; END;
 RETURN "TTestLabG: "+tz_TDf()+" = "+tz_Cm(tz_Fmt(tzT,2))+", p"+tz_Rel(tz_Cm(tz_FmtP(tzP,3)))+", "+tz_TailS();
END;
