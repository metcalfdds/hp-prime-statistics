// ---- pv_Cm: commas in big numbers (round 5, 2 Nov 2026, night) ----
// Roger, 2 Nov 2026: "always put comma delimiters in big numbers".  STRING never
// groups digits, whatever Home's setting, so this inserts a comma every three
// digits in each run of four or more digits that does not follow a decimal
// point: "1234567.25" -> "1,234,567.25", "-98765" -> "-98,765"; "0.00012345",
// "1.8E10" and "1,000" stay as they are.  Only text for the screen comes here,
// never text the CAS or EXPR reads ("1,000" there is two numbers).
// Run on the Virtual Calculator: CommaProbe beta 1, 12 of 12 (2 Nov 2026).
pv_Cm(cmS)
BEGIN
  LOCAL cmO, cmI, cmJ, cmN, cmR, cmK;
  cmN := DIM(cmS);
  IF cmN < 4 THEN RETURN cmS; END;
  cmO := ""; cmI := 1;
  WHILE cmI <= cmN DO
    IF INSTRING("0123456789", MID(cmS, cmI, 1)) > 0 THEN
      // a run of digits from cmI to cmJ
      cmJ := cmI;
      WHILE cmJ < cmN AND INSTRING("0123456789", MID(cmS, cmJ + 1, 1)) > 0 DO cmJ := cmJ + 1; END;
      cmR := MID(cmS, cmI, cmJ - cmI + 1);
      IF DIM(cmR) >= 4 AND (cmI == 1 OR MID(cmS, MAX(1, cmI - 1), 1) <> ".") THEN
        // the first group takes 1 to 3 digits, then a comma before every three
        cmK := DIM(cmR) MOD 3;
        IF cmK == 0 THEN cmK := 3; END;
        cmO := cmO + LEFT(cmR, cmK);
        WHILE cmK < DIM(cmR) DO
          cmO := cmO + "," + MID(cmR, cmK + 1, 3);
          cmK := cmK + 3;
        END;
      ELSE
        cmO := cmO + cmR;
      END;
      cmI := cmJ + 1;
    ELSE
      cmO := cmO + MID(cmS, cmI, 1);
      cmI := cmI + 1;
    END;
  END;
  RETURN cmO;
END;

// 1 for the three tests of means (1, 2 and 5), 0 for the two of
// proportions (3 and 4).  v2.5f asked "TT<=2" for "a means test" and
// "TT>=3" for "a proportion test"; with test 5 both questions go here.
LOCAL pv_MeanT()
BEGIN
 IF TT==1 OR TT==2 OR TT==5 THEN RETURN 1; END;
 RETURN 0;
END;

// The SD that Cohen's d is measured in.  Tests 1 and 2 have one sigma.
// Test 5 has two, and Cohen (1988, p. 44, eq. 2.3.2) uses their root mean
// square, sigma' = sqrt((sigma1^2 + sigma2^2)/2), so d = (mu2 - mu1)/sigma'.
// The arrow-key step (sigma'/20) and the power-vs-effect view use it.
LOCAL pv_SigE()
BEGIN
 // scaled by the larger sigma, so squaring never underflows or overflows
 // the Prime's range (a typed 1E-300 or 1E300; the same in pv_Model)
 IF TT==5 THEN RETURN MAX(SG,SG2)*(((SG/MAX(SG,SG2))^2+(SG2/MAX(SG,SG2))^2)/2)^.5; END;
 RETURN SG;
END;

// Test 5: the size of group 2 when group 1 has nv people.  The student
// types n1 and n2; the program keeps their RATIO (NN2/KN1, the n2 and
// the n1 as typed), so when the arrows, a drag or the Pow-n view change
// n1, n2 follows: n2 = n1 x (typed n2 / typed n1), rounded to a whole
// person, and never below 2.  (G*Power and statsmodels call this the
// allocation ratio.)  n1 x ratio, rounded, never goes down when n1 goes
// up, so power still rises with n1 and the n80 search still works.
LOCAL pv_N2(nv)
BEGIN
 RETURN MAX(2,ROUND(nv*NN2/KN1,0));
END;

// Test 5: the two sample sizes as the readouts write them: "40/20",

// Test 5: take a typed n1 (s5a) and n2 (s5b), from the Edit form or
// the n cell's form.  Each is a whole number from 2 to 5000, and n2 is
// kept between 0.1 and 10 times n1; one box per changed value says
// what was changed and why, as for every other clamp in this program.
// If the pair is what the screen showed (n1 = NN and n2 = pv_N2(NN)),
// nothing changes, so opening a form and pressing OK never nudges the
// ratio by a rounding.  Otherwise the pair becomes the ratio:
// NN2/KN1 = n2/n1.  Returns 1 if it changed anything.
LOCAL pv_Set5(s5a, s5b)
BEGIN
 LOCAL s5c, s5d, s5e;
 s5c:=MIN(5000,MAX(2,IP(s5a)));
 IF s5c<>s5a THEN
  MSGBOX("n₁ must be a whole number from 2 to 5,000 (you entered "+pv_Cm(STRING(s5a,2,2))+"): set to "+pv_Cm(STRING(s5c,1,0)));
 END;
 // n2: a whole number from 2 to 5000 (s5d), then between 0.1 n1 (whole
 // people: rounded up) and 10 n1 (s5e).  One box for n2, naming the
 // rule that changed it.
 s5d:=MIN(5000,MAX(2,IP(s5b)));
 s5e:=MIN(MIN(5000,10*s5c),MAX(MAX(2,CEILING(s5c/10)),s5d));
 IF s5e<>s5b THEN
  IF s5e==s5d THEN
   MSGBOX("n₂ must be a whole number from 2 to 5,000 (you entered "+pv_Cm(STRING(s5b,2,2))+"): set to "+pv_Cm(STRING(s5e,1,0)));
  ELSE
   MSGBOX("n₂ must be 0.1 to 10 times n₁ (you entered n₁ = "+pv_Cm(STRING(s5c,1,0))+", n₂ = "+pv_Cm(STRING(s5b,2,2))+"): n₂ set to "+pv_Cm(STRING(s5e,1,0)));
  END;
 END;
 IF s5c==NN AND s5e==pv_N2(NN) THEN RETURN 0; END;
 NN:=s5c; NN2:=s5e; KN1:=s5c;
 RETURN 1;
END;

// decimal point, no exponent.  Home's STRING() follows the Home
// number format (PyProbe saw grouping marks and a special minus
// sign), which the CAS cannot read, so CAS text is built only from
// this routine.
LOCAL pv_IStr(pv_N)
BEGIN
 LOCAL dS, dV, dD;
 dV := ROUND(ABS(pv_N), 0);
 IF dV == 0 THEN RETURN "0"; END;
 dS := "";
 WHILE dV > 0 DO
  dD := dV MOD 10;
  dS := CHAR(48 + dD) + dS;
  dV := (dV - dD) / 10;
 END;
 IF pv_N < 0 THEN dS := "-" + dS; END;
 RETURN dS;
END;

// The ONE place this program calls the CAS.  It returns the CAS answer as text,
// or "" if the CAS failed, gave undef, an error or an infinity.
// The call form is the one CASProbe beta 1b found on the HP Prime Virtual
// Calculator (build 2026-09-09, 17 Oct 2026): CAS(s) with s a variable hands
// the text back UNEVALUATED, so the text is built into a literal CAS("...")
// that EXPR reads at run time, and the CAS is asked for string(...) of the
// answer, so a big integer crosses into PPL whole, not as a 12-digit real.
LOCAL pv_Cas(pv_S)
BEGIN
 LOCAL cR;
 cR := "";
 IFERR
  cR := STRING(EXPR("CAS(" + CHAR(34) + "string(" + pv_S + ")" + CHAR(34) + ")"));
 THEN
  cR := "";
 END;
 // The answer is a PPL string, and STRING() of it adds one pair of quotes
 // (CASProbe beta 1b T12, emulator): drop that pair.
 IF DIM(cR) >= 2 THEN
  IF LEFT(cR, 1) == CHAR(34) AND MID(cR, DIM(cR), 1) == CHAR(34) THEN
   cR := MID(cR, 2, DIM(cR) - 2);
  END;
 END;
 // undef, an error, or an infinity (CHAR(8734) is the infinity sign; it
 // also catches the Prime's plus-minus infinity, 1/0): no usable answer.
 // An empty answer is already a failure; INSTRING("", ...) is "Invalid input" on
 // the Prime (emulator, 17 Oct 2026), so the checks run only on a non-empty answer
 IF DIM(cR) > 0 THEN
  IF INSTRING(cR, "undef") > 0 OR INSTRING(cR, "rror") > 0 OR INSTRING(cR, CHAR(8734)) > 0 THEN cR := ""; END;
 END;
 RETURN cR;
END;

// ---- helpers for the CAS texts (v2.5) ----

// A typed value as CAS text, a ratio of whole numbers in brackets:
// 15 -> "(15)", 2.5 -> "(25/10)", -5 -> "(-5)", 0.55 -> "(55/100)".
// Every value this program uses is typed, or stepped on a 1-2-5 grid,
// so at most 6 decimals covers it; a value with more returns "" and
// the page leaves the CAS's numbers out (no decimal point may go into
// CAS text).
LOCAL pv_RatS(rsv)
BEGIN
 LOCAL rsd, rss;
 FOR rsd FROM 0 TO 6 DO
  rss := rsv*10^rsd;
  IF ABS(rss - ROUND(rss,0)) < 0.000001 THEN
   IF rsd == 0 THEN RETURN "(" + pv_IStr(rss) + ")"; END;
   RETURN "(" + pv_IStr(rss) + "/" + pv_IStr(10^rsd) + ")";
  END;
 END;
 RETURN "";
END;

// Replace every rpa in rps by rpb (CAS names -> the names on the
// screen).  LEFT is never called with 0 characters (it returns the
// WHOLE string then: emulator, 17 Oct 2026).  v2.5b: INSTRING is asked
// only while text is left -- INSTRING("", ...) is "Invalid input" on
// the Prime, and the text is "" once the pattern ends it -- and an
// empty pattern hands the text back (INSTRING(s, "") is 1, so the
// loop would never end).
LOCAL pv_Rep(rps, rpa, rpb)
BEGIN
 LOCAL rpr, rpt, rpi, rpk;
 IF DIM(rpa) == 0 THEN RETURN rps; END;
 rpr := ""; rpt := rps; rpk := DIM(rpa);
 rpi := 0; IF DIM(rpt) > 0 THEN rpi := INSTRING(rpt, rpa); END;
 WHILE rpi > 0 DO
  IF rpi > 1 THEN rpr := rpr + MID(rpt, 1, rpi - 1); END;
  rpr := rpr + rpb;
  IF rpi + rpk > DIM(rpt) THEN rpt := ""; ELSE rpt := MID(rpt, rpi + rpk, DIM(rpt)); END;
  rpi := 0; IF DIM(rpt) > 0 THEN rpi := INSTRING(rpt, rpa); END;
 END;
 RETURN rpr + rpt;
END;

// A CAS answer without its list wrapping: the Prime answers a solve
// with braces, "{...}" (CASProbe beta 1b T18, emulator 17 Oct 2026);
// desktop Giac writes "list[...]".  Brackets, braces and spaces go;
// a leading "list" (desktop Giac only) goes too.
LOCAL pv_Strip(sts)
BEGIN
 LOCAL stz, sti, stc;
 stz := "";
 FOR sti FROM 1 TO DIM(sts) DO
  stc := MID(sts, sti, 1);
  IF INSTRING("[]{} ", stc) == 0 THEN stz := stz + stc; END;
 END;
 // v2.5b: only of a non-empty stz -- a failed call ("") or an empty
 // answer ("{}") leaves nothing, and INSTRING("", ...) is "Invalid input"
 IF DIM(stz) > 0 THEN
  IF INSTRING(stz, "list") == 1 THEN stz := MID(stz, 5, DIM(stz)); END;
 END;
 RETURN stz;
END;

// 1 if a CAS answer is a usable EXACT formula for n: it still has both
// z symbols, n has been solved away (no "pvn" left, no comma for a
// second root), and it holds no decimal point (a CAS in approximate
// mode would answer "9.0*pvza^2+..."; that is not exact, so it is not
// shown as exact).  Anything else is dropped and the textbook form is
// shown instead, with the badge "decimal".
LOCAL pv_CasUse(cus)
BEGIN
 IF DIM(cus) == 0 THEN RETURN 0; END;
 IF INSTRING(cus, "pvza") == 0 OR INSTRING(cus, "pvzb") == 0 THEN RETURN 0; END;
 IF INSTRING(cus, "pvn") > 0 OR INSTRING(cus, ",") > 0 THEN RETURN 0; END;
 IF INSTRING(cus, ".") > 0 THEN RETURN 0; END;
 RETURN 1;
END;

// The difference the effect is measured by, as the screen names it:
// "(μ₁−μ₀)", "(μ₂−μ₁)" for the two groups, "(p₁−p₀)", "(p₂−p₁)".
LOCAL pv_DName()
BEGIN
 IF TT==1 THEN RETURN "(μ₁−μ₀)"; END;
 IF TT==2 OR TT==5 THEN RETURN "(μ₂−μ₁)"; END;        // v2.5g: test 5 too
 IF TT==3 THEN RETURN "(p₁−p₀)"; END;
 RETURN "(p₂−p₁)";
END;

// A CAS formula for the screen: no list wrapping; the CAS names
// turned into the names on the screen (pvza za, pvzb zb, pvs sigma,
// pvd the difference, pvp/pvr the two proportions, pvm the pooled
// p-hat); sqrt shown with the root glyph (SQG, the one place to
// change it) -- the Prime's CAS already writes the root sign itself
// (CASProbe beta 1b T14b: "√2"), so on the calculator that step
// finds nothing, and only desktop Giac's "sqrt" is changed; * shown
// as the verified times sign.  ^ and / stay.
LOCAL pv_Disp(dss)
BEGIN
 LOCAL dst;
 dst := pv_Strip(dss);
 dst := pv_Rep(dst, "pvza", "zα");
 dst := pv_Rep(dst, "pvzb", "zβ");
 // v2.5g, test 5: two sigmas and the ratio k = n2/n1 (pv_Eq's pvs,
 // pvt, pvk); after this no "pvs" is left for the next line
 IF TT==5 THEN
  dst := pv_Rep(dst, "pvs", "σ₁");
  dst := pv_Rep(dst, "pvt", "σ₂");
  dst := pv_Rep(dst, "pvk", "k");
 END;
 dst := pv_Rep(dst, "pvs", "σ");
 dst := pv_Rep(dst, "pvd", pv_Cm(pv_DName()));
 IF TT==3 THEN
  dst := pv_Rep(dst, "pvp", "p₀");
  dst := pv_Rep(dst, "pvr", "p₁");
 ELSE
  dst := pv_Rep(dst, "pvp", "p₁");
  dst := pv_Rep(dst, "pvr", "p₂");
 END;
 dst := pv_Rep(dst, "pvm", CHAR(57345));
 dst := pv_Rep(dst, "sqrt", SQG);
 dst := pv_Rep(dst, "*", CHAR(215));
 RETURN dst;
END;

// The power equation for n, as CAS text, for the test on screen.
// The program's own model (pv_Model) has SE0 = u/sqrt(n) and
// SE1 = w/sqrt(n) with
//   1 mean      u = w = sigma
//   2 means     u = w = sigma*sqrt(2)          (n per group)
//   1 prop.     u = sqrt(p0(1-p0)),  w = sqrt(p1(1-p1))
//   2 props.    u = sqrt(2 pbar(1-pbar)),  w = sqrt(p1(1-p1) + p2(1-p2))
//   v2.5g, unequal sigma and n:  n2 = k n1, so u = w = sqrt(s1^2 + s2^2/k)
//               (written below as sqrt(k s1^2 + s2^2)/sqrt(k n1))
// and power 1 - beta when H1's centre is zb SE1 past the critical
// line, which is za SE0 past H0's:  (delta - za*SE0)/SE1 = zb.
// The one argument is a list of CAS text: either the symbols {pvs,
// pvd, pvp, pvr, pvm} or the student's numbers as whole-number ratios.  sqrt(pvn) is
// kept apart from the SDs, so the CAS finds the ONE positive root
// (Giac 1.9, 16 Oct 2026: with sqrt(p(1-p)/n) together it returns two
// roots, one false).  No factor(): Giac 1.9's factor gives a WRONG
// answer on the symbolic proportion form.
LOCAL pv_Eq(eqL)
BEGIN
 // eqL = {sigma, delta, p (H0 or group 1), p (H1 or group 2), pooled p}
 // as CAS text; the entries a test does not use may be "".  v2.5g:
 // test 5 adds {..., sigma of group 2, k = n2/n1} as entries 6 and 7.
 LOCAL equ, eqw;
 // v2.5g, test 5.  SE = sqrt(s1^2/n1 + s2^2/(k n1)) is written as
 // sqrt(k s1^2 + s2^2)/sqrt(k n1), with sqrt(k*pvn) apart: desktop Giac
 // 1.9 then answers ((sqrt(k s1^2+s2^2) za + sqrt(k s1^2+s2^2) zb)/
 // delta)^2/k, one positive root.  Written as sqrt(s1^2+s2^2/k)/sqrt(n)
 // it brings in abs(k) and k^2 (31 Oct 2026).
 IF TT==5 THEN
  equ := "sqrt(" + eqL(7) + "*" + eqL(1) + "^2+" + eqL(6) + "^2)";
  RETURN "solve((" + eqL(2) + "-pvza*" + equ + "/sqrt(" + eqL(7) + "*pvn))/(" + equ + "/sqrt(" + eqL(7) + "*pvn))=pvzb,pvn)";
 END;
 IF TT==1 THEN equ := "" + eqL(1); eqw := equ; END;
 IF TT==2 THEN equ := eqL(1) + "*sqrt(2)"; eqw := equ; END;
 IF TT==3 THEN
  equ := "sqrt(" + eqL(3) + "*(1-" + eqL(3) + "))";
  eqw := "sqrt(" + eqL(4) + "*(1-" + eqL(4) + "))";
 END;
 IF TT==4 THEN
  equ := "sqrt(2*" + eqL(5) + "*(1-" + eqL(5) + "))";
  eqw := "sqrt(" + eqL(3) + "*(1-" + eqL(3) + ")+" + eqL(4) + "*(1-" + eqL(4) + "))";
 END;
 RETURN "solve((" + eqL(2) + "-pvza*" + equ + "/sqrt(pvn))/(" + eqw + "/sqrt(pvn))=pvzb,pvn)";
END;

// The closed-form n by PPL (v2.5), from the same z values the plot
// uses: za = z(1 - alpha/tails) (pv_Model's ZC), zb = z(0.80):
//    n = ((za*u + zb*w)/|delta|)^2     (u, w as in pv_Eq)
// Returns -1 when there is no effect (delta = 0: no n is enough).
// Exact for a 1-sided test (the program's power IS
// Phi((delta - za SE0)/SE1)); for a 2-sided test it leaves out the far
// tail, so n80 can be one smaller (the formula page says so).
LOCAL pv_NCF()
BEGIN
 LOCAL nca, ncb, ncu, ncw, ncd, ncp;
 // v2.5g: the mirrored form.  1 - alpha is exactly 1 in 12 digits once
 // alpha is below about 1E-12, and NORMALD_ICDF(0,1,1) is infinity
 // (typed on the G1, 20 Oct 2026); -NORMALD_ICDF(0,1,alpha) keeps its
 // digits.  The same at every other critical z in this file.
 nca := -NORMALD_ICDF(0,1,AL/TL);
 ncb := NORMALD_ICDF(0,1,0.8);
 IF pv_MeanT()==1 THEN
  ncd := MM1-MM0; ncu := SG; ncw := SG;
  IF TT==2 THEN ncu := SG*2^.5; ncw := ncu; END;
  // v2.5g, test 5: u = w = sqrt(s1^2 + s2^2/k), k = NN2/KN1 (the typed
  // ratio), so this is the n1 of the formula; n2 = k n1
  IF TT==5 THEN
   ncp := MAX(SG,SG2);              // scaled, as in pv_Model
   ncu := ncp*((SG/ncp)^2+(SG2/ncp)^2*KN1/NN2)^.5; ncw := ncu;
  END;
 ELSE
  ncd := PP1-PP0;
  IF TT==3 THEN
   ncu := (PP0*(1-PP0))^.5; ncw := (PP1*(1-PP1))^.5;
  ELSE
   ncp := (PP0+PP1)/2;
   ncu := (2*ncp*(1-ncp))^.5; ncw := (PP0*(1-PP0)+PP1*(1-PP1))^.5;
  END;
 END;
 IF ncd == 0 THEN RETURN -1; END;
 // v2.5d (fleet review P1): the formula comes from sqrt(n)*|delta| =
 // za*u + zb*w.  A 1-sided alpha above 0.5 makes za negative, and if the
 // right side is then 0 or below, no sqrt(n) solves it: power is above
 // 0.80 at EVERY n.  Squaring would hide the sign and give a false n
 // (20.2 at alpha 0.999, d = 0.5), so the answer is "none" (-1), as for
 // no effect.
 IF nca*ncu + ncb*ncw <= 0 THEN RETURN -1; END;
 RETURN ((nca*ncu + ncb*ncw)/ABS(ncd))^2;
END;

// The formula's n for the panel: one decimal, or more when one would
// round it UP to the whole number it is just below (304.99 shows as
// "304.99", not "305.0", so the student sees why the answer is 305).
LOCAL pv_NCFS()
BEGIN
 LOCAL ncd;
 ncd := 1;
 WHILE ncd < 4 AND ROUND(NCF,ncd) == CEILING(NCF) AND NCF < CEILING(NCF) - 0.0000001 DO
  ncd := ncd + 1;
 END;
 RETURN pv_FMT(NCF,ncd);
END;

// A number for a DISPLAYED formula, as typed, up to 4 decimals
// (pv_NumS above stops at 3): 0.0005 -> "0.0005".  Display only.
LOCAL pv_NumD(ndv)
BEGIN
 LOCAL ndd, ndk;
 // v2.5l: 8 digits or more in the scientific form, 4 decimals: at the
 // extremes the forms accept the formula page wrote all 50 digits of a
 // sigma of 9.9E49 and ran far off the screen.  Display only (the CAS
 // text is built by pv_IStr and pv_RatS).
 IF ABS(ndv) >= 10000000 THEN RETURN STRING(ndv,3,4); END;
 ndk := 4;
 FOR ndd FROM 0 TO 3 DO
  IF ndk == 4 AND ABS(ndv*10^ndd - ROUND(ndv*10^ndd,0)) < 0.000001 THEN ndk := ndd; END;
 END;
 IF ndk == 0 THEN RETURN pv_IStr(ndv); END;
 // v2.5b: rounded first, so a value under 0.0001 is not printed in
 // scientific form (see pv_FMT)
 RETURN STRING(ROUND(ndv,ndk),2,ndk);
END;

// Long text in lines of at most wrw characters, broken at a space in
// the second half when there is one; continuation lines indented.
LOCAL pv_Wrap(wrs, wrw)
BEGIN
 LOCAL wrl, wrt, wri, wrc, wrb;
 wrl := {}; wrt := wrs;
 WHILE DIM(wrt) > wrw DO
  // the last space in the second half of the line; failing that, the
  // last + , times sign or / (a formula from the CAS has no spaces),
  // so a name such as p2 is never cut in two
  wrc := 0; wrb := 0;
  FOR wri FROM IP(wrw/2) TO wrw DO
   IF INSTRING(MID(wrt, wri, 1), " ") > 0 THEN wrc := wri; END;
   IF INSTRING("+/"+CHAR(215), MID(wrt, wri, 1)) > 0 THEN wrb := wri; END;
  END;
  IF wrc == 0 THEN wrc := wrb; END;
  IF wrc == 0 THEN wrc := wrw; END;
  wrl := CONCAT(wrl, {MID(wrt, 1, wrc)});
  wrt := "    " + MID(wrt, wrc + 1, DIM(wrt));
 END;
 RETURN CONCAT(wrl, {wrt});
END;

// ---------------- small helpers ----------------
// (v2.3: pv_NPDF, a one-line wrapper round NORMALD, is gone.  The
//  draw loop calls NORMALD directly and caches the result.)

// "Nice" number: 1, 2 or 5 x 10^e (Heckbert; same as NormExp)
LOCAL pv_NiceNum(nnval, nnrnd)
BEGIN
 LOCAL nnexp, nnf, nnnf;
 IF nnval <= 0 THEN RETURN 1; END;
 nnexp := FLOOR(LOG(nnval));
 nnf := nnval/10^nnexp;
 IF nnrnd THEN
  nnnf := IFTE(nnf<1.5,1,IFTE(nnf<3,2,IFTE(nnf<7,5,10)));
 ELSE
  nnnf := IFTE(nnf<=1,1,IFTE(nnf<=2,2,IFTE(nnf<=5,5,10)));
 END;
 RETURN nnnf*10^nnexp;
END;

// Effect step for the arrow keys: sigma/20 pushed to 1, 2 or 5.
// N(100,15) steps by 1; N(0,1) by 0.05.  Proportions step 0.01.
// (v2.4: the H1 drag snaps to this same grid.)
LOCAL pv_EffStep()
BEGIN
 // v2.5g: pv_SigE() is sigma, or test 5's sigma' (Cohen's d)
 IF pv_MeanT()==1 THEN RETURN pv_NiceNum(pv_SigE()/20,1); END;
 RETURN 0.01;
END;

// Decimals that show one effect step exactly (1 -> 0, 0.05 -> 2),
// plus two more if mu1 was typed in off the grid (105.5 on a step
// of 1), so an entered value is never shown rounded.
LOCAL pv_EffDec()
BEGIN
 LOCAL edS, edD;
 edS := pv_EffStep();
 edD := MAX(0,-FLOOR(LOG(edS)+0.000001));
 IF ABS(MM1/edS - ROUND(MM1/edS,0)) > 0.000001 THEN edD := edD+2; END;
 RETURN edD;
END;

// v2.5i: decimals for test 5's two means side by side: pv_EffDec's,
// or two more than the step's if group 1's mean (MM0) is off the grid
// too, so a typed 4.8185 shows as 4.819, not 4.8.
LOCAL pv_MuDec()
BEGIN
 LOCAL muS, muD;
 muS := pv_EffStep();
 muD := pv_EffDec();
 IF ABS(MM0/muS - ROUND(MM0/muS,0)) > 0.000001 THEN
  muD := MAX(muD, MAX(0,-FLOOR(LOG(muS)+0.000001))+2);
 END;
 RETURN muD;
END;

// arcsin in RADIANS whatever the Home angle mode (HAngle:
// 0 radians, 1 degrees, 2 gradians).  Needed for Cohen's h.
LOCAL pv_AsinR(av)
BEGIN
 LOCAL ar;
 ar := ASIN(av);
 IF HAngle==1 THEN ar := ar*3.14159265358979/180; END;
 IF HAngle==2 THEN ar := ar*3.14159265358979/200; END;
 RETURN ar;
END;

// sine of an angle given in RADIANS whatever the Home angle mode
// (v2.4; the power-vs-effect view turns h back into p with it).
LOCAL pv_SinR(srad)
BEGIN
 IF HAngle==1 THEN RETURN SIN(srad*180/3.14159265358979); END;
 IF HAngle==2 THEN RETURN SIN(srad*200/3.14159265358979); END;
 RETURN SIN(srad);
END;

// Standardized effect size: Cohen's d (means) or h (proportions)
LOCAL pv_EffSz()
BEGIN
 // v2.5g: test 5 divides by sigma' = sqrt((s1^2+s2^2)/2) (pv_SigE)
 IF pv_MeanT()==1 THEN RETURN (MM1-MM0)/pv_SigE(); END;
 RETURN 2*pv_AsinR(SQRT(PP1)) - 2*pv_AsinR(SQRT(PP0));
END;

// Cohen's word for a standardized effect (v2.4).  Benchmarks for d
// and h alike: 0.2 small, 0.5 medium, 0.8 large; below 0.2 "tiny".
// The caller passes the value ROUNDED as it is shown, so "0.20" never
// reads "tiny".
LOCAL pv_EffWord(ewv)
BEGIN
 LOCAL ewa;
 ewa := ABS(ewv);
 IF ewa >= 0.8 THEN RETURN "large"; END;
 IF ewa >= 0.5 THEN RETURN "medium"; END;
 IF ewa >= 0.2 THEN RETURN "small"; END;
 RETURN "tiny";
END;

// A typed-in value printed as typed: 15 -> "15", 2.5 -> "2.5",
// 0.125 -> "0.125" (at most 3 decimals).  For the SE teaching line.
LOCAL pv_NumS(nsv)
BEGIN
 LOCAL nsd, nsk, nsr;
 nsk := 3;
 FOR nsd FROM 0 TO 2 DO
  IF nsk == 3 AND ABS(nsv*10^nsd - ROUND(nsv*10^nsd,0)) < 0.000001 THEN nsk := nsd; END;
 END;
 nsr := ROUND(nsv,0);
 IF nsk == 0 THEN RETURN STRING(nsr,1,0); END;
 // v2.5b: rounded first, so a value under 0.001 is not printed in
 // scientific form (see pv_FMT)
 RETURN STRING(ROUND(nsv,nsk),2,nsk);
END;

// 1 if two lists hold the same numbers, 0 if not (v2.4: the cache
// keys).  A plain loop with a flag -- no list comparison operator.
LOCAL pv_Same(sma, smb)
BEGIN
 LOCAL smi, smr;
 IF SIZE(sma) <> SIZE(smb) THEN RETURN 0; END;
 smr := 1;
 FOR smi FROM 1 TO SIZE(sma) DO
  IF sma(smi) <> smb(smi) THEN smr := 0; END;
 END;
 RETURN smr;
END;

// Empty the key buffer: the Enter that closed a menu or form
// must not be read as the next keypress (NormExp ZPanel fix).
LOCAL pv_Drain()
BEGIN
 LOCAL dk;
 REPEAT dk := GETKEY; UNTIL dk == -1;
END;

// Wait for the finger to lift -- but never for more than a second.
// MOUSE(1) is -1 while nothing touches the screen.  Without the
// time limit a stuck or phantom touch would hold the program here
// for ever, and the only way out would be a reset.  (ChiTable's
// csDrain, device-verified; v2.3.)
LOCAL pv_TDrain()
BEGIN
 LOCAL drT;
 drT := TICKS;                                  // milliseconds
 REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Wait for any key OR any tap (drain, wait, drain).  A key or a
// finger left over from a menu would otherwise skip the screen
// just drawn, and the key that ends the wait must not reach the
// live loop.  (ChiTable's csWaitAny, device-verified; v2.3.)
LOCAL pv_WaitAny()
BEGIN
 LOCAL waK, waM, waM1;
 REPEAT waK := GETKEY; UNTIL waK == -1;
 pv_TDrain();
 REPEAT
  waK := GETKEY;
  waM := MOUSE;
  waM1 := waM(1);
 UNTIL waK > -1 OR SIZE(waM1) > 0;
 pv_TDrain();
 REPEAT waK := GETKEY; UNTIL waK == -1;
END;

// Fit the window around both curves: +/- 6 SE, peak at 62% of
// the plot height, so there is room to explore before a re-fit.
LOCAL pv_FitWin(fsmx, fpk)
BEGIN
 XMN := MIN(M0v,M1v) - 6*fsmx;
 XMX := MAX(M0v,M1v) + 6*fsmx;
 YSC := (PYB-PYT)*0.62/fpk;
END;

// v2.5b: rounded first.  On the Prime STRING(x, 2, d) prints
// scientific ("9.6E-2") whenever 0 < |x| < 10^-d (emulator, 17 Oct
// 2026); after ROUND(x, d) the value is 0 or at least 10^-d, so the
// fixed form holds: a tiny effect size reads 0.00, not 6.67E-5.
LOCAL pv_FMT(xx, dd)
BEGIN
 RETURN STRING(ROUND(xx,dd),2,dd);         // Fixed mode: keeps trailing zeros
END;

// pv_FMT with a floor, for PROBABILITIES only. At dd decimals anything smaller
// than 10^-dd prints as a row of zeros -- "β = 0.000" reads as "zero chance of
// a Type II error", which is never true. Reports "< 0.001" instead.
// Deliberately NOT folded into pv_FMT: pv_FMT also formats axis tick labels (see the
// tick loop), and "< 0.001" printed along an axis would look broken. Use pv_FMTP
// for power/β/α, plain pv_FMT for ticks, means, effect sizes, zcrit and SE.
LOCAL pv_FMTP(xx, dd)
BEGIN
 LOCAL thr;
 thr := 10^(-dd);
 IF xx <> 0 AND ABS(xx) < thr THEN        // v2.5i: <>, not != (header)
  IF xx > 0 THEN RETURN "< " + STRING(thr,2,dd); END;
  RETURN "> -" + STRING(thr,2,dd);
 END;
 // v2.5d: the top floor, the mirror of the one above.  Power (and beta)
 // is never 1 under a normal model, but a power of 0.9996 rounds to
 // "1.000", and at a large n the power IS exactly 1 in the calculator's
 // 12 digits (1 minus a tail below 5E-13).  Either way it would print
 // "POWER 1.000" beside "β < 0.001".  So a probability that would print
 // as 1 prints "> 0.999" (dd = 3), "> 0.9999" (dd = 4) instead.
 IF xx <= 1 AND ROUND(xx, dd) >= 1 THEN RETURN "> " + STRING(1 - thr,2,dd); END;
 RETURN STRING(xx,2,dd);
END;

// v2.5c: " = 0.050", or " < 0.001" when pv_FMTP has already written the
// relation: "α = < 0.001" said two things at once; "α < 0.001" says the
// one that is true.  The first character decides, read only when there
// is one (DIM first: an empty string has no first character).
LOCAL pv_Rel(rlS)
BEGIN
 IF DIM(rlS) > 0 THEN
  IF LEFT(rlS, 1) == "<" OR LEFT(rlS, 1) == ">" THEN RETURN " " + rlS; END;
 END;
 RETURN " = " + rlS;
END;

// v2.5c: beta -- the chance the test does NOT reject when the effect is
// real -- as a probability of its own, P(XCL < X < XCU) with X ~
// N(M1v, SE1), from the critical values pv_Model set for PW.  v2.5b
// printed 1 - PW, which is exactly 0 once PW rounds to 1 (n 5000, d
// 0.67: "β = 0.000").  When the interval lies right of M1v it is
// mirrored about M1v, so both CDFs are lower tails and a tiny beta keeps
// its digits.  Beta is never 0 under a normal model, so an exact 0 (a
// tail below what NORMALD_CDF can hold) is shown as the floor.
// Returns the text, with pv_FMTP's floor: "0.428", "< 0.001".
LOCAL pv_BetaS(bsD)
BEGIN
 LOCAL bsV;
 IF (XCL+XCU)/2 >= M1v THEN
  bsV := NORMALD_CDF(M1v,SE1,2*M1v-XCL)-NORMALD_CDF(M1v,SE1,2*M1v-XCU);
 ELSE
  bsV := NORMALD_CDF(M1v,SE1,XCU)-NORMALD_CDF(M1v,SE1,XCL);
 END;
 IF bsV <= 0 THEN RETURN "< "+STRING(10^(-bsD),2,bsD); END;
 RETURN pv_FMTP(bsV,bsD);
END;

// ---------------- the engine ----------------
// v2.4: the first half of v2.3's pv_Setup, with n passed in, so the
// main view, the n80 search and both curve views all compute power
// with ONE set of formulas.  Sets M0v, M1v, SE0, SE1, DIRN, ZC, XCL,
// XCU and PW for n = nv per group; touches nothing else.
LOCAL pv_Model(nv)
BEGIN
 LOCAL pb;

 CASE
  IF TT==1 THEN                  // one-sample mean (z)
   M0v:=MM0; M1v:=MM1;
   SE0:=SG/nv^.5; SE1:=SE0;
  END;
  IF TT==2 THEN                  // two-sample means (z, common sigma)
   M0v:=0; M1v:=MM1-MM0;
   SE0:=SG*(2/nv)^.5; SE1:=SE0;
  END;
  IF TT==3 THEN                  // one proportion
   M0v:=PP0; M1v:=PP1;
   SE0:=(PP0*(1-PP0)/nv)^.5;
   SE1:=(PP1*(1-PP1)/nv)^.5;
  END;
  IF TT==4 THEN                  // two proportions (pooled null SE)
   pb:=(PP0+PP1)/2;
   M0v:=0; M1v:=PP1-PP0;
   SE0:=(2*pb*(1-pb)/nv)^.5;
   SE1:=(PP0*(1-PP0)/nv+PP1*(1-PP1)/nv)^.5;
  END;
  IF TT==5 THEN                  // v2.5g: two means, unequal sigma and n (z)
   // each group its own sigma and n; n = nv is group 1's and group 2
   // has pv_N2(nv).  SE = sqrt(s1^2/n1 + s2^2/n2), the same under H0
   // and H1: the sigmas do not change when the means do.  Worked out
   // as m x sqrt((s1/m)^2/n1 + (s2/m)^2/n2), m the larger sigma, so a
   // typed sigma of 1E-300 or 1E300 never squares past the Prime's
   // range (1E-499 to 1E499) into 0 or infinity.
   pb:=MAX(SG,SG2);
   M0v:=0; M1v:=MM1-MM0;
   SE0:=pb*((SG/pb)^2/nv+(SG2/pb)^2/pv_N2(nv))^.5; SE1:=SE0;
  END;
 END;

 DIRN:=SIGN(M1v-M0v);
 IF DIRN==0 THEN DIRN:=1; END;

 ZC:=-NORMALD_ICDF(0,1,AL/TL);          // v2.5g: mirrored (see pv_NCF)
 IF TL==2 THEN
  XCL:=M0v-ZC*SE0; XCU:=M0v+ZC*SE0;
  PW:=NORMALD_CDF(M1v,SE1,XCL)+1-NORMALD_CDF(M1v,SE1,XCU);
 ELSE
  IF DIRN>0 THEN
   XCU:=M0v+ZC*SE0; XCL:=-1E10;
   PW:=1-NORMALD_CDF(M1v,SE1,XCU);
  ELSE
   XCL:=M0v-ZC*SE0; XCU:=1E10;
   PW:=NORMALD_CDF(M1v,SE1,XCL);
  END;
 END;
END;

// n80 (v2.4): the smallest whole n per group, 2 to 5000, at which
// power reaches 0.80, from the SAME engine as the plot.  Returns 0
// when there is no effect (power stays at alpha for every n) and -1
// when even n = 5000 falls short (the cap).  Power rises with n in
// all five tests (test 5: n is n1, and n2 follows it, see pv_N2), so
// halving the interval finds it in about 13
// engine calls; stepping n one at a time could take 5000, and this
// runs again whenever the effect or alpha changes -- on every drag
// sample.  Leaves the scene as it was (pv_Model(NN) at the end).
LOCAL pv_N80()
BEGIN
 LOCAL nlo, nhi, nmd, nres;
 pv_Model(5000);
 IF M1v == M0v THEN
  nres := 0;
 ELSE
  IF PW < 0.8 THEN
   nres := -1;
  ELSE
   pv_Model(2);
   IF PW >= 0.8 THEN
    nres := 2;
   ELSE
    nlo := 2; nhi := 5000;        // power(nlo) < 0.80 <= power(nhi)
    WHILE nhi - nlo > 1 DO
     nmd := IP((nlo+nhi)/2);
     pv_Model(nmd);
     IF PW >= 0.8 THEN nhi := nmd; ELSE nlo := nmd; END;
    END;
    nres := nhi;
   END;
  END;
 END;
 pv_Model(NN);
 RETURN nres;
END;

// ---------------- model setup ----------------
LOCAL pv_Setup()
BEGIN
 LOCAL smx, pk, esc, nk;

 // n80 depends on everything except n: recompute it only when one
 // of those inputs changed since the last time (v2.4).
 nk := {TT,TL,AL,MM0,MM1,SG,PP0,PP1,SG2,NN2,KN1};   // v2.5g: test 5's three
 IF pv_Same(N8K,nk) == 0 THEN
  N8K := nk;
  N80 := pv_N80();
  // v2.5: the formula's n has the same inputs (it does not depend on
  // n either), so it is worked out at the same moment -- by PPL, a
  // few operations, never by the CAS (the CAS answers events, and a
  // drag of H1 or alpha comes through here on every sample)
  NCF := pv_NCF();
 END;

 pv_Model(NN);

 smx:=MAX(SE0,SE1);
 pk:=MAX(1/(SE0*C2PI),1/(SE1*C2PI));
 // The window is HELD between keypresses (NormExp Explorer lesson):
 // if it re-fitted every time, H1 would never appear to move and a
 // larger n would never appear to narrow the curves.  Re-fit only
 // on request (WFIT) or when the scene would leave the frame.
 esc := 0;
 IF MIN(M0v,M1v)-3*smx < XMN THEN esc := 1; END;
 IF MAX(M0v,M1v)+3*smx > XMX THEN esc := 1; END;
 IF pk*YSC > (PYB-PYT)*0.97 THEN esc := 1; END;
 IF WHLD == 1 THEN
  // v2.4: a finger is dragging.  The x-window must not jump under
  // the finger, so it stays exactly as it is even if a curve runs
  // off the edge; only the HEIGHT is re-fitted, and only if the peak
  // would clip (a larger n makes taller curves).  The full re-fit
  // waits for the lift (WHLD back to 0).
  IF pk*YSC > (PYB-PYT)*0.97 THEN YSC := (PYB-PYT)*0.62/pk; END;
 ELSE
  IF WFIT==1 OR esc==1 THEN pv_FitWin(smx,pk); END;
  WFIT := 0;
 END;
END;

// d and h are the STANDARDIZED effect: Cohen's d = (mu1-mu0)/sigma,
// Cohen's h = 2asin(sqrt p1) - 2asin(sqrt p0).  Benchmarks for both:
// 0.2 small, 0.5 medium, 0.8 large.  v2.4: the word is printed too,
// "d = 0.33, small", from the value as shown (2 decimals).
// v2.5l: the width of a line of font-1 text in pixels: about 5.5 px a
// character and 3 px for a decimal point (Lessons Learned v7zi s6: "0.03"
// is 20 px on the Virtual Calculator).
LOCAL pv_TxW(twS)
BEGIN
 LOCAL twI, twW;
 twW:=0;
 FOR twI FROM 1 TO DIM(twS) DO
  twW:=twW+IFTE(MID(twS,twI,1)==".",3,5.5);
 END;
 RETURN twW;
END;

// v2.5l: a readout's text, or the same number in the scientific form
// with as many decimals as fit, when the text is wider than its place
// (ftW px): at the extremes the forms accept, "-1000000.00" ran into the
// next column's "-999999.00".  At everyday values the text is returned
// unchanged.
LOCAL pv_Fit(ftS, ftV, ftW)
BEGIN
 LOCAL ftK;
 IF pv_TxW(ftS)<=ftW THEN RETURN ftS; END;
 ftK:=6;
 REPEAT
  ftK:=ftK-1;
  ftS:=STRING(ftV,3,ftK);
 UNTIL ftK==0 OR pv_TxW(ftS)<=ftW;
 RETURN ftS;
END;

 dlw := ", "+pv_EffWord(dlv);

END;

// The SE teaching line (v2.4): the null curve's standard error with
// the numbers put in, e.g. "SE₀ = σ/√n = 15/√30 = 2.74".  Decimals:
// two, or more for a small SE so three figures show (0.0500).
// v2.5g: slx is the x where the n80 text starts on the same row; test
// 5's line is longer, and gives way to it (the numbers if they fit,
// else the symbols, else the value alone).  Tests 1-4 ignore slx.
LOCAL pv_SELine(slx)
BEGIN
 LOCAL sdd, sqg2, pbl, nst, sel, ses;
 sdd := MAX(2, 2-FLOOR(LOG(SE0)));
 sqg2 := SQG;
 nst := pv_Cm(STRING(NN,1,0));
 // v2.5l: tests 1-4 give way to the n80 text too: the symbols with the
 // value, else the value alone (sigma = 1,000,000 ran "SE0 = sigma/sqrt
 // n = 1000000/sqrt 30 = ..." into "n80 = 2474").  Measured at 5.5 px
 // for every character, decimal points included, as the overlap check
 // does: by pv_TxW (3 px a point) test 4's "SE0 = sqrt(2x0.975x0.025/
 // 4990) = 0.00313" fitted with 8 px to spare, and the check saw it
 // reach "n80 = 3826" (round 4's label check under random use).
 sel := ""; ses := "";
 CASE
  IF TT==1 THEN
   sel := "SE₀ = σ/"+sqg2+"n = "+pv_Cm(pv_NumS(SG))+"/"+sqg2+nst+" = "+pv_Cm(pv_FMT(SE0,sdd));
   ses := "SE₀ = σ/"+sqg2+"n = "+pv_Cm(pv_FMT(SE0,sdd));
  END;
  IF TT==2 THEN
   sel := "SE₀ = σ"+sqg2+"(2/n) = "+pv_Cm(pv_NumS(SG))+sqg2+"(2/"+nst+") = "+pv_Cm(pv_FMT(SE0,sdd));
   ses := "SE₀ = σ"+sqg2+"(2/n) = "+pv_Cm(pv_FMT(SE0,sdd));
  END;
  IF TT==3 THEN
   sel := "SE₀ = "+sqg2+"("+pv_Cm(pv_NumS(PP0))+CHAR(215)+pv_Cm(pv_NumS(1-PP0))+"/"+nst+") = "+pv_Cm(pv_FMT(SE0,sdd));
  END;
  IF TT==4 THEN
   pbl := (PP0+PP1)/2;           // the pooled proportion under H0
   sel := "SE₀ = "+sqg2+"(2"+CHAR(215)+pv_Cm(pv_NumS(pbl))+CHAR(215)+pv_Cm(pv_NumS(1-pbl))+"/"+nst+") = "+pv_Cm(pv_FMT(SE0,sdd));
  END;
  IF TT==5 THEN
   // v2.5g: "SE₀ = √(10²/40+20²/20) = 4.74"; 6.5 px a character
   // (the overlap check's wide estimate) must end 2 px short of slx
   sel := "SE₀ = "+SQG+"("+pv_Cm(pv_NumS(SG))+"²/"+nst+"+"+pv_Cm(pv_NumS(SG2))+"²/"+pv_Cm(STRING(pv_N2(NN),1,0))+") = "+pv_Cm(pv_FMT(SE0,sdd));
   IF 8+6.5*DIM(sel) > slx-2 THEN sel := "SE₀ = "+SQG+"(σ₁²/n₁+σ₂²/n₂) = "+pv_Cm(pv_FMT(SE0,sdd)); END;
   IF 8+6.5*DIM(sel) > slx-2 THEN sel := "SE₀ = "+pv_Cm(pv_FMT(SE0,sdd)); END;
   RETURN sel;
  END;
 END;
 IF DIM(ses) > 0 AND 8+5.5*DIM(sel) > slx-2 THEN sel := ses; END;
 IF DIM(sel) > 0 AND 8+5.5*DIM(sel) > slx-2 THEN sel := "SE₀ = "+pv_Cm(pv_FMT(SE0,sdd)); END;
 RETURN sel;
END;

// ---------------- alpha ladder ----------------
LOCAL pv_SyncA()
BEGIN
 LOCAL i2, bd:=1E10, dv;
 FOR i2 FROM 1 TO SIZE(ALAD) DO
  dv:=ABS(ALAD(i2)-AL);
  IF dv<bd THEN bd:=dv; AIDX:=i2; END;
 END;
END;

// One rung up (asd > 0) or down the ladder (v2.4: used by +/- and the
// two alpha soft keys).  From a rung this is the old AIDX+1 / AIDX-1;
// from a custom alpha (set by dragging the critical line) it goes to
// the next rung above or below it.  Stops at the ends.
LOCAL pv_AStep(asd)
BEGIN
 LOCAL i2, anx;
 anx := 0;
 IF asd > 0 THEN
  FOR i2 FROM 1 TO SIZE(ALAD) DO
   IF anx == 0 AND ALAD(i2) > AL*1.000001 THEN anx := i2; END;
  END;
  IF anx == 0 THEN anx := SIZE(ALAD); END;
 ELSE
  FOR i2 FROM 1 TO SIZE(ALAD) DO
   IF ALAD(i2) < AL*0.999999 THEN anx := i2; END;
  END;
  IF anx == 0 THEN anx := 1; END;
 END;
 AIDX := anx;
 AL := ALAD(anx);
END;

// The rung nearest the current alpha, measured RELATIVE to the rung
// (0.048 is 4 % from 0.05; 0.037 is 26 % from 0.05).
LOCAL pv_ANear()
BEGIN
 LOCAL i2, abr, abi, arr;
 abr := 1E10; abi := 1;
 FOR i2 FROM 1 TO SIZE(ALAD) DO
  arr := ABS(AL-ALAD(i2))/ALAD(i2);
  IF arr < abr THEN abr := arr; abi := i2; END;
 END;
 RETURN abi;
END;

// The status line while the critical line is dragged.
LOCAL pv_ADrag()
BEGIN
 LOCAL abi;
 abi := pv_ANear();
 IF ABS(AL-ALAD(abi))/ALAD(abi) <= 0.15 THEN
  RETURN "α = "+pv_Cm(pv_FMTP(AL,3))+" (lift: snaps to "+pv_Cm(pv_FMTP(ALAD(abi),3))+")";
 END;
 RETURN "α = "+pv_Cm(pv_FMTP(AL,3))+" (off the ladder)";
END;

// On lift after dragging the critical line: snap to the nearest rung
// if it is within 15 %, otherwise keep the custom alpha -- and say
// which on the status line.
LOCAL pv_ASnap()
BEGIN
 LOCAL abi;
 abi := pv_ANear();
 IF ABS(AL-ALAD(abi))/ALAD(abi) <= 0.15 THEN
  AL := ALAD(abi); AIDX := abi;
  HNT := "α = "+pv_Cm(pv_FMTP(AL,3))+": snapped to the ladder";
 ELSE
  pv_SyncA();
  HNT := "α = "+pv_Cm(pv_FMTP(AL,3))+" kept as a custom value (off the ladder)";
 END;
 HNC := ccrt;
END;

// The alpha list (menu item "Set α", and a tap on α or zcrit).
// CHOOSE pattern: index set to 0 first, the index tested after.
LOCAL pv_AChoose()
BEGIN
 LOCAL asel;
 asel:=0;
 CHOOSE(asel,"Alpha",
  {"0.001","0.005","0.01","0.02","0.05","0.10","0.20"});
 IF asel>0 THEN AIDX:=asel; AL:=ALAD(AIDX); END;
END;

// Toggle 1-/2-sided, and set up the one-frame callout that names
// what changed (v2.4): where alpha went and what the critical z is.
LOCAL pv_Tails()
BEGIN
 LOCAL tzo, tzn;
 tzo := -NORMALD_ICDF(0,1,AL/TL);        // v2.5g: mirrored (see pv_NCF)
 TL := 3-TL;
 tzn := -NORMALD_ICDF(0,1,AL/TL);
 IF TL == 2 THEN
  // v2.5c: pv_Rel -- a typed alpha of 0.0001 reads "α < 0.001", not "= < 0.001"
  HB1 := "2-sided: α"+pv_Rel(pv_Cm(pv_FMTP(AL,3)))+" is split, "+
         pv_Cm(pv_FMTP(AL/2,IFTE(AL/2<0.01,4,3)))+" in each tail,";
 ELSE
  // v2.5c: pv_Rel, as above
  HB1 := "1-sided: all of α"+pv_Rel(pv_Cm(pv_FMTP(AL,3)))+" sits in one tail,";
 END;
 HB2 := "so the critical z is "+pv_Cm(pv_FMT(tzn,3))+" instead of "+pv_Cm(pv_FMT(tzo,3))+".";
END;

// Ask for the ghost (v2.4): the next frame that recomputes H1 first
// copies the H1 it is about to overwrite into CYG.  Called before an
// arrow step and when a drag of H1 or of n begins.
LOCAL pv_GSave()
BEGIN
 GREQ := 1;
END;

// v2.5i: the degrees of freedom of the t-test that goes with the test
// on screen (Roger, 31 Oct 2026: "we need to put in the df also -- even
// the fractional one").  PowerViz works with z, so df changes no number
// here; it is shown so the screen can be read beside SPSS's.
//   test 1  the one-sample t-test: n - 1
//   test 2  the pooled t-test: 2n - 2
//   test 5  Welch's t-test: the Welch-Satterthwaite df, fractional,
//           (a + b)^2 / (a^2/(n1 - 1) + b^2/(n2 - 1)), a = s1^2/n1 and
//           b = s2^2/n2, from the sigmas and n's typed.  SPSS's "Equal
//           variances not assumed" row uses the sample SDs the same way,
//           so typing a sample's SDs and n's gives SPSS's df.  The
//           sigmas are divided by the larger first (as in pv_Model):
//           df does not change, and nothing squares out of range.
// -1 for the proportions (tests 3 and 4): z tests, no df.
LOCAL pv_DF()
BEGIN
 LOCAL dfa, dfb, dfm, dfk;
 IF TT==1 THEN RETURN NN-1; END;
 IF TT==2 THEN RETURN 2*NN-2; END;
 IF TT==5 THEN
  dfm := MAX(SG,SG2);
  dfk := pv_N2(NN);
  dfa := (SG/dfm)^2/NN;
  dfb := (SG2/dfm)^2/dfk;
  RETURN (dfa+dfb)^2/(dfa^2/(NN-1)+dfb^2/(dfk-1));
 END;
 RETURN -1;
END;

// v2.5j: THE TEST STATISTIC OBTAINED (Roger, 31 Oct 2026: "we didn't
// put in t obtained!"), as "t(265.9) = 4.35" or "z = 2.00": the t (or
// z) a sample would give if its means (or proportions), SDs and n's
// were exactly the ones typed.  It is (M1v - M0v)/SE0 in every test,
// the H1 centre in SE0 units, because SE0 is the very standard error
// that test divides by: sigma/sqrt(n) (test 1), sigma sqrt(2/n) (test
// 2), sqrt(p0 q0/n) (test 3), the pooled sqrt(2 pbar qbar/n) (test 4)
// and sqrt(s1^2/n1 + s2^2/n2), Welch's (test 5).  With Roger's SPSS
// numbers it is 4.35 on 265.9 df: SPSS's -4.351 on 265.924, the sign
// turned because PowerViz takes group 2 minus group 1.
LOCAL pv_TObt()
BEGIN
 LOCAL toV, toD;
 toV := (M1v-M0v)/SE0;
 toD := pv_DF();
 IF toD < 0 THEN RETURN "z = "+pv_FMT(toV,2); END;
 RETURN "t("+IFTE(TT==5,pv_FMT(toD,1),STRING(toD,1,0))+") = "+pv_FMT(toV,2);
END;

// v2.5j: THE READOUT BOX, every test (Roger, 31 Oct 2026: "display mu 1
// also", "we need to see the std dev's too ... rework the output box",
// "the other tests should have both groups too", "we need to put in
// the df also", "we didn't put in t obtained!").  y 150-217, rows of
// the small font (font 1) every 13 px at y 152, 165, 178, 191 and 204:
//   left    two groups (tests 2, 4, 5): a table, a column per group,
//           rows n, then mu and sigma or p.  Test 2's n and sigma are
//           the same in both columns: that is what "equal sigma and n"
//           means.  One sample (tests 1, 3): n, mu0 and mu1 (or p0 and
//           p1), sigma, one value each.
//   middle  d or h (magenta, as the arrow), zcrit, alpha, beta
//   right   POWER, large, coloured by size; under it the test
//           statistic obtained with its df, "t(265.9) = 4.35" (pv_TObt)
//   bottom  the SE line, and n80 pulled left as long as it needs
// Each number goes in its own TEXTOUT_P at a fixed x, so the columns

// ---------------- rendering: main view ----------------
LOCAL pv_Draw()
BEGIN
 // colours come from the file-level palette set in pv_Init (v2.3)
 LOCAL cpwr, nd0, nd1, cix, lbs;
 LOCAL pxx, vx, y0, y1, rej, span;
 LOCAL py0, py1, pv0, pv1, hg, lw;
 LOCAL cv, cpx, yy, xv, seg, gcp;
 LOCAL rw, ex, ff, stp, dec, tk;
 LOCAL bA, bX, bY, bW, pA, pX;
 LOCAL pY, pWd, lx, ly, ytp, ybt;   // v2.5: pW -> pWd (PW is the power; see CHANGES)
 LOCAL ar0, ar1, ary, arl, ars;     // v2.5h: the d arrow
 LOCAL tlS, tlX, tlR, hx0, hx1;     // v2.5l: a tick label, its x, the last one's right end; H0/H1's x

 span:=XMX-XMN;

 // v2.4: a clean white G1 every frame (the fleet's verified clear)
 DIMGROB_P(G1,320,240,cwht);

 // title (blue)
 TEXTOUT_P("PowerViz  "+pv_TName()+"  "+pv_Cm(STRING(TL,1,0))+"-sided",G1,4,2,2,cnul);

 // status row: the legend chips, or (for one frame) a hint
 IF DIM(HNT) > 0 THEN
  TEXTOUT_P(pv_Cm(HNT),G1,6,19,1,HNC);
 ELSE
  RECT_P(G1,12,20,24,28,calp);
  TEXTOUT_P("α",G1,27,19,1,cgry);
  RECT_P(G1,70,20,82,28,cbet);
  TEXTOUT_P("β",G1,85,19,1,cgry);
  RECT_P(G1,120,20,132,28,cpow);
  TEXTOUT_P("power",G1,135,19,1,cgry);
  TEXTOUT_P("crit",G1,236,19,1,ccrt);
  // short dashed red sample after "crit" (matches the critical line style)
  FOR cv FROM 0 TO 3 DO
   cpx:=258+cv*7;
   LINE_P(G1,cpx,24,cpx+3,24,ccrt);
  END;
 END;

 // ----- pv_NiceNum step (computed early so the grid can use it) -----
 rw:=span/5;
 ex:=FLOOR(LOG(rw));
 ff:=rw/10^ex;
 stp:=IFTE(ff<1.5,1,IFTE(ff<3,2,IFTE(ff<7,5,10)))*10^ex;
 dec:=MAX(0,-FLOOR(LOG(stp)));

 // ----- vertical gridlines at the ticks (drawn first, under fills) -----
 tk:=CEILING(XMN/stp)*stp;
 WHILE tk<=XMX DO
  cpx:=PXL+IP((tk-XMN)/span*(PXR-PXL));
  LINE_P(G1,cpx,PYT,cpx,PYB-1,cgrd);
  tk:=tk+stp;
 END;

 // ----- horizontal gridlines (drawn first, under fills) -----
 FOR cv FROM 1 TO 4 DO
  yy:=PYB-IP(cv/5*(PYB-PYT));
  LINE_P(G1,PXL,yy,PXR,yy,cgrd);
 END;

 // ----- which cached curves are still good? (v2.4) -----
 // A curve's cache is reused when its centre, SE and the window are
 // exactly what they were when it was filled: H0 while H1 is dragged,
 // both curves while alpha is dragged -- no NORMALD calls at all.
 nd0 := 1-pv_Same(CK0,{M0v,SE0,XMN,XMX,YSC});
 nd1 := 1-pv_Same(CK1,{M1v,SE1,XMN,XMX,YSC});
 // the ghost copy happens in this frame if one was asked for and H1
 // is about to be overwritten; it records the window H1 was drawn in
 gcp := 0;
 IF GREQ==1 AND nd1==1 THEN
  gcp := 1;
  IF SIZE(CK1) >= 5 THEN GWIN := {CK1(3),CK1(4),CK1(5)}; ELSE GWIN := {0,0,0}; END;
 END;

 // area-label sums (v2.4): beta (yellow) and the visible green
 bA:=0; bX:=0; bY:=0; bW:=0;
 pA:=0; pX:=0; pY:=0; pWd:=0;

 // ----- pass 1: shaded fills, pixel column by pixel column -----
 // v2.3: each curve's height is evaluated ONCE per column here and
 // kept in CY0 (H0) and CY1 (H1), one entry per pixel column, so
 // pass 2 reads them back instead of calling NORMALD again.  That
 // halves the density work per frame (594 NORMALD calls instead of
 // 1,188) and drops the wrapper calls altogether -- the cost a G1
 // feels while an arrow key is held.  The lists hold the PIXEL y
 // of each curve, already clamped to the plot top.  (Allocated once
 // in pv_Init; element pxx-PXL+1 is column pxx.)
 FOR pxx FROM PXL TO PXR DO
  cix:=pxx-PXL+1;
  vx:=XMN+(pxx-PXL)*span/(PXR-PXL);
  IF nd0 THEN
   y0:=MAX(PYT,PYB-NORMALD(M0v,SE0,vx)*YSC);
   CY0(cix):=y0;
  ELSE
   y0:=CY0(cix);                         // cached: no NORMALD
  END;
  IF nd1 THEN
   IF gcp THEN CYG(cix):=CY1(cix); END;  // the ghost: H1 as it was
   y1:=MAX(PYT,PYB-NORMALD(M1v,SE1,vx)*YSC);
   CY1(cix):=y1;
  ELSE
   y1:=CY1(cix);                         // cached: no NORMALD
  END;
  rej:=(vx<=XCL) OR (vx>=XCU);
  IF rej THEN
   LINE_P(G1,pxx,IP(y1),pxx,PYB,cpow);   // power under H1
   LINE_P(G1,pxx,IP(y0),pxx,PYB,calp);   // alpha under H0 (on top)
   // the green that stays visible is H1's band ABOVE H0 (y1 to y0),
   // on H1's side only (the far tail of a 2-sided test is a sliver)
   IF (DIRN>0 AND vx>=XCU) OR (DIRN<0 AND vx<=XCL) THEN
    hg:=MAX(0,y0-y1);
    pA:=pA+hg; pX:=pX+pxx*hg; pY:=pY+hg*(y0+y1)/2;
    IF hg>=11 THEN pWd:=pWd+1; END;
   END;
  ELSE
   LINE_P(G1,pxx,IP(y1),pxx,PYB,cbet);   // beta under H1
   hg:=PYB-y1;
   bA:=bA+hg; bX:=bX+pxx*hg; bY:=bY+hg*(y1+PYB)/2;
   IF hg>=11 THEN bW:=bW+1; END;
  END;
 END;
 CK0:={M0v,SE0,XMN,XMX,YSC};
 CK1:={M1v,SE1,XMN,XMX,YSC};
 IF gcp THEN GHOK:=1; GREQ:=0; END;

 // ----- the ghost: H1 before the last step or drag (v2.4) -----
 // Pale magenta, from CYG -- no NORMALD.  Only if the window is the
 // one it was drawn in; after a re-fit its pixels mean nothing.
 IF GHOK==1 THEN
  IF pv_Same(GWIN,{XMN,XMX,YSC}) THEN
   FOR pxx FROM PXL+1 TO PXR DO
    LINE_P(G1,pxx-1,IP(CYG(pxx-PXL)),pxx,IP(CYG(pxx-PXL+1)),cgho);
   END;
  END;
 END;

 // ----- pass 2: curve outlines (from the cached heights) -----
 pv0:=PYB; pv1:=PYB;
 FOR pxx FROM PXL TO PXR DO
  py0:=CY0(pxx-PXL+1);
  py1:=CY1(pxx-PXL+1);
  IF pxx>PXL THEN
   LINE_P(G1,pxx-1,IP(pv0),pxx,IP(py0),cnul);
   LINE_P(G1,pxx-1,IP(pv1),pxx,IP(py1),calt);
  END;
  pv0:=py0; pv1:=py1;
 END;

 // ----- H0 / H1 labels at the curve peaks (from PowerCalc) -----
 // v2.5l: side by side, 16 px apart, when the peaks nearly coincide (a
 // small effect drew one label on the other); H1 on its own side of H0
 cpx:=PXL+IP((M0v-XMN)/span*(PXR-PXL));
 hx0:=MAX(PXL,cpx-8);
 cpx:=PXL+IP((M1v-XMN)/span*(PXR-PXL));
 hx1:=MIN(PXR-12,cpx-4);
 IF ABS(hx1-hx0)<16 THEN
  IF (M1v>=M0v AND hx0+16<=PXR-12) OR hx0-16<PXL THEN
   hx1:=hx0+16;
  ELSE
   hx1:=hx0-16;
  END;
 END;
 TEXTOUT_P("H₀",G1,hx0,PYT-8,1,cnul);
 TEXTOUT_P("H₁",G1,hx1,PYT-8,1,calt);

 // ----- dotted color-keyed verticals at the H0 / H1 centers -----
 cpx:=PXL+IP((M0v-XMN)/span*(PXR-PXL));
 yy:=PYT;
 WHILE yy<PYB DO
  PIXON_P(G1,cpx,yy,cnul);
  yy:=yy+4;
 END;
 cpx:=PXL+IP((M1v-XMN)/span*(PXR-PXL));
 yy:=PYT;
 WHILE yy<PYB DO
  PIXON_P(G1,cpx,yy,calt);
  yy:=yy+4;
 END;

 // ----- v2.5h: the effect made visible -----
 // A magenta arrow from the H0 centre to the H1 centre, 6 px below the
 // plot top, labelled "d = 0.32" ("h = ..." for proportions): the gap
 // between the means, and d is that gap measured in SDs.  The ends are
 // kept inside the plot; the label sits under the middle of the arrow,
 // or beside it when the arrow is shorter than the label.  No arrow when
 // the centres are under 6 px apart (no effect, or a tiny one).
 ar0:=MIN(PXR,MAX(PXL,PXL+IP((M0v-XMN)/span*(PXR-PXL))));
 ar1:=MIN(PXR,MAX(PXL,PXL+IP((M1v-XMN)/span*(PXR-PXL))));
 ary:=PYT+6;
 arl:=IFTE(pv_MeanT()==1,"d","h")+" = "+pv_Cm(pv_FMT(ROUND(pv_EffSz(),2),2));
 IF ABS(ar1-ar0)>=6 THEN
  ars:=SIGN(ar1-ar0);
  LINE_P(G1,ar0,ary,ar1,ary,calt);
  LINE_P(G1,ar1,ary,ar1-3*ars,ary-3,calt);
  LINE_P(G1,ar1,ary,ar1-3*ars,ary+3,calt);
  LINE_P(G1,ar0,ary,ar0+3*ars,ary-3,calt);
  LINE_P(G1,ar0,ary,ar0+3*ars,ary+3,calt);
 END;
 // 6 px a character (font 1); centred under the arrow if it fits
 lx:=IP((ar0+ar1)/2-3*DIM(arl));
 IF ABS(ar1-ar0) < 6*DIM(arl)+8 THEN
  lx:=MAX(ar0,ar1)+5;
  IF lx+6*DIM(arl) > PXR THEN lx:=MIN(ar0,ar1)-5-6*DIM(arl); END;
 END;
 lx:=MIN(PXR-6*DIM(arl),MAX(PXL+2,lx));
 TEXTOUT_P(pv_Cm(arl),G1,lx,ary+3,1,calt);

 // baseline axis
 LINE_P(G1,PXL,PYB,PXR,PYB,cgry);

 // ----- labeled tick marks (gray, fixed decimals) -----
 // v2.5l: a label that would overlap the one before it (4 px apart at
 // least) or run past the screen's right edge is left out; every tick
 // mark is still drawn.  Widths from pv_TxW (font 1).  At everyday
 // values every label is drawn, as before.
 tk:=CEILING(XMN/stp)*stp;
 tlR:=-1000;
 WHILE tk<=XMX DO
  cpx:=PXL+IP((tk-XMN)/span*(PXR-PXL));
  LINE_P(G1,cpx,PYB,cpx,PYB+3,cgry);
  tlS:=pv_Cm(pv_FMT(tk,dec));
  tlX:=cpx-9;
  IF tlX>=tlR+4 AND tlX+pv_TxW(tlS)<=319 THEN
   TEXTOUT_P(pv_Cm(tlS),G1,tlX,PYB+5,1,cgry);
   tlR:=tlX+pv_TxW(tlS);
  END;
  tk:=tk+stp;
 END;

 // ----- critical value dashed line(s), drawn manually -----
 // v2.4: each line's pixel x is kept in CLX (1 lower, 2 upper) so a
 // press within 6 px of it grabs alpha (pv_Touch).
 CLX:={-100,-100};
 FOR cv FROM 1 TO 2 DO
  xv:=IFTE(cv==1,XCL,XCU);
  IF (xv>XMN) AND (xv<XMX) THEN
   cpx:=PXL+IP((xv-XMN)/span*(PXR-PXL));
   CLX(cv):=cpx;
   yy:=PYT;
   WHILE yy<PYB DO
    seg:=MIN(yy+3,PYB);
    LINE_P(G1,cpx,yy,cpx,seg,ccrt);
    yy:=yy+7;
   END;
  END;
 END;

 // ----- area labels inside the fills (v2.4) -----
 // At the centroid of the yellow (beta) region and of the visible
 // green (power) region.  Left out when the region is narrower than
 // the label: bW / pWd count the columns at least 11 px tall (one line
 // of the small font), and a label is taken as 5 px a character (an
 // estimate: "abc" in font 2 measured 19 px).  The label is then kept
 // inside its region at the centroid's column.
 IF bA>0 THEN
  // v2.5d: pv_Rel, since pv_FMTP can now write "> 0.99" (beta near 1,
  // power near 0): "β > 0.99", never "β = > 0.99"
  lbs:="β"+pv_Rel(pv_Cm(pv_FMTP(1-PW,2)));
  lw:=5*DIM(lbs);
  IF bW>=lw+2 THEN
   cix:=MIN(PXR-PXL+1,MAX(1,IP(bX/bA)-PXL+1));
   ytp:=IP(CY1(cix))+1; ybt:=PYB-11;
   IF ytp<=ybt THEN
    ly:=MIN(ybt,MAX(ytp,IP(bY/bA)-5));
    lx:=MIN(PXR-lw,MAX(PXL+2,IP(bX/bA-lw/2)));
    TEXTOUT_P(pv_Cm(lbs),G1,lx,ly,1,cgry);
   END;
  END;
 END;
 IF pA>0 THEN
  // v2.5d: pv_Rel, as above: "power > 0.99" where the power region is
  // nearly the whole H1 curve, never "power = > 0.99"
  lbs:="power"+pv_Rel(pv_Cm(pv_FMTP(PW,2)));
  lw:=5*DIM(lbs);
  IF pWd>=lw+2 THEN
   cix:=MIN(PXR-PXL+1,MAX(1,IP(pX/pA)-PXL+1));
   ytp:=IP(CY1(cix))+1; ybt:=IP(CY0(cix))-11;
   IF ytp<=ybt THEN
    ly:=MIN(ybt,MAX(ytp,IP(pY/pA)-5));
    lx:=MIN(PXR-lw,MAX(PXL+2,IP(pX/pA-lw/2)));
    TEXTOUT_P(pv_Cm(lbs),G1,lx,ly,1,cgrn);
   END;
  END;
 END;

 // ----- the one-frame callout (v2.4: what the Tails toggle changed) -----
 IF DIM(HB1) > 0 THEN
  RECT_P(G1,8,PYT+2,312,PYT+30,cfrm,cwht);
  TEXTOUT_P(pv_Cm(HB1),G1,14,PYT+5,1,cblk);
  TEXTOUT_P(pv_Cm(HB2),G1,14,PYT+17,1,cblk);
 END;

 BLIT_P(G0,0,0,G1);
 DRAWMENU({"Edit","Tails","α−","α+","Pow-n","Menu"});

 HNT:=""; HB1:=""; HB2:="";
 IF WHLD==0 THEN GHOK:=0; GREQ:=0; END;
END;

// ---------------- the two curve views (v2.4) ----------------
// Column cc (1..250) of the power-vs-n plot is pixel x = PCL + cc and

BEGIN

END;

// Draw a curve view from the cached sweep PWC (no engine calls).
//   kind 1 = power vs n (x: n from 2 to PNMX)
//   kind 2 = power vs effect size (x: d or h from 0 to 1.5)
//   selc   = the column under the finger (0 = none): a red marker
//            and "at n = ...: power = ..." follow it.
LOCAL pv_PCDraw(kind, selc)
BEGIN
 LOCAL w2, h2, i2, tickx, ticky, x0p;
 LOCAL nv, pyv, lastpy, cc, ev, lbs;
 LOCAL cmk, cmx;
 w2:=PCR-PCL; h2:=PCB-PCT;

 DIMGROB_P(G1,320,240,cwht);
 // title, the test, and what is held fixed
 IF kind==1 THEN
  TEXTOUT_P("Power vs sample size",G1,4,3,2,cblk);
  // v2.5c: pv_Rel -- "α < 0.001" for a typed alpha below the floor
  lbs:=pv_EffLbl()+", α"+pv_Rel(pv_Cm(pv_FMTP(AL,3)))+", "+pv_Cm(STRING(TL,1,0))+"-sided";
 ELSE
  TEXTOUT_P("Power vs effect size",G1,4,3,2,cblk);
  // v2.5c: pv_Rel, as above
  IF TT==5 THEN
   lbs:="n = "+pv_Cm(pv_NStr(NN))+"; α"+pv_Rel(pv_Cm(pv_FMTP(AL,3)))+"; "+pv_Cm(STRING(TL,1,0))+"-sided";   // v2.5g; v2.5i ";"
  ELSE
   // v2.5i: "n = 30 per group" (tests 2, 4), "n = 30" (one sample), no "/"
   lbs:="n = "+pv_Cm(STRING(NN,1,0))+IFTE(TT==2 OR TT==4," per group","")+", α"+pv_Rel(pv_Cm(pv_FMTP(AL,3)))+", "+pv_Cm(STRING(TL,1,0))+"-sided";
  END;
  IF pv_MeanT()==0 THEN lbs:=lbs+", p₀ = "+pv_Cm(pv_FMT(PP0,2)); END;
 END;
 TEXTOUT_P(pv_TName(),G1,196,7,1,cgry);
 TEXTOUT_P(pv_Cm(lbs),G1,4,19,1,cgry);

 // grid: y ticks 0..1 by 0.2
 FOR i2 FROM 0 TO 5 DO
  ticky:=PCB-IP(i2/5*h2);
  IF (i2>0) AND (i2<5) THEN LINE_P(G1,PCL+1,ticky,PCR,ticky,cgrd); END;
  TEXTOUT_P(pv_Cm(STRING(i2/5,2,1)),G1,PCL-24,ticky-4,1,cgry);
 END;

// ---------------- power-vs-n curve view (from PowerCalc) ----------------
// v2.4: the 250 power values are computed ONCE on entry and kept in
// PWC; a finger dragged along the curve only redraws from them.
LOCAL pv_PowNView()
BEGIN
 LOCAL nn0, cc, nv, nlo, go, kk;
 LOCAL mm, mmF, selc, tx;

 nn0:=NN;                        // save current n (restored below)
 // the range: at least 200, and wide enough to show the current n and
 // n80 with room to spare; never past 5000, the largest n allowed
 PNMX:=MAX(200,IP(NN*1.25));
 IF N80>0 THEN PNMX:=MAX(PNMX,IP(N80*1.25)); END;
 PNMX:=MIN(5000,PNMX);

 // the sweep: one whole n per pixel column, all four tests through
 // the shared engine; record the first column at power 0.80
 nv:=0;
 FOR cc FROM 1 TO 250 DO
  pv_Model(pv_NofC(cc));
  PWC(cc):=PW;
  IF nv==0 AND PW>=0.8 THEN nv:=cc; END;
 END;
 // that column's n may overshoot by a few (one column spans several
 // n when PNMX is large): step n down with the same engine to the
 // FIRST whole n at 0.80 -- the same number pv_N80 prints in the panel
 PN80:=-1;
 IF nv>0 THEN
  nlo:=2;
  IF nv>1 THEN nlo:=pv_NofC(nv-1)+1; END;
  PN80:=pv_NofC(nv);
  go:=1;
  WHILE go==1 AND PN80>nlo DO
   pv_Model(PN80-1);
   IF PW>=0.8 THEN PN80:=PN80-1; ELSE go:=0; END;
  END;
 END;
 NN:=nn0;                        // restore and recompute current scene
 pv_Model(NN);

 // show it, then wait for a key or a touch (drain, wait, drain)
 selc:=0;
 pv_PCDraw(1,0);
 REPEAT kk:=GETKEY; UNTIL kk==-1;
 pv_TDrain();
 REPEAT
  kk:=GETKEY;
  mm:=MOUSE;
  mmF:=mm(1);
 UNTIL kk>-1 OR SIZE(mmF)>0;
 IF SIZE(mmF)>0 THEN
  // a touch on the curve's box: follow the finger along the curve
  // until it lifts, redrawing from the cache; on lift use that n.
  // Esc while the finger is down lets go without choosing.
  IF mmF(1)>=PCL AND mmF(1)<=PCR+4 AND mmF(2)>=PCT-12 AND mmF(2)<=PCB+14 THEN
   go:=1;
   WHILE go==1 DO
    tx:=mmF(1);
    IF tx>=0 AND tx<=319 THEN
     cc:=MIN(250,MAX(1,IP(tx)-PCL));
     IF cc<>selc THEN
      selc:=cc;
      pv_PCDraw(1,selc);
     END;
    ELSE
     go:=0;                      // a phantom reading: treat as a lift
    END;
    mm:=MOUSE;
    mmF:=mm(1);
    kk:=GETKEY;
    IF SIZE(mmF)==0 THEN go:=0; END;
    IF kk==4 THEN go:=0; selc:=0; END;
   END;
   IF selc>0 THEN NN:=MIN(5000,MAX(2,pv_NofC(selc))); END;
  END;
  pv_TDrain();                   // capped: never more than a second
 END;
 REPEAT kk:=GETKEY; UNTIL kk==-1;
 WFIT:=1;                        // back on the main view: re-fit
END;

// ---------------- power-vs-effect-size view (v2.4) ----------------
// Power over the standardized effect, 0 to 1.5, at the current n and
// alpha, in the direction of the current effect.  Means: mu1 = mu0 +
// d*sigma (v2.5g: test 5's sigma', pv_SigE).  Proportions: p1 = sin^2(asin(sqrt p0) + h/2), which runs
// out where p1 would pass 0.98 (the program's own limit); the curve
// simply stops there.
LOCAL pv_PowEView()
BEGIN
 LOCAL cc, ev, sm1, sp1, sdir, a0;
 LOCAL aa;
 sm1:=MM1; sp1:=PP1;
 pv_Model(NN);
 sdir:=DIRN;
 a0:=0;
 IF pv_MeanT()==0 THEN a0:=pv_AsinR(SQRT(PP0)); END;
 FOR cc FROM 1 TO 250 DO
  ev:=cc/250*1.5;
  PWC(cc):=-1;
  IF pv_MeanT()==1 THEN
   MM1:=MM0+sdir*ev*pv_SigE();
   pv_Model(NN);
   PWC(cc):=PW;
  ELSE
   aa:=a0+sdir*ev/2;
   IF aa>0 AND aa<1.5707963 THEN
    PP1:=pv_SinR(aa)^2;
    IF PP1>=0.02 AND PP1<=0.98 THEN
     pv_Model(NN);
     PWC(cc):=PW;
    END;
   END;
  END;
 END;
 MM1:=sm1; PP1:=sp1;             // the effect as it was
 pv_Model(NN);
 pv_PCDraw(2,0);
 pv_WaitAny();
 WFIT:=1;
END;

// ---------------- edit forms (with help strings, from PowerCalc) ----------------
// v2.5l: numbers PowerViz cannot draw (the monkey test, 2 Nov 2026), as
// a sentence, or "" when they can be drawn.
// * A mean or sigma above 1,000,000: v2.5k took anything, and a sigma of
//   1E300 squared past the Prime's 1E499 (an infinity with no error; the
//   Levene page then divided by zero), while numbers of 13 digits and
//   more ran out of the readout box, the d label and the formula page.
//   A power analysis needs no bigger numbers: a student can use bigger
//   units.
// * A sigma too small next to the means: the plot's window is the means
//   +/- 6 SE in data units, and with n up to 5000 the SE can be
//   sigma/71; in 12 digits the tick loop (tk := tk + stp) stops moving
//   once the step is below the 12th digit of the means (SampDist v2.3d
//   did so on the Virtual Calculator), and the readouts' decimals,
//   worked out from sigma/20, must stay at 12 or fewer (v2.5k asked
//   STRING for 302 decimals at a sigma of 1E-300).  So sigma >= 0.000001
//   and sigma >= |mu| x 0.000001.
// * Test 5: sigma2 from 0.01 to 100 times sigma1 (beyond, the two
//   curves' labels fell on each other and the best-split line ran off the
//   formula page).
// * Means that differ, but by less than a billionth of the larger
//   sigma: the n for 80% power is ((za u + zb w)/|mu1 - mu0|)^2, and
//   at mu0 = 1E-300, mu1 = 0, sigma = 999999 that is about 1E612, past
//   the largest number the Prime holds (the monkey test, 2 Nov 2026).
//   Above the line, n stays below about 1E25.  Equal means are allowed:
//   no effect, and the panel says no n is enough.
// A sigma of 0 or less is left to the form's own clamp, which says so
// and resets it.
LOCAL pv_BadMsg(bm0, bm1, bs1, bs2)
BEGIN
 LOCAL bmM, bmR, bmD;
 bmM:=MAX(ABS(bm0),ABS(bm1));
 // v2.5q (Round 10): the fleet's limits
 IF bmM>10000 THEN
  RETURN "A mean must be from -10,000 to 10,000.";
 END;
 IF bs1>1000 OR bs2>1000 THEN
  RETURN "σ can be at most 1,000.";
 END;
 IF bs1<=0 OR bs2<=0 THEN RETURN ""; END;
 IF MIN(bs1,bs2)<MAX(0.000001,bmM*0.000001) THEN
  RETURN "σ is too small next to the means for the calculator's 12 digits. Use a larger σ or means nearer 0.";
 END;
 bmR:=bs2/bs1;
 IF bmR<0.01 OR bmR>100 THEN
  RETURN "σ₂ must be from 0.01 to 100 times σ₁.";
 END;
 bmD:=ABS(bm1-bm0);
 IF bmD>0 AND bmD<MAX(bs1,bs2)*0.000000001 THEN
  RETURN "The means differ by under a billionth of σ: no n could tell them apart.";
 END;
 RETURN "";
END;

// v2.5l: means that differ by less than a billionth of the larger
// sigma are taken as equal (no effect).  The arrows and the drag snap
// mu1 to the grid of the effect step; with mu0 off that grid (1E-300)
// mu1 = 0 left a difference of 1E-300, and the n for 80% power,
// ((za u + zb w)/delta)^2, was about 1E612, past the Prime's largest
// number (the monkey test, 2 Nov 2026).  Typed, such means are refused
// with a sentence (pv_BadMsg).
LOCAL pv_Snap0()
BEGIN
 IF pv_MeanT()==1 THEN
  IF MM1<>MM0 AND ABS(MM1-MM0)<MAX(SG,IFTE(TT==5,SG2,SG))*0.000000001 THEN MM1:=MM0; END;
 END;
END;

// A tap on the effect cell (v2.4): a one-field form for mu1 or p1.
LOCAL pv_EditEff()
BEGIN

  eev:=MM1;
  eel:=IFTE(TT==1,"μ₁:","μ grp2:");

// ============================================================
// v2.5h: LEVENE'S TEST AND TWO REAL SAMPLES
// ============================================================

// One value from N(0,1): Marsaglia's polar method (no trig, so the
// Home angle mode cannot touch it; Lessons Learned s11).  Each accepted
// pair gives two values; the second is kept for the next call.
LOCAL pv_RNorm()
BEGIN
 LOCAL rnu, rnv, rns, rnf;
 IF pvGsf==1 THEN pvGsf:=0; RETURN pvGsp; END;
 REPEAT
  rnu:=2*RANDOM-1; rnv:=2*RANDOM-1;
  rns:=rnu*rnu+rnv*rnv;
 UNTIL rns>0 AND rns<1;
 rnf:=(-2*LN(rns)/rns)^.5;
 pvGsp:=rnv*rnf; pvGsf:=1;
 RETURN rnu*rnf;
END;

// Exact power of the two-sided F test of equal variances, F = s2^2/s1^2,
// for normal data: under H1, F/rho2 has the F(fd2, fd1) distribution,
// rho2 = sigma2^2/sigma1^2, fd1 = n1 - 1, fd2 = n2 - 1.  Power = P(F <
// lower critical value) + P(F > upper one).  The upper critical value is
// 1/(a lower quantile of F(fd1, fd2)) and the upper tail P(F > c) is the
// lower tail of 1/F, so every call is a lower tail and keeps its digits
// (Lessons Learned s3: never 1 - CDF, never ICDF(1 - alpha)).
LOCAL pv_FPow(fr2, fd1, fd2, fal)
BEGIN
 LOCAL fcl, fcu;
 fcl:=FISHER_ICDF(fd2,fd1,fal/2);
 fcu:=1/FISHER_ICDF(fd1,fd2,fal/2);
 RETURN FISHER_CDF(fd2,fd1,fcl/fr2)+FISHER_CDF(fd1,fd2,fr2/fcu);
END;

// Levene's power, large samples: Levene's F for two groups is the square
// of a pooled two-sample t on z = |x - group mean|.  For normal data z
// has mean sigma sqrt(2/pi) sqrt((n-1)/n) and variance about sigma^2 (1 -
// 2/pi), and for large n the t is close to normal: power = P(|T| > t
// crit) with T normal, centre delta_z/SE0, spread SE1/SE0 (SE0 the
// pooled SE the test uses, SE1 the true one).  Within 0.01 of simulation
// once n1 + n2 is past 240 (verify_PowerViz_v2.5h, section E); used only
// there.  The sigmas are divided by the larger one first: power does not
// depend on their scale, and nothing is squared out of range.
LOCAL pv_LvApx(lxa, lxb, lna, lnb, lal)
BEGIN
 LOCAL lsc, lca, lcb, lvv, lpc, lsp;
 LOCAL ls0, ls1, ldz, lzc;
 lsc:=MAX(lxa,lxb); lca:=lxa/lsc; lcb:=lxb/lsc;
 lpc:=(2/3.14159265359)^.5;              // sqrt(2/pi)
 lvv:=1-2/3.14159265359;                 // 1 - 2/pi
 ldz:=lpc*(lcb*((lnb-1)/lnb)^.5-lca*((lna-1)/lna)^.5);
 lsp:=((lna-1)*lca^2*lvv+(lnb-1)*lcb^2*lvv)/(lna+lnb-2);
 ls0:=(lsp*(1/lna+1/lnb))^.5;
 ls1:=(lvv*(lca^2/lna+lcb^2/lnb))^.5;
 lzc:=-STUDENT_ICDF(lna+lnb-2,lal/2);    // the t critical value, mirrored
 RETURN NORMALD_CDF(0,1,(ldz-lzc*ls0)/ls1)+NORMALD_CDF(0,1,(-ldz-lzc*ls0)/ls1);
END;

// Levene's power by simulation: lsr pairs of normal samples, sizes n1 =
// NN and n2 = pv_N2(NN), SDs sigma1 and sigma2 (scaled by the larger:
// the result does not depend on the scale), each tested with Levene's F
// on the absolute deviations from the group MEANS (the form SPSS prints)
// at the alpha on screen.  F for two groups is SSB/(SSW/(N - 2)), with
// critical value t(N-2)^2 at alpha/2 (F(1, d) = t(d)^2), so no F quantile
// is needed.  A progress bar every 20 pairs, straight to the screen;
// Esc stops early.  Returns {rejections, pairs done}.
LOCAL pv_LvSim(lsr)
BEGIN
 LOCAL lsn1, lsn2, lsN, lsi, lsj, lsx;
 LOCAL lss, lsm, lsz, lsBt, lsS1, lsQ1;
 LOCAL lsS2, lsQ2, lsF, lsc, lsk, lsA;
 LOCAL lsB, lsg, lsca, lscb, lssw, lszb;
 LOCAL lsTot;
 lsTot:=lsr;                           // a LOCAL copy: Esc shortens it
 lsn1:=NN; lsn2:=pv_N2(NN); lsN:=lsn1+lsn2;
 lsg:=MAX(SG,SG2); lsca:=SG/lsg; lscb:=SG2/lsg;
 lsc:=(-STUDENT_ICDF(lsN-2,AL/2))^2;
 lsA:=MAKELIST(0,lsj,1,lsn1); lsB:=MAKELIST(0,lsj,1,lsn2);
 lsk:=0; lsi:=0;
 REPEAT
  lsi:=lsi+1;
  // group 1: the sample, its mean, then the absolute deviations
  lss:=0;
  FOR lsj FROM 1 TO lsn1 DO lsx:=lsca*pv_RNorm(); lsA(lsj):=lsx; lss:=lss+lsx; END;
  lsm:=lss/lsn1; lsS1:=0; lsQ1:=0;
  FOR lsj FROM 1 TO lsn1 DO lsz:=ABS(lsA(lsj)-lsm); lsS1:=lsS1+lsz; lsQ1:=lsQ1+lsz*lsz; END;
  // group 2, the same
  lss:=0;
  FOR lsj FROM 1 TO lsn2 DO lsx:=lscb*pv_RNorm(); lsB(lsj):=lsx; lss:=lss+lsx; END;
  lsm:=lss/lsn2; lsS2:=0; lsQ2:=0;
  FOR lsj FROM 1 TO lsn2 DO lsz:=ABS(lsB(lsj)-lsm); lsS2:=lsS2+lsz; lsQ2:=lsQ2+lsz*lsz; END;
  // Levene's F: between-groups over within-groups sums of squares of z
  lszb:=(lsS1+lsS2)/lsN;
  lsBt:=lsn1*(lsS1/lsn1-lszb)^2+lsn2*(lsS2/lsn2-lszb)^2;
  lssw:=MAX(0,lsQ1-lsS1*lsS1/lsn1)+MAX(0,lsQ2-lsS2*lsS2/lsn2);
  IF lssw>0 THEN
   lsF:=lsBt*(lsN-2)/lssw;
   IF lsF>lsc THEN lsk:=lsk+1; END;
  END;
  IF lsi MOD 20 == 0 THEN
   RECT_P(G0,40,60,40+IP(240*lsi/lsTot),66,cfrm,cfrm);
   IF GETKEY==4 THEN lsTot:=lsi; END;   // Esc: stop with what is done
  END;
 UNTIL lsi>=lsTot;
 RETURN {lsk, lsi};
END;

// mean of a list (a loop: no list arithmetic, Lessons Learned v7r)
LOCAL pv_LMean(lml)
BEGIN
 LOCAL lmi, lms;
 lms:=0;
 FOR lmi FROM 1 TO SIZE(lml) DO lms:=lms+lml(lmi); END;
 RETURN lms/SIZE(lml);
END;

// SD of a list (n - 1), two passes
LOCAL pv_LSD(lsl, lsmn)
BEGIN
 LOCAL lsi, lss;
 lss:=0;
 FOR lsi FROM 1 TO SIZE(lsl) DO lss:=lss+(lsl(lsi)-lsmn)^2; END;
 RETURN (lss/(SIZE(lsl)-1))^.5;
END;

// median of a list
LOCAL pv_LMed(mdl)
BEGIN
 LOCAL mds, mdn;
 mds:=SORT(mdl); mdn:=SIZE(mds);
 IF mdn MOD 2 == 1 THEN RETURN mds((mdn+1)/2); END;
 RETURN (mds(mdn/2)+mds(mdn/2+1))/2;
END;

// Levene's F for two groups from the centres lfa, lfb (the group means:
// Levene 1960, as SPSS; the medians: Brown & Forsythe 1974, as R's car
// and AnovaLab): z = |x - centre|, F = SSB/(SSW/(N - 2)), two passes.
// Returns {F, p}; p = 2 P(T < -sqrt F) with T ~ t(N - 2), a lower tail
// (F(1, d) = t(d)^2).  {-1, -1} when every z equals its group's mean z
// (SSW = 0: no F).
LOCAL pv_LvF(lfx, lfy, lfa, lfb)
BEGIN
 LOCAL lfi, lf1, lf2, lfn, lfw, lfz;
 LOCAL lfm, lfF;
 lf1:=0; lf2:=0; lfn:=SIZE(lfx)+SIZE(lfy);
 FOR lfi FROM 1 TO SIZE(lfx) DO lf1:=lf1+ABS(lfx(lfi)-lfa); END;
 FOR lfi FROM 1 TO SIZE(lfy) DO lf2:=lf2+ABS(lfy(lfi)-lfb); END;
 lf1:=lf1/SIZE(lfx); lf2:=lf2/SIZE(lfy);
 lfm:=(SIZE(lfx)*lf1+SIZE(lfy)*lf2)/lfn;
 lfw:=0;
 FOR lfi FROM 1 TO SIZE(lfx) DO lfz:=ABS(lfx(lfi)-lfa)-lf1; lfw:=lfw+lfz*lfz; END;
 FOR lfi FROM 1 TO SIZE(lfy) DO lfz:=ABS(lfy(lfi)-lfb)-lf2; lfw:=lfw+lfz*lfz; END;
 IF lfw<=0 THEN RETURN {-1,-1}; END;
 lfF:=(SIZE(lfx)*(lf1-lfm)^2+SIZE(lfy)*(lf2-lfm)^2)*(lfn-2)/lfw;
 RETURN {lfF, 2*STUDENT_CDF(lfn-2,-(lfF^.5))};
END;

// Everything the data page shows, for samples tsa (group 1) and tsb
// (group 2), as one flat list of numbers (so a verify script can check
// it against SciPy in one call):
//   1-6   n1, mean1, s1, n2, mean2, s2
//   7-10  Levene F and p (means), Brown-Forsythe F and p (medians)
//   11-13 pooled t, df, p (2-sided)
//   14-16 Welch t, df (Satterthwaite), p (2-sided)
//   17-18 Cohen's d with the pooled s; with sigma' = sqrt((s1^2+s2^2)/2)
// t and d are group 2 minus group 1.  A p is a lower tail, doubled.  -1
// marks a number that does not exist (no spread: s = 0 in both groups).
// Data beyond 1E50 or under 1E-50 is first divided by a power of 10 near
// the largest |value| (tsk), so that no square of a square over- or
// underflows (Lessons Learned v7u: scale before squaring).  A power of 10
// divides exactly on the Prime, whose numbers are decimal.  t, F, p, df
// and d do not change; the means and SDs are multiplied back at the end.
// Everyday data is left exactly as it is (tsk = 1).
LOCAL pv_TwoS(tsa, tsb)
BEGIN
 LOCAL tn1, tn2, tm1, tm2, ts1, ts2;
 LOCAL tlv, tbf, tsp, ttp, tdf, tv1;
 LOCAL tv2, ttw, twd, tpp, tpw, tdp;
 LOCAL tds, tsk, tsx, tsy, tsj;
 tn1:=SIZE(tsa); tn2:=SIZE(tsb);
 tsk:=0;
 FOR tsj FROM 1 TO tn1 DO tsk:=MAX(tsk,ABS(tsa(tsj))); END;
 FOR tsj FROM 1 TO tn2 DO tsk:=MAX(tsk,ABS(tsb(tsj))); END;
 IF tsk>1E50 OR (tsk>0 AND tsk<1E-50) THEN
  tsk:=10^FLOOR(LOG(tsk));
 ELSE
  tsk:=1;
 END;
 // working copies (a list assigned to a LOCAL is a copy), scaled if need be
 tsx:=tsa; tsy:=tsb;
 IF tsk<>1 THEN
  FOR tsj FROM 1 TO tn1 DO tsx(tsj):=tsa(tsj)/tsk; END;
  FOR tsj FROM 1 TO tn2 DO tsy(tsj):=tsb(tsj)/tsk; END;
 END;
 tm1:=pv_LMean(tsx); tm2:=pv_LMean(tsy);
 ts1:=pv_LSD(tsx,tm1); ts2:=pv_LSD(tsy,tm2);
 tlv:=pv_LvF(tsx,tsy,tm1,tm2);
 tbf:=pv_LvF(tsx,tsy,pv_LMed(tsx),pv_LMed(tsy));
 tdf:=tn1+tn2-2;
 tsp:=(((tn1-1)*ts1^2+(tn2-1)*ts2^2)/tdf)^.5;
 tv1:=ts1^2/tn1; tv2:=ts2^2/tn2;
 IF tsp>0 THEN
  ttp:=(tm2-tm1)/(tsp*(1/tn1+1/tn2)^.5);
  tpp:=2*STUDENT_CDF(tdf,-ABS(ttp));
  ttw:=(tm2-tm1)/(tv1+tv2)^.5;
  twd:=(tv1+tv2)^2/(tv1^2/(tn1-1)+tv2^2/(tn2-1));
  tpw:=2*STUDENT_CDF(twd,-ABS(ttw));
  tdp:=(tm2-tm1)/tsp;
  tds:=(tm2-tm1)/((ts1^2+ts2^2)/2)^.5;
 ELSE
  ttp:=-1; tpp:=-1; ttw:=-1; twd:=-1; tpw:=-1; tdp:=-1; tds:=-1;
 END;
 RETURN {tn1,tm1*tsk,ts1*tsk,tn2,tm2*tsk,ts2*tsk,tlv(1),tlv(2),tbf(1),tbf(2),
         ttp,tdf,tpp,ttw,twd,tpw,tdp,tds};
END;

// ---------------- effect / n stepping ----------------
// Each press moves to the NEXT multiple of the step rather than adding
// to whatever is there, so an odd value (103.7) snaps onto the grid
// (104) and stays round from then on (NormExp beta 26).
LOCAL pv_StepEff(dd)
BEGIN
 LOCAL es;
 es := pv_EffStep();
 IF pv_MeanT()==1 THEN
  IF dd>0 THEN
   MM1:=ROUND((FLOOR(MM1/es+0.000001)+1)*es,6);
  ELSE
   MM1:=ROUND((CEILING(MM1/es-0.000001)-1)*es,6);
  END;
 ELSE
  IF dd>0 THEN
   PP1:=ROUND((FLOOR(PP1/es+0.000001)+1)*es,6);
  ELSE
   PP1:=ROUND((CEILING(PP1/es-0.000001)-1)*es,6);
  END;
  PP1:=MIN(0.98,MAX(0.02,PP1));
 END;
 pv_Snap0();                       // v2.5l: a difference of 1E-300 is none
END;

LOCAL pv_StepN(dd)
BEGIN
 LOCAL sn:=1;
 IF dd>0 THEN
  IF NN>=100 THEN
   sn:=10;
  ELSE
   IF NN>=20 THEN sn:=5; END;
  END;
 ELSE
  IF NN>100 THEN
   sn:=10;
  ELSE
   IF NN>20 THEN sn:=5; END;
  END;
 END;
 // snap to the grid of the step, as pv_StepEff does (33 -> 35 -> 40)
 IF dd>0 THEN
  NN:=(FLOOR(NN/sn+0.000001)+1)*sn;
 ELSE
  NN:=(CEILING(NN/sn-0.000001)-1)*sn;
 END;
 NN:=MIN(5000,MAX(2,NN));
END;

 IF sl==3 THEN pv_AStep(-1); END;                          // α−
 IF sl==4 THEN pv_AStep(1); END;                           // α+
 IF sl==5 THEN pv_Setup(); pv_PowNView(); END;             // Pow-n
 IF sl==6 THEN pv_MainMenu(); pv_Drain(); pv_TDrain(); WFIT:=1; END;    // Menu
 RETURN 1;
END;

// ---------------- one touch on the main view (v2.4) ----------------
// Called by the live loop with the PRESS -- the first MOUSE reading,
// whose x, y it has already checked are there (SIZE > 0).  Follows the
// finger until it lifts and returns 1 if the main view must be
// redrawn, 0 if not.  What the press landed on decides the gesture:
//   y >= 218        a soft key: slot MIN(6, IP(x/53.4)+1)
//   a readout cell  n / effect / alpha (hit boxes recorded by pv_Draw)
//   the plot box    a critical line within 6 px (and nearer than H1's
//                   centre): alpha; otherwise whichever centre is
//                   nearer: H1 (ties too) or H0
// Press or drag: until the finger has moved more than 8 px it is a
// press; then a mostly vertical move (|dy| > 2|dx|) steps n, anything
// else moves the thing pressed.  H0 never moves (it is the null).
// MOUSE here is read as a list of finger lists; only the first
// finger's x and y (elements 1 and 2) are used, after a SIZE check.
LOCAL pv_Touch(tx0, ty0)
BEGIN
 LOCAL tg, sd, md, ns, nt, hb;
 LOCAL tm, tm1, tk, tx, ty, dx;
 LOCAL dy, vx, span, nw, na, i2;
 LOCAL p0x, p1x, gone, es, hz, nw0;

 // ---- 1. the soft-key strip: act once the finger is off ----
 IF ty0>=218 THEN
  pv_TDrain();
  RETURN pv_Soft(MIN(6,IP(tx0/53.4)+1));
 END;

 // ---- 2. a readout cell ----
 hb:=0;
 FOR i2 FROM 1 TO SIZE(HX1) DO
  IF tx0>=HX1(i2) AND tx0<=HX2(i2) AND ty0>=HY1(i2) AND ty0<=HY2(i2) THEN
   hb:=HACT(i2);
  END;
 END;
 IF hb>0 THEN
  pv_TDrain();
  IF hb==1 THEN pv_EditN(); END;
  IF hb==2 THEN pv_EditEff(); END;
  IF hb==3 THEN pv_AChoose(); END;
  IF hb==4 THEN pv_Formula(); END;          // v2.5: where n comes from
  IF hb==5 THEN pv_EditForm(); WFIT:=1; END;   // v2.5i: the sigma row
  pv_Drain();                               // the form's OK press is spent
  pv_TDrain();                              // v2.5i: and the finger on OK
  RETURN 1;
 END;

 // ---- 3. anywhere else outside the plot box: nothing ----
 IF tx0<PXL OR tx0>PXR OR ty0<PYT-10 OR ty0>PYB+4 THEN
  pv_TDrain();
  RETURN 0;
 END;

 // ---- 4. inside the plot box: what did the finger land on? ----
 // tg: 1 = H1, 2 = a critical line (sd: 1 upper, -1 lower), 3 = H0
 span:=XMX-XMN;
 p0x:=PXL+(M0v-XMN)/span*(PXR-PXL);
 p1x:=PXL+(M1v-XMN)/span*(PXR-PXL);
 tg:=0; sd:=0;
 IF ABS(tx0-CLX(2))<=6 THEN tg:=2; sd:=1; END;
 IF ABS(tx0-CLX(1))<=6 THEN
  IF tg==0 OR ABS(tx0-CLX(1))<ABS(tx0-CLX(2)) THEN tg:=2; sd:=-1; END;
 END;
 // the line often sits a few px from H1's centre (4.6 px at the
 // defaults): a press nearer H1's dotted centre line than the dashed
 // critical line is a grab of H1, not of alpha
 IF tg==2 THEN
  IF ABS(tx0-p1x)<ABS(tx0-CLX(IFTE(sd>0,2,1))) THEN tg:=0; sd:=0; END;
 END;
 IF tg==0 THEN
  IF ABS(tx0-p1x)<=ABS(tx0-p0x) THEN tg:=1; ELSE tg:=3; END;
 END;

 // ---- 5. follow the finger until it lifts ----
 // md: 0 not moved yet (a press), 1 H1, 2 alpha, 3 n, 4 H0 (stays)
 WHLD:=1;                                   // pv_Setup holds the window
 md:=0; ns:=0; gone:=0; hz:=0;
 REPEAT
  tm:=MOUSE;
  tm1:=tm(1);
  tk:=GETKEY;
  IF SIZE(tm1)>0 THEN
   tx:=tm1(1); ty:=tm1(2);
   // a reading off the screen is a phantom touch: treat it as a lift
   IF tx<0 OR tx>319 OR ty<0 OR ty>239 THEN gone:=1; END;
  ELSE
   gone:=1;                                 // the finger is off
  END;
  IF tk==4 THEN gone:=1; END;               // Esc lets go
  IF gone==0 THEN
   dx:=tx-tx0; dy:=ty-ty0;
   IF md==0 AND (ABS(dx)>8 OR ABS(dy)>8) THEN
    IF ABS(dy)>2*ABS(dx) THEN
     md:=3;
    ELSE
     md:=4;
     IF tg==1 THEN md:=1; END;
     IF tg==2 THEN md:=2; END;
    END;
    IF md==1 OR md==3 THEN pv_GSave(); END; // the ghost: H1 as it was
   END;
   nw:=0;
   // -- drag H1: the alternative follows the finger, on the grid --
   IF md==1 THEN
    vx:=XMN+(MIN(PXR,MAX(PXL,tx))-PXL)*span/(PXR-PXL);
    es:=pv_EffStep();
    IF pv_MeanT()==1 THEN
     IF TT==2 OR TT==5 THEN vx:=vx+MM0; END;   // the axis is the difference
     vx:=ROUND(ROUND(vx/es,0)*es,6);
     IF vx<>MM1 THEN MM1:=vx; pv_Snap0(); nw:=1; END;   // v2.5l: pv_Snap0
    ELSE
     IF TT==4 THEN vx:=vx+PP0; END;
     vx:=MIN(0.98,MAX(0.02,ROUND(ROUND(vx/es,0)*es,6)));
     IF vx<>PP1 THEN PP1:=vx; nw:=1; END;
    END;
    HNT:="H₀ stays put; H₁ follows your finger"; HNC:=calt;
   END;
   // -- drag a critical line: alpha = the H0 tail beyond the finger --
   IF md==2 THEN
    vx:=XMN+(MIN(PXR,MAX(PXL,tx))-PXL)*span/(PXR-PXL);
    IF sd>0 THEN
     na:=1-NORMALD_CDF(M0v,SE0,vx);         // upper tail
    ELSE
     na:=NORMALD_CDF(M0v,SE0,vx);           // lower tail
    END;
    // 2-sided: the dragged side holds half of alpha (the other mirrors)
    na:=MIN(0.5,MAX(0.001,na*TL));
    IF ABS(na-AL)>0.0000001 THEN AL:=na; nw:=1; END;
    HNT:=pv_ADrag(); HNC:=ccrt;
   END;
   // -- vertical drag: n, one step per 12 px (finger up = larger n) --
   IF md==3 THEN
    nw0:=NN;
    nt:=IP(-dy/12);
    WHILE ns<nt DO pv_StepN(1); ns:=ns+1; END;
    WHILE ns>nt DO pv_StepN(-1); ns:=ns-1; END;
    IF NN<>nw0 THEN nw:=1; END;
    HNT:="n = "+pv_Cm(STRING(NN,1,0))+" per group: drag up = larger n"; HNC:=cnul;
    IF TT==5 THEN HNT:="n = "+pv_Cm(pv_NStr(NN))+": drag up = larger n₁ (n₂ follows)"; END;   // v2.5g
   END;
   // -- sideways from nearer H0: H0 does not move; say so once --
   IF md==4 AND hz==0 THEN
    HNT:="H₀ is the null: it stays put. Drag H₁ instead"; HNC:=cnul;
    nw:=1; hz:=1;
   END;
   IF nw==1 THEN pv_Setup(); pv_Draw(); END;   // from the caches
  END;
 UNTIL gone==1;
 pv_TDrain();                               // capped: never more than 1 s
 WHLD:=0;                                   // the window may re-fit now
 IF md==0 THEN
  HNT:="Drag H₁, the dashed line (α), or up/down (n)"; HNC:=cfot;
 END;
 IF md==2 THEN pv_ASnap(); END;
 RETURN 1;
END;

// ---------------- main (zero-argument signature) ----------------
EXPORT POWERVIZ()
BEGIN
 LOCAL kk, rd, nl, mm, mmF, ns5;

 pv_Init();                                          // palette constants (v2.3)

 IF pv_SelTest()==0 THEN
  RETURN "Cancelled";
 END;
 pv_EditForm();

 DIMGROB_P(G1,320,240,cwht);
 WFIT:=1;
 pv_Setup();
 HNT:="Drag H₁, the dashed line (α), or up/down (n)"; HNC:=cfot;
 pv_Draw();
 pv_Drain();                                         // the form's OK press
 pv_TDrain();                                        // and any finger

 // ---------- live loop: only device-verified key codes ----------
 // v2.4: MOUSE is polled beside GETKEY.  A touch goes to pv_Touch,
 // which follows it to the lift; the key paths are exactly as before.
 REPEAT
  kk:=GETKEY;
  mm:=MOUSE;
  mmF:=mm(1);
  rd:=0;

  IF SIZE(mmF)>0 THEN
   rd:=pv_Touch(mmF(1),mmF(2));
   IF kk<>4 THEN kk:=-1; END;                     // a touch is not a key
  ELSE
   // Arrows: act on the code GETKEY just returned OR on the key still
   // being held.  v2.2 tested ISKEYDOWN alone, so a short tap whose
   // code GETKEY had already consumed could be missed; a held key
   // still steps every pass (ISKEYDOWN stays true while it is down).
   // v2.4: pv_GSave first, so the next frame shows a ghost of H1.
   IF kk==7  OR ISKEYDOWN(7)  THEN pv_GSave(); pv_StepEff(-1); rd:=1; END;  // Left
   IF kk==8  OR ISKEYDOWN(8)  THEN pv_GSave(); pv_StepEff(1);  rd:=1; END;  // Right
   IF kk==2  OR ISKEYDOWN(2)  THEN pv_GSave(); pv_StepN(1);    rd:=1; END;  // Up
   IF kk==12 OR ISKEYDOWN(12) THEN pv_GSave(); pv_StepN(-1);   rd:=1; END;  // Down

   IF kk==50 THEN pv_AStep(1);  rd:=1; END;       // + : alpha ladder up
   IF kk==45 THEN pv_AStep(-1); rd:=1; END;       // - : alpha ladder down
   IF kk==42 THEN WFIT:=1; rd:=1; END;            // 0 : re-fit window
   IF kk==30 THEN                                  // Enter -> menu
    pv_MainMenu();
    pv_Drain();                                    // its Enter is spent
    pv_TDrain();                                   // v2.5i: and a finger on an item
    WFIT:=1; rd:=1;
   END;
  END;

  IF rd THEN
   pv_Setup();
   pv_Draw();
  END;
  WAIT(0.03);
 UNTIL kk==4;                                     // Esc

 // ---------- export & summary ----------
 EP_PWR:=PW;
 EP_NPG:=NN;                      // test 5: n1 (n2 is in the exit box)
 EP_ALPHA:=AL;
 EP_DELTA:=M1v-M0v;
 EP_ZCRIT:=ZC;

 nl:=CHAR(10);
 // v2.5g: test 5 names both sample sizes (EP_NPG holds n1)
 ns5:="n = "+pv_Cm(STRING(NN,1,0))+IFTE(TT==2 OR TT==4," per group","");   // v2.5i: no "/"
 IF TT==5 THEN ns5:="n₁ = "+pv_Cm(STRING(NN,1,0))+", n₂ = "+pv_Cm(STRING(pv_N2(NN),1,0)); END;
 // v2.5c: pv_Rel on alpha and POWER, and beta from pv_BetaS (never 1 - PW = 0)
 MSGBOX(pv_TName()+nl+
  pv_Cm(ns5)+"  α"+pv_Rel(pv_Cm(pv_FMTP(AL,3)))+"  "+pv_Cm(STRING(TL,1,0))+"-sided"+nl+
  "effect = "+pv_Cm(pv_FMT(M1v-M0v,4))+"   "+pv_Cm(IFTE(pv_MeanT()==1,"d","h"))+" = "+pv_Cm(pv_FMT(pv_EffSz(),3))+nl+
  "POWER"+pv_Rel(pv_Cm(pv_FMTP(PW,4)))+"   (β"+pv_Rel(pv_Cm(pv_BetaS(4)))+")"+nl+
  "Exported: EP_PWR EP_NPG EP_ALPHA EP_DELTA EP_ZCRIT");

 RETURN PW;
END;
