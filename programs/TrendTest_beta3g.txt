// Run TrendTest() from the Program Catalog.
// ============================================================
//
// THE QUESTION
//   k exposure levels have a natural ORDER and a score x_i
//   (a dose, hours per day, 0/1/2/3 ...).  At each level a_i of
//   n_i people have the outcome.  Pearson's chi-square (k-1 df)
//   only asks "do the proportions differ?".  The chi-square for
//   trend (1 df) asks "do they rise or fall STEADILY with the
//   score?" -- the question an epidemiologist usually means.
//
// THE ARITHMETIC (Mantel 1963; Epi Info StatCalc's form)
//   N = sum n_i,  A = sum a_i  (all cases),  P = A / N
//   S = sum(x_i a_i) - A * sum(x_i n_i) / N
//   V = A (N-A) / (N^2 (N-1)) * [ N sum(x_i^2 n_i) - (sum x_i n_i)^2 ]
//   X2 trend = S^2 / V,  1 df   ( = (N-1) r^2, r = Pearson
//     correlation of score with the 0/1 outcome over all N )
//   Armitage (1955) uses N r^2 instead; the difference is
//   N/(N-1), invisible at classroom sample sizes.  R's
//   prop.trend.test and SAS PROC FREQ / TREND print Armitage's
//   N r^2 (72.688 for the snoring table, where this program shows
//   Mantel's 72.659; beta 3d, fleet review C-T2).
//   X2 Pearson for the 2 x k table, k-1 df.
//   X2 departure = X2 Pearson - X2 trend, k-2 df: large means the
//     pattern is not a straight line.
//   Weighted least-squares line through the proportions
//     (weights n_i): slope b = S / Sxx, Sxx = sum n_i (x_i - xbar)^2,
//     xbar = sum x_i n_i / N; the line passes through (xbar, P).
//   Its 95% band: fit(x) +/- 1.96 * sqrt( P(1-P) * (1/N +
//     (x - xbar)^2 / Sxx) ) -- the weighted regression with the
//     variance of each p_i taken as P(1-P)/n_i, the null variance
//     the trend statistic itself uses: (b / SE_b)^2 = N r^2.
//   Residual of a level: p_i - fit_i, in units of
//     SE_i = sqrt(P(1-P)/n_i); the squared residuals add up to
//     Pearson - Armitage's N r^2.  The departure statistic the
//     screens show is Pearson - Mantel's (N-1) r^2, which is larger
//     by trend/(N-1): 53.1149 against the residuals' 53.1141 for the
//     sleep example (beta 3d, fleet review C-T1).
//   Midrank scores: level i gets the average rank its n_i people
//     would have if all N were ranked by level (n_1 + ... +
//     n_(i-1) + (n_i + 1)/2).  With midranks the trend test is the
//     Wilcoxon-Mann-Whitney test in disguise.
//   Each level: p_i with a 95% Wilson score interval; the odds
//     ratio against level 1 with a Woolf (log) 95% interval.
//     If any cell is 0, 0.5 is added to every cell (Haldane-
//     Anscombe) and the screens say so.
//
// EXAMPLES (Menu -> Examples); every number below was checked

// A "nice" step (1, 2 or 5 times a power of ten) at least as big
// as the value handed in -- Heckbert's rule, for axis ticks.
trNice(nnV)
BEGIN
  LOCAL nnE, nnF;
  IF nnV <= 0 THEN RETURN 1; END;
  nnE := FLOOR(LOG(nnV));
  nnF := nnV / 10^nnE;
  IF nnF <= 1 THEN RETURN 10^nnE; END;
  IF nnF <= 2 THEN RETURN 2 * 10^nnE; END;
  IF nnF <= 5 THEN RETURN 5 * 10^nnE; END;
  RETURN 10^(nnE + 1);
END;

// The sum of a list (a FOR loop, so nothing outside the verified
// command set is needed).
trSum(smL)
BEGIN
  LOCAL smI, smT;
  smT := 0;
  FOR smI FROM 1 TO SIZE(smL) DO
    smT := smT + smL(smI);
  END;
  RETURN smT;
END;

// A fresh copy of a list, built element by element.  On the Prime
// "L2 := L1" already copies, but writing the copy out keeps the
// saved originals (trOA) safe whatever the list semantics -- the
// PC test bench shares a list between two names after ":=".
trCopy(cyL)
BEGIN
  LOCAL cyI, cyR;
  cyR := {};
  FOR cyI FROM 1 TO SIZE(cyL) DO
    cyR := CONCAT(cyR, {cyL(cyI)});
  END;
  RETURN cyR;
END;

// beta 3c: fixed decimals that stay fixed.  On the Prime STRING(x, 2, d)
// prints SCIENTIFIC form when 0 < |x| < 10^-d: STRING(0.096, 2, 1) is
// "9.6E-2" with a small-capital E (U+1D07; Home, emulator, 17 Oct 2026),
// and the proportions plot showed "(+9.6E-2 SE)".  ROUND first makes the
// value either 0 or at least 10^-d, so the fixed form always holds.  A
// value that rounds to 0 prints 0.000, never -0.000.
trFx(fxV, fxD)
BEGIN
  LOCAL fxR;
  fxR := ROUND(fxV, fxD);
  IF fxR == 0 THEN fxR := 0; END;
  RETURN STRING(fxR, 2, fxD);
END;

// A p-value as text.  Tiny values are reported as "< 0.0001":
// four decimals are all a screen needs, and Pearson's and the
// departure's p, which are 1 - CDF, lose their digits far out anyway.
// (beta 3d: the trend's p is a lower normal tail and keeps its
// digits -- see TR_P in Home -- but it prints the same way here.)
trPStr(psP)
BEGIN
  IF psP < 0.0001 THEN RETURN "p < 0.0001"; END;
  RETURN "p = " + STRING(psP, 2, 4);
END;

// Decimals that show a coefficient's first three significant
// digits (at least 2, at most 6).
trDecOf(dcV)
BEGIN
  IF dcV == 0 THEN RETURN 2; END;
  RETURN MAX(2, MIN(6, 2 - FLOOR(LOG(ABS(dcV)))));
END;

// An odds ratio as text: 3 decimals below 10, 2 below 100, else 1.
// beta 3c: below 0.001 it says "< 0.001", as trPStr does for p.  Three
// decimals printed it in scientific form on the Prime (a level dragged
// to no cases against a level of all cases gives OR 0.0000003).
trOrStr(osV)
BEGIN
  IF osV >= 100 THEN RETURN STRING(osV, 2, 1); END;
  IF osV >= 10 THEN RETURN STRING(osV, 2, 2); END;
  IF osV < 0.001 THEN RETURN "< 0.001"; END;
  RETURN STRING(osV, 2, 3);
END;

// A score as text: no decimals for a whole number, one for a
// half (midranks are whole or end in .5), two otherwise.
// beta 3c: three, rounded first (trFx), for a score below 0.01 (a dose
// of 0.005): two decimals printed it in scientific form on the Prime.
trXStr(xsV)
BEGIN
  IF FP(xsV) == 0 THEN RETURN STRING(xsV, 2, 0); END;
  IF FP(2 * xsV) == 0 THEN RETURN STRING(xsV, 2, 1); END;
  IF ABS(xsV) >= 0.01 THEN RETURN STRING(xsV, 2, 2); END;
  RETURN trFx(xsV, 3);
END;

// A filled dot of radius 3 (7 px across).  beta 3b: three overlapping
// RECT_P bars -- 7 x 3, 3 x 7 and 5 x 5 -- which together cover
// exactly the pixels of a disc of radius 3.  Up to beta 3a this was

trDot(dtG, dtX, dtY, dtCol)

  RECT_P(dtG, dtX - 2, dtY - 2, dtX + 2, dtY + 2, dtCol, dtCol);

// Wait for the finger to lift -- but never for more than a second.
// MOUSE(1) is -1 while nothing touches the screen.  Without the
// time limit a stuck or phantom touch would hold the program here
// for ever, and the only way out would be a reset.
trDrain()
BEGIN
  LOCAL drT;
  drT := TICKS;                                  // milliseconds
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Wait for any key or tap (drain, wait, drain).  The help pages
// use it; the results pages and plots have their own loop (trView).
// beta 3c: returns the key code (-1 for a tap), so help can leave on
// Esc (4) from any page.
trWaitAny()
BEGIN
  LOCAL waK, waM, waM1, waR;
  REPEAT waK := GETKEY; UNTIL waK == -1;
  trDrain();
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  waR := waK;
  trDrain();
  REPEAT waK := GETKEY; UNTIL waK == -1;
  RETURN waR;
END;

// THE CAS KIT (the house shapes, BRIEF_CAS 16 Oct 2026; beta 3)
// beta 3a: trCas in the call form CASProbe beta 1b found on the
// emulator (BRIEF_CAS_v2), not yet run on a calculator.  The kit's
// parameters are
// trIv and trCs here: "trN" would be TRN (the transpose, case is
// ignored) and trS is already the trend sum S.
// ------------------------------------------------------------

// A whole number as plain digits: no digit-grouping marks, no decimal point, no
// exponent.  Home's STRING() follows the Home number format (PyProbe saw grouping
// marks and a special minus sign), which the CAS cannot read, so CAS text is built
// only from this routine.
trIStr(trIv)
BEGIN
  LOCAL dS, dV, dD;
  dV := ROUND(ABS(trIv), 0);
  IF dV == 0 THEN RETURN "0"; END;
  dS := "";
  WHILE dV > 0 DO
    dD := dV MOD 10;
    dS := CHAR(48 + dD) + dS;
    dV := (dV - dD) / 10;
  END;
  IF trIv < 0 THEN dS := "-" + dS; END;
  RETURN dS;
END;

// The ONE place this program calls the CAS.  It returns the CAS answer as text,
// or "" if the CAS failed, gave undef, an error or an infinity.
// The call form is the one CASProbe beta 1b found on the HP Prime Virtual
// Calculator (build 2026-09-09, 17 Oct 2026): CAS(s) with s a variable hands
// the text back UNEVALUATED, so the text is built into a literal CAS("...")
// that EXPR reads at run time, and the CAS is asked for string(...) of the
// answer, so a big integer crosses into PPL whole, not as a 12-digit real.
trCas(trCs)
BEGIN
  LOCAL cR;
  cR := "";
  IFERR
    cR := STRING(EXPR("CAS(" + CHAR(34) + "string(" + trCs + ")" + CHAR(34) + ")"));
  THEN
    cR := "";
  END;
  // The answer is a PPL string, and STRING() of it adds one pair of quotes
  // (CASProbe beta 1b T12, emulator): drop that pair.
  IF DIM(cR) >= 2 THEN
    IF LEFT(cR, 1) == CHAR(34) AND MID(cR, DIM(cR), 1) == CHAR(34) THEN
      cR := MID(cR, 2, DIM(cR) - 2);
    END;
  END;
  // undef, an error, or an infinity (CHAR(8734) is the infinity sign; it
  // also catches the Prime's plus-minus infinity, 1/0): no usable answer
  // an empty answer is already a failure; INSTRING("", ...) is "Invalid input" on
  // the Prime (emulator, 17 Oct 2026), so the checks run only on a non-empty answer
  IF DIM(cR) > 0 THEN
    IF INSTRING(cR, "undef") > 0 OR INSTRING(cR, "rror") > 0 OR INSTRING(cR, CHAR(8734)) > 0 THEN cR := ""; END;
  END;
  RETURN cR;
END;

// A number as EXACT CAS text: a whole number ("24", "(-1)"), or a
// whole number over a power of ten ("(16985/10)" for 1698.5) -- or
// "" when it needs more than six decimals (1/3, say) or is too big
// to write exactly.  Built with trIStr only, so no decimal point and
// no Home number format ever reaches the CAS.
trRat(rtV)
BEGIN
  LOCAL rtD, rtS, rtN;
  FOR rtD FROM 0 TO 6 DO
    rtS := rtV * 10^rtD;
    rtN := ROUND(rtS, 0);
    IF ABS(rtS - rtN) <= 0.000001 + ABS(rtS) * 0.000000000001 THEN
      IF ABS(rtN) >= 100000000000 THEN RETURN ""; END;
      IF rtD == 0 THEN
        IF rtN < 0 THEN RETURN "(" + trIStr(rtN) + ")"; END;
        RETURN trIStr(rtN);
      END;
      RETURN "(" + trIStr(rtN) + "/" + trIStr(10^rtD) + ")";
    END;
  END;
  RETURN "";
END;

// For the page: "1698.5 = 16985/10" -- a value and the exact text
// trRat wrote for it (rnR holds a "/"; its denominator's zeros give
// the decimals to show).
trRatNote(rnV, rnR)
BEGIN
  LOCAL rnP, rnD;
  rnP := INSTRING(rnR, "/");
  rnD := MID(rnR, rnP + 1, DIM(rnR) - rnP - 1);
  RETURN STRING(rnV, 2, DIM(rnD) - 1) + " = " + MID(rnR, 2, DIM(rnR) - 2);
END;

// Every rpO in the text rpT replaced by rpN (INSTRING, LEFT, MID).
// beta 3c: INSTRING("", s) is "Error: Invalid input" on the Prime
// (emulator, 17 Oct 2026), and a pattern at the very end -- the "}"
// of solve's {[a,b]} -- leaves rpS empty, which stopped page 4.  So
// the position is looked for only while text is left; an empty
// pattern gives the text back unchanged (INSTRING(s, "") is 1, which
// would loop for ever).
trRepl(rpT, rpO, rpN)
BEGIN
  LOCAL rpR, rpS, rpI, rpK;
  IF DIM(rpO) == 0 THEN RETURN rpT; END;
  rpR := ""; rpS := rpT; rpK := DIM(rpO);
  rpI := 0; IF DIM(rpS) > 0 THEN rpI := INSTRING(rpS, rpO); END;
  WHILE rpI > 0 DO
    IF rpI > 1 THEN rpR := rpR + LEFT(rpS, rpI - 1); END;
    rpR := rpR + rpN;
    IF rpI + rpK > DIM(rpS) THEN rpS := ""; ELSE rpS := MID(rpS, rpI + rpK); END;
    rpI := 0; IF DIM(rpS) > 0 THEN rpI := INSTRING(rpS, rpO); END;
  END;
  RETURN rpR + rpS;
END;

// Cut a long formula into lines of at most wpW characters, at the
// best place in the second half of each line: just after a closing
// bracket or a times sign, just before a + or - that joins two
// terms, or just after a / that is followed by a bracket -- and of
// those, the one with the fewest brackets open.  A number or a
// fraction such as 63080529409/7933174634 is never split; only a
// formula with no such place is cut at the width.  Returns the list
// of lines.
trWrap(wpT, wpW)
BEGIN
  LOCAL wpL, wpS, wpI, wpB, wpC, wpD;
  LOCAL wpBd, wpP, wpN, wpOk;
  wpL := {}; wpS := wpT;
  WHILE DIM(wpS) > wpW DO
    wpB := 0; wpBd := 999; wpD := 0;
    FOR wpI FROM 1 TO wpW DO
      wpC := MID(wpS, wpI, 1);
      wpP := ""; IF wpI > 1 THEN wpP := MID(wpS, wpI - 1, 1); END;
      wpN := MID(wpS, wpI + 1, 1);
      // a break BEFORE a + or - (one that joins terms, not a sign)
      IF wpI > IP(wpW / 2) AND (wpC == "+" OR wpC == "-") THEN
        IF wpP <> "(" AND wpP <> "^" AND wpP <> CHAR(215) AND wpP <> "/" AND wpD <= wpBd THEN
          wpB := wpI - 1; wpBd := wpD;
        END;
      END;
      IF wpC == "(" THEN wpD := wpD + 1; END;
      IF wpC == ")" THEN wpD := wpD - 1; END;
      // a break AFTER a ), a times sign, or a / followed by a bracket
      wpOk := 0;
      IF wpC == ")" OR wpC == CHAR(215) THEN wpOk := 1; END;
      IF wpC == "/" AND wpN == "(" THEN wpOk := 1; END;
      IF wpI > IP(wpW / 2) AND wpOk == 1 AND wpD <= wpBd THEN
        wpB := wpI; wpBd := wpD;
      END;
    END;
    IF wpB == 0 THEN wpB := wpW; END;
    wpL := CONCAT(wpL, {LEFT(wpS, wpB)});
    wpS := MID(wpS, wpB + 1);
  END;
  IF DIM(wpS) > 0 THEN wpL := CONCAT(wpL, {wpS}); END;
  RETURN wpL;
END;

// The scores in use, from trScM: the scores as entered (trOX), equal
// steps 1, 2, 3 ..., or midranks.  The trend statistic depends on
// the SPACING of the scores; Pearson's chi-square does not use them.
trSetX()
BEGIN
  LOCAL sxI, sxC;
  IF trScM == 0 THEN trX := trCopy(trOX); END;
  IF trScM == 1 THEN
    trX := {};
    FOR sxI FROM 1 TO trK DO
      trX := CONCAT(trX, {sxI});
    END;
  END;
  IF trScM == 2 THEN
    // midrank: the average rank the n_i people at level i would get
    // if all N were lined up by level (ties share their ranks)
    trX := {};
    sxC := 0;                                    // people at the levels below
    FOR sxI FROM 1 TO trK DO
      trX := CONCAT(trX, {sxC + (trTot(sxI) + 1) / 2});
      sxC := sxC + trTot(sxI);
    END;
  END;
END;

// beta 3f: P(chi-square(df) > x) as a finite sum of positive terms, so
// a tiny tail keeps its digits where 1 - CHISQUARE_CDF rounds to 0.
// Even df: e^(-h) (1 + h + h^2/2! + ... + h^(m-1)/(m-1)!), h = x/2,
// m = df/2.  Odd df, m = (df-1)/2: 2 P(Z < -sqrt x) +
// sqrt(2x/pi) e^(-h) (1 + x/3 + x^2/(3*5) + ... ), m terms.
trUpper(upDF, upX)
BEGIN
  LOCAL upH, upT, upS, upJ, upM;
  IF upX <= 0 THEN RETURN 1; END;
  upH := upX / 2;
  IF upDF MOD 2 == 0 THEN
    upM := upDF / 2;
    upT := EXP(-upH); upS := upT;
    FOR upJ FROM 1 TO upM - 1 DO
      upT := upT * upH / upJ; upS := upS + upT;
    END;
    RETURN MIN(1, upS);
  END;
  upM := (upDF - 1) / 2;
  upS := 2 * NORMALD_CDF(0, 1, -(upX^0.5));
  IF upM >= 1 THEN
    upT := (2 * upX / 3.14159265359)^0.5 * EXP(-upH); upS := upS + upT;   // pi to the Prime's 12 digits; ASCII, as the tally wants
    FOR upJ FROM 2 TO upM DO
      upT := upT * upX / (2 * upJ - 1); upS := upS + upT;
    END;
  END;
  RETURN MIN(1, upS);
END;

trCompute()
BEGIN
  LOCAL cpI, cpSxa, cpSxn, cpSxxn, cpEa, cpEb;
  LOCAL cpP, cpDen, cpCen, cpHalf, cpSxx, cpAdj;
  LOCAL cpA1, cpB1, cpAi, cpBi, cpO, cpSe;
  LOCAL cpPad;

  // totals and the three sums the trend statistic needs
  trNT := 0; trAT := 0; cpSxa := 0; cpSxn := 0; cpSxxn := 0;
  FOR cpI FROM 1 TO trK DO
    trNT := trNT + trTot(cpI);
    trAT := trAT + trA(cpI);
    cpSxa := cpSxa + trX(cpI) * trA(cpI);
    cpSxn := cpSxn + trX(cpI) * trTot(cpI);
    cpSxxn := cpSxxn + trX(cpI)^2 * trTot(cpI);
  END;

  // chi-square for trend (Mantel): S^2 / V, 1 df.  V > 0 because
  // trEnter, the drag and Up/Down all keep 0 < A < N.
  trS := cpSxa - trAT * cpSxn / trNT;
  trV := trAT * (trNT - trAT) / (trNT^2 * (trNT - 1)) * (trNT * cpSxxn - cpSxn^2);
  trX2T := trS^2 / trV;
  // beta 3d: the trend p as a LOWER tail, so it keeps its digits.
  // With 1 df, chi-square = Z^2, so P(chi-square > X2) is
  // P(|Z| > sqrt(X2)) = 2 * P(Z < -sqrt(X2)).  The Prime's reals hold
  // 12 digits: once the true tail is below about 5E-13 (X2 above about
  // 50), CHISQUARE_CDF rounds to exactly 1 and the old
  // 1 - CHISQUARE_CDF(1, X2) is exactly 0 -- on a 12-digit model of the
  // Prime the snoring table's TR_P is 0 instead of 1.5E-17.  A lower
  // tail is computed directly and loses nothing (fleet review 19 Oct
  // 2026, TrendTest improvement 1).  trX2T >= 0 (a square over V > 0),
  // and ^0.5 is the house square root.
  trPT := 2 * NORMALD_CDF(0, 1, -(trX2T^0.5));

  // Pearson chi-square for the 2 x k table, k-1 df
  trX2P := 0;
  FOR cpI FROM 1 TO trK DO
    cpEa := trTot(cpI) * trAT / trNT;            // expected cases
    cpEb := trTot(cpI) - cpEa;                   // expected non-cases
    trX2P := trX2P + (trA(cpI) - cpEa)^2 / cpEa + (trTot(cpI) - trA(cpI) - cpEb)^2 / cpEb;
  END;
  trPP := trUpper(trK - 1, trX2P);                 // beta 3f: keeps its digits

  // departure from linearity, k-2 df (needs at least 3 levels)
  trDFD := trK - 2;
  trX2D := MAX(0, trX2P - trX2T);
  IF trDFD >= 1 THEN
    trPD := trUpper(trDFD, trX2D);                 // beta 3f
  ELSE
    trPD := -1;
  END;

  // weighted least-squares line through the proportions
  trXbar := cpSxn / trNT;
  cpSxx := cpSxxn - trNT * trXbar^2;
  trB := trS / cpSxx;
  trA0 := trAT / trNT - trB * trXbar;
  // what the band and the residual colours need: Sxx and P(1-P)
  trSxx := cpSxx;
  trPQ := (trAT / trNT) * (1 - trAT / trNT);
  // the x-window of both plots: the scores, padded by 10% each side
  cpPad := (trX(trK) - trX(1)) * 0.1;
  trXlo := trX(1) - cpPad;
  trXhi := trX(trK) + cpPad;

  // each level: proportion and its 95% Wilson score interval
  trP := {}; trLo := {}; trHi := {};
  trHiMx := 0;                                   // running maximum of trHi
  FOR cpI FROM 1 TO trK DO
    cpP := trA(cpI) / trTot(cpI);
    cpDen := 1 + trZ^2 / trTot(cpI);
    cpCen := (cpP + trZ^2 / (2 * trTot(cpI))) / cpDen;
    cpHalf := trZ * (cpP * (1 - cpP) / trTot(cpI) + trZ^2 / (4 * trTot(cpI)^2))^0.5 / cpDen;
    trP := CONCAT(trP, {cpP});
    trLo := CONCAT(trLo, {MAX(0, cpCen - cpHalf)});
    trHi := CONCAT(trHi, {MIN(1, cpCen + cpHalf)});
    IF trHi(cpI) > trHiMx THEN trHiMx := trHi(cpI); END;
  END;

  // odds ratios against level 1, with Woolf 95% intervals.
  // A zero cell would make an OR 0 or infinite, so then 0.5 is
  // added to every cell (Haldane-Anscombe) and trHald says so.
  trHald := 0;
  FOR cpI FROM 1 TO trK DO
    IF trA(cpI) == 0 OR trA(cpI) == trTot(cpI) THEN trHald := 1; END;
  END;
  cpAdj := 0.5 * trHald;
  cpA1 := trA(1) + cpAdj;
  cpB1 := trTot(1) - trA(1) + cpAdj;
  trOR := {}; trORL := {}; trORH := {};
  FOR cpI FROM 1 TO trK DO
    cpAi := trA(cpI) + cpAdj;
    cpBi := trTot(cpI) - trA(cpI) + cpAdj;
    cpO := (cpAi * cpB1) / (cpBi * cpA1);
    cpSe := (1 / cpAi + 1 / cpBi + 1 / cpA1 + 1 / cpB1)^0.5;
    trOR := CONCAT(trOR, {cpO});
    trORL := CONCAT(trORL, {EXP(LN(cpO) - trZ * cpSe)});
    trORH := CONCAT(trORH, {EXP(LN(cpO) + trZ * cpSe)});
  END;

  // Home exports.  (beta 1b also reset the highlighted level here;
  // beta 2 leaves it, so a dragged dot stays selected -- trEnter
  // and the examples set it instead.)
  TR_X2T := trX2T; TR_P := trPT; TR_X2 := trX2P; TR_DF := trK - 1;
  TR_OR := trOR; TR_B := trB;
  trHas := 1;
END;

// DATA ENTRY

trEnter()
BEGIN
  LOCAL enK, enX, enA, enN, enI, enBad;
  LOCAL enDX, enDA, enDN;
  enK := 0;
  CHOOSE(enK, "How many exposure levels?",
    {"2 levels", "3 levels", "4 levels", "5 levels", "6 levels"});
  IF enK == 0 THEN RETURN 0; END;
  enK := enK + 1;

  // the dialogs open on the data already loaded when it has this
  // many levels (so the Data soft key edits what is on the screen);
  // otherwise on scores 1, 2, 3 ... and zero counts
  IF trHas == 1 AND enK == trK THEN
    enDX := trOX; enDA := trA; enDN := trTot;
  ELSE
    enDX := MAKELIST(K, K, 1, enK, 1);
    enDA := MAKELIST(0, K, 1, enK, 1);
    enDN := MAKELIST(0, K, 1, enK, 1);
  END;

  // scores: 1, 2, 3 ... unless real doses are known
  enX := trAsk(enK, "Scores, in order of exposure", "Score",
    "A number per level: 0,1,2.. or the dose itself", enDX);
  IF SIZE(enX) == 0 THEN RETURN 0; END;

  enA := trAsk(enK, "Cases (with the outcome)", "Cases",
    "How many at this level have the outcome", enDA);
  IF SIZE(enA) == 0 THEN RETURN 0; END;

  enN := trAsk(enK, "Totals (everyone at the level)", "Total n",
    "How many people are at this level in all", enDN);
  IF SIZE(enN) == 0 THEN RETURN 0; END;

  // ---- checks that teach ----
  // beta 3d: cases and totals count people, so a fraction (24.5 cases
  // of 100.5) is refused (enBad 4).  Before, the arithmetic ran on
  // 24.5 while the table printed "25/101" (fleet review 19 Oct 2026,
  // C-T4).  Scores may have fractions: a dose of 0.5 is fine.
  enBad := 0;
  FOR enI FROM 1 TO enK DO
    IF enN(enI) <= 0 THEN enBad := 1; END;
    IF enA(enI) < 0 OR enA(enI) > enN(enI) THEN enBad := 2; END;
    IF enI > 1 THEN
      IF enX(enI) <= enX(enI - 1) THEN enBad := 3; END;
    END;
    IF FP(enA(enI)) <> 0 OR FP(enN(enI)) <> 0 THEN enBad := 4; END;
  END;
  IF enBad == 1 THEN MSGBOX("Every total n must be above 0."); RETURN 0; END;
  IF enBad == 2 THEN MSGBOX("Cases must be between 0 and the total at that level."); RETURN 0; END;
  IF enBad == 3 THEN MSGBOX("Scores must increase from level 1 to level " + STRING(enK, 2, 0) + ": the levels are ordered, and the order is the point of the test."); RETURN 0; END;
  IF enBad == 4 THEN MSGBOX("Cases and totals count people: whole numbers only."); RETURN 0; END;
  IF trSum(enA) == 0 OR trSum(enA) == trSum(enN) THEN
    MSGBOX("Every level has the same outcome (all cases, or none), so there is no trend to test.");
    RETURN 0;
  END;
  IF trSum(enN) < 2 THEN MSGBOX("At least two people are needed."); RETURN 0; END;

  trK := enK; trTot := enN;
  trOX := trCopy(enX);                           // kept for Restore
  trOA := trCopy(enA);
  trA := trCopy(enA);
  trScM := 0; trEdit := 0; trIllu := 0; trSel := 1;
  trSetX();
  trLbl := "Your data: " + STRING(trK, 2, 0) + " levels, N = " + STRING(trSum(enN), 2, 0) + ", " + STRING(trSum(enA), 2, 0) + " cases";
  trLbl2 := "";
  trCompute();
  RETURN 1;
END;

// One small chi-square panel for page 2: the curve for cqDf df
// between pixel columns cqX1 and cqX2, the tail beyond the statistic
// cqSt shaded (its area is the p-value cqPv), the 5% cut-off dotted
// red, the statistic as a solid line in cqCol.
trChiPanel(cqX1, cqX2, cqDf, cqSt, cqPv, cqCol, cqNm)
BEGIN
  LOCAL cqW, cqDen, cqI, cqXmx, cqCr, cqYmx;
  LOCAL cqStp, cqDec, cqV, cqN, cqPx, cqPy;
  LOCAL cqLast, cqSx, cqCx, cqT, cqB;
  cqT := 34; cqB := 120;                         // the panel's top and bottom
  cqW := cqX2 - cqX1;
  cqCr := CHISQUARE_ICDF(cqDf, 0.95);            // the 5% cut-off
  // x from 0 to past the cut-off, far enough to include the
  // statistic -- unless it is more than 4 cut-offs out; then it
  // is marked at the right edge as off the scale
  cqXmx := MAX(cqCr * 1.6, MIN(cqSt * 1.15, cqCr * 4));
  cqStp := trNice(cqXmx / 4);
  cqXmx := CEILING(cqXmx / cqStp - 0.000001) * cqStp;
  cqDec := MAX(0, -FLOOR(LOG(cqStp)));
  // the height: df 1 rises without limit near 0, so it is simply
  // clipped; df 2 starts at 0.5; larger df peak at df - 2
  cqYmx := 0.6;
  IF cqDf == 2 THEN cqYmx := 0.55; END;
  IF cqDf > 2 THEN cqYmx := CHISQUARE(cqDf, cqDf - 2) * 1.15; END;
  // ---- the density, once per pixel column (Lessons Learned 6):
  // the shaded tail and the curve both read cqDen ----
  cqDen := {};
  FOR cqI FROM 1 TO cqW DO
    cqDen := CONCAT(cqDen, {CHISQUARE(cqDf, cqI / cqW * cqXmx)});
  END;
  TEXTOUT_P(cqNm + ": " + trChi + "(" + STRING(cqDf, 2, 0) + ")", G1, cqX1, 22, 1, cqCol, cqW);
  // ---- x ticks and grid ----
  cqV := 0; cqN := 0;
  WHILE cqV <= cqXmx + cqStp / 1000 AND cqN < 10 DO
    cqPx := cqX1 + IP(cqV / cqXmx * cqW + 0.5);
    LINE_P(G1, cqPx, cqT, cqPx, cqB, trGRID);
    LINE_P(G1, cqPx, cqB, cqPx, cqB + 3, #000000h);
    TEXTOUT_P(STRING(cqV, 2, cqDec), G1, cqPx - 5, cqB + 5, 1, trGRAY, 30);
    cqV := cqV + cqStp;
    cqN := cqN + 1;
  END;
  // ---- the p-value: the tail beyond the statistic, column by column ----
  cqSx := cqX1 + IP(MIN(cqSt, cqXmx) / cqXmx * cqW + 0.5);
  IF cqSt < cqXmx THEN
    FOR cqPx FROM MAX(cqSx, cqX1 + 1) TO cqX2 DO
      cqPy := cqB - IP(MIN(cqDen(cqPx - cqX1) / cqYmx, 1) * (cqB - cqT));
      IF cqPy < cqB THEN LINE_P(G1, cqPx, cqB - 1, cqPx, cqPy, trLMAG); END;
    END;
  END;
  // ---- the curve, pixel column by pixel column ----
  cqLast := -1;
  FOR cqPx FROM cqX1 + 1 TO cqX2 DO
    cqPy := cqB - IP(MIN(cqDen(cqPx - cqX1) / cqYmx, 1) * (cqB - cqT));
    IF cqLast >= 0 THEN LINE_P(G1, cqPx - 1, cqLast, cqPx, cqPy, trBLUE); END;
    cqLast := cqPy;
  END;
  LINE_P(G1, cqX1, cqT, cqX1, cqB, #000000h);
  LINE_P(G1, cqX1, cqB, cqX2, cqB, #000000h);
  // ---- the 5% cut-off, dotted red ----
  cqCx := cqX1 + IP(cqCr / cqXmx * cqW + 0.5);
  FOR cqI FROM cqT TO cqB STEP 3 DO
    PIXON_P(G1, cqCx, cqI, trRED);
  END;
  TEXTOUT_P("5%", G1, cqCx + 3, cqT + 14, 1, trRED, 24);
  // ---- the statistic: a solid line, or at the edge if off the scale ----
  IF cqSt <= cqXmx THEN
    LINE_P(G1, cqSx, cqT, cqSx, cqB, cqCol);
    LINE_P(G1, cqSx + 1, cqT, cqSx + 1, cqB, cqCol);
  ELSE
    LINE_P(G1, cqX2, cqT, cqX2, cqB, cqCol);
    LINE_P(G1, cqX2 - 1, cqT, cqX2 - 1, cqB, cqCol);
    TEXTOUT_P("further " + CHAR(8594), G1, cqX2 - 52, cqT + 2, 1, cqCol, 50);
  END;
  TEXTOUT_P(trChi + " = " + trFx(cqSt, 2) + ", " + trPStr(cqPv), G1, cqX1, 140, 1, cqCol, cqW + 6);
END;

// ------------------------------------------------------------
// PAGE 4 (beta 3): LEAST SQUARES AS CALCULUS
// The line on the proportions plot is the weighted least-squares
// line: of all lines p = a + b x it makes
//     SSE(a, b) = sum over the levels of  n (p - a - b x)^2
// smallest (p = cases/n, x = the score in use, weights n).  At the
// bottom of that bowl both partial derivatives are 0 -- the two
// "normal equations" -- and solving them gives the program's
//     b = S / Sxx  and  a = P - b xbar.
// The CAS does the algebra on the table on screen, exactly.
// ------------------------------------------------------------

// A CAS answer that is no use as an EXACT answer: one with a
// decimal point (the CAS is in its approximate mode) counts as no
// answer, and the page falls back to the decimals.
// beta 3c: "" (the CAS did not answer) stays "" without INSTRING, which
// stops the program on an empty text ("Invalid input", emulator).
trCxClean(ccR)
BEGIN
  IF DIM(ccR) == 0 THEN RETURN ""; END;
  IF INSTRING(ccR, ".") > 0 THEN RETURN ""; END;
  RETURN ccR;
END;

// Read a and b out of solve's answer, whatever brackets the CAS
// puts round them -- {[a,b]} on the Prime (CASProbe beta 1b T36,
// emulator 17 Oct 2026), [[a,b]] or list[[a,b]] in desktop Giac, or
// [a=...,b=...] -- into trCxSa and trCxSb; both stay "" if the
// answer is not exactly two values.
trCxSol(soR)
BEGIN
  LOCAL soS, soK;
  trCxSa := ""; trCxSb := "";
  IF DIM(soR) == 0 THEN RETURN; END;
  soS := trRepl(soR, "list", "");
  soS := trRepl(soS, "[", ""); soS := trRepl(soS, "]", "");
  soS := trRepl(soS, "{", ""); soS := trRepl(soS, "}", "");
  soS := trRepl(soS, " ", "");
  // beta 3c: nothing left (solve gave {} or []): no answer.  INSTRING
  // on an empty text stops the program on the Prime (emulator).
  IF DIM(soS) == 0 THEN RETURN; END;
  soK := INSTRING(soS, ",");
  IF soK == 0 THEN soK := INSTRING(soS, ";"); END;
  IF soK < 2 OR soK >= DIM(soS) THEN RETURN; END;
  trCxSa := LEFT(soS, soK - 1);
  trCxSb := MID(soS, soK + 1);
  IF INSTRING(trCxSb, ",") > 0 OR INSTRING(trCxSb, ";") > 0 THEN
    trCxSa := ""; trCxSb := "";                  // more than two values
    RETURN;
  END;
  IF INSTRING(trCxSa, "=") > 0 THEN trCxSa := MID(trCxSa, INSTRING(trCxSa, "=") + 1); END;
  IF INSTRING(trCxSb, "=") > 0 THEN trCxSb := MID(trCxSb, INSTRING(trCxSb, "=") + 1); END;
  // beta 3c: "a=" with nothing after it leaves an empty text: no answer
  IF DIM(trCxSa) == 0 OR DIM(trCxSb) == 0 THEN trCxSa := ""; trCxSb := ""; RETURN; END;
  // a symbol left in it means no single solution: not usable
  IF INSTRING(trCxSa, "tr") > 0 OR INSTRING(trCxSb, "tr") > 0 THEN trCxSa := ""; trCxSb := ""; END;
END;

// Ask the CAS for page 4, if the table or the scores changed since
// it last answered.  The SSE is written out for the data on screen,
//   n1*(a1/n1-tra-trb*x1)^2 + n2*(a2/n2-tra-trb*x2)^2 + ...
// in the symbols tra (a) and trb (b), every number exact (trRat).
// Then four calls, through trCas only:
//   normal(SSE)                     the SSE in order, a polynomial
//   normal(diff(SSE,tra))           its slope along a
//   normal(diff(SSE,trb))           its slope along b
//   solve([d/da = 0, d/db = 0], [tra, trb])   the normal equations
trCxMake()
BEGIN
  LOCAL cxE, cxI, cxN, cxA, cxX, cxR;
  cxE := ""; trCxNt := "";
  FOR cxI FROM 1 TO trK DO
    cxN := trRat(trTot(cxI));
    cxA := trRat(trA(cxI));
    cxX := trRat(trX(cxI));
    IF DIM(cxN) == 0 OR DIM(cxA) == 0 OR DIM(cxX) == 0 THEN
      // a value with more than six decimals: no exact text
      trCxSt := 3; trCxK := ""; trCasOK := 0;
      RETURN;
    END;
    // one example of a value written over a power of ten, for the page
    IF DIM(trCxNt) == 0 AND INSTRING(cxX, "/") > 0 THEN
      trCxNt := trRatNote(trX(cxI), cxX);
    END;
    IF cxI > 1 THEN cxE := cxE + "+"; END;
    cxE := cxE + cxN + "*(" + cxA + "/" + cxN + "-tra-trb*" + cxX + ")^2";
  END;
  IF cxE == trCxK THEN RETURN; END;              // answered already
  trCxK := cxE;
  trCxE := ""; trCxA := ""; trCxB := ""; trCxSa := ""; trCxSb := "";
  trCasOK := 0;
  // the longest call (solve) holds the SSE twice
  IF 2 * DIM(cxE) + 60 > trCasMx THEN trCxSt := 4; RETURN; END;
  // the G1's CAS speed is not known: say what is happening
  RECT_P(G0, 60, 100, 260, 124, trFOOT, #FFFFFFh);
  TEXTOUT_P("asking the CAS ...", G0, 106, 107, 1, trFOOT);
  trCxE := trCxClean(trCas("normal(" + cxE + ")"));
  trCxA := trCxClean(trCas("normal(diff(" + cxE + ",tra))"));
  trCxB := trCxClean(trCas("normal(diff(" + cxE + ",trb))"));
  cxR := trCxClean(trCas("solve([diff(" + cxE + ",tra)=0,diff(" + cxE + ",trb)=0],[tra,trb])"));
  trCxSol(cxR);
  trCxSt := 2;
  IF DIM(trCxE) > 0 AND DIM(trCxA) > 0 AND DIM(trCxB) > 0 AND DIM(trCxSa) > 0 THEN
    trCxSt := 1; trCasOK := 1;
  END;
END;

// A CAS answer written for people: tra -> a, trb -> b, * -> times.
trCxShow(csT)
BEGIN
  LOCAL csS;
  csS := trRepl(csT, "tra", "a");
  csS := trRepl(csS, "trb", "b");
  csS := trRepl(csS, "*", CHAR(215));
  RETURN csS;
END;

// Draw clT wrapped at 49 characters from y clY, in colour clC, in at
// most clMx lines; if it needs more, say how long it is instead (an
// exact answer is never cut silently).  Returns the next free y.
trCxLines(clT, clY, clMx, clC, clNm)
BEGIN
  LOCAL clL, clI, clYy;
  clL := trWrap(clT, 49);
  clYy := clY;
  IF SIZE(clL) > clMx THEN
    TEXTOUT_P(clNm + ": " + STRING(DIM(clT), 2, 0) + " characters, too long to show", G1, 10, clYy, 1, trGRAY, 300);
    RETURN clYy + 11;
  END;
  FOR clI FROM 1 TO SIZE(clL) DO
    TEXTOUT_P(clL(clI), G1, 10, clYy, 1, clC, 302);
    clYy := clYy + 11;
  END;
  RETURN clYy;
END;

// One line: the exact fraction beside the program's decimal,
// "b = 95339/4757909   (program 0.020038)" -- or, when the fraction
// is too long for the line, how many digits it has (an exact answer
// is never cut silently).  Returns the next free y.
trCxFrac(cfNm, cfF, cfD, cfY)
BEGIN
  LOCAL cfI, cfN, cfS;
  cfS := cfNm + " = " + cfF + "   (program " + trDec(cfD) + ")";
  IF DIM(cfS) > 49 THEN
    cfN := 0;
    FOR cfI FROM 1 TO DIM(cfF) DO
      IF INSTRING("0123456789", MID(cfF, cfI, 1)) > 0 THEN cfN := cfN + 1; END;
    END;
    cfS := cfNm + ": exact fraction, " + STRING(cfN, 2, 0) + " digits (program " + trDec(cfD) + ")";
  END;
  TEXTOUT_P(cfS, G1, 10, cfY, 1, trMAG, 302);
  RETURN cfY + 11;
END;

// The program's own decimal beside an exact fraction: about seven
// significant digits (0.02003799, 0.0000528181, 16.63624).
// beta 3c: rounded first (trFx), so a value below 1E-10 -- a slope that
// is 0 but for the last digit -- prints "0", never in scientific form.
trDec(deV)
BEGIN
  LOCAL deD;
  IF deV == 0 THEN RETURN "0"; END;
  deD := MAX(2, MIN(10, 6 - FLOOR(LOG(ABS(deV)))));
  IF ROUND(deV, deD) == 0 THEN RETURN "0"; END;
  RETURN trFx(deV, deD);
END;

// A number for the decimal page: whole numbers as they are, others
// with about six significant digits and no trailing zeros (260.8,
// not 260.800).  MID, not RIGHT: RIGHT is not in the verified list.
// beta 3c: rounded first (trFx).  A sum below 1E-6 (scores such as -2/3,
// 1/3, 1/3 leave -1E-11 on the Prime) printed in scientific form, and
// the zero strip below then cut the exponent's last 0 (1E-10 -> 1E-1).
trNum(nmV)
BEGIN
  LOCAL nmS, nmD;
  IF FP(nmV) == 0 AND ABS(nmV) < 1000000000 THEN RETURN STRING(nmV, 2, 0); END;
  IF nmV == 0 THEN RETURN "0"; END;
  // beta 3c: from 1E12 up the Prime prints scientific form whatever the
  // decimals (it holds 12 digits; midranks on N = 20000 give sum n x^2 =
  // 7E13).  Five decimals keep six significant digits, and the zero
  // strip below must not reach the exponent.
  IF ABS(nmV) >= 1000000000000 THEN RETURN trFx(nmV, 5); END;
  nmD := MAX(1, MIN(6, 5 - FLOOR(LOG(ABS(nmV)))));
  IF ROUND(nmV, nmD) == 0 THEN RETURN "0"; END;
  nmS := trFx(nmV, nmD);
  WHILE DIM(nmS) > 1 AND MID(nmS, DIM(nmS), 1) == "0" DO nmS := LEFT(nmS, DIM(nmS) - 1); END;
  IF MID(nmS, DIM(nmS), 1) == "." THEN nmS := LEFT(nmS, DIM(nmS) - 1); END;
  RETURN nmS;
END;

// PLOT 1: proportions by level, Wilson bars, WLS trend line and
// its 95% band; a dot can be dragged

// pixel x of a score, for the plot window xlo..xhi
trPxOf(pxV, pxLo, pxHi)
BEGIN
  RETURN trPL + IP((pxV - pxLo) / (pxHi - pxLo) * (trPR - trPL) + 0.5);
END;

// pixel y of a proportion on plot 1's scale (0 .. trYhi), kept
// inside the plot box -- during a drag the scale is frozen, so a
// Wilson bar or the band can run past the top and is cut there
trPyOf(pyV)
BEGIN
  LOCAL pyY;
  pyY := trPB - IP(pyV / trYhi * (trPB - trPTop) + 0.5);
  RETURN MAX(trPTop, MIN(trPB, pyY));
END;

// plot 1's y-scale: a nice step, the top a whole number of steps
// above the largest Wilson upper limit (never above 1)
trFitY()
BEGIN
  trYStp := trNice(trHiMx / 4);
  trYhi := CEILING(trHiMx / trYStp - 0.000001) * trYStp;
  IF trYhi > 1 THEN trYhi := 1; END;
  trYDec := MAX(0, -FLOOR(LOG(trYStp)));
END;

// Compose plot 1 in G1.  dpF is the fraction of the fitted slope to
// draw: 1 normally; 0 .. 1 while the fit animates (trAnimate).
trDrawProp(dpF)
BEGIN
  LOCAL dpI, dpX, dpY, dpY1, dpY2, dpV;
  LOCAL dpCol, dpS, dpPx, dpXv, dpN, dpB;
  LOCAL dpA, dpFit, dpH, dpSe, dpR, dpT;
  LOCAL dpV1, dpV2, dpTa, dpTb, dpLo, dpHi;
  // ---- the y window: refitted unless a drag has frozen it ----
  IF trFrz == 0 THEN trFitY(); END;
  // the line drawn in this frame: slope trB * dpF, pivoting on the
  // point (xbar, overall p) that every weighted line passes through
  dpB := trB * dpF;
  dpA := trAT / trNT - dpB * trXbar;

  DIMGROB_P(G1, 320, 240, #FFFFFFh);
  IF trIllu == 1 THEN
    TEXTOUT_P("Made-up data: proportion by level", G1, 10, 4, 2, trRED);
  ELSE
    TEXTOUT_P("Proportion with the outcome, by level", G1, 10, 4, 2, #000000h);
  END;
  // ---- grid and y ticks ----
  dpN := 0;
  dpV := 0;
  WHILE dpV <= trYhi + 0.000001 AND dpN < 12 DO
    dpY := trPyOf(dpV);
    LINE_P(G1, trPL, dpY, trPR, dpY, trGRID);
    TEXTOUT_P(STRING(dpV, 2, trYDec), G1, 6, dpY - 5, 1, trGRAY, 38);
    dpV := dpV + trYStp;
    dpN := dpN + 1;
  END;
  // ---- the 95% band round the fitted line, light magenta, one
  // pixel column at a time (only for the fitted line, not while it
  // animates): fit +/- 1.96 * sqrt(P(1-P) (1/N + (x-xbar)^2/Sxx)).
  // Narrowest at xbar, where the line is pinned by all N people. ----
  IF dpF >= 1 THEN
    FOR dpPx FROM trPL + 1 TO trPR DO
      dpXv := trXlo + (dpPx - trPL) / (trPR - trPL) * (trXhi - trXlo);
      dpFit := trA0 + trB * dpXv;
      dpH := trZ * (trPQ * (1 / trNT + (dpXv - trXbar)^2 / trSxx))^0.5;
      dpY1 := trPyOf(dpFit + dpH);
      dpY2 := trPyOf(dpFit - dpH);
      IF dpY2 > dpY1 THEN LINE_P(G1, dpPx, dpY1, dpPx, dpY2, trLMAG); END;
    END;
  END;
  // ---- axes ----
  LINE_P(G1, trPL, trPTop, trPL, trPB, #000000h);
  LINE_P(G1, trPL, trPB, trPR, trPB, #000000h);
  // ---- x ticks: one per level, labelled with its score ----
  FOR dpI FROM 1 TO trK DO
    dpX := trPxOf(trX(dpI), trXlo, trXhi);
    LINE_P(G1, dpX, trPB, dpX, trPB + 3, #000000h);
    TEXTOUT_P(trXStr(trX(dpI)), G1, dpX - 8, trPB + 5, 1, trGRAY, 40);
  END;
  TEXTOUT_P("score:", G1, 6, trPB + 5, 1, trGRAY);
  // ---- overall proportion, dotted ----
  dpY := trPyOf(trAT / trNT);
  trDotLine(G1, trPL, trPR, dpY, trGRAY);
  TEXTOUT_P("overall", G1, trPR - 44, dpY - 11, 1, trGRAY);
  // ---- the line, 2 px thick: one LINE_P between the points where
  // it enters and leaves the window 0..trYhi.  Along the box, t runs
  // 0 (left edge, value dpV1) to 1 (right edge, value dpV2). ----
  dpV1 := dpA + dpB * trXlo;
  dpV2 := dpA + dpB * trXhi;
  dpLo := 0; dpHi := 1;
  IF dpV2 == dpV1 THEN
    IF dpV1 < 0 OR dpV1 > trYhi THEN dpHi := -1; END;    // flat, outside
  ELSE
    dpTa := (0 - dpV1) / (dpV2 - dpV1);                  // where it crosses 0
    dpTb := (trYhi - dpV1) / (dpV2 - dpV1);              // ... and the top
    dpLo := MAX(0, MIN(dpTa, dpTb));
    dpHi := MIN(1, MAX(dpTa, dpTb));
  END;
  IF dpHi > dpLo THEN
    dpX := trPL + IP(dpLo * (trPR - trPL) + 0.5);
    dpT := trPL + IP(dpHi * (trPR - trPL) + 0.5);
    dpY1 := trPyOf(dpV1 + dpLo * (dpV2 - dpV1));
    dpY2 := trPyOf(dpV1 + dpHi * (dpV2 - dpV1));
    LINE_P(G1, dpX, dpY1, dpT, dpY2, trMAG);
    LINE_P(G1, dpX, MAX(trPTop, dpY1 - 1), dpT, MAX(trPTop, dpY2 - 1), trMAG);
  END;
  // ---- Wilson bars and the dots, coloured by the residual from
  // this frame's line: blue above it by more than 1 SE, red below
  // by more than 1 SE, grey within; SE = sqrt(P(1-P)/n).  The dots'
  // pixel centres go into trDX, trDY for the touch handler. ----
  trDX := {}; trDY := {};
  FOR dpI FROM 1 TO trK DO
    dpX := trPxOf(trX(dpI), trXlo, trXhi);
    dpY1 := trPyOf(trHi(dpI));
    dpY2 := trPyOf(trLo(dpI));
    dpY := trPyOf(trP(dpI));
    dpR := trP(dpI) - (dpA + dpB * trX(dpI));
    dpSe := (trPQ / trTot(dpI))^0.5;
    dpCol := trGRAY;
    IF dpR > dpSe THEN dpCol := trBLUE; END;
    IF dpR < -dpSe THEN dpCol := trRED; END;
    LINE_P(G1, dpX, dpY1, dpX, dpY2, dpCol);
    LINE_P(G1, dpX - 3, dpY1, dpX + 3, dpY1, dpCol);
    LINE_P(G1, dpX - 3, dpY2, dpX + 3, dpY2, dpCol);
    trDot(G1, dpX, dpY, dpCol);
    IF dpI == trSel THEN                         // the selected level: a black ring
      ARC_P(G1, dpX, dpY, 6, #000000h);
      ARC_P(G1, dpX, dpY, 7, #000000h);
    END;
    trDX := CONCAT(trDX, {dpX});
    trDY := CONCAT(trDY, {dpY});
  END;
  // ---- legend ----
  TEXTOUT_P("above", G1, trPL + 4, trPTop + 2, 1, trBLUE);
  TEXTOUT_P("below", G1, trPL + 38, trPTop + 2, 1, trRED);
  TEXTOUT_P("within 1 SE", G1, trPL + 72, trPTop + 2, 1, trGRAY);
  TEXTOUT_P("WLS line, 95% band", G1, trPL + 146, trPTop + 2, 1, trMAG);
  // ---- footer, four lines above the soft keys (y 170-217) ----
  // the selected level: its count, p, the line's value there, and
  // the residual in SE units
  dpFit := dpA + dpB * trX(trSel);
  dpR := (trP(trSel) - dpFit) / (trPQ / trTot(trSel))^0.5;
  IF dpR < 0 THEN dpS := "-"; ELSE dpS := "+"; END;
  // beta 3c: trFx (ROUND, then fixed decimals), so a small value never
  // prints in scientific form -- the emulator showed "(+9.6E-2 SE)"
  dpS := "L" + STRING(trSel, 2, 0) + ": x = " + trXStr(trX(trSel)) + ", " + STRING(trA(trSel), 2, 0) + "/" + STRING(trTot(trSel), 2, 0) + " = " + trFx(trP(trSel), 3) + ", line " + trFx(dpFit, 3) + " (" + dpS + trFx(ABS(dpR), 1) + " SE)";
  TEXTOUT_P(dpS, G1, 6, 170, 1, #000000h, 310);
  // the trend statistic: while the fit animates, the value it would
  // have for the slope drawn (it grows with the slope squared)
  IF dpF >= 1 THEN
    TEXTOUT_P(trChi + "trend = " + trFx(trX2T, 2) + ", " + trPStr(trPT) + ", scores " + trScName(), G1, 6, 182, 1, trMAG, 310);
  ELSE
    TEXTOUT_P("fitting: slope " + CHAR(215) + " " + STRING(dpF, 2, 2) + ", " + trChi + "trend = " + trFx(trX2T * dpF^2, 2), G1, 6, 182, 1, trMAG, 310);
  END;
  TEXTOUT_P("Pearson " + trChi + " = " + trFx(trX2P, 2), G1, 6, 194, 1, trBLUE, 120);
  IF trDFD >= 1 THEN
    TEXTOUT_P("departure = " + trFx(trX2D, 2) + ", " + trPStr(trPD), G1, 130, 194, 1, trGRN, 186);
  ELSE
    TEXTOUT_P("departure: needs 3+ levels", G1, 130, 194, 1, trGRN, 186);
  END;
  TEXTOUT_P("drag a dot or ⇅ = cases, ⇆ = level, Esc = menu", G1, 6, 206, 1, trFOOT, 310);
END;

// The fit, animated: the line starts flat at the overall proportion
// and swings to the fitted slope over 15 frames while the trend
// statistic counts up.  Any key or tap skips to the end.
trAnimate()
BEGIN
  LOCAL anI, anGo, anK, anM, anM1;
  anI := 0; anGo := 1;
  WHILE anGo == 1 DO
    trDrawProp(anI / 15);
    trShow();
    anK := GETKEY;
    anM := MOUSE;
    anM1 := anM(1);
    IF anK > -1 OR SIZE(anM1) > 0 THEN anGo := 0; END;
    anI := anI + 1;
    IF anI >= 15 THEN anGo := 0; END;
  END;
  REPEAT anK := GETKEY; UNTIL anK == -1;         // a skipping key or
  trDrain();                                     // tap does nothing else
END;

BEGIN

  doLlo := 0; doLhi := 0;                        // always include OR = 1
  FOR doI FROM 2 TO trK DO
    doLlo := MIN(doLlo, LOG(trORL(doI)));
    doLhi := MAX(doLhi, LOG(trORH(doI)));
  END;

  doLlo := doLlo - 0.08 * doSpan; doLhi := doLhi + 0.08 * doSpan;

    TEXTOUT_P("Odds ratio vs level 1 (log scale)", G1, 10, 4, 2, #000000h);
  END;
  // ---- ticks at 1, 2, 5 in each decade ----
  doMul := {1, 2, 5};
  FOR doD FROM FLOOR(doLlo) TO CEILING(doLhi) DO
    FOR doM FROM 1 TO 3 DO
      doTk := doD + LOG(doMul(doM));
      IF doTk >= doLlo AND doTk <= doLhi THEN

        LINE_P(G1, trPL, doY, trPR, doY, trGRID);

        IF doV >= 1 THEN
          TEXTOUT_P(STRING(doV, 2, 0), G1, 6, doY - 5, 1, trGRAY, 38);
        ELSE
          TEXTOUT_P(STRING(doV, 2, MAX(1, -doD)), G1, 6, doY - 5, 1, trGRAY, 38);
        END;
      END;

// ------------------------------------------------------------
// TOUCH ON THE PLOTS
// ------------------------------------------------------------

// A finger came down at (ptX0, ptY0) on plot ptW (4 proportions,
// 5 odds ratios).  On plot 1, within 10 px of a dot: that dot is
// grabbed -- if the finger then moves more than 8 px it is a drag,
// and every sample sets the level's cases from the finger's height
// on the FROZEN y-scale; if it lifts without moving it was a tap,
// which selects the level.  Anywhere else in the plot box, a tap
// selects the nearest level in x (within 14 px).
trPlotTouch(ptW, ptX0, ptY0)
BEGIN
  LOCAL ptI, ptHit, ptD, ptX, ptM, ptM1;
  LOCAL ptOn, ptMv, ptNew, ptCLo, ptCHi, ptP;
  ptHit := 0;
  IF ptW == 4 THEN
    ptD := 11;
    FOR ptI FROM 1 TO trK DO
      ptX := MAX(ABS(ptX0 - trDX(ptI)), ABS(ptY0 - trDY(ptI)));
      IF ptX <= 10 AND ptX < ptD THEN ptD := ptX; ptHit := ptI; END;
    END;
  END;
  IF ptHit > 0 THEN
    trSel := ptHit;
    // the other levels' cases stay fixed; keep at least one case and
    // one non-case in the whole table, or V = 0 and the test has
    // nothing to compare
    ptCLo := MAX(0, 1 - (trAT - trA(ptHit)));
    ptCHi := MIN(trTot(ptHit), trNT - 1 - (trAT - trA(ptHit)));
    ptMv := 0;
    trFrz := 1;                                  // freeze the y-scale
    REPEAT
      ptM := MOUSE;
      ptM1 := ptM(1);
      ptOn := 0;                                 // finger still on the screen?
      IF SIZE(ptM1) > 0 THEN
        IF ptM1(1) >= 0 AND ptM1(1) <= 319 AND ptM1(2) >= 0 AND ptM1(2) <= 239 THEN ptOn := 1; END;
      END;
      IF ptOn == 1 THEN
        IF ABS(ptM1(1) - ptX0) > 8 OR ABS(ptM1(2) - ptY0) > 8 THEN ptMv := 1; END;
        IF ptMv == 1 THEN
          // finger height -> proportion on the frozen scale -> cases
          ptP := (trPB - ptM1(2)) / (trPB - trPTop) * trYhi;
          ptP := MAX(0, MIN(trYhi, ptP));
          ptNew := MAX(ptCLo, MIN(ptCHi, ROUND(ptP * trTot(ptHit), 0)));
          IF ptNew <> trA(ptHit) THEN            // redraw only on a change
            trA(ptHit) := ptNew;
            trEdit := 1;
            trCompute();
            trDrawProp(1);
            trShow();
          END;
        END;
      END;
    UNTIL ptOn == 0;
    trFrz := 0;                                  // refit on lift
  ELSE
    ptD := 15;
    IF ptY0 >= trPTop AND ptY0 <= trPB THEN
      FOR ptI FROM 1 TO trK DO
        ptX := trPxOf(trX(ptI), trXlo, trXhi);
        IF ABS(ptX0 - ptX) < ptD THEN ptD := ABS(ptX0 - ptX); ptHit := ptI; END;
      END;
    END;
    IF ptHit > 0 THEN trSel := ptHit; END;
  END;
  trDrain();
END;


// ------------------------------------------------------------
// THE VIEW LOOP: results pages 1-4 and the two plots, one loop,
// one soft-key strip.  vwV: 1-3 results pages, 4 proportions plot,
// 5 odds-ratio plot, 6 results page 4 (least squares as calculus,
// beta 3 -- numbered 6 so that the plots keep 4 and 5 and the drag
// code is untouched).
//   Soft keys: Prop, OR, Scores, Data, Restore, Menu.
//   Keys:  results  Left/Right = page, Enter = next (menu after 4)
//          plots    Left/Right = level, Up/Down = cases (plot 1),
//                   Enter = menu, - = the other plot
//          all      + = Scores, 0 = Restore, Esc = menu
//   Taps:  results  left half = back, right half = on
//          plots    a dot = grab and drag it (plot 1); else select
// ------------------------------------------------------------

// The results pages in the order the student turns them: 1, 2, 3,
// then 6 (page 4).  trPgStep(v, d) is the page d steps (+1 or -1)
// from v, staying at the first and the last.
trPgStep(psV, psD)
BEGIN
  LOCAL psO, psJ, psK;
  psO := {1, 2, 3, 6};
  psK := 1;
  FOR psJ FROM 1 TO 4 DO
    IF psO(psJ) == psV THEN psK := psJ; END;
  END;
  psK := MAX(1, MIN(4, psK + psD));
  RETURN psO(psK);
END;

trDrawView(dwV)
BEGIN
  CASE
    IF dwV == 1 THEN trPage1(); END;
    IF dwV == 2 THEN trPageChi(); END;
    IF dwV == 3 THEN trPage3(); END;
    IF dwV == 4 THEN trDrawProp(1); END;
    IF dwV == 6 THEN
      // page 4: the CAS is asked here only if the table or the scores
      // changed since it last answered (trCxMake keeps its answers)
      trCxMake();
      trPage4();
    END;
    DEFAULT trDrawOR();
  END;
  trShow();
END;

trView(vwStart)
BEGIN
  LOCAL vwV, vwOld, vwGo, vwK, vwM, vwM1;
  LOCAL vwSl, vwStp, vwLo, vwHi, vwA;
  vwV := vwStart; vwOld := 0; vwGo := 1;
  trFrz := 0;
  REPEAT vwK := GETKEY; UNTIL vwK == -1;
  trDrain();
  WHILE vwGo == 1 DO
    // entering the proportions plot from anywhere else: animate the fit
    IF vwV == 4 AND vwOld <> 4 THEN trAnimate(); END;
    vwOld := vwV;
    trDrawView(vwV);
    // wait for a key or a touch (poll both)
    REPEAT
      vwK := GETKEY;
      vwM := MOUSE;
      vwM1 := vwM(1);
    UNTIL vwK > -1 OR SIZE(vwM1) > 0;
    vwSl := 0;                                   // the soft key to act on
    IF SIZE(vwM1) > 0 THEN
      IF vwM1(2) >= 218 THEN
        vwSl := MIN(6, IP(vwM1(1) / 53.4) + 1);  // the soft-key strip
        trDrain();
      ELSE
        IF vwV <= 3 OR vwV == 6 THEN
          // a results page: left half = back, right half = on
          IF vwM1(1) < 160 THEN vwV := trPgStep(vwV, -1); ELSE vwV := trPgStep(vwV, 1); END;
          trDrain();
        ELSE
          trPlotTouch(vwV, vwM1(1), vwM1(2));    // drains on lift
        END;
      END;
    ELSE
      IF vwK == 4 THEN vwGo := 0; END;           // Esc = menu
      IF vwK == 50 THEN vwSl := 3; END;          // +   = Scores
      IF vwK == 42 THEN vwSl := 5; END;          // 0   = Restore
      IF vwV <= 3 OR vwV == 6 THEN
        IF vwK == 7 THEN vwV := trPgStep(vwV, -1); END;
        IF vwK == 8 THEN vwV := trPgStep(vwV, 1); END;
        IF vwK == 30 THEN
          IF vwV == 6 THEN vwGo := 0; ELSE vwV := trPgStep(vwV, 1); END;
        END;
        IF vwK == 45 THEN vwV := 4; END;         // -   = the plots
      ELSE
        IF vwK == 7 THEN trSel := MAX(1, trSel - 1); END;
        IF vwK == 8 THEN trSel := MIN(trK, trSel + 1); END;
        IF vwK == 30 THEN vwGo := 0; END;
        // Up/Down on plot 1: the keyboard's drag.  One press moves the
        // dot about 1/40 of the y-scale (at least one case).
        IF vwV == 4 THEN
          IF vwK == 2 OR vwK == 12 THEN
            vwStp := MAX(1, ROUND(trTot(trSel) * trYhi / 40, 0));
            vwLo := MAX(0, 1 - (trAT - trA(trSel)));
            vwHi := MIN(trTot(trSel), trNT - 1 - (trAT - trA(trSel)));
            vwA := trA(trSel);
            IF vwK == 2 THEN vwA := MIN(vwHi, vwA + vwStp); END;
            IF vwK == 12 THEN vwA := MAX(vwLo, vwA - vwStp); END;
            IF vwA <> trA(trSel) THEN
              trA(trSel) := vwA;
              trEdit := 1;
              trCompute();
            END;
          END;
        END;
        IF vwK == 45 THEN vwV := 9 - vwV; END;   // -   = the other plot
      END;
    END;
    // ---- the soft keys (and their hardware shortcuts) ----
    IF vwSl == 1 THEN vwV := 4; END;             // Prop
    IF vwSl == 2 THEN vwV := 5; END;             // OR
    IF vwSl == 3 THEN                            // Scores: next set, recompute
      trScM := trScM + 1;
      IF trScM > 2 THEN trScM := 0; END;
      trSetX();
      trCompute();
    END;
    IF vwSl == 4 THEN                            // Data: the entry dialogs
      IF trEnter() == 1 THEN vwV := 1; END;
      // the tap that closed a dialog can arrive again on the screen
      // underneath, so drain keys and touch before the next frame
      REPEAT vwK := GETKEY; UNTIL vwK == -1;
      trDrain();
    END;
    IF vwSl == 5 THEN                            // Restore the data as entered
      trA := trCopy(trOA);
      trScM := 0;
      trSetX();
      trEdit := 0;
      trCompute();
    END;
    IF vwSl == 6 THEN vwGo := 0; END;            // Menu
  END;
  REPEAT vwK := GETKEY; UNTIL vwK == -1;
END;

EXPORT TrendTest()
BEGIN
  LOCAL mnI, mnRun, mnS;
  trInit();
  mnRun := 1;
  WHILE mnRun == 1 DO
    mnI := 0;
    // beta 3d: "Plot: proportions + trend line" (30 characters) was at
    // the edge of the list-form box, which cut a 35-character line
    // after about 31 on the emulator; now 29 (fleet review 19 Oct
    // 2026, C-T3).  Comments go above the CHOOSE, never inside the
    // list (a Syntax Error on the Prime).
    CHOOSE(mnI, "χ² for linear trend", {
      "Enter data (2-6 levels)",
      "Results",
      "Plot: proportions, trend line",
      "Plot: odds ratios vs level 1",
      "Examples",
      "Help",
      "Quit"});
    CASE
      IF mnI == 0 OR mnI == 7 THEN mnRun := 0; END;
      IF mnI == 1 THEN
        IF trEnter() == 1 THEN trView(1); END;
      END;
      IF mnI >= 2 AND mnI <= 4 AND trHas == 0 THEN
        MSGBOX("No data yet.  Use Enter data, or load one of the Examples, first.");
      END;
      IF mnI == 2 AND trHas == 1 THEN trView(1); END;
      IF mnI == 3 AND trHas == 1 THEN trView(4); END;
      IF mnI == 4 AND trHas == 1 THEN trView(5); END;
      IF mnI == 5 THEN
        IF trExamples() == 1 THEN trView(1); END;
      END;
      IF mnI == 6 THEN trHelp(); END;
    END;
  END;
  // Hand the summary back as the program's value: Home shows it in
  // the history and no Terminal screen opens.  The numbers stay in
  // TR_X2T TR_P TR_X2 TR_DF TR_OR TR_B.
  IF trHas == 1 THEN
    mnS := "TrendTest: χ² trend = " + trFx(trX2T, 4) + " (1 df), " + trPStr(trPT) + "; Pearson χ² = " + trFx(trX2P, 4) + " (" + STRING(trK - 1, 2, 0) + " df)";
    IF trScM <> 0 THEN mnS := mnS + "; scores " + trScName(); END;
    IF trEdit == 1 THEN mnS := mnS + "; counts edited"; END;
    RETURN mnS;
  END;
  RETURN "TrendTest: no table";
END;
