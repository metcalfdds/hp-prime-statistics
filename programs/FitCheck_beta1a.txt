//   set of counts follow the proportions a hypothesis predicts (chi-square
//   goodness of fit, Benford's law, a Poisson or binomial model), and could
//   a sample of measurements have come from a normal distribution (Q-Q

//   title page, then the menu).  Menu: Chi-square goodness of fit;

//   Normality: chi-square on bins; Exact binomial test; Load a preset;
//   Use my data (D1, D2); Help; Quit.
// Every result has pictures you can touch: observed vs expected bars with
//   each cell's chi-square contribution in a strip below (tap a cell for O,
//   E, (O-E)^2/E and (O-E)/sqrt(E)); the Benford curve over the observed
//   first digits; a table of the cells; the chi-square curve with the
//   statistic marked and its tail shaded; a normal Q-Q plot (tap a point
//   for its value and normal score); the ECDF against the fitted normal CDF
//   with the largest gap D marked; the binomial distribution with the
//   observed count and the p-value region, on a scale of SDs.
// Home exports (after a run): FC_X2, FC_DF, FC_P (the last chi-square fit),
//   FC_O, FC_E (its observed and expected counts, as shown), FC_W, FC_PW
//   (Shapiro-Wilk), FC_A2, FC_PA (Anderson-Darling: A^2 uncorrected, and p),

// Written for the 2026-09-09 firmware (HP Prime G2 and G1).  beta 1a has
//   NOT been run on hardware or the emulator, only in the PC harness.
// Sources:
//   chi-square goodness of fit: Pearson (1900); the expected-count rules:
//     every E >= 5, or Cochran's (1954, Biometrics 10:417) "at most 20% of
//     the E below 5 and none below 1"; df = k - 1 - (parameters estimated)
//     (Fisher 1924).  Upper tail: 2 Phi(-sqrt x) for df 1, e^(-x/2) for
//     df 2, else the regularized upper incomplete gamma Q(df/2, x/2) by its
//     series or continued fraction (Press et al., Numerical Recipes, 2nd ed.,
//     section 6.2), so a tiny p keeps its digits (Lessons Learned 3).
//   Benford's law: P(first digit d) = log10(1 + 1/d); first two digits
//     P(ab) = log10(1 + 1/(10a + b)) (Benford 1938; Nigrini, "Forensic
//     Analytics", 2011 and 2020).  Mean absolute deviation MAD = (1/K) sum
//     |actual - expected proportion| (Nigrini).  Nigrini's MAD conformity
//     ranges are NOT shown: no file we can read here states them, so the
//     program shows the MAD and says why the ranges are missing.

//     A^2 (1 + 0.75/n + 2.25/n^2) and its p-value: D'Agostino & Stephens

//   Exact binomial test: two-sided p = the sum of every outcome no more

// Wait for the finger to lift -- never more than one second (Lessons
// Learned 7: a stuck touch reading must not freeze the program).
fc_Drain()
BEGIN
  LOCAL drT;
  drT := TICKS;
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Drain, wait for a key OR a tap, drain.  Returns the key code, or -1 for
// a tap (whose y is then returned as -2 - y, so a page can tell where).
fc_WaitAny()
BEGIN
  LOCAL waK, waM, waM1, waR;
  REPEAT waK := GETKEY; UNTIL waK == -1;
  fc_Drain();
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  waR := waK;
  IF SIZE(waM1) > 0 THEN waR := -2 - waM1(2); END;
  fc_Drain();
  REPEAT waK := GETKEY; UNTIL waK == -1;
  RETURN waR;
END;

// A "nice" step at least as big as the value: 1, 2 or 5 times a power of
// ten (Heckbert), for axis ticks.
fc_Nice(nnV)
BEGIN
  LOCAL nnE, nnF;
  IF nnV <= 0 THEN RETURN 1; END;
  nnE := FLOOR(LOG(nnV));
  nnF := nnV / 10^nnE;
  IF nnF <= 1 THEN RETURN 10^nnE; END;
  IF nnF <= 2 THEN RETURN 2 * 10^nnE; END;
  IF nnF <= 5 THEN RETURN 5 * 10^nnE; END;
  RETURN 10^(nnE + 1);
END;

// The 1, 2 or 5 times a power of ten NEAREST the value (on a log scale):
// the bin width for a sample's histogram bins.
fc_NiceNear(nrV)
BEGIN
  LOCAL nrE, nrF;
  IF nrV <= 0 THEN RETURN 1; END;
  nrE := FLOOR(LOG(nrV));
  nrF := nrV / 10^nrE;
  // the boundaries are the geometric means: sqrt(2), sqrt(10), sqrt(50)
  IF nrF < 1.41421356 THEN RETURN 10^nrE; END;
  IF nrF < 3.16227766 THEN RETURN 2 * 10^nrE; END;
  IF nrF < 7.07106781 THEN RETURN 5 * 10^nrE; END;
  RETURN 10^(nrE + 1);
END;

// A number with d fixed decimals, rounded FIRST: STRING(x, 2, d) turns
// scientific when 0 < |x| < 10^-d (Lessons Learned 3); "+ 0" turns a
// rounded -0 into 0.
fc_F(fxV, fxD)
BEGIN
  RETURN STRING(ROUND(fxV, fxD) + 0, 2, fxD);
END;

// A whole number as text.
fc_I(fiV)
BEGIN
  RETURN STRING(ROUND(fiV, 0) + 0, 2, 0);
END;

// A number with its sign: "+1.67" or "-3.63".
fc_Sgn(sgV, sgD)
BEGIN
  LOCAL sgR;
  sgR := ROUND(sgV, sgD) + 0;
  IF sgR >= 0 THEN RETURN "+" + STRING(sgR, 2, sgD); END;
  RETURN STRING(sgR, 2, sgD);
END;

// A number with as few decimals as it needs, up to fnD: 17, 142.2, 0.25.
fc_Num(fnV, fnD)
BEGIN
  LOCAL fnI, fnR;
  FOR fnI FROM 0 TO fnD DO
    fnR := ROUND(fnV, fnI);
    IF ABS(fnV - fnR) < 0.0000001 * MAX(1, ABS(fnV)) THEN RETURN fc_F(fnV, fnI); END;
  END;
  RETURN fc_F(fnV, fnD);
END;

// A p-value as text: "p = 0.0123", or "p < 0.0001" below that (the house
// style; the exact tiny value is shown separately by fc_Pow).
fc_PStr(psP)
BEGIN
  IF psP < 0.0001 THEN RETURN "p < 0.0001"; END;
  IF psP > 0.9999 THEN RETURN "p > 0.9999"; END;
  RETURN "p = " + fc_F(psP, 4);
END;

// A positive number from its natural log, as "1.23 x 10^-186": works for
// numbers far below the Prime's 1E-499, because only the log is used.
fc_Pow(pwL)
BEGIN
  LOCAL pwT, pwE, pwM;
  pwT := pwL / LN(10);
  pwE := FLOOR(pwT);
  pwM := ROUND(10^(pwT - pwE), 2);
  IF pwM >= 10 THEN pwM := pwM / 10; pwE := pwE + 1; END;
  RETURN fc_F(pwM, 2) + " " + fcTm + " 10^" + fc_I(pwE);
END;

// A p-value with its tiny value spelled out from its log (plL = LN(p))
// when it is below 0.0001: "p = 0.0078", or "p = 1.60 x 10^-186".  plL >= 0
// means "no log known" (then "p < 0.0001").
fc_PLong(plP, plL)
BEGIN
  IF plP >= 0.0001 THEN RETURN fc_PStr(plP); END;
  IF plL < 0 THEN RETURN "p = " + fc_Pow(plL); END;
  RETURN "p < 0.0001";
END;

// Long text cut into lines of at most wrW characters, broken at a space
// (BayesTree's bt_Wrap shape).  Returns a list of strings.
fc_Wrap(wrS, wrW)
BEGIN
  LOCAL wrL, wrT, wrI, wrC;
  wrL := {}; wrT := wrS;
  WHILE DIM(wrT) > wrW DO
    wrC := 0;
    FOR wrI FROM IP(wrW / 2) TO wrW + 1 DO
      IF MID(wrT, wrI, 1) == " " THEN wrC := wrI; END;
    END;
    IF wrC == 0 THEN wrC := wrW; END;
    wrL := CONCAT(wrL, {MID(wrT, 1, wrC)});
    wrT := MID(wrT, wrC + 1, DIM(wrT));
  END;
  RETURN CONCAT(wrL, {wrT});
END;

// Start a new text page list, and add a paragraph to it (wrapped to 46
// characters, all in one colour).  fc_TLb adds an empty line.
fc_TL0()
BEGIN
  fcTL := {}; fcTC := {};
END;

fc_TL(tlS, tlC)
BEGIN
  LOCAL tlW, tlI;
  tlW := fc_Wrap(tlS, 46);
  FOR tlI FROM 1 TO SIZE(tlW) DO
    fcTL := CONCAT(fcTL, {tlW(tlI)});
    fcTC := CONCAT(fcTC, {tlC});
  END;
END;

fc_TLb()
BEGIN
  fcTL := CONCAT(fcTL, {""});
  fcTC := CONCAT(fcTC, {0});
END;

// A polynomial c(1) + c(2) x + c(3) x^2 + ... (Horner), for AS R94.
fc_Poly(poC, poX)
BEGIN
  LOCAL poI, poR;
  poR := 0;
  poI := SIZE(poC);
  WHILE poI >= 1 DO
    poR := poR * poX + poC(poI);
    poI := poI - 1;
  END;
  RETURN poR;
END;

// ============================================================
// SPECIAL FUNCTIONS

// e^v, or 0 when v is so negative that e^v would fall below the Prime's
// range (1E-499; what the calculator does there has not been measured,
// so it is never asked).
fc_ExpG(egV)
BEGIN
  IF egV < -1100 THEN RETURN 0; END;
  RETURN EXP(egV);
END;

// ln Gamma(z) for z >= 0.5: Lanczos (1964), g = 7, the nine coefficients
// of Numerical Recipes (3rd ed.) / Godfrey, rounded to 12 digits.  There
// is no LNGAMMA outside the CAS (Lessons Learned 3).
//   Gamma(z) = sqrt(2 pi) t^(z - 1/2) e^-t A(z),  t = z + 6.5,
//   A(z) = c0 + sum c_i / (z - 1 + i)
fc_LnGam(lgZ)
BEGIN
  LOCAL lgC, lgX, lgT, lgI, lgW;
  lgC := {1, 676.520368122, -1259.13921672, 771.323428778, -176.615029162,
    12.5073432787, -0.138571095266, 9.98436957802E-6, 1.50563273515E-7};
  lgW := lgZ - 1;
  lgX := lgC(1);
  FOR lgI FROM 1 TO 8 DO lgX := lgX + lgC(lgI + 1) / (lgW + lgI); END;
  lgT := lgW + 7.5;
  // 0.918938533205 = ln sqrt(2 pi)
  RETURN 0.918938533205 + (lgW + 0.5) * LN(lgT) - lgT + LN(lgX);
END;

// Phi(z), the standard normal CDF, held at exactly 0 or 1 beyond
// |z| = 37.5, where it is 0 or 1 to far more than 12 digits (the built-in
// is never asked for a value that might underflow).
fc_Phi(phZ)
BEGIN
  IF phZ < -37.5 THEN RETURN 0; END;
  IF phZ > 37.5 THEN RETURN 1; END;
  RETURN NORMALD_CDF(0, 1, phZ);
END;

// ln Phi(z).  Below z = -30 the asymptotic series of Mills' ratio:
//   Phi(z) = phi(z) / |z| (1 - 1/z^2 + 3/z^4 - 15/z^6 ...)
// (Abramowitz & Stegun 26.2.12), so ln Phi never meets a 0.
fc_LnPhi(lpZ)
BEGIN
  LOCAL lpQ;
  IF lpZ > -30 THEN RETURN LN(NORMALD_CDF(0, 1, lpZ)); END;
  lpQ := 1 / (lpZ * lpZ);
  RETURN -lpZ * lpZ / 2 - 0.918938533205 - LN(-lpZ) + LN(1 - lpQ + 3 * lpQ^2 - 15 * lpQ^3);
END;

// ln Q(a, x), the regularized UPPER incomplete gamma function, so that
// P(chi-square with df > x) = Q(df/2, x/2).  Numerical Recipes (2nd ed.)
// section 6.2: for x < a + 1 the series for P and Q = 1 - P (Q is not
// small there); otherwise Lentz's continued fraction for Q itself, which
// keeps every digit of a tiny tail (1 - CDF would be exactly 0 on the
// Prime once the tail is below about 5E-13, Lessons Learned 3).
fc_LnGamQ(gqA, gqX)
BEGIN
  LOCAL gqS, gqD, gqAp, gqN, gqLf, gqB;
  LOCAL gqC, gqE, gqH, gqAn, gqDl, gqGo;
  IF gqX <= 0 THEN RETURN 0; END;
  // ln of x^a e^-x / Gamma(a), the common factor
  gqLf := -gqX + gqA * LN(gqX) - fc_LnGam(gqA);
  IF gqX < gqA + 1 THEN
    gqAp := gqA; gqS := 1 / gqA; gqD := gqS; gqN := 0; gqGo := 1;
    WHILE gqGo == 1 DO
      gqAp := gqAp + 1;
      gqD := gqD * gqX / gqAp;
      gqS := gqS + gqD;
      gqN := gqN + 1;
      IF ABS(gqD) < ABS(gqS) * 2E-12 OR gqN > 1000 THEN gqGo := 0; END;
    END;
    gqS := 1 - gqS * EXP(gqLf);
    IF gqS <= 0 THEN RETURN -1E99; END;
    RETURN LN(gqS);
  END;
  gqB := gqX + 1 - gqA; gqC := 1E300; gqE := 1 / gqB; gqH := gqE;
  gqN := 1; gqGo := 1;
  WHILE gqGo == 1 DO
    gqAn := -gqN * (gqN - gqA);
    gqB := gqB + 2;
    gqE := gqAn * gqE + gqB;
    IF ABS(gqE) < 1E-300 THEN gqE := 1E-300; END;
    gqC := gqB + gqAn / gqC;
    IF ABS(gqC) < 1E-300 THEN gqC := 1E-300; END;
    gqE := 1 / gqE;
    gqDl := gqE * gqC;
    gqH := gqH * gqDl;
    gqN := gqN + 1;
    // 2E-11: on 12-digit reals a ratio this close to 1 is 1 (or one step off)
    IF ABS(gqDl - 1) < 2E-11 OR gqN > 1000 THEN gqGo := 0; END;
  END;
  RETURN gqLf + LN(gqH);
END;

// ln P(chi-square(df) > x), the upper tail, never as 1 - CDF:
//   df 1: 2 Phi(-sqrt x);  df 2: e^(-x/2) exactly;  else ln Q(df/2, x/2).
fc_LnChiUp(cuDF, cuX)
BEGIN
  IF cuX <= 0 THEN RETURN 0; END;
  IF cuDF == 1 THEN RETURN LN(2) + fc_LnPhi(-(cuX^0.5)); END;
  IF cuDF == 2 THEN RETURN -cuX / 2; END;
  RETURN fc_LnGamQ(cuDF / 2, cuX / 2);
END;

// The upper tail itself (0 when it is below the Prime's range).
fc_ChiUp(chDF, chX)
BEGIN
  RETURN MIN(1, fc_ExpG(fc_LnChiUp(chDF, chX)));
END;


// ============================================================
// THE CHI-SQUARE FIT ENGINE
//   raw cells fcO (observed), fcPr (expected proportions), fcLab;
//   the cells SHOWN (merged or not) fcVO, fcVE, fcVL; for each:
//     E = N x (its proportion),  contribution (O - E)^2 / E,
//     residual (O - E) / sqrt(E)  (its square is the contribution);
//   X2 = sum of the contributions,  df = k - 1 - m  (m = parameters
//   estimated from the same data: Fisher 1924),  p = P(chi2(df) > X2).
//   The check: every E >= 5; or Cochran's (1954) rule: at most 20% of
//   the E below 5 and none below 1.  EVERY cell is checked (GOFBIN,
//   GOFARB and GOFNORML looked at the first cell only).
// ============================================================

// The label of merged cells s..e: "6-9"; a Poisson group that takes the
// tail cell is "at least s" (">=2").
fc_GLab(glS, glE)
BEGIN
  IF glS == glE THEN RETURN fcLab(glS); END;
  IF fcKind == 4 AND glE == fcK THEN RETURN fcGe + fcLab(glS); END;
  RETURN fcLab(glS) + "-" + fcLab(glE);
END;

// Compute the cells as shown, the statistic, df, p and the check, and
// put the results in Home's FC_ variables.
fc_Calc()
BEGIN
  LOCAL ccG, ccI, ccO, ccQ, ccE, ccK;
  LOCAL ccS, ccEd;
  fcN := 0;
  FOR ccI FROM 1 TO fcK DO fcN := fcN + fcO(ccI); END;
  ccK := fcK;
  IF fcMrg == 1 THEN ccK := SIZE(fcMS); END;
  fcVK := ccK;
  fcVO := MAKELIST(0, ccI, 1, ccK, 1);
  fcVE := MAKELIST(0, ccI, 1, ccK, 1);
  fcVC := MAKELIST(0, ccI, 1, ccK, 1);
  fcVR := MAKELIST(0, ccI, 1, ccK, 1);
  fcVL := {};
  fcX2 := 0; fcBad := 0; fcN5 := 0; fcMinE := 1E300;
  FOR ccG FROM 1 TO ccK DO
    ccS := ccG; ccEd := ccG;
    IF fcMrg == 1 THEN ccS := fcMS(ccG); ccEd := fcME(ccG); END;
    ccO := 0; ccQ := 0;
    FOR ccI FROM ccS TO ccEd DO
      ccO := ccO + fcO(ccI);
      ccQ := ccQ + fcPr(ccI);
    END;
    ccE := fcN * ccQ;
    fcVO(ccG) := ccO; fcVE(ccG) := ccE;
    fcVL := CONCAT(fcVL, {fc_GLab(ccS, ccEd)});
    IF ccE <= 0 THEN
      fcBad := 1;                        // an impossible cell: E = 0
    ELSE
      fcVC(ccG) := (ccO - ccE)^2 / ccE;
      fcVR(ccG) := (ccO - ccE) / ccE^0.5;
      fcX2 := fcX2 + fcVC(ccG);
    END;
    IF ccE < 5 THEN fcN5 := fcN5 + 1; END;
    IF ccE < fcMinE THEN fcMinE := ccE; END;
  END;
  fcDF := ccK - 1 - fcNPar;
  fcP := -1; fcLnP := 1; fcCrit := 0;
  IF fcDF >= 1 AND fcBad == 0 THEN
    fcLnP := fc_LnChiUp(fcDF, fcX2);
    fcP := MIN(1, fc_ExpG(fcLnP));
    fcCrit := CHISQUARE_ICDF(fcDF, 0.95);
  END;
  fcChk := 0;
  IF fcN5 == 0 THEN
    fcChk := 1;
  ELSE
    IF fcN5 <= 0.2 * ccK AND fcMinE >= 1 THEN fcChk := 2; END;
  END;
  IF fcSel > ccK OR fcSel < 1 THEN fcSel := 1; END;
  FC_X2 := fcX2; FC_DF := fcDF; FC_P := fcP;
  FC_O := fcVO; FC_E := fcVE;
END;

// A new fit from raw cells: O, proportions, labels, parameters estimated,
// kind, title and source line.  The view starts unmerged.
fc_SetFit(sfO, sfPr, sfLab, sfNP, sfKd, sfTt, sfSr)
BEGIN
  LOCAL sfI;
  fcO := sfO; fcPr := sfPr; fcLab := sfLab;
  fcK := SIZE(sfO); fcNPar := sfNP; fcKind := sfKd;
  fcTtl := sfTt; fcSrc := sfSr;
  fcMrg := 0; fcFixG := 0; fcNote := ""; fcSel := 1; fcHasF := 1;
  fcMS := {}; fcME := {};
  FOR sfI FROM 1 TO fcK DO
    fcMS := CONCAT(fcMS, {sfI}); fcME := CONCAT(fcME, {sfI});
  END;
  fcMrgN := 0;
  fc_Calc();
END;

// Automatic merging: while some expected count is below 5, the cell with
// the SMALLEST E joins its neighbour with the smaller E (the left one on
// a tie; an end cell joins its only neighbour), so only ADJACENT cells
// are merged and an ordered scale stays ordered.  It stops early rather
// than let df fall below 1.  The groups go to fcMS / fcME.
fc_AutoMerge()
BEGIN
  LOCAL amS, amE, amX, amK, amI, amJ;
  LOCAL amNb, amGo, amM, amLo;
  amK := fcK;
  amS := {}; amE := {}; amX := {};
  FOR amI FROM 1 TO fcK DO
    amS := CONCAT(amS, {amI}); amE := CONCAT(amE, {amI});
    amX := CONCAT(amX, {fcN * fcPr(amI)});
  END;
  amGo := 1;
  WHILE amGo == 1 DO
    amGo := 0;
    IF amK > fcNPar + 2 THEN
      amJ := 0; amM := 5;
      FOR amI FROM 1 TO amK DO
        IF amX(amI) < amM THEN amM := amX(amI); amJ := amI; END;
      END;
      IF amJ > 0 THEN
        IF amJ == 1 THEN
          amNb := 2;
        ELSE
          IF amJ == amK THEN
            amNb := amK - 1;
          ELSE
            amNb := amJ + 1;
            IF amX(amJ - 1) <= amX(amJ + 1) THEN amNb := amJ - 1; END;
          END;
        END;
        amLo := MIN(amJ, amNb);
        amE(amLo) := amE(amLo + 1);
        amX(amLo) := amX(amLo) + amX(amLo + 1);
        // close the gap the merged cell leaves
        FOR amI FROM amLo + 1 TO amK - 1 DO
          amS(amI) := amS(amI + 1);
          amE(amI) := amE(amI + 1);
          amX(amI) := amX(amI + 1);
        END;
        amK := amK - 1;
        amGo := 1;
      END;
    END;
  END;
  fcMS := {}; fcME := {};
  FOR amI FROM 1 TO amK DO
    fcMS := CONCAT(fcMS, {amS(amI)}); fcME := CONCAT(fcME, {amE(amI)});
  END;
END;

// How many groups of the current grouping hold more than one cell.
fc_NMerged()
BEGIN
  LOCAL nmI, nmC;
  nmC := 0;
  FOR nmI FROM 1 TO SIZE(fcMS) DO
    IF fcME(nmI) > fcMS(nmI) THEN nmC := nmC + 1; END;
  END;
  RETURN nmC;
END;

// The Merge soft key: unmerged -> merged (Triola's grouping for the Nelson
// preset, else fc_AutoMerge), merged -> back to every cell.  Says exactly
// why when there is nothing it can merge.
fc_ToggleMerge()
BEGIN
  IF fcKind == 7 THEN
    MSGBOX("A 2 " + fcTm + " 2 table has nothing to merge.");
    RETURN;
  END;
  IF fcMrg == 1 THEN
    fcMrg := 0; fc_Calc();
    RETURN;
  END;
  IF fcFixG == 0 THEN
    fc_Calc();
    fc_AutoMerge();
  END;
  fcMrgN := fc_NMerged();
  IF fcMrgN == 0 THEN
    IF fcN5 == 0 THEN
      MSGBOX("Nothing to merge: every expected count is already 5 or more.");
    ELSE
      MSGBOX("Too few cells to merge: one more merge would leave df = 0.");
    END;
    RETURN;
  END;
  fcMrg := 1;
  fc_Calc();
END;

// The merged groups, said exactly, as a paragraph of the text page:
// "Merged: {6, 7, 8, 9} -> 6-9; {2, >=3} -> >=2."
fc_MrgLines()
BEGIN
  LOCAL mlI, mlJ, mlS, mlG;
  IF fcMrg == 0 THEN RETURN; END;
  mlS := "Merged (neighbouring cells): ";
  IF fcFixG == 1 THEN mlS := "Merged (Triola's grouping): "; END;
  mlG := 0;
  FOR mlI FROM 1 TO SIZE(fcMS) DO
    IF fcME(mlI) > fcMS(mlI) THEN
      IF mlG > 0 THEN mlS := mlS + "; "; END;
      mlS := mlS + "{";
      FOR mlJ FROM fcMS(mlI) TO fcME(mlI) DO
        mlS := mlS + fcLab(mlJ);
        IF mlJ < fcME(mlI) THEN mlS := mlS + ", "; END;
      END;
      mlS := mlS + "} " + fcAr + " " + fc_GLab(fcMS(mlI), fcME(mlI));
      mlG := mlG + 1;
    END;
  END;
  fc_TL(mlS + ".  Now " + fc_I(fcVK) + " cells (from " + fc_I(fcK) + ").", fcMAG);
END;


// ============================================================
// THE FITS: Benford, Poisson, binomial, normal on bins
// ============================================================

// The leading digit (fdTwo = 0, 1..9) or the first two digits (fdTwo = 1,
// 10..99) of a number's significand; 0 for a zero.  The ROUND guards a
// 2.9999999999 from a division (the PC's binary arithmetic; the Prime's
// decimal arithmetic gives 3 exactly): a number with ten or more 9s in a
// row would be read one too high.
fc_FirstDig(fdV, fdTwo)
BEGIN
  LOCAL fdA, fdE, fdM;
  fdA := ABS(fdV);
  IF fdA == 0 THEN RETURN 0; END;
  fdE := FLOOR(LOG(fdA));
  fdM := fdA / 10^fdE;
  IF fdM >= 10 THEN fdM := fdM / 10; END;
  IF fdM < 1 THEN fdM := fdM * 10; END;
  IF fdTwo == 1 THEN
    fdM := ROUND(fdM * 10, 8);
    IF fdM >= 100 THEN fdM := 10; END;
    RETURN IP(fdM);
  END;
  fdM := ROUND(fdM, 9);
  IF fdM >= 10 THEN fdM := 1; END;
  RETURN IP(fdM);
END;

// Benford's law for a list of numbers: first digit (bfTwo = 0, 9 cells,
// P(d) = log10(1 + 1/d)) or first two digits (bfTwo = 1, 90 cells,
// P(ab) = log10(1 + 1/(10a + b))).  Zeros are skipped and counted.  No
// parameter is estimated: df = k - 1.  Also Nigrini's MAD = (1/K) sum
// |actual proportion - Benford's proportion| over all K cells.
// beta 1a (review 4): in the first-two-digits test a number below 10 has
// no second digit (5 is not "50"), so it is left out and counted (fcBenS).
// Returns 0 if no number was usable.
fc_Benford(bfV, bfTwo, bfTt, bfSr)
BEGIN
  LOCAL bfO, bfPr, bfLab, bfI, bfD, bfK;
  LOCAL bfOff;
  bfK := 9; bfOff := 0;
  IF bfTwo == 1 THEN bfK := 90; bfOff := 9; END;
  bfO := MAKELIST(0, bfI, 1, bfK, 1);
  bfPr := MAKELIST(0, bfI, 1, bfK, 1);
  bfLab := {};
  FOR bfI FROM 1 TO bfK DO
    bfPr(bfI) := LOG(1 + 1 / (bfI + bfOff));
    bfLab := CONCAT(bfLab, {fc_I(bfI + bfOff)});
  END;
  fcBenZ := 0; fcBenU := 0; fcBenS := 0;
  FOR bfI FROM 1 TO SIZE(bfV) DO
    bfD := fc_FirstDig(bfV(bfI), bfTwo);
    IF bfTwo == 1 AND bfD > 0 AND ABS(bfV(bfI)) < 10 THEN
      fcBenS := fcBenS + 1;
      bfD := -1;
    END;
    IF bfD == 0 THEN
      fcBenZ := fcBenZ + 1;
    END;
    IF bfD > 0 THEN
      bfO(bfD - bfOff) := bfO(bfD - bfOff) + 1;
      fcBenU := fcBenU + 1;
    END;
  END;
  IF fcBenU == 0 THEN RETURN 0; END;
  fc_SetFit(bfO, bfPr, bfLab, 0, 2 + bfTwo, bfTt, bfSr);
  fcMAD := 0;
  FOR bfI FROM 1 TO bfK DO
    fcMAD := fcMAD + ABS(bfO(bfI) / fcN - bfPr(bfI));
  END;
  fcMAD := fcMAD / bfK;
  RETURN 1;
END;

// A Poisson fit to counts: pfF(1) units saw 0 events, pfF(2) saw 1, ...
// lambda-hat = (sum j f_j) / N.  Cells 0, 1, ..., K-1 and the TAIL
// ">= K" (K = the largest count seen), whose E is N P(X >= K), a sum of
// positive terms -- so the expected counts add up to N (GOFPOISS left the
// tail out).  One parameter estimated: df = k - 2.  Returns 0 when there
// is nothing to fit (no units, or no events at all).
fc_PoisFit(pfF, pfTt, pfSr)
BEGIN
  LOCAL pfK, pfN, pfS, pfI, pfO, pfPr;
  LOCAL pfLab, pfL, pfJ, pfGo, pfTl, pfLL;
  pfK := SIZE(pfF);
  WHILE pfK > 1 AND pfF(pfK) == 0 DO pfK := pfK - 1; END;
  pfN := 0; pfS := 0;
  FOR pfI FROM 1 TO pfK DO
    pfN := pfN + pfF(pfI);
    pfS := pfS + (pfI - 1) * pfF(pfI);
  END;
  IF pfN <= 0 OR pfS <= 0 OR pfK < 2 THEN RETURN 0; END;
  fcLmb := pfS / pfN;
  pfLL := LN(fcLmb);
  pfO := MAKELIST(0, pfI, 1, pfK, 1);
  pfPr := MAKELIST(0, pfI, 1, pfK, 1);
  pfLab := {};
  FOR pfI FROM 1 TO pfK DO
    pfO(pfI) := pfF(pfI);
    IF pfI < pfK THEN
      // P(X = j), j = pfI - 1: e^-lambda lambda^j / j!, in logs
      pfPr(pfI) := fc_ExpG(-fcLmb + (pfI - 1) * pfLL - fc_LnGam(pfI));
      pfLab := CONCAT(pfLab, {fc_I(pfI - 1)});
    ELSE
      // P(X >= K): add terms from K upward until they no longer matter
      pfTl := 0; pfJ := pfK - 1; pfGo := 1;
      WHILE pfGo == 1 DO
        pfL := fc_ExpG(-fcLmb + pfJ * pfLL - fc_LnGam(pfJ + 1));
        pfTl := pfTl + pfL;
        pfJ := pfJ + 1;
        IF pfJ > fcLmb + 1 AND pfL <= 1E-15 * pfTl THEN pfGo := 0; END;
        IF pfJ > pfK + 5000 THEN pfGo := 0; END;
      END;
      pfPr(pfI) := pfTl;
      pfLab := CONCAT(pfLab, {fcGe + fc_I(pfK - 1)});
    END;
  END;
  fc_SetFit(pfO, pfPr, pfLab, 1, 4, pfTt, pfSr);
  RETURN 1;
END;

// A binomial fit: bnF(1) groups had 0 successes, ..., bnF(n + 1) had n.
// bnP = p0 if typed (0 < p0 < 1: df = k - 1), or -1 to estimate it,
// p-hat = (sum j f_j) / (n N) (one parameter: df = k - 2).
// E_j = N BINOMIAL(n, p, j) (n, p, k: the order the Prime takes).
// Returns 0 when p-hat is 0 or 1 (nothing varies) or there is no group.
fc_BinFit(bnF, bnP, bnTt, bnSr)
BEGIN
  LOCAL bnN, bnS, bnI, bnPr, bnLab, bnNt;
  bnNt := SIZE(bnF) - 1;
  IF bnNt < 1 THEN RETURN 0; END;
  bnN := 0; bnS := 0;
  FOR bnI FROM 1 TO bnNt + 1 DO
    bnN := bnN + bnF(bnI);
    bnS := bnS + (bnI - 1) * bnF(bnI);
  END;
  IF bnN <= 0 THEN RETURN 0; END;
  fcBEst := 0;
  fcBp := bnP;
  IF bnP < 0 THEN
    fcBEst := 1;
    fcBp := bnS / (bnNt * bnN);
    IF fcBp <= 0 OR fcBp >= 1 THEN RETURN 0; END;
  END;
  fcBn := bnNt;
  bnPr := MAKELIST(0, bnI, 1, bnNt + 1, 1);
  bnLab := {};
  FOR bnI FROM 1 TO bnNt + 1 DO
    bnPr(bnI) := BINOMIAL(bnNt, fcBp, bnI - 1);
    bnLab := CONCAT(bnLab, {fc_I(bnI - 1)});
  END;
  fc_SetFit(bnF, bnPr, bnLab, fcBEst, 5, bnTt, bnSr);
  RETURN 1;
END;

// A normal fit to binned counts: midpoints nbM, counts nbF, bin width
// nbW, and the mean and SD to use.  The END cells are open (the first
// takes everything below its upper edge, the last everything above its
// lower edge), so the proportions add to 1.  Each middle cell's
// proportion is Phi(z_hi) - Phi(z_lo), or for a cell above the mean
// Phi(-z_lo) - Phi(-z_hi), its mirror (no difference of two numbers near
// 1).  Mean and SD estimated: df = k - 3.
fc_NormBinFit(nbM, nbF, nbW, nbXb, nbSd, nbTt, nbSr)
BEGIN
  LOCAL nbK, nbI, nbPr, nbLab, nbZ, nbZo;
  nbK := SIZE(nbM);
  nbPr := MAKELIST(0, nbI, 1, nbK, 1);
  nbLab := {};
  nbZo := 0;
  FOR nbI FROM 1 TO nbK DO
    nbLab := CONCAT(nbLab, {fc_Num(nbM(nbI), 2)});
    nbZ := (nbM(nbI) + nbW / 2 - nbXb) / nbSd;     // this cell's upper edge, as z
    IF nbI == 1 THEN
      nbPr(1) := fc_Phi(nbZ);
    ELSE
      IF nbI == nbK THEN
        nbPr(nbI) := fc_Phi(-nbZo);
      ELSE
        IF nbZo >= 0 THEN
          nbPr(nbI) := fc_Phi(-nbZo) - fc_Phi(-nbZ);
        ELSE
          nbPr(nbI) := fc_Phi(nbZ) - fc_Phi(nbZo);
        END;
      END;
    END;
    nbZo := nbZ;
  END;
  fc_SetFit(nbF, nbPr, nbLab, 2, 6, nbTt, nbSr);
  fcGxb := nbXb; fcGsd := nbSd; fcMid := nbM;
  RETURN 1;
END;

// The sample in fcX put into bins for the chi-square normal fit: about
// Sturges' number of bins (log2 n + 1), a width that is 1, 2 or 5 times a
// power of ten, bins (a, a + w] (a value on an edge goes to the LOWER
// bin), and the mean and SD of the RAW values.
fc_RawBins()
BEGIN
  LOCAL rbW, rbA, rbK, rbI, rbF, rbM;
  LOCAL rbJ, rbK0;
  rbK0 := CEILING(LN(fcNX) / LN(2)) + 1;
  rbW := fc_NiceNear((fcX(fcNX) - fcX(1)) / rbK0);
  rbA := FLOOR(ROUND(fcX(1) / rbW, 8)) * rbW;
  IF ROUND((fcX(1) - rbA) / rbW, 8) == 0 THEN rbA := rbA - rbW; END;
  rbK := MAX(1, CEILING(ROUND((fcX(fcNX) - rbA) / rbW, 8)));
  rbF := MAKELIST(0, rbI, 1, rbK, 1);
  rbM := MAKELIST(0, rbI, 1, rbK, 1);
  FOR rbI FROM 1 TO rbK DO rbM(rbI) := rbA + (rbI - 0.5) * rbW; END;
  FOR rbI FROM 1 TO fcNX DO
    rbJ := CEILING(ROUND((fcX(rbI) - rbA) / rbW, 8));
    rbJ := MAX(1, MIN(rbK, rbJ));
    rbF(rbJ) := rbF(rbJ) + 1;
  END;
  fc_NormBinFit(rbM, rbF, rbW, fcXb, fcSd, fcNTtl + ": " + fcX2S + " on bins", fcNSrc);
  fcNote := "Bins (a, a + " + fc_Num(rbW, 4) + "], labelled by their midpoints; the end cells are open.  " + fcXbar + " and s come from the " + fc_I(fcNX) + " raw values, so the true reference lies " +
      "between " + fcX2S + "(k - 3) and " + fcX2S + "(k - 1) (Chernoff & Lehmann 1954); df = k - 3 is the usual textbook rule.";
END;


// ============================================================
// NORMALITY: Blom scores, Shapiro-Wilk W, Anderson-Darling A^2,
// Lilliefors D
// ============================================================

// Shapiro-Wilk W and p: Royston (1995), algorithm AS R94, exactly as
// R's and SciPy's swilk compute it (checked against scipy.stats.shapiro
// for n = 3 to 50 and up to 5000; build/FitCheck/verify).  With
// m_i = Phi^-1((i - 3/8)/(n + 1/4)) (the Blom scores, fcZq), the
// coefficients a_i of the largest two values come from Royston's (1992)
// polynomials in 1/sqrt(n), the rest are m_i scaled so sum a^2 = 1, and
//   W = (sum a_i (x(n+1-i) - x(i)))^2 / sum (x - x-bar)^2.
// p: n = 3 exactly, p = (6/pi)(asin sqrt W - pi/3), written below as
//   3 ASIN(sqrt W)/ASIN(1) - 2, which is the same in radians, degrees or
//   grads; n = 4..11: y = -ln(gamma - ln(1 - W)), gamma = -2.273 + 0.459 n;
//   n >= 12: y = ln(1 - W); then p = P(Z > (y - m)/s) with Royston's
//   polynomials for m and s (computed as a lower tail).
fc_SWcalc()
BEGIN
  LOCAL swN, swH, swI, swA, swS2, swSS;
  LOCAL swRs, swA1, swA2, swFc, swI1, swNum;
  LOCAL swSx, swW1, swY, swGm, swMu, swSg;
  swN := fcNX; swH := IP(swN / 2);
  swA := MAKELIST(0, swI, 1, swH, 1);
  IF swN == 3 THEN
    swA(1) := 0.5^0.5;
  ELSE
    swS2 := 0;
    FOR swI FROM 1 TO swH DO swS2 := swS2 + fcZq(swI)^2; END;
    swS2 := 2 * swS2;
    swSS := swS2^0.5;
    swRs := 1 / swN^0.5;
    swA1 := fc_Poly({0, 0.221157, -0.147981, -2.07119, 4.434685, -2.706056}, swRs) - fcZq(1) / swSS;
    IF swN > 5 THEN
      swI1 := 3;
      swA2 := -fcZq(2) / swSS + fc_Poly({0, 0.042981, -0.293762, -1.752461, 5.682633, -3.582633}, swRs);
      swFc := ((swS2 - 2 * fcZq(1)^2 - 2 * fcZq(2)^2) / (1 - 2 * swA1^2 - 2 * swA2^2))^0.5;
      swA(2) := swA2;
    ELSE
      swI1 := 2;
      swFc := ((swS2 - 2 * fcZq(1)^2) / (1 - 2 * swA1^2))^0.5;
    END;
    swA(1) := swA1;
    FOR swI FROM swI1 TO swH DO swA(swI) := -fcZq(swI) / swFc; END;
  END;
  swNum := 0; swSx := 0;
  // beta 1a (review 8): both sums in units of fcSc (the largest |x - x-bar|),
  // so a value near 1E300 cannot push a square past the Prime's 1E499
  FOR swI FROM 1 TO swH DO
    swNum := swNum + swA(swI) * ((fcX(swN + 1 - swI) - fcX(swI)) / fcSc);
  END;
  FOR swI FROM 1 TO swN DO swSx := swSx + ((fcX(swI) - fcXb) / fcSc)^2; END;
  fcW := MIN(1, swNum^2 / swSx);
  swW1 := 1 - fcW;
  fcLnPW := 1;
  IF swN == 3 THEN
    fcPW := MAX(0, 3 * ASIN(fcW^0.5) / ASIN(1) - 2);
    RETURN;
  END;
  IF swW1 <= 0 THEN fcPW := 1; RETURN; END;
  swY := LN(swW1);
  IF swN <= 11 THEN
    swGm := -2.273 + 0.459 * swN;
    IF swY >= swGm THEN fcPW := 1E-99; RETURN; END;   // AS R94: "obviously" tiny
    swY := -LN(swGm - swY);
    swMu := fc_Poly({0.544, -0.39978, 0.025054, -0.0006714}, swN);
    swSg := EXP(fc_Poly({1.3822, -0.77857, 0.062767, -0.0020322}, swN));
  ELSE
    swMu := fc_Poly({-1.5861, -0.31082, -0.083751, 0.0038915}, LN(swN));
    swSg := EXP(fc_Poly({-0.4803, -0.082676, 0.0030302}, LN(swN)));
  END;
  fcLnPW := fc_LnPhi(-(swY - swMu) / swSg);
  fcPW := fc_ExpG(fcLnPW);
END;

// Anderson-Darling, mean and SD estimated (D'Agostino & Stephens 1986):
//   A^2 = -n - (1/n) sum (2i - 1)[ln Phi(z(i)) + ln(1 - Phi(z(n+1-i)))],
//   z = (x - x-bar)/s, and ln(1 - Phi(z)) = ln Phi(-z) (a lower tail);
//   A*^2 = A^2 (1 + 0.75/n + 2.25/n^2), and p from their four formulas
//   (as statsmodels' normal_ad; above A*^2 = 13 it says p = 0, which is
//   below 5E-31: shown as "p < 0.0001").
fc_ADcalc()
BEGIN
  LOCAL adI, adS, adN, adA;
  adN := fcNX; adS := 0;
  FOR adI FROM 1 TO adN DO
    IF fcAbort == 0 THEN
      adS := adS + (2 * adI - 1) * (fc_LnPhi((fcX(adI) - fcXb) / fcSd) + fc_LnPhi(-(fcX(adN + 1 - adI) - fcXb) / fcSd));
      IF adN > 200 THEN fc_Prog(adI, adN, "A" + fcSq + ":"); END;
    END;
  END;
  fcA2 := -adN - adS / adN;
  fcA2s := fcA2 * (1 + 0.75 / adN + 2.25 / adN^2);
  adA := fcA2s;
  fcLnPA := 1;
  IF adA < 0.2 THEN
    fcPA := 1 - EXP(-13.436 + 101.14 * adA - 223.73 * adA^2);
  ELSE
    IF adA < 0.34 THEN
      fcPA := 1 - EXP(-8.318 + 42.796 * adA - 59.938 * adA^2);
    ELSE
      IF adA < 0.6 THEN
        fcPA := EXP(0.9177 - 4.279 * adA - 1.38 * adA^2);
      ELSE
        IF adA <= 13 THEN
          fcLnPA := 1.2937 - 5.709 * adA + 0.0186 * adA^2;
          fcPA := EXP(fcLnPA);
        ELSE
          fcPA := 0;
        END;
      END;
    END;
  END;
  fcPA := MIN(1, MAX(0, fcPA));
END;

// Lilliefors (1967): the K-S distance between the ECDF and the normal
// with the SAMPLE mean and SD,
//   D = max over i of  max( i/n - Phi(z(i)),  Phi(z(i)) - (i-1)/n ),
// and p from Dallal & Wilkinson's (1986) approximation (for n > 100,
// D (n/100)^0.49 with n = 100), valid only for p < 0.10: above that the
// pages say "p > 0.10" (fcPD = -1).  beta 1a (review 3): p is given only
// for n >= 5, the start of Dallal & Wilkinson's range as implemented here;
// for n = 3 or 4 fcPD = -2 and the pages say "p not available for n < 5"
// (in grey, and the test is left out of the verdict).  KOLSMIR used the plain K-S p with
// estimated parameters, which is far too large.
fc_LFcalc()
BEGIN
  LOCAL lfI, lfF, lfDp, lfDm, lfN, lfK;
  LOCAL lfNd;
  lfN := fcNX; fcLD := 0; fcLDi := 1; fcLDy := 0;
  FOR lfI FROM 1 TO lfN DO
    IF lfN > 200 AND fcAbort == 0 THEN fc_Prog(lfI, lfN, "D:"); END;
    lfF := fc_Phi((fcX(lfI) - fcXb) / fcSd);
    lfDp := lfI / lfN - lfF;
    lfDm := lfF - (lfI - 1) / lfN;
    IF lfDp > fcLD THEN fcLD := lfDp; fcLDi := lfI; fcLDy := lfI / lfN; END;
    IF lfDm > fcLD THEN fcLD := lfDm; fcLDi := lfI; fcLDy := (lfI - 1) / lfN; END;
  END;
  fcPD := -1; fcLnPD := 1;
  IF lfN < 5 THEN fcPD := -2; END;
  IF lfN >= 5 THEN
    lfK := fcLD; lfNd := lfN;
    IF lfN > 100 THEN lfK := fcLD * (lfN / 100)^0.49; lfNd := 100; END;
    fcLnPD := -7.01256 * lfK^2 * (lfNd + 2.78019) + 2.99587 * lfK * (lfNd + 2.78019)^0.5 - 0.122119 + 0.974598 / lfNd^0.5 + 1.67997 / lfNd;
    IF fcLnPD <= LN(0.1) THEN
      fcPD := fc_ExpG(fcLnPD);
    ELSE
      fcLnPD := 1;
    END;
  END;
END;

// Load a sample for the normality tests: sorts it, mean, SD (n - 1,
// two passes), skewness g1 and excess kurtosis g2 (moment forms), the
// Blom scores (half computed, half mirrored: z(n+1-i) = -z(i), and the
// middle of an odd n is exactly 0 -- NMLPROB left it out), then W, A^2
// and D.  Returns 0 (with a message) for fewer than 3 values, more than
// 5000, or no spread.
fc_NormSet(nsV, nsTt, nsSr)
BEGIN
  LOCAL nsN, nsI, nsS, nsM2, nsM3, nsM4;
  LOCAL nsD;
  nsN := SIZE(nsV);
  IF nsN < 3 THEN MSGBOX("The normality tests need at least 3 values."); RETURN 0; END;
  IF nsN > 5000 THEN MSGBOX("At most 5000 values: Royston's W is defined up to n = 5000."); RETURN 0; END;
  fcX := SORT(nsV); fcNX := nsN; fcBin := 0;
  nsS := 0;
  FOR nsI FROM 1 TO nsN DO nsS := nsS + fcX(nsI); END;
  fcXb := nsS / nsN;
  // beta 1a (review 8): the moments of (x - x-bar)/fcSc, fcSc = the largest
  // |x - x-bar|, so no power can overflow; s, g1 and g2 are rescaled after
  fcSc := 0;
  FOR nsI FROM 1 TO nsN DO fcSc := MAX(fcSc, ABS(fcX(nsI) - fcXb)); END;
  IF fcSc <= 0 THEN MSGBOX("Every value is the same: s = 0, so there is no shape to test."); RETURN 0; END;
  nsM2 := 0; nsM3 := 0; nsM4 := 0;
  FOR nsI FROM 1 TO nsN DO
    nsD := (fcX(nsI) - fcXb) / fcSc;
    nsM2 := nsM2 + nsD^2; nsM3 := nsM3 + nsD^3; nsM4 := nsM4 + nsD^4;
  END;
  IF nsM2 <= 0 THEN MSGBOX("Every value is the same: s = 0, so there is no shape to test."); RETURN 0; END;
  fcSd := fcSc * (nsM2 / (nsN - 1))^0.5;
  fcSkw := (nsM3 / nsN) / (nsM2 / nsN)^1.5;
  fcKur := (nsM4 / nsN) / (nsM2 / nsN)^2 - 3;
  fcAbort := 0;
  IF nsN > 200 THEN fc_Busy("W, A" + fcSq + " and D for n = " + fc_I(nsN)); END;
  fcZq := MAKELIST(0, nsI, 1, nsN, 1);
  FOR nsI FROM 1 TO IP(nsN / 2) DO
    IF fcAbort == 0 THEN
      fcZq(nsI) := NORMALD_ICDF(0, 1, (nsI - 0.375) / (nsN + 0.25));
      fcZq(nsN + 1 - nsI) := -fcZq(nsI);
      IF nsN > 200 THEN fc_Prog(nsI, IP(nsN / 2), "scores:"); END;
    END;
  END;
  IF fcAbort == 0 THEN fc_SWcalc(); END;
  IF fcAbort == 0 THEN fc_ADcalc(); END;
  IF fcAbort == 0 THEN fc_LFcalc(); END;
  IF fcAbort == 1 THEN
    // Esc: nothing half-computed is kept
    fcHasN := 0; fcAbort := 0;
    MSGBOX("Stopped (Esc): no sample is loaded.");
    RETURN 0;
  END;
  fcNTtl := nsTt; fcNSrc := nsSr; fcHasN := 1; fcNSel := 1;
  FC_W := fcW; FC_PW := fcPW; FC_A2 := fcA2; FC_PA := fcPA;
  FC_D := fcLD; FC_PD := fcPD;
  RETURN 1;
END;

// A BINNED sample (crimtab): midpoints, counts, width.  The mean and SD
// come from the midpoints (no Sheppard correction).  W, A^2 and D are NOT
// computed: thousands of tied values (every height rounded to its bin)
// are not what those tests assume.  The Q-Q plot shows one point per
// bin; the chi-square fit on bins is the test.
fc_NormSetBin(sbM, sbF, sbW, sbTt, sbSr)
BEGIN
  LOCAL sbI, sbN, sbS, sbD, sbM2, sbM3;
  LOCAL sbM4;
  sbN := 0; sbS := 0;
  FOR sbI FROM 1 TO SIZE(sbM) DO sbN := sbN + sbF(sbI); sbS := sbS + sbF(sbI) * sbM(sbI); END;
  fcXb := sbS / sbN;
  // beta 1a: in units of the largest |midpoint - x-bar|, as fc_NormSet
  fcSc := 0;
  FOR sbI FROM 1 TO SIZE(sbM) DO
    IF sbF(sbI) > 0 THEN fcSc := MAX(fcSc, ABS(sbM(sbI) - fcXb)); END;
  END;
  sbM2 := 0; sbM3 := 0; sbM4 := 0;
  FOR sbI FROM 1 TO SIZE(sbM) DO
    sbD := (sbM(sbI) - fcXb) / fcSc;
    sbM2 := sbM2 + sbF(sbI) * sbD^2; sbM3 := sbM3 + sbF(sbI) * sbD^3; sbM4 := sbM4 + sbF(sbI) * sbD^4;
  END;
  fcSd := fcSc * (sbM2 / (sbN - 1))^0.5;
  fcSkw := (sbM3 / sbN) / (sbM2 / sbN)^1.5;
  fcKur := (sbM4 / sbN) / (sbM2 / sbN)^2 - 3;
  fcBin := 1; fcBM := sbM; fcBF := sbF; fcBW := sbW; fcNX := sbN;
  fcX := {}; fcZq := {};
  fcW := -1; fcPW := -1; fcA2 := -1; fcA2s := -1; fcPA := -1; fcLD := -1; fcPD := -1;
  fcLnPW := 1; fcLnPA := 1; fcLnPD := 1;
  fcNTtl := sbTt; fcNSrc := sbSr; fcHasN := 1; fcNSel := 1;
  RETURN 1;
END;


// ============================================================
// THE EXACT BINOMIAL TEST, in logs
//   X ~ Binomial(n, p0);  ln P(X = k) = ln n! - ln k! - ln (n-k)!
//     + k ln p0 + (n - k) ln(1 - p0);
//   one-sided P(X <= x), P(X >= x), each summed from x AWAY from the mode
//   (the terms shrink, so the sum stops once they no longer matter);
//   two-sided p = the sum of P(X = k) over every k with P(X = k) <=
//   P(X = x) (1 + 1E-7): R's binom.test and SciPy's binomtest ("minlike").
//   Everything stays in logs, so p = 10^-186 is a number, not 0.
// ============================================================

fc_LPmf(lmK)
BEGIN
  RETURN fcTLC - fc_LnGam(lmK + 1) - fc_LnGam(fcTn - lmK + 1) + lmK * fcTLp + (fcTn - lmK) * fcTLq;
END;

// ln(e^a + e^b), without overflow or underflow.
fc_LAdd(laA, laB)
BEGIN
  LOCAL laH, laL;
  laH := MAX(laA, laB); laL := MIN(laA, laB);
  IF laL - laH < -40 THEN RETURN laH; END;
  RETURN laH + LN(1 + EXP(laL - laH));
END;

// ln of the sum of P(X = k) for k = tk0, tk0 + dir, tk0 + 2 dir, ... to 0
// or n, starting on the side AWAY from the mode, so every next term is
// smaller: a term below e^-40 of the first ends the sum.  Each term
// comes from the one before: P(k + 1)/P(k) = (n - k)/(k + 1) p0/(1 - p0).
fc_LTail(tkK0, tkDir)
BEGIN
  LOCAL tkK, tkL, tkL0, tkS, tkGo;
  tkL0 := fc_LPmf(tkK0); tkL := tkL0; tkS := 1; tkK := tkK0; tkGo := 1;
  WHILE tkGo == 1 DO
    IF tkDir > 0 THEN
      IF tkK >= fcTn THEN
        tkGo := 0;
      ELSE
        tkL := tkL + LN((fcTn - tkK) / (tkK + 1)) + fcTLp - fcTLq;
        tkK := tkK + 1;
      END;
    ELSE
      IF tkK <= 0 THEN
        tkGo := 0;
      ELSE
        tkL := tkL + LN(tkK / (fcTn - tkK + 1)) + fcTLq - fcTLp;
        tkK := tkK - 1;
      END;
    END;
    IF tkGo == 1 THEN
      IF tkL - tkL0 < -40 THEN tkGo := 0; ELSE tkS := tkS + EXP(tkL - tkL0); END;
    END;
  END;
  RETURN tkL0 + LN(tkS);
END;

// ln P(X <= k) and ln P(X >= k); -1E99 stands for ln 0.
fc_LLo(loK)
BEGIN
  LOCAL loM;
  IF loK < 0 THEN RETURN -1E99; END;
  IF loK >= fcTn THEN RETURN 0; END;
  loM := FLOOR((fcTn + 1) * fcTp0);                // the mode
  IF loK <= loM THEN RETURN fc_LTail(loK, -1); END;
  RETURN LN(MAX(1E-300, 1 - EXP(fc_LTail(loK + 1, 1))));
END;

fc_LHi(hiK)
BEGIN
  LOCAL hiM;
  IF hiK <= 0 THEN RETURN 0; END;
  IF hiK > fcTn THEN RETURN -1E99; END;
  hiM := FLOOR((fcTn + 1) * fcTp0);
  IF hiK >= hiM THEN RETURN fc_LTail(hiK, 1); END;
  RETURN LN(MAX(1E-300, 1 - EXP(fc_LTail(hiK - 1, -1))));
END;

// The test for x successes in n trials against p0: E, SD, z, the
// one-sided tails, the two-sided p and its rejection region (fcTkL,
// fcTkU), and FC_BP.  The far side of the region is found by bisection:
// on each side of the mode the probabilities are in order.
fc_BTest(btX, btN, btP)
BEGIN
  LOCAL btD, btT, btA, btB, btM, btNP;
  fcTx := btX; fcTn := btN; fcTp0 := btP;
  fcTLC := fc_LnGam(btN + 1); fcTLp := LN(btP); fcTLq := LN(1 - btP);
  btNP := btN * btP;
  fcTE := btNP; fcTSD := (btN * btP * (1 - btP))^0.5;
  fcTz := (btX - fcTE) / fcTSD;
  fcTlL := fc_LLo(btX); fcTlU := fc_LHi(btX);
  btD := fc_LPmf(btX);
  btT := btD + LN(1 + 1E-7);                         // SciPy's relative tolerance
  IF btX == btNP THEN
    fcTl2 := 0; fcTkL := btN; fcTkU := 0;          // p = 1: every k counts
  ELSE
    IF btX < btNP THEN
      // the other side: the first k >= ceil(n p0) with P(k) <= P(x)
      fcTkL := btX;
      IF fc_LPmf(btN) > btT THEN
        fcTkU := btN + 1;
      ELSE
        btA := CEILING(btNP); btB := btN;
        WHILE btA < btB DO
          btM := IP((btA + btB) / 2);
          IF fc_LPmf(btM) <= btT THEN btB := btM; ELSE btA := btM + 1; END;
        END;
        fcTkU := btA;
      END;
    ELSE
      // the other side: the last k <= floor(n p0) with P(k) <= P(x)
      fcTkU := btX;
      IF fc_LPmf(0) > btT THEN
        fcTkL := -1;
      ELSE
        btA := 0; btB := FLOOR(btNP);
        WHILE btA < btB DO
          btM := IP((btA + btB + 1) / 2);
          IF fc_LPmf(btM) <= btT THEN btA := btM; ELSE btB := btM - 1; END;
        END;
        fcTkL := btA;
      END;
    END;
    fcTl2 := fc_LAdd(fc_LLo(fcTkL), fc_LHi(fcTkU));
  END;
  fcTl2 := MIN(0, fcTl2);
  fcTp2 := fc_ExpG(fcTl2);
  fcHasT := 1; fcTSel := -1;
  FC_BP := fcTp2;
END;

// ============================================================
// beta 1a (review 2): FISHER'S EXACT TEST for a 2 x 2 table, in logs
//   N items, K of them marked (the 183 low draft numbers among 366),
//   n drawn (the 182 Jan-Jun birth dates), k marked among them (73).
//   Drawn WITHOUT replacement, k is hypergeometric:
//   P(k) = C(K, k) C(N - K, n - k) / C(N, n).
//   Two-sided p = the sum of P(j) over every j with P(j) <= P(k) (1 + 1E-7),
//   the "minlike" rule of R's fisher.test and SciPy's fisher_exact.
//   Each P(j) comes from the one before:
//   P(j + 1)/P(j) = (K - j)(n - j) / ((j + 1)(N - K - n + j + 1)).
// ============================================================

// ln P(j) for the hypergeometric (N, K, n).
fc_LHyp(lhN, lhK, lhn, lhj)
BEGIN
  RETURN fc_LnGam(lhK + 1) - fc_LnGam(lhj + 1) - fc_LnGam(lhK - lhj + 1)
    + fc_LnGam(lhN - lhK + 1) - fc_LnGam(lhn - lhj + 1) - fc_LnGam(lhN - lhK - lhn + lhj + 1)
    - fc_LnGam(lhN + 1) + fc_LnGam(lhn + 1) + fc_LnGam(lhN - lhn + 1);
END;

// Fisher's two-sided p for k marked among n drawn: fcFshP, and its log
// fcFshL (so a tiny p can still be printed).
fc_Fisher(fhN, fhK, fhn, fhk)
BEGIN
  LOCAL fhJ, fhLo, fhHi, fhD, fhL, fhS;
  fhLo := MAX(0, fhn + fhK - fhN); fhHi := MIN(fhn, fhK);
  fhD := fc_LHyp(fhN, fhK, fhn, fhk) + LN(1 + 1E-7);
  fhL := fc_LHyp(fhN, fhK, fhn, fhLo);
  fhS := 0;
  FOR fhJ FROM fhLo TO fhHi DO
    IF fhL <= fhD AND fhL - fhD > -700 THEN fhS := fhS + EXP(fhL - fhD); END;
    IF fhJ < fhHi THEN
      fhL := fhL + LN((fhK - fhJ) * (fhn - fhJ)) - LN((fhJ + 1) * (fhN - fhK - fhn + fhJ + 1));
    END;
  END;
  // fhS counts in units of P(k) (1 + 1E-7)
  fcFshL := MIN(0, fhD + LN(fhS));
  fcFshP := fc_ExpG(fcFshL);
END;

// beta 1a: a p with six decimals between 0.000001 and 0.001 (Fisher's
// 0.000243 beside Pearson's 0.000168 would both read 0.0002), else as
// fc_PLong.
fc_P6(p6P, p6L)
BEGIN
  IF p6P < 0.001 AND p6P >= 0.000001 THEN RETURN "p = " + fc_F(p6P, 6); END;
  RETURN fc_PLong(p6P, p6L);
END;

// The chi-square curve's window and its cache: one pixel row per column,
// computed ONCE when the page opens (Lessons Learned 6).  The window
// covers the 5% critical value and the statistic, unless the statistic
// is far out (more than 2.5 times the window), which is then said on the
// page instead of squashing the curve.
fc_CurveFit()
BEGIN
  LOCAL cfR, cfS, cfI, cfXv, cfY, cfW;
  IF fcDF < 1 OR fcBad == 1 THEN RETURN; END;
  cfR := MAX(fcCrit * 1.4, fcDF + 4 * (2 * fcDF)^0.5, 3);
  IF fcX2 <= 2.5 * cfR THEN cfR := MAX(cfR, fcX2 * 1.12); END;
  cfS := fc_Nice(cfR / 5);
  fcCvX := CEILING(cfR / cfS) * cfS;
  // the y scale: df 1 rises without limit near 0 (clipped), df 2 starts
  // at 0.5, larger df peak at x = df - 2
  cfY := 0.6;
  IF fcDF == 2 THEN cfY := 0.55; END;
  IF fcDF > 2 THEN cfY := CHISQUARE(fcDF, fcDF - 2) * 1.15; END;
  cfW := fcPR - fcPL;
  fcCvP := MAKELIST(0, cfI, 1, cfW, 1);
  FOR cfI FROM 1 TO cfW DO
    cfXv := cfI / cfW * fcCvX;
    fcCvP(cfI) := fcPB - IP(MIN(CHISQUARE(fcDF, cfXv) / cfY, 1) * (fcPB - fcPT));
  END;
END;

// The curve page.
fc_DrawCurve()
BEGIN
  LOCAL dcS, dcV, dcN, dcPx, dcX2, dcXc;
  LOCAL dcLx, dcLast, dcPy, dcDec, dcOff;
  fc_Frame(fcX2S + " curve, df = " + fc_I(MAX(0, fcDF)));
  IF fcDF < 1 OR fcBad == 1 THEN
    TEXTOUT_P(fc_VStr(), G1, 6, 40, 1, fcGRAY, 308);
    TEXTOUT_P("df = k - 1 - m = " + fc_I(fcVK) + " - 1 - " + fc_I(fcNPar) + " = " + fc_I(fcDF) + ": too few cells", G1, 6, 60, 1, #000000h, 308);
    TEXTOUT_P("for a test.  Unmerge, or fit fewer parameters.", G1, 6, 73, 1, #000000h, 308);
    RETURN;
  END;
  TEXTOUT_P(fc_VStr(), G1, 6, 21, 1, fc_VCol(), 308);
  TEXTOUT_P("5% critical value = " + fc_F(fcCrit, 2) + " (dotted red)", G1, 6, 34, 1, fcRED, 308);
  dcS := fc_Nice(fcCvX / 5);
  dcDec := MAX(0, -FLOOR(LOG(dcS)));
  dcV := 0; dcN := 0;
  WHILE dcV <= fcCvX + dcS / 1000 AND dcN < 14 DO
    dcPx := fcPL + IP(dcV / fcCvX * (fcPR - fcPL) + 0.5);
    LINE_P(G1, dcPx, fcPT, dcPx, fcPB, fcGRID);
    LINE_P(G1, dcPx, fcPB, dcPx, fcPB + 3, #000000h);
    TEXTOUT_P(STRING(ROUND(dcV, dcDec), 2, dcDec), G1, dcPx - 6, fcPB + 5, 1, fcGRAY, 36);
    dcV := dcV + dcS; dcN := dcN + 1;
  END;
  // the p-value: the tail beyond the statistic, shaded column by column
  dcOff := 0;
  IF fcX2 > fcCvX THEN dcOff := 1; END;
  IF dcOff == 0 THEN
    dcX2 := fcPL + IP(fcX2 / fcCvX * (fcPR - fcPL) + 0.5);
    FOR dcPx FROM MAX(dcX2, fcPL + 1) TO fcPR DO
      dcPy := fcCvP(dcPx - fcPL);
      IF dcPy < fcPB THEN LINE_P(G1, dcPx, fcPB - 1, dcPx, dcPy, fcSHD); END;
    END;
  END;
  // the curve
  dcLast := -1;
  FOR dcPx FROM fcPL + 1 TO fcPR DO
    dcPy := fcCvP(dcPx - fcPL);
    IF dcLast >= 0 THEN LINE_P(G1, dcPx - 1, dcLast, dcPx, dcPy, fcBLUE); END;
    dcLast := dcPy;
  END;
  LINE_P(G1, fcPL, fcPT, fcPL, fcPB, #000000h);
  LINE_P(G1, fcPL, fcPB, fcPR, fcPB, #000000h);
  // the 5% critical value
  dcXc := MIN(fcPR, fcPL + IP(fcCrit / fcCvX * (fcPR - fcPL) + 0.5));
  fc_DotV(dcXc, fcPT, fcPB, fcRED);
  // the statistic
  IF dcOff == 0 THEN
    LINE_P(G1, dcX2, fcPT, dcX2, fcPB, fcMAG);
    dcLx := dcX2 + 3;
    IF dcX2 > fcPR - 80 THEN dcLx := dcX2 - 80; END;
    TEXTOUT_P(fcX2S + " = " + fc_F(fcX2, 2), G1, dcLx, fcPT + 4, 1, fcMAG, 78);
  ELSE
    TEXTOUT_P(fcX2S + " = " + fc_F(fcX2, 1) + " " + fcAr, G1, 190, fcPT + 4, 1, fcMAG, 128);
    TEXTOUT_P("(far off the chart)", G1, 190, fcPT + 17, 1, fcMAG, 128);
  END;
  TEXTOUT_P("Shaded tail = p-value: " + fc_PLong(fcP, fcLnP), G1, 6, 168, 1, fcMAG, 308);
  IF fcP < 0.05 THEN
    TEXTOUT_P(fcX2S + " is past the 5% critical value: reject " + fcH0, G1, 6, 181, 1, fcRED, 308);
  ELSE
    TEXTOUT_P(fcX2S + " is short of the 5% value: do not reject " + fcH0, G1, 6, 181, 1, fcGRN, 308);
  END;
  IF fcKind == 7 THEN
    TEXTOUT_P("df = (2-1)(2-1) = 1,  Fisher exact " + fc_PLong(fcFshP, fcFshL), G1, 6, 194, 1, #000000h, 308);
  ELSE
    TEXTOUT_P("df = k - 1 - m = " + fc_I(fcVK) + " - 1 - " + fc_I(fcNPar) + " = " + fc_I(fcDF) + "  (m: fitted)", G1, 6, 194, 1, #000000h, 308);
  END;
  IF fcChk == 0 THEN TEXTOUT_P("E too small: do not trust this p (Merge).", G1, 6, 206, 1, fcRED, 308); END;
END;

  fc_TL(fcX2S + " = sum of (O - E)" + fcSq + "/E = " + fc_F(fcX2, 3), #000000h);
  IF fcKind == 7 THEN
    fc_TL("df = (rows - 1)(columns - 1) = 1", #000000h);
  ELSE
    fc_TL("df = k - 1 - m = " + fc_I(fcVK) + " - 1 - " + fc_I(fcNPar) + " = " + fc_I(fcDF) + "   (m = parameters estimated from these data)", #000000h);
  END;
  IF fcBad == 1 THEN
    fc_TL("A cell has expected count 0 (proportion 0): a count there is impossible under " + fcH0 + ", so no " + fcX2S + " can be computed.", fcRED);
  ELSE
    IF fcDF < 1 THEN
      fc_TL("df < 1: too few cells for a test.  Unmerge, or fit fewer parameters.", fcRED);
    ELSE
      tsC := fcGRN; IF fcP < 0.05 THEN tsC := fcRED; END;
      tsS := fc_PLong(fcP, fcLnP);
      IF fcKind == 7 THEN tsS := fc_P6(fcP, fcLnP); END;
      fc_TL("P(" + fcX2S + "(" + fc_I(fcDF) + ") " + fcGe + " " + fc_F(fcX2, 3) + ") : " + tsS, tsC);
      IF fcKind == 7 THEN
        IF fcP < 0.05 THEN
          fc_TL("At " + fcAl + " = 0.05: reject " + fcH0 + ".  The data give evidence that birth date and draft number are related.", fcRED);
        ELSE
          fc_TL("At " + fcAl + " = 0.05: do not reject " + fcH0 + ".  No evidence of a relation -- which does not prove there is none.", fcGRN);
        END;
      ELSE
      IF fcP < 0.05 THEN
        fc_TL("At " + fcAl + " = 0.05: reject " + fcH0 + ".  The data give evidence that the counts do not follow these proportions.", fcRED);
      ELSE
        fc_TL("At " + fcAl + " = 0.05: do not reject " + fcH0 + ".  No evidence of a departure from these proportions -- which does not prove they are right.", fcGRN);
      END;
      END;
    END;
  END;
  // the check, every cell
  IF fcChk == 1 THEN
    fc_TL("Expected counts: every E " + fcGe + " 5 (smallest " + fc_F(fcMinE, 2) + "): the " + fcX2S + " curve is a fair guide.", fcGRN);
  ELSE
    tsS := "Expected counts: " + fc_I(fcN5) + " of " + fc_I(fcVK) + " cells have E < 5 (" + fc_I(ROUND(100 * fcN5 / fcVK, 0)) + "%); smallest E = " + fc_F(fcMinE, 2) + ".";
    IF fcChk == 2 THEN
      fc_TL(tsS + "  Cochran's rule (at most 20% below 5, none below 1) holds.", fcGRN);
    ELSE
      fc_TL(tsS + "  That breaks Cochran's rule (at most 20% below 5, none below 1): the p-value is not trustworthy.  Merge neighbouring cells (Merge key).", fcRED);
    END;
  END;
  fc_MrgLines();
  // beta 1a (review 2): the draft lottery's exact test, and why the simple
  // goodness-of-fit test with fixed shares understates the evidence
  IF fcKind == 7 THEN
    tsC := fcGRN; IF fcFshP < 0.05 THEN tsC := fcRED; END;
    fc_TL("Fisher's exact test: the 183 low numbers were drawn WITHOUT replacement from 366, so the number landing on the 182 Jan-Jun dates is hypergeometric.  Two-sided " + fc_P6(fcFshP, fcFshL) + ".", tsC);
    fc_TL("Why not a goodness-of-fit test with shares 182 : 184?  It treats the 183 low numbers as 183 independent draws and gives " + fcX2S + " = " + fc_F(fcGx2, 2) + ", " + fc_PStr(fcGp) + ".  But " +
        "once a low number goes to a January date it cannot go to a July one: the counts vary only about half as much as that test assumes ((366 - 183)/365 = 0.50), so its " + fcX2S + " is about half " +
            "of this one and it understates the evidence.", fcMAG);
  END;
  // what belongs to this kind of fit
  IF fcKind == 2 OR fcKind == 3 THEN
    tsS := "";
    IF fcKind == 3 THEN tsS := "; below 10 left out, as they have no second digit: " + fc_I(fcBenS); END;
    fc_TL("Numbers used: " + fc_I(fcBenU) + " (zeros skipped: " + fc_I(fcBenZ) + tsS + ").  Benford: P(d) = log" + CHAR(8321) + CHAR(8320) + "(1 + 1/d).", #000000h);
    fc_TL("MAD = (1/K) sum |actual - Benford proportion| = " + fc_F(fcMAD, 4) + " over all K = " + fc_I(fcK) + " cells (Nigrini, Forensic Analytics, 2011 and 2020).", fcMAG);
    fc_TL("Nigrini gives conformity ranges for the MAD; their values are not in any source this program could check, so none is shown.  Compare the MAD with his table yourself.", fcGRAY);
  END;
  IF fcKind == 4 THEN
    fc_TL("estimated " + fcGLm + " = (sum of x f) / N = " + fc_F(fcLmb, 4) + ".  The last cell is the tail " + fcGe + " K, with E = N P(X " + fcGe + " K), so the E add up to N.  One parameter estimated: df = k - 2.", #000000h);
  END;
  IF fcKind == 5 THEN
    IF fcBEst == 1 THEN
      fc_TL(fcPh + " = (sum of x f) / (n N) = " + fc_F(fcBp, 4) + ": one parameter estimated, df = k - 2.", #000000h);
    ELSE
      fc_TL("p = " + fc_Num(fcBp, 6) + " was given, not estimated: df = k - 1.", #000000h);
    END;
  END;
  IF fcKind == 6 THEN
    fc_TL(fcXbar + " = " + fc_F(fcGxb, 3) + ", s = " + fc_F(fcGsd, 3) + ".  Each cell's E = N (normal CDF at its upper edge - at its lower edge); the end cells are open.  Two parameters estimated: df = k - 3.", #000000h);
  END;
  IF DIM(fcNote) > 0 THEN fc_TL(fcNote, fcGRAY); END;
  // the presets' talking points
  IF fcPre == 1 THEN
    fc_TL("Triola: with all 9 digits most E are below 5, so combine 1 / 2-5 / 6-9 (Merge key) and test again.", fcBLUE);
    fc_TL("Triola's question (p. 653, paraphrased): what argument could you make as the defendant's attorney?  Points a defence could raise (teaching points, not the trial record): only 23 cheques; " +
        "Benford's law suits amounts spread over several powers of ten, but 21 of these 23 lie between 71,552.16 and 97,473.96, just under 100,000; and a poor fit shows the amounts are unusual, not " +
            "who wrote them or why.", #000000h);
  END;
  IF fcPre == 2 THEN
    fc_TL("Two cells: this " + fcX2S + " is z" + fcSq + " of the normal approximation, z = (339 - 688.2)/12.0.  The exact binomial test is the other menu item.", #000000h);
  END;
END;

// The chi-square fit's screens: 1 Bars, 2 Table, 3 Curve, 4 Test; the
// soft keys move between them, Merge toggles the merged view, Menu (or
// Esc, Enter) goes back to the main menu.
fc_GofLoop(glScr)
BEGIN
  LOCAL glGo, glK, glM, glM1, glSl, glPg;
  LOCAL glMx, glX, glY, glR, glMl;
  glGo := 1; glPg := 1;
  IF glScr == 2 THEN glPg := fc_SelPage(); END;
  IF glScr == 3 THEN fc_CurveFit(); END;
  IF glScr == 4 THEN fc_TestLines(); END;
  REPEAT glK := GETKEY; UNTIL glK == -1;
  fc_Drain();
  WHILE glGo == 1 DO
    CASE
      IF glScr == 1 THEN fc_DrawBars(); END;
      IF glScr == 2 THEN fc_DrawTable(glPg); END;
      IF glScr == 3 THEN fc_DrawCurve(); END;
      DEFAULT
        fc_TextPage(fcTtl + ": test", glPg, "page " + fc_I(glPg) + " of " + fc_I(fc_NPg()) + ", " + fcUD + " or tap = page");
    END;
    glMl := "Merge";
    IF fcMrg == 1 THEN glMl := "Unmrg"; END;
    fc_Show({"Bars", "Table", "Curve", "Test", glMl, "Menu"});
    REPEAT
      glK := GETKEY;
      glM := MOUSE;
      glM1 := glM(1);
    UNTIL glK > -1 OR SIZE(glM1) > 0;
    IF SIZE(glM1) > 0 THEN
      glX := glM1(1); glY := glM1(2);
      fc_Drain();
      IF glY >= 218 THEN
        glSl := MIN(6, IP(glX / 53.4) + 1);
        IF glSl <= 4 THEN
          glScr := glSl; glPg := 1;
          IF glScr == 2 THEN glPg := fc_SelPage(); END;
          IF glScr == 3 THEN fc_CurveFit(); END;
          IF glScr == 4 THEN fc_TestLines(); END;
        END;
        IF glSl == 5 THEN
          fc_ToggleMerge();
          glPg := 1;
          IF glScr == 2 THEN glPg := fc_SelPage(); END;
          IF glScr == 3 THEN fc_CurveFit(); END;
          IF glScr == 4 THEN fc_TestLines(); END;
        END;
        IF glSl == 6 THEN glGo := 0; END;
      ELSE
        IF glScr == 1 AND glX >= fcPL AND glX <= fcPR AND glY >= fcPT AND glY <= 188 THEN
          fcSel := MIN(fcVK, MAX(1, IP((glX - fcPL) / ((fcPR - fcPL) / fcVK)) + 1));
        END;
        IF glScr == 2 THEN
          IF glY >= 34 AND glY < 190 THEN
            glR := (glPg - 1) * 12 + IP((glY - 35) / 13) + 1;
            IF glR >= 1 AND glR <= MIN(fcVK, glPg * 12) THEN fcSel := glR; END;
          ELSE
            glPg := glPg + 1;
            IF glPg > CEILING(fcVK / 12) THEN glPg := 1; END;
          END;
        END;
        IF glScr == 4 THEN
          glPg := glPg + 1;
          IF glPg > fc_NPg() THEN glPg := 1; END;
        END;
      END;
    ELSE
      IF glK == 4 OR glK == 30 THEN glGo := 0; END;
      IF glK == 7 OR glK == 8 THEN
        IF glK == 7 THEN fcSel := MAX(1, fcSel - 1); ELSE fcSel := MIN(fcVK, fcSel + 1); END;
        IF glScr == 2 THEN glPg := fc_SelPage(); END;
      END;
      IF glK == 2 OR glK == 12 THEN
        glMx := 1;
        IF glScr == 2 THEN glMx := CEILING(fcVK / 12); END;
        IF glScr == 4 THEN glMx := fc_NPg(); END;
        IF glK == 2 THEN glPg := MAX(1, glPg - 1); ELSE glPg := MIN(glMx, glPg + 1); END;
      END;
      REPEAT glK := GETKEY; UNTIL glK == -1;
    END;
  END;
  REPEAT glK := GETKEY; UNTIL glK == -1;
END;

// ============================================================
// PICTURES AND PAGES OF THE NORMALITY TESTS
//   Q-Q: each value against its normal score (Blom), with the line
//     x-bar + s z; tap a point for its value, rank and z.  A binned
//     sample shows one point per bin.
//   ECDF: the sample's step function against the normal CDF with the
//     sample's x-bar and s, the largest gap D marked in red.
//   Tests: W, A^2, D, their p, the shape, the verdict in words.
// ============================================================

// The Q-Q points: for raw data (z(i), x(i)); for a binned sample one point
// per non-empty bin: its midpoint against the Blom score of its MIDDLE
// rank, ((c + (f + 1)/2) - 3/8) / (N + 1/4), c = the count below the bin.
fc_QQPts()
BEGIN
  LOCAL qtI, qpC, qpR;
  IF fcBin == 0 THEN
    fcQZ := fcZq; fcQV := fcX;
    fcQR := {};
  ELSE
    fcQZ := {}; fcQV := {}; fcQR := {};
    qpC := 0;
    FOR qtI FROM 1 TO SIZE(fcBM) DO
      IF fcBF(qtI) > 0 THEN
        qpR := qpC + (fcBF(qtI) + 1) / 2;
        fcQZ := CONCAT(fcQZ, {NORMALD_ICDF(0, 1, (qpR - 0.375) / (fcNX + 0.25))});
        fcQV := CONCAT(fcQV, {fcBM(qtI)});
        fcQR := CONCAT(fcQR, {qtI});
      END;
      qpC := qpC + fcBF(qtI);
    END;
  END;
  fcQN := SIZE(fcQZ);
  IF fcNSel > fcQN OR fcNSel < 1 THEN fcNSel := 1; END;
END;

// The Q-Q page.  Box: x 44..312, y 52..158.

BEGIN

  LOCAL qqX1, qqY1, qqX2, qqY2, qqZ, qqB;
  qqN := fcQN;
  qqZa := fcQZ(1); qqZb := fcQZ(qqN);

  qqVa := fcQV(1); qqVb := fcQV(1);

  qqT := qqVb - qqVa;

  qqVa := qqVa - 0.06 * qqT; qqVb := qqVb + 0.06 * qqT;
  fc_Frame("Q-Q: " + fcNTtl);
  IF fcBin == 0 THEN
    qqC := fcGRN; IF fcPW < 0.05 THEN qqC := fcRED; END;
    TEXTOUT_P("n = " + fc_I(fcNX) + ",  W = " + fc_F(fcW, 4) + ",  " + fc_PLong(fcPW, fcLnPW), G1, 6, 21, 1, qqC, 308);
  ELSE
    TEXTOUT_P("n = " + fc_I(fcNX) + " in " + fc_I(SIZE(fcBM)) + " bins: one point per bin", G1, 6, 21, 1, fcBLUE, 308);
  END;
  TEXTOUT_P("x: normal score z;  y: value;  line: " + fcXbar + " + s z", G1, 6, 34, 1, fcGRAY, 308);
  // x ticks (z)
  qqS := fc_Nice((qqZb - qqZa) / 6);
  qqDc := MAX(0, -FLOOR(LOG(qqS)));
  qqV := CEILING(qqZa / qqS) * qqS;
  WHILE qqV <= qqZb DO
    qqP := 44 + IP((qqV - qqZa) / (qqZb - qqZa) * 268 + 0.5);

    TEXTOUT_P(STRING(ROUND(qqV, qqDc), 2, qqDc), G1, qqP - 6, 161, 1, fcGRAY, 30);
    qqV := qqV + qqS;
  END;
  // y ticks (values)
  qqS := fc_Nice((qqVb - qqVa) / 5);
  qqDc := MAX(0, -FLOOR(LOG(qqS)));
  qqV := CEILING(qqVa / qqS) * qqS;
  WHILE qqV <= qqVb DO
    qqP := 158 - IP((qqV - qqVa) / (qqVb - qqVa) * 106 + 0.5);
    LINE_P(G1, 44, qqP, 312, qqP, fcGRID);
    TEXTOUT_P(STRING(ROUND(qqV, qqDc), 2, qqDc), G1, 2, qqP - 5, 1, fcGRAY, 40);
    qqV := qqV + qqS;
  END;
  LINE_P(G1, 44, 52, 44, 158, #000000h);
  LINE_P(G1, 44, 158, 312, 158, #000000h);
  // the line x-bar + s z, clipped to the box
  qqX1 := qqZa; qqY1 := fcXb + fcSd * qqZa;
  qqX2 := qqZb; qqY2 := fcXb + fcSd * qqZb;
  IF qqY1 < qqVa THEN qqX1 := (qqVa - fcXb) / fcSd; qqY1 := qqVa; END;
  IF qqY2 > qqVb THEN qqX2 := (qqVb - fcXb) / fcSd; qqY2 := qqVb; END;
  IF qqX1 < qqX2 THEN
    LINE_P(G1, 44 + IP((qqX1 - qqZa) / (qqZb - qqZa) * 268), 158 - IP((qqY1 - qqVa) / (qqVb - qqVa) * 106),
      44 + IP((qqX2 - qqZa) / (qqZb - qqZa) * 268), 158 - IP((qqY2 - qqVa) / (qqVb - qqVa) * 106), fcGRN);
  END;
  // the points, remembered in pixels for tapping
  fcQX := MAKELIST(0, qqI, 1, qqN, 1);
  fcQY := MAKELIST(0, qqI, 1, qqN, 1);
  qqB := 1; IF fcBin == 1 THEN qqB := 2; END;
  FOR qqI FROM 1 TO qqN DO
    fcQX(qqI) := 44 + IP((fcQZ(qqI) - qqZa) / (qqZb - qqZa) * 268 + 0.5);
    fcQY(qqI) := 158 - IP((fcQV(qqI) - qqVa) / (qqVb - qqVa) * 106 + 0.5);
    fc_Dot(fcQX(qqI), fcQY(qqI), qqB, fcBLUE);
  END;
  qqI := fcNSel;
  fc_Dot(fcQX(qqI), fcQY(qqI), 2, fcRED);

  IF fcBin == 0 THEN

    TEXTOUT_P("One point per bin: its midpoint against the", G1, 6, 189, 1, #000000h, 308);
    TEXTOUT_P("Blom score of the bin's middle rank.", G1, 6, 202, 1, #000000h, 308);
  END;
END;

// The ECDF window (raw: the data's range; binned: the bins' edges, plus
// 5%) and the normal CDF cached once per pixel column.
fc_EcdfFit()
BEGIN
  LOCAL efI, efT, efV;
  IF fcBin == 0 THEN
    fcEa := fcX(1); fcEb := fcX(fcNX);
  ELSE
    fcEa := fcBM(1) - fcBW / 2; fcEb := fcBM(SIZE(fcBM)) + fcBW / 2;
  END;
  efT := MAX(fcEb - fcEa, 0.000001);
  fcEa := fcEa - 0.05 * efT; fcEb := fcEb + 0.05 * efT;
  fcCvP := MAKELIST(0, efI, 1, 268, 1);
  FOR efI FROM 1 TO 268 DO
    efV := fcEa + (efI - 0.5) / 268 * (fcEb - fcEa);
    fcCvP(efI) := 158 - IP(fc_Phi((efV - fcXb) / fcSd) * 106 + 0.5);
  END;
  fcEPx := -1;
END;

// The ECDF at value ev: raw, the share of values <= ev; binned, the share
// in bins whose upper edge is <= ev.
fc_Ecdf(evV)
BEGIN
  LOCAL evI, evC;
  evC := 0;
  IF fcBin == 0 THEN
    FOR evI FROM 1 TO fcNX DO
      IF fcX(evI) <= evV THEN evC := evC + 1; END;
    END;
  ELSE
    FOR evI FROM 1 TO SIZE(fcBM) DO
      IF fcBM(evI) + fcBW / 2 <= evV + 0.0000001 THEN evC := evC + fcBF(evI); END;
    END;
  END;
  RETURN evC / fcNX;
END;

// beta 1a (review 3): Lilliefors' p as text, and its colour.  -2 (n < 5):
// "p not available for n < 5", grey -- never green for a p not computed.
fc_LDStr()
BEGIN
  IF fcPD == -2 THEN RETURN "p not available for n < 5"; END;
  IF fcPD == -1 THEN RETURN "p > 0.10"; END;
  RETURN fc_PLong(fcPD, fcLnPD);
END;

fc_LDCol()
BEGIN
  IF fcPD == -2 THEN RETURN fcGRAY; END;
  IF fcPD >= 0 AND fcPD < 0.05 THEN RETURN fcRED; END;
  RETURN fcGRN;
END;

// ============================================================
// PICTURES AND PAGES OF THE EXACT BINOMIAL TEST
//   Plot: P(X = k) over a window that holds the bell (E +/- 4.5 SD) and
//     the observed x however far out it is; red = x, magenta = every k no
//     more likely than x (their total is the two-sided p), blue = the
//     rest; a second axis counts SDs from E.  Tap a bar for P(X = k).
//   Why: the reasoning in words (for Castaneda, the Court's SDs).
// ============================================================

// 1 when k is in the two-sided rejection region (no more likely than x).
fc_InRej(irK)
BEGIN
  IF irK <= fcTkL OR irK >= fcTkU THEN RETURN 1; END;
  RETURN 0;
END;

// A probability from its log: four decimals, or "1.23 x 10^-186".
fc_ProbStr(prL)
BEGIN
  IF prL < -1E90 THEN RETURN "0"; END;
  IF prL >= LN(0.0001) THEN RETURN fc_F(EXP(prL), 4); END;
  RETURN fc_Pow(prL);
END;

// The k drawn in pixel column c (1..278) when there are more k than columns.
fc_BtK(bkC)
BEGIN
  RETURN MIN(fcTHi, fcTLo + FLOOR((bkC - 0.5) / (fcPR - fcPL) * (fcTHi - fcTLo + 1)));
END;

// The pixel column (centre) of k.
fc_BtPx(bpK)
BEGIN
  RETURN fcPL + IP((bpK - fcTLo + 0.5) / (fcTHi - fcTLo + 1) * (fcPR - fcPL));
END;

// The plot's window and its bar heights (relative to the mode), computed
// once per test.  Up to 70 values of k: one bar each; more: one pixel
// column each, at the k that column stands for.
fc_BtFit()
BEGIN
  LOCAL bfLo, bfHi, bfMd, bfLm, bfI, bfNk;
  bfLo := MAX(0, MIN(fcTx, FLOOR(fcTE - 4.5 * fcTSD)));
  bfHi := MIN(fcTn, MAX(fcTx, CEILING(fcTE + 4.5 * fcTSD)));
  IF bfHi - bfLo < 8 THEN bfLo := MAX(0, bfLo - 4); bfHi := MIN(fcTn, bfHi + 4); END;
  fcTLo := bfLo; fcTHi := bfHi;
  bfMd := MIN(fcTn, MAX(0, FLOOR((fcTn + 1) * fcTp0)));
  bfLm := fc_LPmf(bfMd);
  bfNk := bfHi - bfLo + 1;
  IF bfNk <= 70 THEN
    fcTCol := MAKELIST(0, bfI, 1, bfNk, 1);
    FOR bfI FROM 1 TO bfNk DO fcTCol(bfI) := fc_ExpG(fc_LPmf(bfLo + bfI - 1) - bfLm); END;
  ELSE
    fcTCol := MAKELIST(0, bfI, 1, fcPR - fcPL, 1);
    FOR bfI FROM 1 TO fcPR - fcPL DO fcTCol(bfI) := fc_ExpG(fc_LPmf(fc_BtK(bfI)) - bfLm); END;
  END;
END;

// The plot page.  Box: x 34..312, y 46..146; the k axis under it, the

BEGIN

  END;
  // the bars
  IF dtNk <= 70 THEN

  END;
  LINE_P(G1, fcPL, 146, fcPR, 146, #000000h);

  // above them, each on a white patch
  dtP := fc_BtPx(fcTx);

  TEXTOUT_P("E = " + fc_F(fcTE, 1), G1, dtLx, 59, 1, fcGRN, 62);

  LINE_P(G1, fcPL, 162, fcPR, 162, fcGRAY);

// The test's screens: 1 the plot, 2 Why; the chi-square key opens the
// two-category fit; Menu (or Esc, Enter) goes back.
fc_BtLoop(btScr)
BEGIN
  LOCAL bpGo, bpK, bpM, bpM1, bpSl, bpPg;
  LOCAL bpX, bpY, bpNk;
  bpGo := 1; bpPg := 1;
  fc_BtFit();
  IF btScr == 2 THEN fc_BtLines(); END;
  REPEAT bpK := GETKEY; UNTIL bpK == -1;
  fc_Drain();
  WHILE bpGo == 1 DO
    IF btScr == 1 THEN
      fc_DrawBt();
    ELSE
      fc_TextPage("Why: " + IFTE(fcTPre == 1, "Castaneda v. Partida", "exact binomial test"), bpPg, "page " + fc_I(bpPg) + " of " + fc_I(fc_NPg()) + ", " + fcUD + " or tap = page");
    END;
    fc_Show({"Plot", "Why", fcX2S, "", "", "Menu"});
    REPEAT
      bpK := GETKEY;
      bpM := MOUSE;
      bpM1 := bpM(1);
    UNTIL bpK > -1 OR SIZE(bpM1) > 0;
    IF SIZE(bpM1) > 0 THEN
      bpX := bpM1(1); bpY := bpM1(2);
      fc_Drain();
      IF bpY >= 218 THEN
        bpSl := MIN(6, IP(bpX / 53.4) + 1);
        IF bpSl == 1 THEN btScr := 1; END;
        IF bpSl == 2 THEN btScr := 2; bpPg := 1; fc_BtLines(); END;
        IF bpSl == 3 THEN fc_BtChi(); bpGo := 0; END;
        IF bpSl == 6 THEN bpGo := 0; END;
      ELSE
        IF btScr == 1 AND bpX >= fcPL AND bpX <= fcPR AND bpY >= 40 AND bpY <= 170 THEN
          bpNk := fcTHi - fcTLo + 1;
          fcTSel := MIN(fcTHi, fcTLo + FLOOR((bpX - fcPL) / (fcPR - fcPL) * bpNk));
        END;
        IF btScr == 2 THEN
          bpPg := bpPg + 1;
          IF bpPg > fc_NPg() THEN bpPg := 1; END;
        END;
      END;
    ELSE
      IF bpK == 4 OR bpK == 30 THEN bpGo := 0; END;
      IF btScr == 1 AND (bpK == 7 OR bpK == 8) THEN
        IF fcTSel < 0 THEN fcTSel := fcTx; END;
        IF bpK == 7 THEN fcTSel := MAX(fcTLo, fcTSel - 1); ELSE fcTSel := MIN(fcTHi, fcTSel + 1); END;
      END;
      IF btScr == 2 AND (bpK == 2 OR bpK == 12) THEN
        IF bpK == 2 THEN bpPg := MAX(1, bpPg - 1); ELSE bpPg := MIN(fc_NPg(), bpPg + 1); END;
      END;
      REPEAT bpK := GETKEY; UNTIL bpK == -1;
    END;
  END;
  REPEAT bpK := GETKEY; UNTIL bpK == -1;
END;

END;

BEGIN
  LOCAL rdL;

// 1 when every entry is a whole number >= 0 (a count), else 0.
fc_IsCounts(icL)
BEGIN
  LOCAL icI;
  FOR icI FROM 1 TO SIZE(icL) DO
    IF icL(icI) < 0 OR FP(icL(icI)) <> 0 THEN RETURN 0; END;
  END;
  RETURN 1;
END;

// The routes from D1 (and D2):
//   1 values: normality;  2 values: Benford first digit;  3 first two
//   digits;  4 counts, equal shares;  5 counts with shares in D2;
//   6 counts of 0, 1, 2 ... events: Poisson;  7 counts of 0..n
//   successes: binomial (p estimated or typed).
fc_MyD(mdR)
BEGIN
  LOCAL mdL, mdQ, mdI, mdS, mdLab, mdP;
  LOCAL mdC, mdOK, mdPr, mdLb, mdHp;
  mdL := fc_ReadD1();
  IF SIZE(mdL) == 0 THEN
    MSGBOX("List D1 is empty (or the Statistics 1Var app has never been opened).  Put your data in D1 first.");
    RETURN;
  END;
  IF mdR <= 3 THEN
    IF SIZE(mdL) > 5000 THEN MSGBOX("D1 has more than 5000 values; FitCheck takes at most 5000."); RETURN; END;
    IF mdR == 1 THEN
      IF fc_NormSet(mdL, "My data (D1)", "your own values, list D1") == 1 THEN fcNPre := 0; fc_NorLoop(1); END;
      RETURN;
    END;
    IF fc_Benford(mdL, mdR - 2, "Benford: my data (D1)", "your own values, list D1") == 0 THEN
      IF mdR == 3 THEN
        MSGBOX("No value in D1 is 10 or more: the first-two-digits test needs two digits.");
      ELSE
        MSGBOX("Every value in D1 is 0: there are no leading digits.");
      END;
      RETURN;
    END;
    // beta 1a (review 4): say how many values the two-digit test left out
    IF mdR == 3 AND fcBenS > 0 THEN
      MSGBOX(fc_I(fcBenS) + " of the values are below 10: they have no second digit, so they are left out.");
    END;
    fcPre := 0;
    fc_GofLoop(1);
    RETURN;
  END;
  // the count routes
  IF fc_IsCounts(mdL) == 0 THEN MSGBOX("D1 must hold counts: whole numbers, none negative."); RETURN; END;
  IF SIZE(mdL) > 120 THEN MSGBOX("At most 120 cells in D1."); RETURN; END;
  mdS := 0;
  FOR mdI FROM 1 TO SIZE(mdL) DO mdS := mdS + mdL(mdI); END;
  IF mdS == 0 THEN MSGBOX("Every count in D1 is 0: nothing to test."); RETURN; END;
  IF mdR == 4 OR mdR == 5 THEN
    IF SIZE(mdL) < 2 THEN MSGBOX("A goodness-of-fit test needs at least 2 cells."); RETURN; END;
    mdLab := {}; mdPr := {};
    FOR mdI FROM 1 TO SIZE(mdL) DO
      mdLab := CONCAT(mdLab, {fc_I(mdI)});
      mdPr := CONCAT(mdPr, {1 / SIZE(mdL)});
    END;
    IF mdR == 5 THEN
      mdQ := fc_ReadD2();
      IF SIZE(mdQ) <> SIZE(mdL) THEN MSGBOX("D2 must hold one expected share for each count in D1 (D1 has " + fc_I(SIZE(mdL)) + ", D2 has " + fc_I(SIZE(mdQ)) + ")."); RETURN; END;
      mdC := 0; mdOK := 1;
      FOR mdI FROM 1 TO SIZE(mdQ) DO
        IF mdQ(mdI) <= 0 THEN mdOK := 0; END;
        mdC := mdC + mdQ(mdI);
      END;
      IF mdOK == 0 THEN MSGBOX("Every share in D2 must be above 0 (a cell with share 0 would be impossible)."); RETURN; END;
      FOR mdI FROM 1 TO SIZE(mdQ) DO mdPr(mdI) := mdQ(mdI) / mdC; END;
      fc_SetFit(mdL, mdPr, mdLab, 0, 1, "My counts (D1), shares (D2)", "your counts, list D1; expected shares, list D2");
      IF ABS(mdC - 1) > 0.000001 THEN fcNote := "The shares in D2 added up to " + fc_Num(mdC, 6) + "; each was divided by that, so they add to 1."; END;
    ELSE
      fc_SetFit(mdL, mdPr, mdLab, 0, 1, "My counts (D1), equal shares", "your counts, list D1; every cell equally likely");
    END;
    fcPre := 0;
    fc_GofLoop(1);
    RETURN;
  END;
  IF mdR == 6 THEN
    IF fc_PoisFit(mdL, "Poisson: my counts (D1)", "your counts, list D1: D1(1) units with 0 events, D1(2) with 1, ...") == 0 THEN
      MSGBOX("Nothing to fit: D1 needs units with 1 or more events (D1(2) onward).");
      RETURN;
    END;
    fcPre := 0;
    fc_GofLoop(1);
    RETURN;
  END;
  // the binomial: estimate p, or type it
  IF SIZE(mdL) < 2 THEN MSGBOX("D1 needs n + 1 counts: groups with 0, 1, ..., n successes."); RETURN; END;
  mdI := 0;
  CHOOSE(mdI, "Binomial: p", {"Estimate p from the data", "Type p" + CHAR(8320)});
  IF mdI == 0 THEN RETURN; END;
  mdP := -1;
  IF mdI == 2 THEN
    mdP := 0.5;
    // beta 1a (review 10): the label built in a variable first, the verified form
    mdLb := "p" + CHAR(8320) + ":";
    mdHp := "The success probability under " + fcH0 + ", 0 < p" + CHAR(8320) + " < 1";
    IF INPUT({mdP}, "Binomial fit", {mdLb}, {mdHp}) == 0 THEN RETURN; END;
    IF mdP <= 0 OR mdP >= 1 THEN MSGBOX("p" + CHAR(8320) + " must be strictly between 0 and 1."); RETURN; END;
  END;
  IF fc_BinFit(mdL, mdP, "Binomial: my counts (D1)", "your counts, list D1: D1(1) groups with 0 successes, ..., D1(n+1) with n") == 0 THEN
    MSGBOX("Nothing to fit: the estimated p is 0 or 1 (no group had any successes, or every trial succeeded).");
    RETURN;
  END;
  fcPre := 0;
  fc_GofLoop(1);
END;

// Typed numbers for the exact binomial test.
fc_TypeBT()
BEGIN
  LOCAL tbX, tbN, tbP, tbL, tbH;
  tbX := 0; tbN := 10; tbP := 0.5;
  IF fcHasT == 1 THEN tbX := fcTx; tbN := fcTn; tbP := fcTp0; END;
  tbL := "p" + CHAR(8320) + ":";
  tbH := "The share under " + fcH0 + ": 0 < p" + CHAR(8320) + " < 1";
  IF INPUT({tbX, tbN, tbP}, "Exact binomial test", {"x:", "n:", tbL}, {"Successes seen: a whole number from 0 to n", "Trials: a whole number from 1 to 100000", tbH}) == 0 THEN RETURN 0; END;
  IF tbN < 1 OR tbN > 100000 OR FP(tbN) <> 0 THEN MSGBOX("n must be a whole number from 1 to 100000."); RETURN 0; END;
  IF tbX < 0 OR tbX > tbN OR FP(tbX) <> 0 THEN MSGBOX("x must be a whole number from 0 to n."); RETURN 0; END;
  IF tbP <= 0 OR tbP >= 1 THEN MSGBOX("p" + CHAR(8320) + " must be strictly between 0 and 1."); RETURN 0; END;
  fcTSrc := "your numbers";
  fc_BTest(tbX, tbN, tbP);
  fcTPre := 0;
  RETURN 1;
END;

// Chi-square goodness of fit: the last fit, your counts, or a preset.
fc_MGof()
BEGIN
  LOCAL mgI;
  mgI := 0;
  IF fcHasF == 1 THEN
    CHOOSE(mgI, fcX2S + " goodness of fit", {"The last fit again", "My counts D1, equal shares", "My counts D1, shares in D2", "Preset: draft lottery 1970", "Preset: Castaneda, 2 cells"});
  ELSE
    CHOOSE(mgI, fcX2S + " goodness of fit", {"My counts D1, equal shares", "My counts D1, shares in D2", "Preset: draft lottery 1970", "Preset: Castaneda, 2 cells"});
    IF mgI > 0 THEN mgI := mgI + 1; END;
  END;
  CASE
    IF mgI == 1 THEN fc_GofLoop(1); END;
    IF mgI == 2 THEN fc_MyD(4); END;
    IF mgI == 3 THEN fc_MyD(5); END;
    IF mgI == 4 THEN fc_PreDraft(); fc_GofLoop(1); END;
    IF mgI == 5 THEN fc_PreCast(); fc_CastSrc(); fc_BtChi(); END;
  END;
END;

// ============================================================
// FitCheck(): the menu
// ============================================================
EXPORT FitCheck()
BEGIN
  LOCAL mnI, mnRun, mnS;
  fc_Init();
  fc_Opening();
  mnRun := 1;
  WHILE mnRun == 1 DO
    mnI := 0;
    // CHOOSE numbers the items itself (1-9, then A, B); the index is
    // tested, never CHOOSE's return value
    CHOOSE(mnI, "FitCheck: goodness of fit", {
      fcX2S + " goodness of fit",
      "Benford's law",
      "Poisson fit",
      "Binomial fit",
      "Normality: Q-Q, W, A" + fcSq + ", D",
      "Normality: " + fcX2S + " on bins",
      "Exact binomial test",
      "Load a preset",
      "Use my data (D1, D2)",
      "Help",
      "Quit"});
    CASE
      IF mnI == 0 OR mnI == 11 THEN mnRun := 0; END;
      IF mnI == 1 THEN fc_MGof(); END;
      IF mnI == 2 THEN fc_MBen(); END;
      IF mnI == 3 THEN fc_MPois(); END;
      IF mnI == 4 THEN fc_MBin(); END;
      IF mnI == 5 THEN fc_MNorm(); END;
      IF mnI == 6 THEN fc_MNBin(); END;
      IF mnI == 7 THEN fc_MBTest(); END;
      IF mnI == 8 THEN fc_Presets(); END;
      IF mnI == 9 THEN fc_MMy(); END;
      IF mnI == 10 THEN fc_Help(); END;
    END;
  END;
  // the summary goes back to Home as the program's value (no Terminal)
  mnS := "FitCheck:";
  IF fcHasF == 1 THEN
    mnS := mnS + " " + fcX2S + "(" + fc_I(fcDF) + ") = " + fc_F(fcX2, 3);
    IF fcDF >= 1 AND fcBad == 0 THEN mnS := mnS + ", " + fc_PStr(fcP); END;
    mnS := mnS + ";";
  END;
  IF fcHasN == 1 AND fcBin == 0 THEN mnS := mnS + " W = " + fc_F(fcW, 4) + ", " + fc_PStr(fcPW) + ";"; END;
  IF fcHasT == 1 THEN mnS := mnS + " binomial " + fc_PStr(fcTp2) + ";"; END;
  IF mnS == "FitCheck:" THEN mnS := "FitCheck: nothing computed"; END;
  RETURN mnS;
END;
