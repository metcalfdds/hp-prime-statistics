// Firmware: written for the 2026-09-09 firmware (2.4.2).  This
//   version has run in the PC harness (ppl_run_v3e) and on the
//   HP Prime Virtual Calculator; not yet on a G1 or a G2.  The
//   heights come from the Prime's own STUDENT(df, t) density, which
//   takes any df (Lessons Learned s3); the critical values from
//   STUDENT_ICDF in the mirrored form, and the tail probabilities
//   as lower tails, so nothing cancels.  verify_TDist3D_beta1.py
//   checks the projection against a Python copy and the readouts
//   against SciPy.
//
// Run TDist3D() from the Program Catalog.
//
// beta 1 (8 Nov 2026): first version.  Roger: "can we do a scaled
//   down rotatable version of that 3D t distribution plot in
//   PPL/MicroPython?"  The plot is the one drawn in Maple for the
//   book (Statisticles, Figure 9.3): the density f(t; nu) as a
//   surface over t and the degrees of freedom nu.
//   - PPL THROUGHOUT.  A frame is about 510 LINE_P calls and a
//     projection of 275 points; MicroPython would have to hand
//     hundreds of numbers back across the bridge every frame
//     (BurrLab beta 1e's lesson, Lessons Learned s9), so there is
//     no Python in this program.
//   - BUILT ON BINORM3D beta 1e: its projection (bnFrame), its
//     drag (bnDrag3D), its wait, drain, page and comma routines,
//     re-tagged td and reworked for a rectangular floor (t across,
//     nu away) and a surface whose peak moves with nu.
//   - WIREFRAME ONLY, LINE_P alone, as BiNorm3D: the 3-D
//     TRIANGLE_P mesh drew nothing in GfxProbe T07 and is unused.
//   - ANGLE MODE.  Every angle is kept in RADIANS and converted at
//     each SIN / COS with tdAU = ASIN(1) / (pi / 2), whatever the
//     calculator's angle setting is (BiNorm3D's rule).
//   - COARSE GRID WHILE DRAGGING: 13 t values by 6 nu rows (137
//     segments) while a finger turns the surface, the full 25 by
//     11 grid (514 segments) when it lifts.  Menu > Grid while
//     dragging turns this off; Menu > Frame timing measures both.
// ============================================================
//
// WHAT YOU SEE
//   3-D view: f(t; nu), Student's t density, over t from -6 to 6
//     (across) and nu from 0.5 to 10 (away from you).  Each row is
//     one t curve, with area 1.  Colour is height: blue low,
//     magenta, red at the top (0.4 is the top of the scale).  The
//     front rows (small nu) are low in the middle and lifted in the
//     tails; the back rows rise toward the normal curve, drawn in
//     gray dots just beyond the back edge as "nu = infinity".
//   Slice: one row, nu = nu0, in red, with its numbers in the
//     corner: the peak f(0), the 2.5 % critical value (the t that
//     replaces 1.960), and the chance of landing beyond +/-2 (0.046
//     for the normal).  Red posts mark +/- t crit on the slice when
//     they fall inside t = +/-6.
//   Curves view (Curves): the family seen end on, as a textbook
//     draws it: the eleven rows stacked, N(0, 1) in black dots, the
//     slice in red, +/-1.960 in grey dots.
//
// THE ARITHMETIC (checked against SciPy, verify_TDist3D_beta1.py)
//   f(t; nu) = STUDENT(nu, t)
//            = G((nu+1)/2) / ((nu pi)^0.5 G(nu/2)) (1 + t^2/nu)^(-(nu+1)/2)
//   t crit (2.5 % in each tail) = -STUDENT_ICDF(nu, 0.025)
//   P(|T| > 2) = 2 STUDENT_CDF(nu, -2)
//   peaks: 0.270 (nu 0.5), 0.318 (1), 0.354 (2), 0.368 (3),
//     0.380 (5), 0.389 (10), 0.399 (normal)
//   3-D view, orthographic, azimuth theta and elevation phi, with
//     X = 14 t pixels across and Y = 14 (nu - 5) pixels away:
//     screen x = 160 + X cos theta - Y sin theta
//     screen y = cy - (X sin theta + Y cos theta) sin phi
//                   - k f cos phi,  k = 250 px per unit of density
//
// KEYS AND TOUCH
//   3-D view: drag sideways = turn (0.6 degrees per pixel); drag
//     down = look from higher up, drag up = lower (phi 10..80
//     degrees); Left / Right turn and Up / Down raise or lower the
//     eye in 10-degree steps.
//   Curves view: Up / Down move the slice up and down the ladder of
//     nu values.
//   Soft keys: nu- nu+ (also the - and + keys) | Curves (3-D) |
//     Slice | Reset (theta 35, phi 30, nu 3) | Menu (also Enter)
//   Menu: set nu exactly (0.5 to 1,000), the views, slice, reset,
//     the grid while dragging, frame timing, help, quit.  Esc = quit.
//
// HOME EXPORT (readable after the program ends)
//   TD_NU  the last nu of the slice
//
// SECTIONS
//   state and geometry; small helpers; the cache (once per run);
//   the slice (once per nu); the projection; the 3-D view; the
//   curves view; dragging; setting nu; menu, timing, help; init
//   and main.
//
// LAYOUT NOTE
//   Font 1 is about 6.5 px per character.  Help pages start at
//   x = 12 and keep each line to 46 characters; every TEXTOUT_P
//   passes a width so nothing runs off the right-hand edge.  The
//   soft-key strip takes y 218-239, so the views stop at y 216.
// ============================================================


EXPORT TD_NU;

// ------------------------------------------------------------
// STATE SHARED BY EVERY ROUTINE (private to this program; all of
// it is set in tdInit at the start of every run)
// ------------------------------------------------------------
LOCAL tdNu, tdTh, tdPh, tdView, tdSlc, tdDrag;
//  tdNu   the slice's degrees of freedom nu0, 0.5 .. 1,000
//  tdTh   azimuth theta in RADIANS: the turn about the vertical axis
//  tdPh   elevation phi in RADIANS (10..80 degrees): the eye's height
//  tdView 1 = the 3-D surface, 2 = the curves end on
//  tdSlc  1 = the slice is drawn     tdDrag 1 while a finger turns it
LOCAL tdCrs, tdMsg, tdTX, tdTY, tdAU, tdPI;
//  tdCrs  1 = draw the coarse grid while dragging
//  tdMsg  a one-frame message, "" for none
//  tdTX, tdTY  where the last tap landed (set by tdWaitAny)
//  tdAU   angle unit: radians times tdAU is what SIN and COS expect
LOCAL tdDR, tdLad, tdNL;
//  tdDR one degree in radians;  tdLad the ladder of nu values the
//  nu- and nu+ keys step through, tdNL its length

// ---- the cache: computed once per run in tdCache ----
LOCAL tdTV, tdNV, tdZ, tdSCX, tdSCY, tdRamp;
//  tdTV  t values: 25 fine (-6 .. 6 by 0.5), then 13 coarse (by 1)
//  tdNV  nu values: 11 fine rows (0.5, 1, 1.5, 2, 2.5, 3, 4, 5, 6,
//        8, 10), then 6 coarse rows (0.5, 1, 2, 3, 5, 10)
//  tdZ   heights f(t; nu): 275 fine, row by row (nu outer, t
//        inner), then 78 coarse
//  tdSCX colour of every segment along t: 264 fine, then 72 coarse
//  tdSCY colour of every segment along nu: 250 fine, then 65 coarse
//  tdRamp 16 colours, blue .. magenta .. red
LOCAL tdNZ, tdCV, tdCN, tdCW;
//  tdNZ  the normal density at the 25 fine t values (the ghost row)
//  tdCV  the curves view: 11 curves at tdCW columns each, f(t; nu)
//  tdCN  the normal density at those columns;  tdCW = 136 columns

// ---- the slice: computed in tdSlice whenever nu changes ----
LOCAL tdSZ, tdSPk, tdSCr, tdSP2, tdSCv;
//  tdSZ  f(t; nu0) at 49 values of t (-6 .. 6 by 0.25)
//  tdSPk its peak f(0; nu0)    tdSCr its 2.5 % critical value
//  tdSP2 P(|T| > 2)            tdSCv the slice at the curves view's columns

// ---- one frame of the 3-D view (set by tdFrame and tdDraw3D) ----
LOCAL tdPX, tdPY, tdWA, tdWB, tdWC, tdWD;
//  tdPX, tdPY screen x and y of every grid point
//  tdWA .. tdWD  the parts of those that depend on t or on nu alone
LOCAL tdFA, tdFB, tdFC, tdFD, tdFE, tdFY;
LOCAL tdFCt, tdFSt;
//  screen x = 160 + FA X - FB Y,  screen y = FY - FC X - FD Y - FE z
//  with X and Y in pixel units (tdFrame explains)

// ---- geometry, shared by the drawing and the touch code ----
LOCAL tdST, tdSN, tdK, tdCX3, tdR3, tdHM;
LOCAL tdYB;
LOCAL tdBT, tdBB, tdCL, tdCR, tdCT, tdCB;
LOCAL tdCU, tdCH, tdCX0;
//  3-D: tdST pixels per unit of t, tdSN pixels per unit of nu, tdK
//       pixels per unit of density; the floor's centre (t = 0,
//       nu = 5) is at x = tdCX3; tdR3 is half the floor's diagonal,
//       tdYB the back edge's depth and tdHM the tallest peak, all in
//       pixels; the picture fits
//       between y = tdBT and y = tdBB
//  curves view: plot box tdCL..tdCR by tdCT..tdCB, tdCU pixels per
//       unit of t, tdCH pixels per unit of density, t = 0 at x = tdCX0

// ---- house colours and display glyphs ----
LOCAL tdBLUE, tdMAG, tdRED, tdGRN, tdORNG, tdGRAY;
LOCAL tdFOOT, tdGRID, tdFLR, tdLBLU, tdLMAG, tdLORN;
LOCAL tdLRED, tdLGRN, tdLGRY;
LOCAL tdNS, tdTS, tdFS, tdDg, tdMin, tdInf;
LOCAL tdPM, tdSq;
//  tdNS nu, tdTS theta, tdFS phi, tdDg degree sign, tdMin a minus
//  sign, tdInf infinity, tdPM plus-or-minus, tdSq a superscript 2 --
//  built with CHAR() in tdInit so the code stays ASCII.


// ------------------------------------------------------------
// SMALL HELPERS
// ------------------------------------------------------------

// ---- tdCm: commas in big numbers (the fleet's routine, round 5) ----
// Roger, 2 Nov 2026: "always put comma delimiters in big numbers".  STRING never
// groups digits, whatever Home's setting, so this inserts a comma every three
// digits in each run of four or more digits that does not follow a decimal
// point: "1234567.25" -> "1,234,567.25".  Only text for the screen comes here.
// Run on the Virtual Calculator: CommaProbe beta 1, 12 of 12 (2 Nov 2026).
tdCm(cmS)
BEGIN
  LOCAL cmO, cmI, cmJ, cmN, cmR, cmK;
  cmN := DIM(cmS);
  IF cmN < 4 THEN RETURN cmS; END;
  cmO := ""; cmI := 1;
  WHILE cmI <= cmN DO
    IF INSTRING("0123456789", MID(cmS, cmI, 1)) > 0 THEN
      cmJ := cmI;
      WHILE cmJ < cmN AND INSTRING("0123456789", MID(cmS, cmJ + 1, 1)) > 0 DO cmJ := cmJ + 1; END;
      cmR := MID(cmS, cmI, cmJ - cmI + 1);
      IF DIM(cmR) >= 4 AND (cmI == 1 OR MID(cmS, MAX(1, cmI - 1), 1) <> ".") THEN
        cmK := DIM(cmR) MOD 3;
        IF cmK == 0 THEN cmK := 3; END;
        cmO := cmO + LEFT(cmR, cmK);
        WHILE cmK < DIM(cmR) DO
          cmO := cmO + "," + MID(cmR, cmK + 1, 3);
          cmK := cmK + 3;
        END;
      ELSE
        cmO := cmO + cmR;
      END;
      cmI := cmJ + 1;
    ELSE
      cmO := cmO + MID(cmS, cmI, 1);
      cmI := cmI + 1;
    END;
  END;
  RETURN cmO;
END;

// A number as text with a FIXED number of decimals.  Mode 2 keeps
// trailing zeros; mode 1 would ignore the digits argument.  The
// value is rounded first, so a small value never turns scientific
// (Lessons Learned s3).
tdF(fmV, fmD)
BEGIN
  RETURN tdCm(STRING(ROUND(fmV, fmD), 2, fmD));
END;

// nu as text: whole numbers without decimals ("3"), one decimal when
// one is enough ("2.5"), else two ("7.13").  The decimals are counted,
// never stripped (Lessons Learned s3: a stripper ate "140000000000").
tdNuS(nsV)
BEGIN
  IF ABS(nsV - ROUND(nsV, 0)) < 0.001 THEN RETURN tdF(nsV, 0); END;
  IF ABS(nsV - ROUND(nsV, 1)) < 0.001 THEN RETURN tdF(nsV, 1); END;
  RETURN tdF(nsV, 2);
END;

// Wait for the finger to lift -- but never for more than a second.
// MOUSE(1) is -1 while nothing touches the screen.  Without the
// time limit a stuck or phantom touch would hold the program here
// for ever, and the only way out would be a reset.
tdDrain()
BEGIN
  LOCAL drT;
  drT := TICKS;                                  // milliseconds
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Drain, wait for a key OR a tap, drain.  Returns the key code, or
// -1 for a tap (its x and y are left in tdTX and tdTY).
tdWaitAny()
BEGIN
  LOCAL wk, wm, wm1;
  REPEAT wk := GETKEY; UNTIL wk == -1;
  tdDrain();
  REPEAT
    wk := GETKEY;
    wm := MOUSE;
    wm1 := wm(1);
  UNTIL wk > -1 OR SIZE(wm1) > 0;
  tdTX := -1; tdTY := -1;
  IF SIZE(wm1) > 0 THEN tdTX := wm1(1); tdTY := wm1(2); wk := -1; END;
  tdDrain();
  REPEAT UNTIL GETKEY == -1;
  RETURN wk;
END;

// Keep theta between 0 and 360 degrees (2 pi) after a turn.
tdNormTh()
BEGIN
  WHILE tdTh < 0 DO tdTh := tdTh + 2 * tdPI; END;
  WHILE tdTh >= 2 * tdPI DO tdTh := tdTh - 2 * tdPI; END;
END;

// An angle held in radians, shown in whole degrees.
tdDeg(adV)
BEGIN
  LOCAL adD;
  adD := ROUND(adV / tdDR, 0);
  IF adD >= 360 THEN adD := adD - 360; END;
  RETURN STRING(adD, 2, 0) + tdDg;
END;

// The colour for a height given as a share of the top of the scale
// (0 to 1): one of 16 steps from blue through magenta to red.
tdRampC(rcT)
BEGIN
  RETURN tdRamp(1 + IP(15 * MAX(0, MIN(1, rcT)) + 0.5));
END;


// ------------------------------------------------------------
// THE CACHE.  The surface never changes (nu is an axis, not a
// setting), so everything but the slice is computed ONCE per run:
// the heights on both grids and the segment colours, the normal
// ghost row, and the curves view's eleven curves.
// ------------------------------------------------------------
tdCache()
BEGIN
  LOCAL chI, chJ, chNT, chNN, chOT, chON;
  LOCAL chOf, chSo, chP, chM, chT, chN;
  FOR chM FROM 1 TO 2 DO
    // chM = 1: the fine 25 x 11 grid; chM = 2: the coarse 13 x 6
    // grid, stored after it in the same lists
    IF chM == 1 THEN
      chNT := 25; chNN := 11; chOT := 0; chON := 0; chOf := 0; chSo := 0;
    ELSE
      chNT := 13; chNN := 6; chOT := 25; chON := 11; chOf := 275; chSo := 264;
    END;
    // the heights, row by row: point (i, j) is t = value i, nu = value j
    FOR chJ FROM 1 TO chNN DO
      chN := tdNV(chON + chJ);
      FOR chI FROM 1 TO chNT DO
        chT := tdTV(chOT + chI);
        tdZ(chOf + (chJ - 1) * chNT + chI) := STUDENT(chN, chT);
      END;
    END;
    // segment colours from the mean height of the two ends, as a
    // share of 0.4.  Along t: point p to point p + 1.
    FOR chJ FROM 1 TO chNN DO
      FOR chI FROM 1 TO chNT - 1 DO
        chP := chOf + (chJ - 1) * chNT + chI;
        tdSCX(chSo + (chJ - 1) * (chNT - 1) + chI) := tdRampC((tdZ(chP) + tdZ(chP + 1)) / 0.8);
      END;
    END;
    // along nu: point p to point p + chNT (the next row); the coarse
    // grid's colours follow the fine grid's in the same list
    IF chM == 1 THEN chSo := 0; ELSE chSo := 250; END;
    FOR chI FROM 1 TO chNT DO
      FOR chJ FROM 1 TO chNN - 1 DO
        chP := chOf + (chJ - 1) * chNT + chI;
        tdSCY(chSo + (chI - 1) * (chNN - 1) + chJ) := tdRampC((tdZ(chP) + tdZ(chP + chNT)) / 0.8);
      END;
    END;
  END;
  // the normal ghost row: N(0, 1) at the 25 fine t values
  FOR chI FROM 1 TO 25 DO
    tdNZ(chI) := NORMALD(0, 1, tdTV(chI));
  END;
  // the curves view: f(t; nu) at tdCW columns, 2 px apart, for the
  // 11 fine rows, and the normal at the same columns
  FOR chJ FROM 1 TO 11 DO
    chN := tdNV(chJ);
    FOR chI FROM 1 TO tdCW DO
      chT := (tdCL + 2 * (chI - 1) - tdCX0) / tdCU;
      tdCV((chJ - 1) * tdCW + chI) := STUDENT(chN, chT);
    END;
  END;
  FOR chI FROM 1 TO tdCW DO
    chT := (tdCL + 2 * (chI - 1) - tdCX0) / tdCU;
    tdCN(chI) := NORMALD(0, 1, chT);
  END;
END;

// The slice: the row nu = nu0 at 49 values of t, its peak, its
// critical value and its tail chance, and the same row at the
// curves view's columns.  Recomputed whenever nu changes.
tdSlice()
BEGIN
  LOCAL slI, slT;
  FOR slI FROM 1 TO 49 DO
    slT := -6 + (slI - 1) * 0.25;
    tdSZ(slI) := STUDENT(tdNu, slT);
  END;
  tdSPk := STUDENT(tdNu, 0);
  // the mirrored form: never ICDF(1 - alpha) (Lessons Learned s3)
  tdSCr := -STUDENT_ICDF(tdNu, 0.025);
  // a lower tail, doubled: 1 - CDF would lose its digits
  tdSP2 := 2 * STUDENT_CDF(tdNu, -2);
  FOR slI FROM 1 TO tdCW DO
    slT := (tdCL + 2 * (slI - 1) - tdCX0) / tdCU;
    tdSCv(slI) := STUDENT(tdNu, slT);
  END;
END;


// ------------------------------------------------------------
// THE PROJECTION (orthographic: no perspective), BiNorm3D's.
// The floor is measured in pixels before it is turned: X = tdST t
// across, Y = tdSN (nu - 5) away.  Turn by theta about the
// vertical axis, then look from elevation phi:
//   screen x = 160 + X cos theta - Y sin theta
//   screen y = cy  - (X sin theta + Y cos theta) sin phi
//                  - k z cos phi
// X cos theta - Y sin theta is the left-right position after the
// turn; X sin theta + Y cos theta is the depth (far points sit
// higher on the screen when we look down at angle phi); the height
// z shows by cos phi.  tdFrame works out the constants once per
// frame; its four SIN / COS calls convert radians with tdAU.
// ------------------------------------------------------------
tdFrame()
BEGIN
  LOCAL frCt, frSt, frSp, frCp, frT, frB;
  frCt := COS(tdTh * tdAU); frSt := SIN(tdTh * tdAU);
  frSp := SIN(tdPh * tdAU); frCp := COS(tdPh * tdAU);
  tdFA := frCt;
  tdFB := frSt;
  tdFC := frSt * frSp;
  tdFD := frCt * frSp;
  tdFE := tdK * frCp;
  tdFCt := frCt; tdFSt := frSt;
  // Where the floor's centre goes.  It depends on phi only, so the
  // picture does not jump while theta changes: room above for the
  // far corner of the floor or for the tallest peak on the back
  // edge (t = 0, nu = 10, at most tdYB away), whichever is higher;
  // room below for the near corner; centred between tdBT and tdBB
  // when it all fits.
  frT := MAX(tdR3 * frSp, tdYB * frSp + tdHM * frCp);
  frB := tdR3 * frSp;
  IF frT + frB <= tdBB - tdBT THEN
    tdFY := tdBT + frT + (tdBB - tdBT - frT - frB) / 2;
  ELSE
    tdFY := tdBB - frB;
  END;
END;

// One point (t, nu, z) of the 3-D view on the screen: the floor,
// the axes, the ghost row and the slice use these; the grid uses
// tdDraw3D's lists.
tdPrjX(pjT, pjN)
BEGIN
  RETURN IP(tdCX3 + 0.5 + tdFA * tdST * pjT - tdFB * tdSN * (pjN - 5));
END;

tdPrjY(pjT, pjN, pjZ)
BEGIN
  RETURN IP(tdFY + 0.5 - tdFC * tdST * pjT - tdFD * tdSN * (pjN - 5) - tdFE * pjZ);
END;

// The soft keys, painted on the screen after every blit.  The
// labels never produce a key code; a tap with y >= 218 is read as
// slot MIN(6, IP(x / 53.4) + 1) by the main loop.
tdMenuBar()
BEGIN
  IF tdView == 1 THEN
    DRAWMENU({tdNS + tdMin, tdNS + "+", "Curves", "Slice", "Reset", "Menu"});
  ELSE
    DRAWMENU({tdNS + tdMin, tdNS + "+", "3-D", "Slice", "Reset", "Menu"});
  END;
END;

// The slice's numbers: four lines from x = rdX, at the top.  The 3-D
// view puts them top right (at the starting angles the surface's
// back edge rises into the top-left corner); the curves view puts
// them top left, above its plot box.
tdReadout(rdX)
BEGIN
  TEXTOUT_P(tdNS + " = " + tdNuS(tdNu), G1, rdX, 3, 2, tdRED, 110);
  TEXTOUT_P("peak " + tdF(tdSPk, 3), G1, rdX, 18, 1, tdGRAY, 110);
  TEXTOUT_P("t crit " + tdF(tdSCr, 3) + " (z 1.960)", G1, rdX, 30, 1, tdRED, 143);
  TEXTOUT_P("beyond " + tdPM + "2: " + tdF(tdSP2, 3) + " (z 0.046)", G1, rdX, 42, 1, tdRED, 143);
END;


// ------------------------------------------------------------
// THE 3-D VIEW: floor, wireframe, the normal ghost, the slice and
// the readouts -- one frame in G1, one blit.  While a finger drags
// (tdDrag = 1) and the coarse option is on, the 13 x 6 grid is
// drawn instead of 25 x 11.
// ------------------------------------------------------------
tdDraw3D()
BEGIN
  LOCAL d3NT, d3NN, d3OT, d3ON, d3Of, d3SX;
  LOCAL d3SY, d3I, d3J, d3Q, d3P, d3V;
  LOCAL d3Lx, d3Ly, d3X, d3Y, d3Cx, d3Cy;
  LOCAL d3M, d3T;
  tdFrame();
  DIMGROB_P(G1, 320, 240, #FFFFFFh);
  // ---- which grid ----
  d3NT := 25; d3NN := 11; d3OT := 0; d3ON := 0; d3Of := 0; d3SX := 0; d3SY := 0;
  IF tdDrag == 1 AND tdCrs == 1 THEN
    d3NT := 13; d3NN := 6; d3OT := 25; d3ON := 11; d3Of := 275; d3SX := 264; d3SY := 250;
  END;
  // ---- project every grid point (t, nu, f) ----
  // The t-part and the nu-part of each coordinate are worked out
  // once per grid value; a point then costs two lookups and a sum.
  FOR d3I FROM 1 TO d3NT DO
    d3V := tdST * tdTV(d3OT + d3I);
    tdWA(d3I) := tdFA * d3V;
    tdWC(d3I) := tdFC * d3V;
  END;
  FOR d3J FROM 1 TO d3NN DO
    d3V := tdSN * (tdNV(d3ON + d3J) - 5);
    tdWB(d3J) := tdFB * d3V;
    tdWD(d3J) := tdFD * d3V;
  END;
  d3Cx := tdCX3 + 0.5; d3Cy := tdFY + 0.5;
  FOR d3J FROM 1 TO d3NN DO
    FOR d3I FROM 1 TO d3NT DO
      d3P := (d3J - 1) * d3NT + d3I;
      tdPX(d3P) := IP(d3Cx + tdWA(d3I) - tdWB(d3J));
      tdPY(d3P) := IP(d3Cy - tdWC(d3I) - tdWD(d3J) - tdFE * tdZ(d3Of + d3P));
    END;
  END;
  // ---- the floor (z = 0): t from -6 to 6 by nu from 0 to 10, and
  //      the two axes with their names ----
  LINE_P(G1, tdPrjX(-6, 0), tdPrjY(-6, 0, 0), tdPrjX(6, 0), tdPrjY(6, 0, 0), tdFLR);
  LINE_P(G1, tdPrjX(6, 0), tdPrjY(6, 0, 0), tdPrjX(6, 10), tdPrjY(6, 10, 0), tdFLR);
  LINE_P(G1, tdPrjX(6, 10), tdPrjY(6, 10, 0), tdPrjX(-6, 10), tdPrjY(-6, 10, 0), tdFLR);

// ------------------------------------------------------------
// SETTING nu
// ------------------------------------------------------------

// A new nu for the slice: rounded to two decimals, kept within
// 0.5 .. 1,000 (with a word on screen when it stops), then the
// slice is recomputed.
tdSetNu(snV)
BEGIN
  LOCAL snN;
  snN := ROUND(snV, 2);
  IF snN > 1000 THEN
    snN := 1000;
    tdMsg := tdNS + " stops at 1,000 here";
  END;
  IF snN < 0.5 THEN
    snN := 0.5;
    tdMsg := tdNS + " stops at 0.5 here";
  END;
  IF snN <> tdNu THEN
    tdNu := snN;
    tdSlice();
  END;
  TD_NU := tdNu;
END;

// nu- and nu+: the next value of the ladder below or above nu0.
tdStepNu(stD)
BEGIN
  LOCAL stI, stV;
  stV := tdNu;
  IF stD > 0 THEN
    FOR stI FROM 1 TO tdNL DO
      IF tdLad(stI) > tdNu + 0.001 AND stV == tdNu THEN stV := tdLad(stI); END;
    END;
    IF stV == tdNu THEN tdMsg := tdNS + " stops at 1,000 here"; END;
  ELSE
    FOR stI FROM tdNL DOWNTO 1 DO
      IF tdLad(stI) < tdNu - 0.001 AND stV == tdNu THEN stV := tdLad(stI); END;
    END;
    IF stV == tdNu THEN tdMsg := tdNS + " stops at 0.5 here"; END;
  END;
  tdSetNu(stV);
END;

// Reset: theta 35 degrees, phi 30 degrees, nu 3, the slice on.
tdReset()
BEGIN
  tdTh := 35 * tdDR;
  tdPh := 30 * tdDR;
  tdSlc := 1;
  tdSetNu(3);
END;


// ------------------------------------------------------------
// DRAGGING (BiNorm3D's bnDrag3D)
// ------------------------------------------------------------

// Turn the surface with a finger.  dgPx, dgPy is the first MOUSE
// sample (the press).  While the finger stays down, every new sample
// sets theta and phi from the distance moved since the press, 0.6
// degrees per pixel; movement under 8 px is a tap, not a drag.  The
// view is redrawn per sample from the cache (coarse if tdCrs = 1)
// and once more at full detail by the main loop after the lift.
// Any key ends the drag too, so a stuck touch cannot hold it.
tdDrag3D(dgPx, dgPy)
BEGIN
  LOCAL dgM, dgM1, dgK, dgT0, dgP0, dgMv;
  LOCAL dgDx, dgDy, dgNt, dgNp;
  dgT0 := tdTh; dgP0 := tdPh; dgMv := 0;
  REPEAT
    dgM := MOUSE;
    dgM1 := dgM(1);
    dgK := GETKEY;
    IF SIZE(dgM1) > 0 THEN
      dgDx := dgM1(1) - dgPx;
      dgDy := dgM1(2) - dgPy;
      IF ABS(dgDx) > 8 OR ABS(dgDy) > 8 THEN dgMv := 1; END;
      IF dgMv == 1 THEN
        // sideways = turn; down = look from higher up (phi grows)
        dgNt := dgT0 + 0.6 * dgDx * tdDR;
        dgNp := MAX(10 * tdDR, MIN(80 * tdDR, dgP0 + 0.6 * dgDy * tdDR));
        IF dgNt <> tdTh OR dgNp <> tdPh THEN
          tdTh := dgNt; tdPh := dgNp;
          tdDrag := 1;
          tdDraw3D();
        END;
      END;
    END;
  UNTIL SIZE(dgM1) == 0 OR dgK > -1;
  tdDrag := 0;
  tdNormTh();
  tdDrain();                         // capped: at most one second
  RETURN dgMv;
END;

// In the curves view a tap on the plot picks the row whose curve
// passes nearest the finger, at the finger's own column (near the
// peak the rows are ordered by height, in the tails the other way
// round).  The top rows sit within a pixel of each other, so a tap
// there picks one of them.  A tap elsewhere does nothing.
tdTapCv(tcX, tcY)
BEGIN
  LOCAL tcI, tcH, tcB, tcD, tcV, tcC;
  IF tcX < tdCL OR tcX > tdCR OR tcY < tdCT OR tcY > tdCB THEN RETURN 0; END;
  tcH := (tdCB - tcY) / tdCH;
  tcC := 1 + IP((tcX - tdCL) / 2 + 0.5);
  tcC := MAX(1, MIN(tdCW, tcC));
  tcB := 1; tcD := 10;
  FOR tcI FROM 1 TO 11 DO
    tcV := ABS(tdCV((tcI - 1) * tdCW + tcC) - tcH);
    IF tcV < tcD THEN tcD := tcV; tcB := tcI; END;
  END;
  tdSlc := 1;
  tdSetNu(tdNV(tcB));
  RETURN 1;
END;

// Type nu exactly.  INPUT needs a variable NAME, so the value goes
// through the local anV; the field takes a real number only.
tdAskNu()
BEGIN
  LOCAL anV, anOK, anL;
  anV := tdNu;
  anL := tdNS + " =";
  anOK := INPUT({{anV,[0]}}, "Degrees of freedom", {anL}, {"From 0.5 to 1,000; two decimals are kept"});
  IF anOK == 1 THEN
    IF anV < 0.5 OR anV > 1000 THEN
      MSGBOX(tdNS + " must be from 0.5 to 1,000 here.  Below 0.5 the curve is all tail, and past 1,000 it is the normal curve to three decimals.");
    ELSE
      tdSlc := 1;
      tdSetNu(anV);
    END;
  END;
END;

// Frame timing, for the G1 test: five 3-D frames at 25 x 11, five
// at 13 x 6 (as while dragging), then five curves frames, in
// milliseconds (TICKS).
tdTiming()
BEGIN
  LOCAL tmI, tmT, tmF, tmC, tmV, tmK;
  LOCAL tmP;
  tmV := tdView; tmK := tdCrs;
  tdView := 1;
  tdDrag := 0;
  tmT := TICKS;
  FOR tmI FROM 1 TO 5 DO
    tdTh := tdTh + 2 * tdDR;
    tdDraw3D();
  END;
  tmF := (TICKS - tmT) / 5;
  tdDrag := 1; tdCrs := 1;
  tmT := TICKS;
  FOR tmI FROM 1 TO 5 DO
    tdTh := tdTh + 2 * tdDR;
    tdDraw3D();
  END;
  tmC := (TICKS - tmT) / 5;
  tdDrag := 0;
  tdView := 2;
  tmT := TICKS;
  FOR tmI FROM 1 TO 5 DO tdDrawCv(); END;
  tmP := (TICKS - tmT) / 5;
  tdCrs := tmK; tdView := tmV;
  tdNormTh();
  tmF := ROUND(tmF, 0); tmC := ROUND(tmC, 0); tmP := ROUND(tmP, 0);
  MSGBOX("3-D frame: " + tdCm(STRING(tmF, 2, 0)) + " ms at 25" + CHAR(215) + "11, " + tdCm(STRING(tmC, 2, 0)) + " ms at 13" + CHAR(215) + "6." + CHAR(10) + "Curves frame: " + tdCm(STRING(tmP, 2, 0)) + " ms." + CHAR(10) + "Over about 250 ms a drag feels sticky.");
END;

// The Enter / Menu choices.  Returns 0 to quit, 1 to carry on.
tdMenu()
BEGIN
  LOCAL muI, muS1, muS2, muS3;
  muS1 := "Curves view (end on)";
  IF tdView == 2 THEN muS1 := "3-D view"; END;
  muS2 := "Slice on";
  IF tdSlc == 1 THEN muS2 := "Slice off"; END;
  muS3 := "Grid while dragging: 25" + CHAR(215) + "11";
  IF tdCrs == 1 THEN muS3 := "Grid while dragging: 13" + CHAR(215) + "6"; END;
  muI := 0;
  CHOOSE(muI, "TDist3D", {
    "Set " + tdNS + " exactly (now " + tdNuS(tdNu) + ")",
    muS1,
    muS2,
    "Reset the view",
    muS3,
    "Frame timing (G1 test)",
    "Help",
    "Quit"});
  IF muI == 1 THEN tdAskNu(); END;
  IF muI == 2 THEN tdView := 3 - tdView; END;
  IF muI == 3 THEN tdSlc := 1 - tdSlc; END;
  IF muI == 4 THEN tdReset(); END;
  IF muI == 5 THEN tdCrs := 1 - tdCrs; END;
  IF muI == 6 THEN tdTiming(); END;
  IF muI == 7 THEN tdHelp(); END;
  IF muI == 8 THEN RETURN 0; END;
  RETURN 1;
END;

tdInit()
BEGIN
  LOCAL ivI, ivT, ivU;
  // ---- house colours ----
  tdBLUE := #003CC8h; tdMAG := #C800C8h; tdRED := #FF0000h;
  tdGRN := #008C3Ch; tdORNG := #FF4500h; tdGRAY := #5A5A5Ah;
  tdFOOT := #1E90FFh; tdGRID := #E1E4EBh; tdLBLU := #EBF2FFh;
  tdLMAG := #FAEBFAh; tdLORN := #FFF0DCh; tdLRED := #FFEBEBh;
  tdLGRN := #E8F6EBh; tdLGRY := #F0F0F0h;
  tdFLR := #A0A0A0h;                     // floor and axes: a mid grey
  // ---- display glyphs, built here so the code stays ASCII:
  //      nu, theta, phi, the degree sign, the minus sign, infinity,
  //      plus-or-minus, superscript 2 (all on the verified list) ----
  tdNS := CHAR(957); tdTS := CHAR(952); tdFS := CHAR(966); tdDg := CHAR(176);
  tdMin := CHAR(8722); tdInf := CHAR(8734); tdPM := CHAR(177); tdSq := CHAR(178);
  // ---- constants and the angle unit ----
  tdPI := 3.14159265358979;
  tdDR := tdPI / 180;
  // ASIN(1) is a right angle in the user's own unit: 90 (degrees),
  // 1.5707963267949 (radians) or 100 (grads).  Radians times tdAU
  // is therefore what SIN and COS expect, whatever the setting.
  tdAU := ASIN(1) / 1.5707963267949;
  // ---- geometry ----
  tdST := 14; tdSN := 14; tdK := 250; tdCX3 := 160;
  tdR3 := ((6 * tdST)^2 + (5 * tdSN)^2)^0.5;      // half the floor's diagonal, 109 px
  tdYB := 5 * tdSN;                                // the back edge's depth, 70 px
  tdHM := 0.4 * tdK;                               // the top of the height scale, 100 px
  tdBT := 3; tdBB := 215;
  tdCL := 30; tdCR := 300; tdCT := 54; tdCB := 182;      // the box: readouts above it
  tdCU := 27; tdCX0 := 165;                        // t from -5 to 5 across 270 px
  tdCH := 305;                                     // 0.42 of density is 128 px
  tdCW := 136;                                     // columns 2 px apart, x = 30 .. 300
  // ---- the t values: 25 fine (by 0.5), 13 coarse (by 1) ----
  tdTV := MAKELIST(0, ivI, 1, 38, 1);
  FOR ivI FROM 1 TO 25 DO tdTV(ivI) := -6 + (ivI - 1) * 0.5; END;
  FOR ivI FROM 1 TO 13 DO tdTV(25 + ivI) := -7 + ivI; END;
  // ---- the nu rows: 11 fine, 6 coarse ----
  tdNV := {0.5, 1, 1.5, 2, 2.5, 3, 4, 5, 6, 8, 10, 0.5, 1, 2, 3, 5, 10};
  // ---- the ladder the nu- and nu+ keys climb ----
  tdLad := {0.5, 1, 1.5, 2, 2.5, 3, 4, 5, 6, 8, 10, 15, 20, 30, 50, 100, 1000};
  tdNL := 17;
  // ---- room for the cache and the frame lists ----
  tdZ := MAKELIST(0, ivI, 1, 353, 1);
  tdSCX := MAKELIST(0, ivI, 1, 336, 1); tdSCY := MAKELIST(0, ivI, 1, 315, 1);
  tdPX := MAKELIST(0, ivI, 1, 275, 1); tdPY := MAKELIST(0, ivI, 1, 275, 1);
  tdWA := MAKELIST(0, ivI, 1, 25, 1); tdWC := MAKELIST(0, ivI, 1, 25, 1);
  tdWB := MAKELIST(0, ivI, 1, 11, 1); tdWD := MAKELIST(0, ivI, 1, 11, 1);
  tdNZ := MAKELIST(0, ivI, 1, 25, 1); tdSZ := MAKELIST(0, ivI, 1, 49, 1);
  tdCV := MAKELIST(0, ivI, 1, 11 * tdCW, 1);
  tdCN := MAKELIST(0, ivI, 1, tdCW, 1); tdSCv := MAKELIST(0, ivI, 1, tdCW, 1);
  // ---- the height colours: blue (0,60,200) to magenta (200,0,200)
  //      to red (255,0,0), in 16 steps ----
  tdRamp := MAKELIST(0, ivI, 1, 16, 1);
  FOR ivI FROM 1 TO 16 DO
    ivT := (ivI - 1) / 15;
    IF ivT <= 0.5 THEN
      ivU := 2 * ivT;
      tdRamp(ivI) := RGB(IP(200 * ivU + 0.5), IP(60 * (1 - ivU) + 0.5), 200);
    ELSE
      ivU := 2 * ivT - 1;
      tdRamp(ivI) := RGB(IP(200 + 55 * ivU + 0.5), 0, IP(200 * (1 - ivU) + 0.5));
    END;
  END;
  // ---- the starting state ----
  tdView := 1; tdSlc := 1; tdDrag := 0; tdCrs := 1; tdMsg := "";
  tdTX := -1; tdTY := -1;
  tdTh := 35 * tdDR; tdPh := 30 * tdDR;
  tdNu := 3;
  tdCache();
  tdSlice();
  TD_NU := tdNu;
END;

// ------------------------------------------------------------
EXPORT TDist3D()
BEGIN
  LOCAL mnK, mnM, mnM1, mnGo, mnAct, mnX;
  LOCAL mnY;
  tdInit();
  tdWelcome();
  mnGo := 1;
  WHILE mnGo == 1 DO
    // a key or a finger left over from a dialog or a drag must not
    // count as a new press: drain both before the next frame
    REPEAT mnK := GETKEY; UNTIL mnK == -1;
    tdDrain();
    tdShow();
    tdMsg := "";                       // a message lasts one frame
    REPEAT
      mnK := GETKEY;
      mnM := MOUSE;
      mnM1 := mnM(1);                  // the first finger, if any
    UNTIL mnK > -1 OR SIZE(mnM1) > 0;
    // mnAct: 1 nu down, 2 nu up, 3 Curves / 3-D, 4 Slice, 5 Reset,
    // 6 Menu; 0 nothing
    mnAct := 0;
    IF SIZE(mnM1) > 0 THEN
      mnX := mnM1(1); mnY := mnM1(2);
      IF mnY >= 218 THEN
        // the soft-key strip: six 53-px slots, left to right
        mnAct := MIN(6, IP(mnX / 53.4) + 1);
        tdDrain();
      ELSE
        IF tdView == 1 THEN tdDrag3D(mnX, mnY); ELSE tdTapCv(mnX, mnY); tdDrain(); END;
      END;
    ELSE
      // the keys remain as shortcuts for the soft keys
      IF mnK == 4 THEN mnGo := 0; END;                 // Esc   = quit
      IF mnK == 45 THEN mnAct := 1; END;               // -     = nu down
      IF mnK == 50 THEN mnAct := 2; END;               // +     = nu up
      IF mnK == 30 THEN mnAct := 6; END;               // Enter = Menu
      IF tdView == 1 THEN
        // arrows turn (theta) and tilt (phi) in 10-degree steps
        IF mnK == 7 THEN tdTh := tdTh - 10 * tdDR; tdNormTh(); END;
        IF mnK == 8 THEN tdTh := tdTh + 10 * tdDR; tdNormTh(); END;
        IF mnK == 2 THEN tdPh := MIN(80 * tdDR, tdPh + 10 * tdDR); END;
        IF mnK == 12 THEN tdPh := MAX(10 * tdDR, tdPh - 10 * tdDR); END;
      ELSE
        // end on, Up and Down move the slice along the ladder
        IF mnK == 2 THEN mnAct := 2; END;
        IF mnK == 12 THEN mnAct := 1; END;
      END;
    END;
    IF mnAct == 1 THEN tdSlc := 1; tdStepNu(-1); END;
    IF mnAct == 2 THEN tdSlc := 1; tdStepNu(1); END;
    IF mnAct == 3 THEN tdView := 3 - tdView; END;
    IF mnAct == 4 THEN tdSlc := 1 - tdSlc; END;
    IF mnAct == 5 THEN tdReset(); END;
    IF mnAct == 6 THEN mnGo := tdMenu(); END;
  END;
  TD_NU := tdNu;
  // Hand the summary back as the program's value: Home shows it in
  // the history and no Terminal screen opens.  TD_NU holds nu.
  RETURN "TDist3D: " + tdNS + " = " + tdNuS(tdNu) + ", t crit " + tdF(tdSCr, 3) + ", P(|t| > 2) = " + tdF(tdSP2, 3) + "; " + tdNS + " in TD_NU";
END;
