// Wait for the finger to lift -- never for more than a second (a stuck
// touch reading must not freeze the program: Lessons Learned 7).
av_Drain()
BEGIN
  LOCAL drT;
  drT := TICKS;
  REPEAT UNTIL MOUSE(1) == -1 OR TICKS - drT > 1000;
END;

// Empty the key buffer.
av_KDrain()
BEGIN
  LOCAL kdK;
  REPEAT kdK := GETKEY; UNTIL kdK == -1;
END;

// Drain, wait for a key OR a tap, drain.  Returns the key, or -1 for a
// tap (its x and y left in avTapX, avTapY).  Holds a page for as long as
// it takes (WAIT(-1) would give up after about a minute).
av_WaitAny()
BEGIN
  LOCAL waK, waM, waM1;
  av_KDrain();
  av_Drain();
  REPEAT
    waK := GETKEY;
    waM := MOUSE;
    waM1 := waM(1);
  UNTIL waK > -1 OR SIZE(waM1) > 0;
  avTapX := -1; avTapY := -1;
  IF SIZE(waM1) > 0 THEN
    avTapX := waM1(1); avTapY := waM1(2); waK := -1;
  END;
  av_Drain();
  av_KDrain();
  RETURN waK;
END;

// One event for a view (drain, wait, drain): a key, or -1 for a tap
// (x, y in avTapX, avTapY).
// The GETKEY + MOUSE polling loop of ChiTable: MOUSE sees a tap on the
// soft-key strip like any other tap (y >= 218).
av_Poll()
BEGIN
  LOCAL poK, poM, poM1;
  // drain first: a key left by a CHOOSE, INPUT or MSGBOX must not end
  // the wait before the page is seen (Lessons Learned 7)
  av_KDrain();
  REPEAT
    poK := GETKEY;
    poM := MOUSE;
    poM1 := poM(1);
  UNTIL poK > -1 OR SIZE(poM1) > 0;
  avTapX := -1; avTapY := -1;
  IF SIZE(poM1) > 0 THEN
    avTapX := poM1(1); avTapY := poM1(2); poK := -1;
    av_Drain();
  ELSE
    av_KDrain();
  END;
  RETURN poK;
END;

// Fixed decimals, rounded first: STRING(x, 2, d) prints SCIENTIFIC when
// 0 < |x| < 10^-d (Lessons Learned 3), and a rounded -0 prints as 0.
av_Fx(fxV, fxD)
BEGIN
  LOCAL fxR;
  fxR := ROUND(fxV, fxD);
  IF fxR == 0 THEN fxR := 0; END;
  RETURN STRING(fxR, 2, fxD);
END;

// A whole number as text.
av_I(itV)
BEGIN
  RETURN av_Fx(itV, 0);
END;

// A number with as many decimals as its size deserves (about 5 digits):
// 0.1234, 1.234, 12.34, 1234.5, 123456.
av_Gn(gnV)
BEGIN
  LOCAL gnA, gnD;
  gnA := ABS(gnV);
  gnD := 4;
  IF gnA >= 1 THEN gnD := 3; END;
  IF gnA >= 10 THEN gnD := 2; END;
  IF gnA >= 1000 THEN gnD := 1; END;
  IF gnA >= 100000 THEN gnD := 0; END;
  RETURN av_Fx(gnV, gnD);
END;

// The same with its sign: "+2.13", "-0.567".
av_Sgn(sgV)
BEGIN
  IF ROUND(sgV, 4) > 0 THEN RETURN "+" + av_Gn(sgV); END;
  RETURN av_Gn(sgV);
END;

// A p-value's digits: "< 0.0001" below what 4 decimals can show.
av_PT(ptP)
BEGIN
  IF ptP < 0.0001 THEN RETURN "< 0.0001"; END;
  RETURN av_Fx(ptP, 4);
END;

// "p = 0.0670" or "p < 0.0001" (never "p = < ...").
av_PE(peP)
BEGIN
  IF peP < 0.0001 THEN RETURN "p < 0.0001"; END;
  RETURN "p = " + av_Fx(peP, 4);
END;

// "= 0.0670" or "< 0.0001": a p-value after a name, with its own relation.
av_Rel(rlP)
BEGIN
  IF rlP < 0.0001 THEN RETURN "< 0.0001"; END;
  RETURN "= " + av_Fx(rlP, 4);
END;

// alpha as text: 0.05, 0.01, 0.001 ...
av_AS()
BEGIN
  IF ABS(avAlpha - ROUND(avAlpha, 2)) < 0.000000001 THEN RETURN av_Fx(avAlpha, 2); END;
  RETURN av_Fx(avAlpha, 3);
END;

// A confidence level as a percentage: 0.95 -> "95%", 0.995 -> "99.5%".
av_CS(clC)
BEGIN
  LOCAL clV;
  clV := 100*clC;
  IF ABS(clV - ROUND(clV, 0)) < 0.000001 THEN RETURN av_Fx(clV, 0) + "%"; END;
  RETURN av_Fx(clV, 1) + "%";
END;

// A "nice" step, 1, 2 or 5 times a power of ten, at least the value
// handed in (Heckbert), for axis ticks.
av_Nice(nnV)
BEGIN
  LOCAL nnE, nnF;
  IF nnV <= 0 THEN RETURN 1; END;
  nnE := FLOOR(LOG(nnV));
  nnF := nnV/10^nnE;
  IF nnF <= 1 THEN RETURN 10^nnE; END;
  IF nnF <= 2 THEN RETURN 2*10^nnE; END;
  IF nnF <= 5 THEN RETURN 5*10^nnE; END;
  RETURN 10^(nnE + 1);
END;

// Median of a list (sorted copy; the middle value or the mean of the
// two middle values).
av_Med(mdL)
BEGIN
  LOCAL mdS, mdN;
  mdS := SORT(mdL);
  mdN := SIZE(mdS);
  IF mdN MOD 2 == 1 THEN RETURN mdS((mdN + 1)/2); END;
  RETURN (mdS(mdN/2) + mdS(mdN/2 + 1))/2;
END;

// A name cut to fit cuW pixels of font 1 (about 6.4 px a character on
// the Prime: ChiTable keeps 46 characters to 296 px).  LEFT(s, 0) would
// be the whole string, so at least one character is kept.
av_Cut(cuS, cuW)
BEGIN
  LOCAL cuN;
  cuN := MAX(1, IP(cuW/6.4));
  IF DIM(cuS) <= cuN THEN RETURN cuS; END;
  RETURN LEFT(cuS, cuN);
END;

// A number in at most cmN characters: as many decimals (3 at most) as
// fit, else whole, else in thousands with a "k".
av_Cmp(cmV, cmN)
BEGIN
  LOCAL cmD, cmS, cmI;
  FOR cmI FROM 0 TO 3 DO
    cmD := 3 - cmI;
    cmS := av_Fx(cmV, cmD);
    IF DIM(cmS) <= cmN THEN RETURN cmS; END;
  END;
  RETURN av_Fx(cmV/1000, 0) + "k";
END;

END;

// Log-gamma, Lanczos g = 7 (LNGAMMA is CAS-only).  StudRange beta 15d's
// sr_LGAM: accurate to about 1E-13 for x >= 0.5.  0.918938533204673 is
// 0.5*LN(2*pi).
av_LGam(lgX)
BEGIN
  LOCAL lgC, lgA, lgT, lgJ, lgY;
  lgC := {676.5203681218851, -1259.1392167224028,
      771.32342877765313, -176.61502916214059,
      12.507343278686905, -0.13857109526572012,
      9.9843695780195716E-6, 1.5056327351493116E-7};
  lgY := lgX - 1; lgA := 0.99999999999980993;
  FOR lgJ FROM 1 TO 8 DO
    lgA := lgA + lgC(lgJ)/(lgY + lgJ);
  END;
  lgT := lgY + 7.5;
  RETURN 0.918938533204673 + (lgY + 0.5)*LN(lgT) - lgT + LN(lgA);
END;

// The continued fraction inside the incomplete beta (Numerical Recipes
// 6.4, modified Lentz): StatDist v2.7f's st_betacf.
av_BetaCF(bcA, bcB, bcX)
BEGIN
  LOCAL bcQab, bcQap, bcQam, bcC, bcD, bcH;
  LOCAL bcM, bcM2, bcAa, bcDel, bcGo;
  bcQab := bcA + bcB; bcQap := bcA + 1; bcQam := bcA - 1;
  bcC := 1; bcD := 1 - bcQab*bcX/bcQap;
  IF ABS(bcD) < 1E-300 THEN bcD := 1E-300; END;
  bcD := 1/bcD; bcH := bcD;
  bcM := 0; bcGo := 1;
  WHILE bcGo == 1 AND bcM < 300 DO
    bcM := bcM + 1; bcM2 := 2*bcM;
    bcAa := bcM*(bcB - bcM)*bcX/((bcQam + bcM2)*(bcA + bcM2));
    bcD := 1 + bcAa*bcD;
    IF ABS(bcD) < 1E-300 THEN bcD := 1E-300; END;
    bcC := 1 + bcAa/bcC;
    IF ABS(bcC) < 1E-300 THEN bcC := 1E-300; END;
    bcD := 1/bcD; bcH := bcH*bcD*bcC;
    bcAa := -(bcA + bcM)*(bcQab + bcM)*bcX/((bcA + bcM2)*(bcQap + bcM2));
    bcD := 1 + bcAa*bcD;
    IF ABS(bcD) < 1E-300 THEN bcD := 1E-300; END;
    bcC := 1 + bcAa/bcC;
    IF ABS(bcC) < 1E-300 THEN bcC := 1E-300; END;
    bcD := 1/bcD; bcDel := bcD*bcC; bcH := bcH*bcDel;
    IF ABS(bcDel - 1) < 0.00000000001 THEN bcGo := 0; END;
  END;
  RETURN bcH;
END;

// Regularized incomplete beta I_x(a, b) (StatDist's st_betaI), with the
// symmetry I_x(a,b) = 1 - I_(1-x)(b,a) where the fraction is slow.
av_BetaI(biA, biB, biX)
BEGIN
  LOCAL biL;
  IF biX <= 0 THEN RETURN 0; END;
  IF biX >= 1 THEN RETURN 1; END;
  biL := av_LGam(biA + biB) - av_LGam(biA) - av_LGam(biB) + biA*LN(biX) + biB*LN(1 - biX);
  IF biX < (biA + 1)/(biA + biB + 2) THEN
    RETURN EXP(biL)*av_BetaCF(biA, biB, biX)/biA;
  END;
  RETURN 1 - EXP(biL)*av_BetaCF(biB, biA, 1 - biX)/biB;
END;

// Regularized lower incomplete gamma P(a, x), the series (x < a + 1);
// StatDist's st_gammaP, series branch.
av_GamP(gpA, gpX)
BEGIN
  LOCAL gpS, gpD, gpAp, gpI, gpGo;
  IF gpX <= 0 THEN RETURN 0; END;
  gpAp := gpA; gpS := 1/gpA; gpD := gpS;
  gpI := 0; gpGo := 1;
  WHILE gpGo == 1 AND gpI < 5000 DO
    gpI := gpI + 1;
    gpAp := gpAp + 1;
    gpD := gpD*gpX/gpAp;
    gpS := gpS + gpD;
    IF ABS(gpD) < ABS(gpS)*0.000000000001 THEN gpGo := 0; END;
  END;
  RETURN MIN(1, gpS*EXP(-gpX + gpA*LN(gpX) - av_LGam(gpA)));
END;

// Regularized UPPER incomplete gamma Q(a, x), every digit kept in the far
// tail (StatDist's st_gammaQ): the continued fraction for x >= a + 1,
// 1 - P (safe there) below.  The chi-square's right tail is Q(df/2, x/2).
av_GamQ(gqA, gqX)
BEGIN
  LOCAL gqB, gqC, gqD, gqH, gqI, gqAn;
  LOCAL gqDel, gqGo;
  IF gqX <= 0 THEN RETURN 1; END;
  IF gqX < gqA + 1 THEN RETURN 1 - av_GamP(gqA, gqX); END;
  gqB := gqX + 1 - gqA; gqC := 1E300; gqD := 1/gqB; gqH := gqD;
  gqI := 0; gqGo := 1;
  WHILE gqGo == 1 AND gqI < 5000 DO
    gqI := gqI + 1;
    gqAn := -gqI*(gqI - gqA);
    gqB := gqB + 2;
    gqD := gqAn*gqD + gqB;
    IF ABS(gqD) < 1E-300 THEN gqD := 1E-300; END;
    gqC := gqB + gqAn/gqC;
    IF ABS(gqC) < 1E-300 THEN gqC := 1E-300; END;
    gqD := 1/gqD;
    gqDel := gqD*gqC;
    gqH := gqH*gqDel;
    IF ABS(gqDel - 1) < 0.00000000001 THEN gqGo := 0; END;
  END;
  RETURN MIN(1, EXP(-gqX + gqA*LN(gqX) - av_LGam(gqA))*gqH);
END;

// P(F > f) for F(a, b), computed as a LOWER tail (12-digit reals:
// 1 - CDF is exactly 0 below about 5E-13).  Whole df: the mirror
// P(F(a,b) > f) = P(F(b,a) < 1/f) with the built-in FISHER_CDF.  A
// fractional df (Welch) is "Invalid input" for FISHER_CDF, so then
// z = b/(a f + b) is Beta(b/2, a/2) and P(F > f) = I_z(b/2, a/2).
av_FUp(fuF, fuA, fuB)
BEGIN
  IF fuF <= 0 THEN RETURN 1; END;
  IF FP(fuA) == 0 AND FP(fuB) == 0 THEN RETURN FISHER_CDF(fuB, fuA, 1/fuF); END;
  RETURN av_BetaI(fuB/2, fuA/2, fuB/(fuA*fuF + fuB));
END;

// P(F < f) for F(a, b).
av_FLo(flF, flA, flB)
BEGIN
  IF flF <= 0 THEN RETURN 0; END;
  IF FP(flA) == 0 AND FP(flB) == 0 THEN RETURN FISHER_CDF(flA, flB, flF); END;
  RETURN av_BetaI(flA/2, flB/2, flA*flF/(flA*flF + flB));
END;

// P(chi-square(df) > x) as a lower-tail form: 2 Phi(-sqrt x) for 1 df,
// e^(-x/2) for 2 df, Q(df/2, x/2) otherwise.
av_ChiUp(cuX, cuD)
BEGIN
  LOCAL cuU;
  IF cuX <= 0 THEN RETURN 1; END;
  IF cuD == 1 THEN RETURN 2*NORMALD_CDF(0, 1, -SQRT(cuX)); END;
  IF cuD == 2 THEN RETURN EXP(-cuX/2); END;
  // beta 1a (second review, item 4): up to x = df the upper tail is at
  // least about one half, so 1 - CHISQUARE_CDF keeps its digits there
  // (Lessons Learned 3); beyond, it is used while it keeps 6 or more
  // digits (above 1E-6), and only the far tail goes to Q(a, x), whose
  // continued fraction converges fast out there.  Beta 1's series for
  // 1 - P stopped at 500 terms and was wrong in the 2nd decimal for
  // df near 1E5-1E6.
  IF cuX <= cuD THEN RETURN 1 - CHISQUARE_CDF(cuD, cuX); END;
  cuU := 1 - CHISQUARE_CDF(cuD, cuX);
  IF cuU > 0.000001 THEN RETURN cuU; END;
  RETURN av_GamQ(cuD/2, cuX/2);
END;

// P(chi-square(df) < x), the built-in (whole df only, which is all this
// program meets).
av_ChiLo(ckX, ckD)
BEGIN
  IF ckX <= 0 THEN RETURN 0; END;
  RETURN CHISQUARE_CDF(ckD, ckX);
END;

// Two-sided p for a t statistic: 2 P(T < -|t|).
av_TP2(tpT, tpD)
BEGIN
  RETURN MIN(1, 2*STUDENT_CDF(tpD, -ABS(tpT)));
END;

// A probability kept inside (0, 1) before any _ICDF (an ICDF at 0 or 1
// returns infinity with no error: Lessons Learned 3).
av_Clp(cpP)
BEGIN
  RETURN MIN(MAX(cpP, 1E-12), 0.999999999999);
END;

// Right-tail t quantile: P(T > t) = a, as -STUDENT_ICDF(df, a), never
// STUDENT_ICDF(df, 1 - a).
av_TQ(tqA, tqD)
BEGIN
  RETURN -STUDENT_ICDF(tqD, av_Clp(tqA));
END;

// Right-tail F quantile: P(F(n, d) > x) = a, as 1 / FISHER_ICDF(d, n, a)
// (the mirror), never FISHER_ICDF(n, d, 1 - a).
av_FQ(fqA, fqN, fqD)
BEGIN
  RETURN 1/FISHER_ICDF(fqD, fqN, av_Clp(fqA));
END;


// ============================================================
// 3. THE STUDENTIZED RANGE (Tukey's q): StudRange beta 15d's PPL engine,
//    copied with this program's tag, made faster in beta 1a.
//    - after Dunlap, Powell & Konnerth (1977), with NORMALD_CDF for their
//      polynomial normal and a u^2 substitution (s = u^2) that makes low
//      df work; df 3 and up; above df 100 the outer integral runs over
//      the spike s in 1 +/- 12/sqrt(2 df) (StudRange's notes, kept);
//    - beta 1a (second review, item 1): the inner integral P_k(x), the
//      CDF of the range of k standard normals, depends on k only.  It is
//      tabulated ONCE per k at x = 0, 0.1, ..., 12 (121 points, each by
//      StudRange's Simpson rule over z in [-7, 4], 32 intervals, 48 from
//      k = 6) and read in the outer 64-interval Simpson loop by 4-point
//      cubic (Lagrange) interpolation; P_k(x) = 1 for x >= 12.  The
//      table survives Set alpha and the Tukey p button.  Cost: 121 x 33
//      (or 49) NORMALD_CDF calls once, then 65 interpolations per CDF
//      instead of 65 x 33 (or 49) NORMALD_CDF calls.
//    - accuracy, measured against scipy.stats.studentized_range over k
//      2-12, df 3-1000 and alpha 0.10, 0.05, 0.01, 0.001: q crit within 0.022 % (worst k 12,
//      df 3, alpha 0.001), the tail P(Q > q crit) within 0.067 % of alpha
//      (worst k 4, df 100, alpha 0.001), and a pair's p = 1 - F(q) within
//      4.2E-6 absolute over q 0.25-9 (verify_AnovaLab_beta1a.py; Python
//      replica of this exact engine, checked against the PPL in the
//      harness)
//    - Esc is read inside the outer loop (every 8 nodes), with a step
//      bar that counts the steps (q crit takes 6 to 23).
//    - The MicroPython engine of StudRange is NOT carried in beta 1a.
// ============================================================

// Node tables for the inner integral over z in [-7, 4]: Phi(z) and
// phi(z) times the Simpson weight do not depend on x, so they are
// computed once per resolution.
av_BuildZ(bzN)
BEGIN
  LOCAL bzJ, bzH, bzZ, bzW, bzI;
  IF bzN == avNZ THEN RETURN 0; END;
  avZN := MAKELIST(0, bzI, 1, bzN + 1);
  avZW := MAKELIST(0, bzI, 1, bzN + 1);
  avZC := MAKELIST(0, bzI, 1, bzN + 1);
  bzH := 11/bzN;
  FOR bzJ FROM 0 TO bzN DO
    bzZ := -7 + bzJ*bzH;
    IF bzJ == 0 OR bzJ == bzN THEN
      bzW := 1;
    ELSE
      IF bzJ MOD 2 == 1 THEN bzW := 4; ELSE bzW := 2; END;
    END;
    avZN(bzJ + 1) := bzZ;
    avZW(bzJ + 1) := bzW*bzH/3*NORMALD(0, 1, bzZ);
    avZC(bzJ + 1) := NORMALD_CDF(0, 1, bzZ);
  END;
  avNZ := bzN;
  RETURN 0;
END;

// P_k(x): the CDF of the range of k independent standard normals,
// k * INT phi(z) [Phi(z + x) - Phi(z)]^(k-1) dz (Simpson).
av_RngCDF(rcX, rcK)
BEGIN
  LOCAL rcJ, rcA, rcD;
  IF rcX <= 0 THEN RETURN 0; END;
  rcA := 0;
  FOR rcJ FROM 1 TO avNZ + 1 DO
    rcD := NORMALD_CDF(0, 1, avZN(rcJ) + rcX) - avZC(rcJ);
    IF rcD > 0 THEN rcA := rcA + avZW(rcJ)*rcD^(rcK - 1); END;
  END;
  RETURN rcK*rcA;
END;

// The progress page for the slow engine, composed on G1: a title, what

  av_Band("Working: Tukey's studentized range");

  av_Tx("The q distribution is an integral, done here", 12, 50, #000000h);

  av_Tx("with a table of the inner integral per k).", 12, 76, #000000h);
  av_Tx(ppE, 12, 92, avMAG);
  av_Tx("Esc stops (read every few moments).", 12, 108, avRED);
  RECT_P(G1, 50, 140, 270, 152, #000000h, #FFFFFFh);
  RECT_P(G1, 50, 156, 270, 160, avGRAY, #FFFFFFh);
  BLIT_P(G0, 0, 0, G1);
END;

// Inside a step: the thin inner bar, and a look for Esc (avStop := 1).
// A key other than Esc pressed meanwhile is simply used up here.
av_PrgTick(ptI, ptN)
BEGIN
  LOCAL ptK;
  RECT_P(G0, 51, 157, 51 + IP(218*MIN(1, ptI/ptN)), 159, avBLUE, avBLUE);
  ptK := GETKEY;

END;

// The table of P_k(x), x = 0, 0.1, ..., 12, for k = btK (avTbP, 121
// values); kept in avTbK.  Returns 0 if Esc stopped it (no table then).
av_BuildT(btK)
BEGIN
  LOCAL btI;
  IF btK == avTbK THEN RETURN 1; END;
  IF btK > 5 THEN av_BuildZ(48); ELSE av_BuildZ(32); END;
  avTbP := MAKELIST(0, btI, 1, 121);
  FOR btI FROM 2 TO 121 DO
    avTbP(btI) := av_RngCDF((btI - 1)/10, btK);
    IF btI MOD 8 == 1 THEN
      av_PrgTick(btI, 121);
      IF avStop == 1 THEN RETURN 0; END;
    END;
  END;
  avTbK := btK;
  RETURN 1;
END;

// P_k(x) read from the table by 4-point cubic (Lagrange) interpolation:
// the four grid points around x, t = x in grid steps from the first.
av_PkI(pkX)
BEGIN
  LOCAL pkI, pkT, pkA, pkB, pkC, pkD;
  IF pkX <= 0 THEN RETURN 0; END;
  IF pkX >= 12 THEN RETURN 1; END;
  pkI := MIN(MAX(IP(pkX*10) - 1, 0), 117);
  pkT := pkX*10 - pkI;
  pkA := avTbP(pkI + 1); pkB := avTbP(pkI + 2); pkC := avTbP(pkI + 3); pkD := avTbP(pkI + 4);
  pkA := -pkA*(pkT - 1)*(pkT - 2)*(pkT - 3)/6 + pkB*pkT*(pkT - 2)*(pkT - 3)/2 - pkC*pkT*(pkT - 1)*(pkT - 3)/2 + pkD*pkT*(pkT - 1)*(pkT - 2)/6;
  RETURN MIN(1, MAX(0, pkA));
END;

// F(q): the studentized range CDF for v df (the table holds k),
// F(q) = INT f_s(s) P_k(q s) ds with s = sqrt(chi-square_v / v); the
// density's constant in logs (the 1977 underflow trick).  ns = outer
// Simpson intervals (even).  avStop = 1 (Esc pressed) makes it return
// at once.
av_SRCDF(scQ, scV, scNs)
BEGIN
  LOCAL scLo, scHi, scH, scLc, scJ, scU;
  LOCAL scS, scW, scA, scSg;
  IF avStop == 1 THEN RETURN 0; END;
  IF scQ <= 0 THEN RETURN 0; END;
  IF scV > 100 THEN
    scSg := 1/SQRT(2*scV);
    scLo := MAX(1E-9, 1 - 12*scSg); scHi := 1 + 12*scSg; scH := (scHi - scLo)/scNs;
    scLc := (1 - scV/2)*LN(2) + (scV/2)*LN(scV) - av_LGam(scV/2);
    scA := 0;
    FOR scJ FROM 0 TO scNs DO
      scS := scLo + scJ*scH;
      IF scJ == 0 OR scJ == scNs THEN
        scW := 1;
      ELSE
        IF scJ MOD 2 == 1 THEN scW := 4; ELSE scW := 2; END;
      END;
      scA := scA + scW*EXP(scLc + (scV - 1)*LN(scS) - scV*scS*scS/2)*av_PkI(scQ*scS);
      IF scJ MOD 8 == 0 THEN
        av_PrgTick(scJ, scNs);
        IF avStop == 1 THEN RETURN 0; END;
      END;
    END;
    RETURN MIN(1, scH/3*scA);
  END;
  scHi := SQRT(1 + 9/SQRT(scV)); scLo := 0.001; scH := (scHi - scLo)/scNs;
  scLc := (1 - scV/2)*LN(2) + (scV/2)*LN(scV) - av_LGam(scV/2);
  scA := 0;
  FOR scJ FROM 0 TO scNs DO
    scU := scLo + scJ*scH; scS := scU*scU;
    IF scJ == 0 OR scJ == scNs THEN
      scW := 1;
    ELSE
      IF scJ MOD 2 == 1 THEN scW := 4; ELSE scW := 2; END;
    END;
    scA := scA + scW*EXP(scLc + (scV - 1)*LN(scS) - scV*scS*scS/2)*2*scU*av_PkI(scQ*scS);
    IF scJ MOD 8 == 0 THEN
      av_PrgTick(scJ, scNs);
      IF avStop == 1 THEN RETURN 0; END;
    END;
  END;
  RETURN MIN(1, scH/3*scA);
END;

// q crit: the q with F(q) = 1 - alpha, by Illinois-modified regula
// falsi (StudRange's sr_SRCrit; 6 to 23 CDF evaluations, median 13).  Each
// evaluation is one step of the bar, counted honestly.  Returns -2 if
// stopped.
av_SRCrit(crA, crV, crNs)
BEGIN
  LOCAL crT, crQl, crQh, crFl, crFh, crQm;
  LOCAL crFm, crIt, crSd, crE;
  crT := 1 - crA; crE := 0;
  crQl := 0.6; crFl := av_SRCDF(crQl, crV, crNs) - crT;
  crE := crE + 1; av_PrgStep(crE, 23, "step " + av_I(crE) + " (q crit takes 6 to 23)");
  crQh := 4; crFh := av_SRCDF(crQh, crV, crNs) - crT;
  crE := crE + 1; av_PrgStep(crE, 23, "step " + av_I(crE) + " (q crit takes 6 to 23)");
  WHILE crFh < 0 AND crQh < 60 AND avStop == 0 DO
    crQh := crQh*1.5; crFh := av_SRCDF(crQh, crV, crNs) - crT;
    crE := crE + 1; av_PrgStep(crE, 23, "step " + av_I(crE) + " (q crit takes 6 to 23)");
  END;
  crQm := (crQl + crQh)/2; crSd := 0;
  FOR crIt FROM 1 TO 40 DO
    IF avStop == 1 THEN BREAK; END;
    crQm := (crQl*crFh - crQh*crFl)/(crFh - crFl);
    crFm := av_SRCDF(crQm, crV, crNs) - crT;
    crE := crE + 1; av_PrgStep(crE, 23, "step " + av_I(crE) + " (q crit takes 6 to 23)");
    IF ABS(crFm) < 1E-7 OR ABS(crQh - crQl) < 5E-5 THEN BREAK; END;
    IF crFm*crFl > 0 THEN
      crQl := crQm; crFl := crFm;
      IF crSd == -1 THEN crFh := crFh/2; END;
      crSd := -1;
    ELSE
      crQh := crQm; crFh := crFm;
      IF crSd == 1 THEN crFl := crFl/2; END;
      crSd := 1;
    END;
  END;
  avSRE := crE;
  IF avStop == 1 THEN RETURN -2; END;
  RETURN crQm;
END;

// The G1 estimate shown before a Tukey computation, in seconds, as
// "about a to b s": table points still to build (121 x (nz + 1) inner
// steps) plus ev outer evaluations of 65 nodes (one interpolation costs
// about 3 inner steps), at 5 to 15 ms per inner step on a G1 (the second
// review's estimate from StudRange's G2 and G1 timings; to be replaced by
// Roger's stopwatch).
av_TEst(teEv)
BEGIN
  LOCAL teU;
  teU := teEv*65*3;
  IF avTbK <> avOK THEN
    IF avOK > 5 THEN teU := teU + 120*49; ELSE teU := teU + 120*33; END;
  END;
  RETURN "about " + av_I(MAX(1, ROUND(teU*0.005, 0))) + "-" + av_I(MAX(2, ROUND(teU*0.015, 0))) + " s on a G1";
END;

// The alpha-dependent constants of the one-way pages: t for each
// group's CI (pooled s), Bonferroni's t, Scheffe's multiplier.  Tukey's
// q crit is forgotten (it is recomputed when a page needs it).
av_OCrit()
BEGIN
  avTcr := av_TQ(avAlpha/2, avDf2);
  avBonT := av_TQ(avAlpha/(2*avNP), avDf2);
  avDotOK := 0;
  avScM := SQRT(avDf1*av_FQ(avAlpha, avDf1, avDf2));
  avQc := -1;
END;

// Every pair of groups i < j: the difference of means m_i - m_j, its
// SE from MS within, the raw two-sided t p (df N - k), Bonferroni
// (p x m, at most 1), Scheffe (F = t^2/(k-1) on F(k-1, N-k)), the q
// statistic for Tukey-Kramer (|t| x sqrt 2), and Holm's step-down.
// Tukey p-values are left at -1 (computed only on request).
av_PostHoc()
BEGIN
  LOCAL phI, phJ, phM, phD, phSE, phT;
  LOCAL phU, phK, phO, phMx, phR, phB;
  phK := avOK; phM := phK*(phK - 1)/2; avNP := phM;
  avPI := MAKELIST(0, phI, 1, phM); avPJ := MAKELIST(0, phI, 1, phM);
  avPD := MAKELIST(0, phI, 1, phM); avPSE := MAKELIST(0, phI, 1, phM);
  avPRw := MAKELIST(0, phI, 1, phM); avPBon := MAKELIST(0, phI, 1, phM);
  avPHm := MAKELIST(0, phI, 1, phM); avPSc := MAKELIST(0, phI, 1, phM);
  avPTk := MAKELIST(-1, phI, 1, phM); avPQ := MAKELIST(0, phI, 1, phM);
  phR := 0;
  FOR phI FROM 1 TO phK - 1 DO
    FOR phJ FROM phI + 1 TO phK DO
      phR := phR + 1;
      avPI(phR) := phI; avPJ(phR) := phJ;
      phD := avRm(phI) - avRm(phJ);
      phSE := SQRT(avMSW*(1/avRn(phI) + 1/avRn(phJ)));
      phT := phD/phSE;
      avPD(phR) := phD; avPSE(phR) := phSE;
      avPRw(phR) := av_TP2(phT, avDf2);
      avPBon(phR) := MIN(1, phM*avPRw(phR));
      avPSc(phR) := av_FUp(phT*phT/(phK - 1), phK - 1, avDf2);
      avPQ(phR) := ABS(phT)*SQRT(2);
    END;
  END;
  // Holm (1979): take the raw p-values from the smallest up; the r-th
  // smallest is multiplied by (m - r + 1), and an adjusted p never
  // falls below the one before it.
  phU := MAKELIST(0, phI, 1, phM);
  phMx := 0;
  FOR phO FROM 1 TO phM DO
    phB := 0;
    FOR phI FROM 1 TO phM DO
      IF phU(phI) == 0 THEN
        IF phB == 0 THEN
          phB := phI;
        ELSE
          IF avPRw(phI) < avPRw(phB) THEN phB := phI; END;
        END;
      END;
    END;
    phU(phB) := 1;
    phMx := MAX(phMx, MIN(1, (phM - phO + 1)*avPRw(phB)));
    avPHm(phB) := phMx;
  END;
  avPTop := 1; avPIx := 0; avSelA := 0; avSelB := 0;
  av_OCrit();
END;

// The whole one-way analysis.  Returns 1, or 0 (with a message) when
// it cannot be done.
//   SS between = sum n_i (mean_i - grand mean)^2
//   SS within  = sum over values (y - its group mean)^2
//   F = (SSB/(k-1)) / (SSW/(N-k)),  p = P(F(k-1, N-k) > F)
//   eta^2 = SSB/SST;  omega^2 = (SSB - (k-1) MSW)/(SST + MSW) (Hays)
//   Levene (Brown-Forsythe): the one-way F of z = |y - group median|
//   Bartlett: [(N-k) ln s_p^2 - sum (n_i-1) ln s_i^2] / C,
//     C = 1 + (sum 1/(n_i-1) - 1/(N-k)) / (3(k-1)), on chi-square(k-1)
//   Welch: w_i = n_i/s_i^2, the weighted mean m_w,
//     A = sum w_i (m_i - m_w)^2/(k-1),
//     L = sum (1 - w_i/W)^2/(n_i - 1),  B = 1 + 2(k-2) L/(k^2-1),
//     F = A/B on (k-1, (k^2-1)/(3L)) df
av_OCompute()
BEGIN
  LOCAL ocI, ocG, ocV, ocK, ocN, ocS;
  LOCAL ocL, ocZ, ocZm, ocZb, ocA, ocB;
  LOCAL ocW, ocWs, ocMw, ocLam, ocC, ocMn;
  LOCAL ocMx;
  ocK := avOK; ocN := SIZE(avOY);
  IF ocK < 2 THEN MSGBOX("ANOVA needs at least two groups."); RETURN 0; END;
  avRn := MAKELIST(0, ocI, 1, ocK); avRm := MAKELIST(0, ocI, 1, ocK);
  avRv := MAKELIST(0, ocI, 1, ocK); avRmd := MAKELIST(0, ocI, 1, ocK);
  ocS := 0;
  FOR ocI FROM 1 TO ocN DO
    ocG := avOG(ocI);
    avRn(ocG) := avRn(ocG) + 1;
    avRm(ocG) := avRm(ocG) + avOY(ocI);
    ocS := ocS + avOY(ocI);
  END;
  FOR ocG FROM 1 TO ocK DO
    IF avRn(ocG) == 0 THEN MSGBOX("Group " + avOSh(ocG) + " has no values."); RETURN 0; END;
  END;
  IF ocN - ocK < 1 THEN
    MSGBOX("ANOVA needs more values than groups (N - k " + avGE + " 1).");
    RETURN 0;
  END;
  FOR ocG FROM 1 TO ocK DO avRm(ocG) := avRm(ocG)/avRn(ocG); END;
  avNN := ocN; avGM := ocS/ocN;
  // the second pass: deviations from the group means and the grand mean
  avSST := 0;
  FOR ocI FROM 1 TO ocN DO
    ocG := avOG(ocI);
    ocV := avOY(ocI) - avRm(ocG);
    avRv(ocG) := avRv(ocG) + ocV*ocV;
    avSST := avSST + (avOY(ocI) - avGM)^2;
  END;
  avSSW := 0; avSSB := 0;
  FOR ocG FROM 1 TO ocK DO
    avSSW := avSSW + avRv(ocG);
    avSSB := avSSB + avRn(ocG)*(avRm(ocG) - avGM)^2;
    IF avRn(ocG) > 1 THEN avRv(ocG) := avRv(ocG)/(avRn(ocG) - 1); ELSE avRv(ocG) := 0; END;
  END;
  IF avSSW <= 0 THEN
    MSGBOX("Every value equals its group's mean: no variation within groups, so F cannot be computed.");
    RETURN 0;
  END;
  avDf1 := ocK - 1; avDf2 := ocN - ocK;
  avMSB := avSSB/avDf1; avMSW := avSSW/avDf2;
  avFF := avMSB/avMSW;
  avPP := av_FUp(avFF, avDf1, avDf2);
  avEta := 0;
  IF avSST > 0 THEN avEta := avSSB/avSST; END;
  avOmg := (avSSB - avDf1*avMSW)/(avSST + avMSW);
  // ---- medians (for Levene / Brown-Forsythe) ----
  // beta 1a (second review, item 6): one pass puts every value in its
  // group's block of a pre-sized list (no growing CONCAT), then each
  // block is copied out once and sorted.  Above 500 values a Working
  // page shows a bar per group, and Esc stops.
  avStop := 0;
  IF ocN > 500 THEN av_Busy("one-way ANOVA", ocN); END;
  ocZm := MAKELIST(0, ocI, 1, ocK);
  ocC := 0;
  FOR ocG FROM 1 TO ocK DO ocZm(ocG) := ocC; ocC := ocC + avRn(ocG); END;
  ocL := MAKELIST(0, ocI, 1, ocN);
  FOR ocI FROM 1 TO ocN DO
    ocG := avOG(ocI);
    ocZm(ocG) := ocZm(ocG) + 1;
    ocL(ocZm(ocG)) := avOY(ocI);
  END;
  ocC := 0;
  FOR ocG FROM 1 TO ocK DO
    ocZ := MAKELIST(0, ocI, 1, avRn(ocG));
    FOR ocI FROM 1 TO avRn(ocG) DO ocZ(ocI) := ocL(ocC + ocI); END;
    ocC := ocC + avRn(ocG);
    avRmd(ocG) := av_Med(ocZ);
    IF ocN > 500 THEN
      av_BusyStep(ocG, ocK);
      IF avStop == 1 THEN MSGBOX("Stopped (Esc): nothing was computed."); RETURN 0; END;
    END;
  END;
  // ---- Levene's test on |y - median| (Brown & Forsythe 1974) ----
  ocZm := MAKELIST(0, ocI, 1, ocK); ocZb := 0;
  FOR ocI FROM 1 TO ocN DO
    ocG := avOG(ocI); ocZ := ABS(avOY(ocI) - avRmd(ocG));
    ocZm(ocG) := ocZm(ocG) + ocZ; ocZb := ocZb + ocZ;
  END;
  FOR ocG FROM 1 TO ocK DO ocZm(ocG) := ocZm(ocG)/avRn(ocG); END;
  ocZb := ocZb/ocN;
  ocA := 0; ocB := 0;
  FOR ocG FROM 1 TO ocK DO ocA := ocA + avRn(ocG)*(ocZm(ocG) - ocZb)^2; END;
  FOR ocI FROM 1 TO ocN DO
    ocG := avOG(ocI); ocZ := ABS(avOY(ocI) - avRmd(ocG));
    ocB := ocB + (ocZ - ocZm(ocG))^2;
  END;
  avLvOK := 0;
  IF ocB > 0 THEN
    avLvF := (ocA/avDf1)/(ocB/avDf2); avLvP := av_FUp(avLvF, avDf1, avDf2); avLvOK := 1;
  END;
  // ---- Bartlett and Welch need n >= 2 and s > 0 in every group ----
  avBrOK := 1;
  FOR ocG FROM 1 TO ocK DO
    IF avRn(ocG) < 2 THEN avBrOK := 0; END;
    IF avRv(ocG) <= 0 THEN avBrOK := 0; END;
  END;
  avWOK := avBrOK;
  IF avBrOK == 1 THEN
    ocA := avDf2*LN(avMSW); ocC := 0;
    FOR ocG FROM 1 TO ocK DO
      ocA := ocA - (avRn(ocG) - 1)*LN(avRv(ocG));
      ocC := ocC + 1/(avRn(ocG) - 1);
    END;
    ocC := 1 + (ocC - 1/avDf2)/(3*avDf1);
    avBrT := ocA/ocC; avBrP := av_ChiUp(avBrT, avDf1);
    // Welch
    ocWs := 0; ocMw := 0;
    FOR ocG FROM 1 TO ocK DO
      ocW := avRn(ocG)/avRv(ocG); ocWs := ocWs + ocW; ocMw := ocMw + ocW*avRm(ocG);
    END;
    ocMw := ocMw/ocWs;
    ocA := 0; ocLam := 0;
    FOR ocG FROM 1 TO ocK DO
      ocW := avRn(ocG)/avRv(ocG);
      ocA := ocA + ocW*(avRm(ocG) - ocMw)^2;
      ocLam := ocLam + (1 - ocW/ocWs)^2/(avRn(ocG) - 1);
    END;
    ocA := ocA/avDf1;
    ocB := 1 + 2*(ocK - 2)/(ocK^2 - 1)*ocLam;
    avWF := ocA/ocB; avWd2 := (ocK^2 - 1)/(3*ocLam);
    avWP := av_FUp(avWF, avDf1, avWd2);
  END;
  // ---- largest s over smallest s (groups with n >= 2) ----
  ocMn := -1; ocMx := -1;
  FOR ocG FROM 1 TO ocK DO
    IF avRn(ocG) > 1 THEN
      ocV := SQRT(avRv(ocG));
      IF ocMn < 0 OR ocV < ocMn THEN ocMn := ocV; END;
      IF ocV > ocMx THEN ocMx := ocV; END;
    END;
  END;
  avSDr := -1;
  IF ocMn > 0 THEN avSDr := ocMx/ocMn; END;
  av_PostHoc();
  // ---- Home ----
  AV_F := avFF; AV_P := avPP; AV_DF := {avDf1, avDf2}; AV_MSE := avMSW; AV_ETA2 := avEta;
  AV_TAB := MAKEMAT(0, 3, 5);
  AV_TAB(1, 1) := avSSB; AV_TAB(1, 2) := avDf1; AV_TAB(1, 3) := avMSB; AV_TAB(1, 4) := avFF; AV_TAB(1, 5) := avPP;
  AV_TAB(2, 1) := avSSW; AV_TAB(2, 2) := avDf2; AV_TAB(2, 3) := avMSW;
  AV_TAB(3, 1) := avSST; AV_TAB(3, 2) := ocN - 1;
  avLast := "one-way F(" + av_I(avDf1) + ", " + av_I(avDf2) + ") = " + av_Fx(avFF, 3) + ", " + av_PE(avPP) + " (" + avOTtl + ")";
  avODone := 1;
  RETURN 1;
END;

// Tukey's q crit for the current one-way data, with the progress page.
// avQc: the value; -1 not asked yet; -2 stopped by Esc (or "not now");
// -3 df below 3 (the PPL engine starts at df 3).  It runs only when the
// student has picked Tukey and said yes to the estimate (av_TukeyAsk).
av_TukeyCrit()
BEGIN
  LOCAL tkT;
  IF avDf2 < 3 THEN avQc := -3; RETURN 0; END;
  av_PrgPage("q crit for k = " + av_I(avOK) + ", df = " + av_I(avDf2) + ", " + avAl + " = " + av_AS(), "Estimate: " + av_TEst(14) + ".");
  avStop := 0;
  IF av_BuildT(avOK) == 0 THEN
    avQc := -2;
  ELSE
    tkT := av_SRCrit(avAlpha, avDf2, 64);
    IF avStop == 1 THEN avQc := -2; ELSE avQc := tkT; END;
  END;
  av_KDrain();
  RETURN 0;
END;

// Before any Tukey computation: a one-line warning with the estimate,
// as a CHOOSE.  Returns 1 = go ahead, 0 = not now.  teEv: how many
// outer evaluations the job needs (q crit about 14; one per pair).
av_TukeyAsk(taT, taEv)
BEGIN
  LOCAL taI;
  taI := 0;
  CHOOSE(taI, taT + ": " + av_TEst(taEv), {"Compute it now", "Not now"});
  IF taI == 1 THEN RETURN 1; END;
  RETURN 0;
END;

// Tukey-Kramer p for every pair not yet done, P(Q > q) = 1 - F(q): one
// CDF per pair (a bar step per pair; Esc stops and keeps the pairs done
// so far).  Asks first, with the estimate.
av_TukeyP()
BEGIN
  LOCAL tpR, tpF, tpN;
  IF avDf2 < 3 THEN
    MSGBOX("Tukey needs df within " + avGE + " 3 here (the PPL engine). Use Bonferroni, Holm or Scheffe.");
    RETURN 0;
  END;
  tpN := 0;
  FOR tpR FROM 1 TO avNP DO
    IF avPTk(tpR) < 0 THEN tpN := tpN + 1; END;
  END;
  IF avQc < 0 THEN tpN := tpN + 14; END;
  IF tpN == 0 THEN avMeth := 1; RETURN 0; END;
  IF av_TukeyAsk("Tukey p", tpN) == 0 THEN RETURN 0; END;
  avMeth := 1;
  IF avQc < 0 THEN av_TukeyCrit(); END;
  IF avQc < 0 THEN RETURN 0; END;
  av_PrgPage("Tukey p for " + av_I(avNP) + " pairs (one step each)", "Estimate: " + av_TEst(tpN) + ".");
  avStop := 0;
  IF av_BuildT(avOK) == 0 THEN RETURN 0; END;
  FOR tpR FROM 1 TO avNP DO
    IF avStop == 0 AND avPTk(tpR) < 0 THEN
      tpF := av_SRCDF(avPQ(tpR), avDf2, 64);
      IF avStop == 0 THEN avPTk(tpR) := MAX(0, 1 - tpF); END;
    END;
    IF avStop == 0 THEN av_PrgStep(tpR, avNP, "pair " + av_I(tpR) + " of " + av_I(avNP)); END;
  END;
  av_KDrain();
  RETURN 0;
END;

// Tukey-Kramer p for ONE pair (the Dots page's pair): one CDF.
av_TukeyP1(t1R)
BEGIN
  LOCAL t1F;
  IF t1R < 1 OR t1R > avNP THEN MSGBOX("Pick a pair first (tap two groups)."); RETURN 0; END;
  IF avDf2 < 3 THEN
    MSGBOX("Tukey needs df within " + avGE + " 3 here (the PPL engine).");
    RETURN 0;
  END;
  IF avPTk(t1R) >= 0 THEN RETURN 0; END;
  av_PrgPage("Tukey p for " + avOSh(avPI(t1R)) + " - " + avOSh(avPJ(t1R)), "Estimate: " + av_TEst(1) + ".");
  avStop := 0;
  IF av_BuildT(avOK) == 0 THEN RETURN 0; END;
  t1F := av_SRCDF(avPQ(t1R), avDf2, 64);
  IF avStop == 0 THEN avPTk(t1R) := MAX(0, 1 - t1F); END;
  av_KDrain();
  RETURN 0;
END;

// The inverse of an n x n matrix by Gauss-Jordan elimination with
// partial pivoting.  avSing := 1 if a pivot is (nearly) zero.
av_Inv(ivM, ivN)
BEGIN
  LOCAL ivA, ivR, ivI, ivJ, ivK, ivP;
  LOCAL ivMx, ivT, ivF, ivSc;
  ivA := ivM;
  ivR := MAKEMAT(0, ivN, ivN);
  ivSc := 0;
  FOR ivI FROM 1 TO ivN DO
    ivR(ivI, ivI) := 1;
    ivSc := MAX(ivSc, ABS(ivA(ivI, ivI)));
  END;
  IF ivSc <= 0 THEN avSing := 1; RETURN ivR; END;
  FOR ivK FROM 1 TO ivN DO
    ivP := ivK; ivMx := ABS(ivA(ivK, ivK));
    FOR ivI FROM ivK + 1 TO ivN DO
      IF ABS(ivA(ivI, ivK)) > ivMx THEN ivMx := ABS(ivA(ivI, ivK)); ivP := ivI; END;
    END;
    IF ivMx < ivSc*1E-10 THEN avSing := 1; RETURN ivR; END;
    IF ivP <> ivK THEN
      FOR ivJ FROM 1 TO ivN DO
        ivT := ivA(ivK, ivJ); ivA(ivK, ivJ) := ivA(ivP, ivJ); ivA(ivP, ivJ) := ivT;
        ivT := ivR(ivK, ivJ); ivR(ivK, ivJ) := ivR(ivP, ivJ); ivR(ivP, ivJ) := ivT;
      END;
    END;
    ivT := ivA(ivK, ivK);
    FOR ivJ FROM 1 TO ivN DO
      ivA(ivK, ivJ) := ivA(ivK, ivJ)/ivT; ivR(ivK, ivJ) := ivR(ivK, ivJ)/ivT;
    END;
    FOR ivI FROM 1 TO ivN DO
      IF ivI <> ivK THEN
        ivF := ivA(ivI, ivK);
        IF ivF <> 0 THEN
          FOR ivJ FROM 1 TO ivN DO
            ivA(ivI, ivJ) := ivA(ivI, ivJ) - ivF*ivA(ivK, ivJ);
            ivR(ivI, ivJ) := ivR(ivI, ivJ) - ivF*ivR(ivK, ivJ);
          END;
        END;
      END;
    END;
  END;
  RETURN ivR;
END;

// The effect-coded row of the two-way design for cell (i, j): 1, then
// a-1 columns for A, b-1 for B, (a-1)(b-1) for A x B.  Effect (sum-to-
// zero) coding: level l of A gives 1 in its column, the LAST level -1
// in every A column, other levels 0 (the same for B); A x B columns are
// the products.
av_TRow(trI, trJ, trA, trB)
BEGIN
  LOCAL trR, trL, trM, trEA, trEB, trK;
  trR := MAKELIST(0, trK, 1, trA*trB);
  trR(1) := 1;
  trEA := MAKELIST(0, trK, 1, trA - 1);
  trEB := MAKELIST(0, trK, 1, trB - 1);
  FOR trL FROM 1 TO trA - 1 DO
    IF trI == trL THEN trEA(trL) := 1; END;
    IF trI == trA THEN trEA(trL) := -1; END;
    trR(1 + trL) := trEA(trL);
  END;
  FOR trM FROM 1 TO trB - 1 DO
    IF trJ == trM THEN trEB(trM) := 1; END;
    IF trJ == trB THEN trEB(trM) := -1; END;
    trR(trA + trM) := trEB(trM);
  END;
  FOR trL FROM 1 TO trA - 1 DO
    FOR trM FROM 1 TO trB - 1 DO
      trR(trA + trB - 1 + (trL - 1)*(trB - 1) + trM) := trEA(trL)*trEB(trM);
    END;
  END;
  RETURN trR;
END;

// The two-way analysis.  Returns 1, or 0 with a message.
//   Cells: n, mean, s.  SSE = sum (y - cell mean)^2 (the full model
//   fits every cell mean), df N - ab.
//   Type III: b = (X'X)^-1 X'y with the effect-coded X; for an effect E
//   with columns E, SS_E = b_E' [ ((X'X)^-1)_EE ]^-1 b_E (a Wald SS;
//   statsmodels anova_lm(typ=3) with Sum contrasts gives the same).
//   X'X and X'y are built cell by cell (n_c r r', n_c (mean_c - M) r).
//   Balanced (every n_c = n): the classical SS as well,
//     SSA = b n sum (A_i - M)^2, SSB = a n sum (B_j - M)^2,
//     SSAB = n sum (m_ij - A_i - B_j + M)^2,
//   and the largest gap between the two, as a share of SST, in avTAgr.
//   Margins A_i, B_j are means of the CELL means (unweighted): that is
//   what Type III compares.
av_TCompute()
BEGIN
  LOCAL tcA, tcB, tcC, tcI, tcJ, tcN;
  LOCAL tcS, tcX, tcY, tcR, tcK, tcL;
  LOCAL tcM, tcCi, tcBv, tcE, tcQ, tcSub;
  LOCAL tcT, tcSS, tcU, tcV, tcLo, tcN0;
  LOCAL tcSE;
  tcA := avTNA; tcB := avTNB; tcC := tcA*tcB; tcN := SIZE(avTY);
  IF tcC > 12 THEN
    av_Band("Working: Type III sums of squares");
    av_Tx("Solving a " + av_I(tcC) + " " + avTm + " " + av_I(tcC) + " system (effect coding);", 12, 40, avBLUE);
    av_Tx("a G1 may take several seconds.", 12, 53, avBLUE);
    BLIT_P(G0, 0, 0, G1);
  END;
  avTCN := MAKELIST(0, tcI, 1, tcC); avTCM := MAKELIST(0, tcI, 1, tcC);
  avTCS := MAKELIST(0, tcI, 1, tcC);
  tcS := 0;
  FOR tcI FROM 1 TO tcN DO
    tcK := (avTA(tcI) - 1)*tcB + avTB(tcI);
    avTCN(tcK) := avTCN(tcK) + 1;
    avTCM(tcK) := avTCM(tcK) + avTY(tcI);
    tcS := tcS + avTY(tcI);
  END;
  FOR tcK FROM 1 TO tcC DO
    IF avTCN(tcK) == 0 THEN
      MSGBOX("Cell " + avTAN(IP((tcK - 1)/tcB) + 1) + ", " + avTBN(((tcK - 1) MOD tcB) + 1) + " is empty: every cell needs a value.");
      RETURN 0;
    END;
  END;
  IF tcN - tcC < 1 THEN MSGBOX("Two-way ANOVA needs more values than cells."); RETURN 0; END;
  FOR tcK FROM 1 TO tcC DO avTCM(tcK) := avTCM(tcK)/avTCN(tcK); END;
  avTNN := tcN; avTGM := tcS/tcN;
  tcT := 0; tcSS := 0;
  FOR tcI FROM 1 TO tcN DO
    tcK := (avTA(tcI) - 1)*tcB + avTB(tcI);
    avTCS(tcK) := avTCS(tcK) + (avTY(tcI) - avTCM(tcK))^2;
    tcT := tcT + (avTY(tcI) - avTGM)^2;
  END;
  FOR tcK FROM 1 TO tcC DO
    tcSS := tcSS + avTCS(tcK);
    IF avTCN(tcK) > 1 THEN avTCS(tcK) := SQRT(avTCS(tcK)/(avTCN(tcK) - 1)); ELSE avTCS(tcK) := 0; END;
  END;
  IF tcSS <= 0 THEN MSGBOX("No variation within cells: F cannot be computed."); RETURN 0; END;
  // ---- X'X and X'y, cell by cell (y centred on the grand mean) ----
  tcX := MAKEMAT(0, tcC, tcC); tcY := MAKELIST(0, tcI, 1, tcC);
  FOR tcI FROM 1 TO tcA DO
    FOR tcJ FROM 1 TO tcB DO
      tcR := av_TRow(tcI, tcJ, tcA, tcB);
      tcK := (tcI - 1)*tcB + tcJ;
      FOR tcL FROM 1 TO tcC DO
        IF tcR(tcL) <> 0 THEN
          tcY(tcL) := tcY(tcL) + tcR(tcL)*avTCN(tcK)*(avTCM(tcK) - avTGM);
          FOR tcM FROM 1 TO tcC DO
            tcX(tcL, tcM) := tcX(tcL, tcM) + avTCN(tcK)*tcR(tcL)*tcR(tcM);
          END;
        END;
      END;
    END;
  END;
  avSing := 0;
  tcCi := av_Inv(tcX, tcC);
  IF avSing == 1 THEN MSGBOX("The design cannot be solved (singular X'X)."); RETURN 0; END;
  tcBv := MAKELIST(0, tcI, 1, tcC);
  FOR tcL FROM 1 TO tcC DO
    FOR tcM FROM 1 TO tcC DO tcBv(tcL) := tcBv(tcL) + tcCi(tcL, tcM)*tcY(tcM); END;
  END;
  // ---- Type III SS for A (columns 2..a), B (a+1..a+b-1), AB (a+b..ab) ----
  avTSS := {0, 0, 0, tcSS, tcT};
  FOR tcE FROM 1 TO 3 DO
    IF tcE == 1 THEN tcLo := 2; tcQ := tcA - 1; END;
    IF tcE == 2 THEN tcLo := tcA + 1; tcQ := tcB - 1; END;
    IF tcE == 3 THEN tcLo := tcA + tcB; tcQ := (tcA - 1)*(tcB - 1); END;
    tcSub := MAKEMAT(0, tcQ, tcQ);
    FOR tcU FROM 1 TO tcQ DO
      FOR tcV FROM 1 TO tcQ DO tcSub(tcU, tcV) := tcCi(tcLo + tcU - 1, tcLo + tcV - 1); END;
    END;
    tcSub := av_Inv(tcSub, tcQ);
    IF avSing == 1 THEN MSGBOX("The design cannot be solved (singular block)."); RETURN 0; END;
    tcSE := 0;
    FOR tcU FROM 1 TO tcQ DO
      FOR tcV FROM 1 TO tcQ DO
        tcSE := tcSE + tcBv(tcLo + tcU - 1)*tcSub(tcU, tcV)*tcBv(tcLo + tcV - 1);
      END;
    END;
    avTSS(tcE) := tcSE;
  END;
  avTDF := {tcA - 1, tcB - 1, (tcA - 1)*(tcB - 1), tcN - tcC, tcN - 1};
  avTMS := MAKELIST(0, tcI, 1, 4);
  FOR tcE FROM 1 TO 4 DO avTMS(tcE) := avTSS(tcE)/avTDF(tcE); END;
  avTF := {0, 0, 0}; avTP := {1, 1, 1}; avTPE := {0, 0, 0};
  FOR tcE FROM 1 TO 3 DO
    avTF(tcE) := avTMS(tcE)/avTMS(4);
    avTP(tcE) := av_FUp(avTF(tcE), avTDF(tcE), avTDF(4));
    avTPE(tcE) := avTSS(tcE)/(avTSS(tcE) + tcSS);
  END;
  // ---- unweighted margins: means of the cell means ----
  avTMA := MAKELIST(0, tcI, 1, tcA); avTMB := MAKELIST(0, tcI, 1, tcB);
  FOR tcI FROM 1 TO tcA DO
    FOR tcJ FROM 1 TO tcB DO
      tcK := (tcI - 1)*tcB + tcJ;
      avTMA(tcI) := avTMA(tcI) + avTCM(tcK)/tcB;
      avTMB(tcJ) := avTMB(tcJ) + avTCM(tcK)/tcA;
    END;
  END;
  // ---- balanced?  then the classical SS too, and the check ----
  avTBal := 1;
  FOR tcK FROM 1 TO tcC DO
    IF avTCN(tcK) <> avTCN(1) THEN avTBal := 0; END;
  END;
  avTCl := {0, 0, 0}; avTAgr := 0;
  IF avTBal == 1 THEN
    tcN0 := avTCN(1);
    FOR tcI FROM 1 TO tcA DO avTCl(1) := avTCl(1) + tcB*tcN0*(avTMA(tcI) - avTGM)^2; END;
    FOR tcJ FROM 1 TO tcB DO avTCl(2) := avTCl(2) + tcA*tcN0*(avTMB(tcJ) - avTGM)^2; END;
    FOR tcI FROM 1 TO tcA DO
      FOR tcJ FROM 1 TO tcB DO
        tcK := (tcI - 1)*tcB + tcJ;
        avTCl(3) := avTCl(3) + tcN0*(avTCM(tcK) - avTMA(tcI) - avTMB(tcJ) + avTGM)^2;
      END;
    END;
    FOR tcE FROM 1 TO 3 DO avTAgr := MAX(avTAgr, ABS(avTCl(tcE) - avTSS(tcE))/tcT); END;
  END;
  AV_F := avTF(3); AV_P := avTP(3); AV_DF := {avTDF(3), avTDF(4)}; AV_MSE := avTMS(4); AV_ETA2 := avTPE(3);
  AV_TAB := MAKEMAT(0, 5, 5);
  FOR tcE FROM 1 TO 5 DO
    AV_TAB(tcE, 1) := avTSS(tcE); AV_TAB(tcE, 2) := avTDF(tcE);
    IF tcE <= 4 THEN AV_TAB(tcE, 3) := avTMS(tcE); END;
    IF tcE <= 3 THEN AV_TAB(tcE, 4) := avTF(tcE); AV_TAB(tcE, 5) := avTP(tcE); END;
  END;
  avLast := "two-way (Type III) A" + avTm + "B F(" + av_I(avTDF(3)) + ", " + av_I(avTDF(4)) + ") = " + av_Fx(avTF(3), 3) + ", " + av_PE(avTP(3));
  avTDone := 1;
  RETURN 1;
END;

// The randomized block analysis (one value per block x treatment).
//   SSTr = nb sum (T_j - M)^2, SSBl = kt sum (B_i - M)^2,
//   SSE = sum (y_ij - T_j - B_i + M)^2, df (kt-1)(nb-1)
//   F for treatments = MSTr/MSE;  partial eta^2 = SSTr/(SSTr + SSE)
//   for comparison, the one-way F that ignores the blocks:
//   (SSTr/(kt-1)) / ((SST - SSTr)/(N - kt))
av_BCompute()
BEGIN
  LOCAL bcI, bcJ, bcV, bcN, bcW;
  bcN := avBNb*avBKt;
  IF avBNb < 2 OR avBKt < 2 THEN MSGBOX("A block design needs 2 or more blocks and treatments."); RETURN 0; END;
  avBTM := MAKELIST(0, bcI, 1, avBKt); avBSD := MAKELIST(0, bcI, 1, avBKt);
  avBBM := MAKELIST(0, bcI, 1, avBNb);
  avBGM := 0;
  FOR bcI FROM 1 TO avBNb DO
    FOR bcJ FROM 1 TO avBKt DO
      bcV := avBY((bcI - 1)*avBKt + bcJ);
      avBTM(bcJ) := avBTM(bcJ) + bcV/avBNb;
      avBBM(bcI) := avBBM(bcI) + bcV/avBKt;
      avBGM := avBGM + bcV/bcN;
    END;
  END;
  avBSS := {0, 0, 0, 0};
  FOR bcI FROM 1 TO avBNb DO
    FOR bcJ FROM 1 TO avBKt DO
      bcV := avBY((bcI - 1)*avBKt + bcJ);
      avBSS(3) := avBSS(3) + (bcV - avBTM(bcJ) - avBBM(bcI) + avBGM)^2;
      avBSS(4) := avBSS(4) + (bcV - avBGM)^2;
      avBSD(bcJ) := avBSD(bcJ) + (bcV - avBTM(bcJ))^2;
    END;
  END;
  FOR bcJ FROM 1 TO avBKt DO
    avBSS(1) := avBSS(1) + avBNb*(avBTM(bcJ) - avBGM)^2;
    avBSD(bcJ) := SQRT(avBSD(bcJ)/(avBNb - 1));
  END;
  FOR bcI FROM 1 TO avBNb DO avBSS(2) := avBSS(2) + avBKt*(avBBM(bcI) - avBGM)^2; END;
  IF avBSS(3) <= 0 THEN MSGBOX("No variation left after blocks and treatments: F cannot be computed."); RETURN 0; END;
  avBDF := {avBKt - 1, avBNb - 1, (avBKt - 1)*(avBNb - 1), bcN - 1};
  avBMS := {avBSS(1)/avBDF(1), avBSS(2)/avBDF(2), avBSS(3)/avBDF(3)};
  avBF := {avBMS(1)/avBMS(3), avBMS(2)/avBMS(3)};
  avBP := {av_FUp(avBF(1), avBDF(1), avBDF(3)), av_FUp(avBF(2), avBDF(2), avBDF(3))};
  avBPE := {avBSS(1)/(avBSS(1) + avBSS(3)), avBSS(2)/(avBSS(2) + avBSS(3))};
  bcW := avBSS(4) - avBSS(1);
  avBOF := -1; avBOP := 1;
  IF bcW > 0 THEN
    avBOF := (avBSS(1)/avBDF(1))/(bcW/(bcN - avBKt));
    avBOP := av_FUp(avBOF, avBDF(1), bcN - avBKt);
  END;
  AV_F := avBF(1); AV_P := avBP(1); AV_DF := {avBDF(1), avBDF(3)}; AV_MSE := avBMS(3); AV_ETA2 := avBPE(1);
  AV_TAB := MAKEMAT(0, 4, 5);
  FOR bcI FROM 1 TO 4 DO
    AV_TAB(bcI, 1) := avBSS(bcI); AV_TAB(bcI, 2) := avBDF(bcI);
    IF bcI <= 3 THEN AV_TAB(bcI, 3) := avBMS(bcI); END;
    IF bcI <= 2 THEN AV_TAB(bcI, 4) := avBF(bcI); AV_TAB(bcI, 5) := avBP(bcI); END;
  END;
  avLast := "blocks: treatments F(" + av_I(avBDF(1)) + ", " + av_I(avBDF(3)) + ") = " + av_Fx(avBF(1), 3) + ", " + av_PE(avBP(1));
  avBDone := 1;
  RETURN 1;
END;

// The chi-square right-tail quantile: P(chi-square(df) > x) = p.  For
// p of 1E-6 or more, CHISQUARE_ICDF(df, 1 - p) (1 - p keeps its digits);
// below that (beta 1a, second review item 10.3) the Lessons' rule: never
// ICDF(1 - p) -- bisect on the logarithm of the upper tail instead.
av_ChiHi(chP, chD)
BEGIN
  LOCAL chLo, chHi, chM, chI;
  IF chP >= 0.000001 THEN RETURN CHISQUARE_ICDF(chD, 1 - chP); END;
  chLo := CHISQUARE_ICDF(chD, 0.999999);
  chHi := chLo*2 + 10;
  WHILE av_ChiUp(chHi, chD) > chP DO chHi := chHi*2; END;
  FOR chI FROM 1 TO 60 DO
    chM := (chLo + chHi)/2;
    IF LN(av_ChiUp(chM, chD)) > LN(chP) THEN chLo := chM; ELSE chHi := chM; END;
  END;
  RETURN (chLo + chHi)/2;
END;

// A lower quantile of the variance page's curve: area pp to the LEFT.
av_VLoQ(vlP)
BEGIN
  IF avVMode == 1 THEN RETURN CHISQUARE_ICDF(avVDf1, av_Clp(vlP)); END;
  RETURN FISHER_ICDF(avVDf1, avVDf2, av_Clp(vlP));
END;

// An upper quantile: area pp to the RIGHT (F by its mirror; the
// chi-square has none, so its probability is clamped first).
av_VHiQ(vhP)
BEGIN
  IF avVMode == 1 THEN RETURN av_ChiHi(av_Clp(vhP), avVDf1); END;
  RETURN av_FQ(vhP, avVDf1, avVDf2);
END;

// The variance test and interval (Triola's S2TEST / S2INT reworked).
//   One sigma: chi^2 = (n-1) s^2 / sigma0^2 on n-1 df;
//     CI for sigma^2: ((n-1)s^2/chi^2_upper, (n-1)s^2/chi^2_lower)
//   Two sigmas: F = s1^2/s2^2 on (n1-1, n2-1) df;
//     CI for sigma1^2/sigma2^2: (F x F_(c/2)(df2, df1), F / F_(c/2)(df1, df2))
//     where F_(c/2)(a, b) is the LOWER c/2 quantile of F(a, b)
//     (FINTRVAL swapped these two)
//   p: two-sided = 2 min(lower tail, upper tail) (STDTEST's rule), or the
//   tail the alternative names.  The two-sided picture shades the
//   observed tail and the other tail with the SAME area (STDTEST).
av_VCompute()
BEGIN
  LOCAL vcC;
  vcC := 1 - avVCL;
  IF avVMode == 1 THEN
    avVDf1 := avVN1 - 1; avVDf2 := 0;
    avVStat := avVDf1*avVS1^2/avVSg0^2;
    avVPL := av_ChiLo(avVStat, avVDf1); avVPU := av_ChiUp(avVStat, avVDf1);
    avVLo := avVDf1*avVS1^2/av_VHiQ(vcC/2);
    avVHi := avVDf1*avVS1^2/av_VLoQ(vcC/2);
  ELSE
    avVDf1 := avVN1 - 1; avVDf2 := avVN2 - 1;
    avVStat := avVS1^2/avVS2^2;
    avVPL := av_FLo(avVStat, avVDf1, avVDf2); avVPU := av_FUp(avVStat, avVDf1, avVDf2);
    avVLo := avVStat*FISHER_ICDF(avVDf2, avVDf1, av_Clp(vcC/2));
    avVHi := avVStat/FISHER_ICDF(avVDf1, avVDf2, av_Clp(vcC/2));
  END;
  avVCrL := -1; avVCrH := -1; avVRa := -1; avVRb := -1;
  IF avVAlt == 1 THEN
    avVP := MIN(1, 2*MIN(avVPL, avVPU));
    avVCrL := av_VLoQ(avAlpha/2); avVCrH := av_VHiQ(avAlpha/2);
    IF avVPU <= avVPL THEN
      avVRb := avVStat; avVRa := av_VLoQ(avVPU);
    ELSE
      avVRa := avVStat; avVRb := av_VHiQ(avVPL);
    END;
  END;
  IF avVAlt == 2 THEN avVP := avVPL; avVCrL := av_VLoQ(avAlpha); avVRa := avVStat; END;
  IF avVAlt == 3 THEN avVP := avVPU; avVCrH := av_VHiQ(avAlpha); avVRb := avVStat; END;
  AV_F := avVStat; AV_P := avVP; AV_DF := {avVDf1, avVDf2};
  IF avVMode == 1 THEN
    avLast := "one " + avSg + ": " + avX2 + "(" + av_I(avVDf1) + ") = " + av_Fx(avVStat, 3) + ", " + av_PE(avVP);
  ELSE
    avLast := "two " + avSg + ": F(" + av_I(avVDf1) + ", " + av_I(avVDf2) + ") = " + av_Fx(avVStat, 3) + ", " + av_PE(avVP);
  END;
  avVHas := 1;
END;

// The window and the curve, cached: the density once per pixel column
// (Lessons Learned 6), as pixel rows in avCvP.  cfT 1 = F(cfA, cfB),
// 2 = chi-square(cfA).  The height is fitted to the curve away from 0
// (an F with 1 df, or a chi-square with 1 df, rises without limit
// there, so its first columns are simply clipped).
av_CvFit(cfT, cfA, cfB, cfX)
BEGIN
  LOCAL cfI, cfV, cfD, cfMx, cfW, cfL;
  avCvX := cfX;
  avCvS := av_Nice(cfX/5);
  avCvDc := MAX(0, -FLOOR(LOG(avCvS)));
  cfW := avPR - avPL;
  cfL := MAKELIST(0, cfI, 1, cfW);
  cfMx := 0;
  FOR cfI FROM 1 TO cfW DO
    cfV := cfI/cfW*cfX;
    IF cfT == 1 THEN cfD := FISHER(cfA, cfB, cfV); ELSE cfD := CHISQUARE(cfA, cfV); END;
    cfL(cfI) := cfD;
    IF cfI > cfW/25 THEN cfMx := MAX(cfMx, cfD); END;
  END;
  IF cfMx <= 0 THEN cfMx := 1; END;
  avCvY := cfMx*1.1;
  avCvP := MAKELIST(0, cfI, 1, cfW);
  FOR cfI FROM 1 TO cfW DO
    avCvP(cfI) := avPB - IP(MIN(cfL(cfI)/avCvY, 1)*(avPB - avPT));
  END;
END;

// The pixel column of a value on the curve's x axis.
av_CvPx(cxV)
BEGIN
  RETURN avPL + IP(cxV/avCvX*(avPR - avPL) + 0.5);
END;

// One curve page on G1 from the cache.  The p region is [0, cdPa]
// (when cdPa > 0) and [cdPb, infinity) (when cdPb >= 0); the critical
// region likewise [0, cdCa] and [cdCb, infinity); -1 = not used.  A
// column in the p region is shaded blue, in the critical region pale
// red, in both purple.  The observed statistic is a red line, or an
// arrow at the right edge when it is off the plot (and the page says
// so).  The caller writes its readout lines and the footer.
av_CvDraw(cdTtl, cdObs, cdPa, cdPb, cdCa, cdCb, cdOL)
BEGIN
  LOCAL cdPx, cdV, cdN, cdY, cdIp, cdIc;
  LOCAL cdC, cdLast, cdX, cdLx;
  av_Band(cdTtl);
  RECT_P(G1, 6, 24, 14, 32, avSHD, avSHD);
  av_Tx("p area", 18, 23, avBLUE);
  RECT_P(G1, 66, 24, 74, 32, avCRIT, avCRIT);
  av_Tx("critical region", 78, 23, avMAG);
  RECT_P(G1, 168, 24, 176, 32, avBOTH, avBOTH);
  av_Tx("both  (" + avAl + " = " + av_AS() + ")", 180, 23, #6E28B4h);
  // ---- x ticks and grid ----
  cdV := 0; cdN := 0;
  WHILE cdV <= avCvX + avCvS/1000 AND cdN < 16 DO
    cdPx := av_CvPx(cdV);
    LINE_P(G1, cdPx, avPT, cdPx, avPB, avGRID);
    LINE_P(G1, cdPx, avPB, cdPx, avPB + 3, #000000h);
    TEXTOUT_P(av_Fx(cdV, avCvDc), G1, cdPx - 6, avPB + 5, 1, avGRAY, 40);
    cdV := cdV + avCvS; cdN := cdN + 1;
  END;
  // ---- shading, one pixel column at a time ----
  FOR cdPx FROM avPL + 1 TO avPR DO
    cdX := (cdPx - avPL)/(avPR - avPL)*avCvX;
    cdIp := 0; cdIc := 0;
    IF cdPa > 0 AND cdX <= cdPa THEN cdIp := 1; END;
    IF cdPb >= 0 AND cdX >= cdPb THEN cdIp := 1; END;
    IF cdCa > 0 AND cdX <= cdCa THEN cdIc := 1; END;
    IF cdCb >= 0 AND cdX >= cdCb THEN cdIc := 1; END;
    IF cdIp + cdIc > 0 THEN
      cdC := avSHD;
      IF cdIp == 0 THEN cdC := avCRIT; END;
      IF cdIp == 1 AND cdIc == 1 THEN cdC := avBOTH; END;
      cdY := avCvP(cdPx - avPL);
      IF cdY < avPB THEN LINE_P(G1, cdPx, avPB - 1, cdPx, cdY, cdC); END;
    END;
  END;
  // ---- the curve ----
  cdLast := -1;
  FOR cdPx FROM avPL + 1 TO avPR DO
    cdY := avCvP(cdPx - avPL);
    IF cdLast >= 0 THEN LINE_P(G1, cdPx - 1, cdLast, cdPx, cdY, avBLUE); END;
    cdLast := cdY;
  END;
  LINE_P(G1, avPL, avPT, avPL, avPB, #000000h);
  LINE_P(G1, avPL, avPB, avPR, avPB, #000000h);
  // ---- critical values: dotted magenta ----
  IF cdCa > 0 AND cdCa <= avCvX THEN av_DotV(av_CvPx(cdCa), avPT, avPB, avMAG); END;
  IF cdCb >= 0 AND cdCb <= avCvX THEN av_DotV(av_CvPx(cdCb), avPT, avPB, avMAG); END;
  // ---- the observed statistic ----
  IF cdObs <= avCvX THEN
    cdX := av_CvPx(cdObs);
    LINE_P(G1, cdX, avPT, cdX, avPB, avRED);
    LINE_P(G1, cdX + 1, avPT, cdX + 1, avPB, avRED);
    cdLx := cdX + 4;
    IF cdX > avPR - 96 THEN cdLx := cdX - 96; END;
    TEXTOUT_P(cdOL, G1, cdLx, avPT + 3, 1, avRED, 92);
  ELSE
    LINE_P(G1, avPR - 30, avPT + 8, avPR, avPT + 8, avRED);
    LINE_P(G1, avPR - 5, avPT + 4, avPR, avPT + 8, avRED);
    LINE_P(G1, avPR - 5, avPT + 12, avPR, avPT + 8, avRED);
    TEXTOUT_P(cdOL, G1, avPR - 130, avPT + 14, 1, avRED, 128);
    TEXTOUT_P("is off the plot, to the right", G1, avPR - 190, avPT + 26, 1, avRED, 190);
  END;
END;

// The window for an F page: wide enough for the critical value, and
// for the observed F unless it is far out (then the arrow says so).
av_FWin(fwF, fwC)
BEGIN
  LOCAL fwX;
  fwX := MAX(fwC*1.6, 3);
  IF fwF*1.1 > fwX AND fwF*1.1 <= 3*fwX THEN fwX := fwF*1.1; END;
  RETURN fwX;
END;

// An F curve page for an effect: F(fpA, fpB), observed fpF, its p.
// Draws the page (not the footer) and writes the two readout lines.
av_FPage(fpTtl, fpF, fpA, fpB, fpP)
BEGIN
  LOCAL fpC, fpX, fpK, fpCol;
  fpC := av_FQ(avAlpha, fpA, fpB);
  fpX := av_FWin(fpF, fpC);
  fpK := "F" + av_Fx(fpA, 0) + "," + av_Fx(fpB, 0) + "," + av_Fx(fpX, 6);
  IF fpK <> avCvK THEN av_CvFit(1, fpA, fpB, fpX); avCvK := fpK; END;
  av_CvDraw(fpTtl, fpF, -1, fpF, -1, fpC, "F = " + av_Fx(fpF, 3));
  fpCol := avGRN;
  IF fpP < avAlpha THEN fpCol := avRED; END;
  av_Tx("Shaded = p = P(F " + avGE + " " + av_Fx(fpF, 3) + " | " + avH0 + ") " + av_Rel(fpP), 4, 180, avBLUE);
  IF fpF >= fpC THEN
    av_Tx("F " + avGE + " crit " + av_Fx(fpC, 3) + ": reject " + avH0 + " (" + avAl + " = " + av_AS() + ")", 4, 192, fpCol);
  ELSE
    av_Tx("F < crit " + av_Fx(fpC, 3) + ": do not reject " + avH0 + " (" + avAl + " = " + av_AS() + ")", 4, 192, fpCol);
  END;
END;

// The method's simultaneous CI half-width for pair r, or -1 (none yet).
av_HW(hwR)
BEGIN
  IF avMeth == 1 THEN
    IF avQc > 0 THEN RETURN avQc/SQRT(2)*avPSE(hwR); END;
    RETURN -1;
  END;
  IF avMeth == 4 THEN RETURN avScM*avPSE(hwR); END;
  RETURN avBonT*avPSE(hwR);
END;

// The method's adjusted p for pair r, or -1 (Tukey's, not computed yet).
av_PAdj(paR)
BEGIN
  IF avMeth == 1 THEN RETURN avPTk(paR); END;
  IF avMeth == 2 THEN RETURN avPBon(paR); END;
  IF avMeth == 3 THEN RETURN avPHm(paR); END;
  RETURN avPSc(paR);
END;

// 1 when the method says pair r differs at alpha.  Tukey without its p:
// q > q crit, the same decision.
av_Sig(sgR)
BEGIN
  LOCAL sgP;
  sgP := av_PAdj(sgR);
  IF sgP >= 0 THEN
    IF sgP < avAlpha THEN RETURN 1; END;
    RETURN 0;
  END;
  IF avMeth == 1 AND avQc > 0 THEN
    IF avPQ(sgR) > avQc THEN RETURN 1; END;
  END;
  RETURN 0;
END;

// The index of pair (i, j) in the pair lists, either order.
av_PairOf(pfA, pfB)
BEGIN
  LOCAL pfI, pfJ;
  pfI := MIN(pfA, pfB); pfJ := MAX(pfA, pfB);
  RETURN (pfI - 1)*avOK - (pfI - 1)*pfI/2 + (pfJ - pfI);
END;

    av_Band("Dots: " + avOTtl);

    ddLo := avOY(1); ddHi := avOY(1);
    FOR ddI FROM 1 TO SIZE(avOY) DO
      ddLo := MIN(ddLo, avOY(ddI)); ddHi := MAX(ddHi, avOY(ddI));
    END;
    FOR ddG FROM 1 TO avOK DO
      ddH := avTcr*SQRT(avMSW/avRn(ddG));
      ddLo := MIN(ddLo, avRm(ddG) - ddH); ddHi := MAX(ddHi, avRm(ddG) + ddH);
    END;
    IF ddHi - ddLo <= 0 THEN ddHi := ddLo + 1; END;
    ddS := av_Nice((ddHi - ddLo)/5);
    ddLo := FLOOR(ddLo/ddS)*ddS; ddHi := CEILING(ddHi/ddS)*ddS;
    ddDc := MAX(0, -FLOOR(LOG(ddS)));
    // ---- grid and y ticks ----
    ddV := ddLo;
    WHILE ddV <= ddHi + ddS/1000 DO
      ddY := av_DY(ddV, ddLo, ddHi);
      LINE_P(G1, avDL, ddY, avDR, ddY, avGRID);
      TEXTOUT_P(av_Fx(ddV, ddDc), G1, 2, ddY - 5, 1, avGRAY, 32);
      ddV := ddV + ddS;
    END;
    // ---- the dots: a little to the left of centre, spread over 5 lanes ----
    ddCnt := MAKELIST(0, ddI, 1, avOK);
    FOR ddI FROM 1 TO SIZE(avOY) DO
      ddG := avOG(ddI);
      ddCnt(ddG) := ddCnt(ddG) + 1;
      ddX := IP(avDL + (ddG - 0.5)*ddW - ddW*0.15) + ((ddCnt(ddG) - 1) MOD 5 - 2)*2;
      ddY := av_DY(avOY(ddI), ddLo, ddHi);
      ddC := avGCol(((ddG - 1) MOD 12) + 1);
      RECT_P(G1, ddX - 1, ddY - 1, ddX + 1, ddY + 1, ddC);
    END;
    // ---- each mean and its CI ----
    FOR ddG FROM 1 TO avOK DO
      ddM := avRm(ddG);
      ddH := avTcr*SQRT(avMSW/avRn(ddG));
      ddX := IP(avDL + (ddG - 0.5)*ddW + ddW*0.22);
      ddT := av_DY(ddM + ddH, ddLo, ddHi); ddR := av_DY(ddM - ddH, ddLo, ddHi);
      RECT_P(G1, ddX - 1, ddT, ddX + 1, ddR, avMAG);
      LINE_P(G1, ddX - 3, ddT, ddX + 3, ddT, avMAG);
      LINE_P(G1, ddX - 3, ddR, ddX + 3, ddR, avMAG);
      ddY := av_DY(ddM, ddLo, ddHi);
      RECT_P(G1, ddX - 6, ddY - 1, ddX + 6, ddY + 1, avGRN);
      TEXTOUT_P(avOSh(ddG), G1, IP(avDL + (ddG - 1)*ddW) + 2, avDB + 3, 1, avGCol(((ddG - 1) MOD 12) + 1), MAX(8, IP(ddW) - 2));
    END;
    LINE_P(G1, avDL, avDT, avDL, avDB, #000000h);
    LINE_P(G1, avDL, avDB, avDR, avDB, #000000h);
    av_Tx("green = mean, magenta = its " + av_CS(1 - avAlpha) + " CI (pooled s)", 4, 172, avGRAY);
    DIMGROB_P(G2, 320, 240, #FFFFFFh);
    BLIT_P(G2, 0, 0, G1);
    avDotOK := 1;
  END;
  // ---- the selected groups' columns, framed in orange (2 px) ----
  FOR ddG FROM 1 TO avOK DO
    IF ddG == avSelA OR ddG == avSelB THEN
      ddX := IP(avDL + (ddG - 1)*ddW) + 1; ddX2 := IP(avDL + ddG*ddW) - 1;
      RECT_P(G1, ddX, avDT, ddX2, avDT + 1, avORNG, avORNG);
      RECT_P(G1, ddX, avDB - 1, ddX2, avDB, avORNG, avORNG);
      RECT_P(G1, ddX, avDT, ddX + 1, avDB, avORNG, avORNG);
      RECT_P(G1, ddX2 - 1, avDT, ddX2, avDB, avORNG, avORNG);
    END;
  END;
  // ---- the pair ----
  IF avSelA > 0 AND avSelB > 0 THEN
    ddPr := av_PairOf(avSelA, avSelB);
    ddX := IP(avDL + (avSelA - 0.5)*ddW); ddX2 := IP(avDL + (avSelB - 0.5)*ddW);
    LINE_P(G1, ddX, avDT + 2, ddX2, avDT + 2, avORNG);
    LINE_P(G1, ddX, avDT + 2, ddX, avDT + 7, avORNG);
    LINE_P(G1, ddX2, avDT + 2, ddX2, avDT + 7, avORNG);
    ddH := av_HW(ddPr);
    ddC := #000000h;
    IF av_Sig(ddPr) == 1 THEN ddC := avRED; END;
    IF ddH > 0 THEN
      av_Tx(avOSh(avPI(ddPr)) + " - " + avOSh(avPJ(ddPr)) + ": diff " + av_Sgn(avPD(ddPr)) + ", CI (" + av_Gn(avPD(ddPr) - ddH) + ", " + av_Gn(avPD(ddPr) + ddH) + ")", 4, 184, ddC);
    ELSE
      av_Tx(avOSh(avPI(ddPr)) + " - " + avOSh(avPJ(ddPr)) + ": diff " + av_Sgn(avPD(ddPr)) + " (no CI yet)", 4, 184, ddC);
    END;
    ddV := av_PAdj(ddPr);
    IF ddV >= 0 THEN
      IF ddV < avAlpha THEN
        av_Tx(av_MShort() + " " + av_PE(ddV) + ": evidence they differ", 4, 195, avRED);
      ELSE
        av_Tx(av_MShort() + " " + av_PE(ddV) + ": no evidence they differ", 4, 195, avGRN);
      END;
    ELSE
      IF avQc > 0 THEN
        IF avPQ(ddPr) > avQc THEN
          av_Tx("q = " + av_Fx(avPQ(ddPr), 3) + " > q crit " + av_Fx(avQc, 3) + ": evidence they differ", 4, 195, avRED);
        ELSE
          av_Tx("q = " + av_Fx(avPQ(ddPr), 3) + " " + avLE + " q crit " + av_Fx(avQc, 3) + ": no evidence they differ", 4, 195, avGRN);
        END;
      ELSE
        av_Tx("No q crit yet: Menu > Tukey p for this pair", 4, 195, avGRAY);
      END;
    END;
  ELSE
    IF avSelA > 0 THEN
      av_Tx("First group: " + avON(avSelA) + ".  Now tap a second one.", 4, 184, #000000h);
    ELSE
      av_Tx("Tap two groups (or " + avLR + ") to compare a pair.", 4, 184, #000000h);
    END;
    av_Tx("Method: " + av_MName() + " (+/- or the Pairs tabs)", 4, 195, avGRAY);
  END;
  av_Foot("tap 2 groups, " + avLR + " = next pair, +/- = method");
END;

// Set alpha (used by every verdict, critical value and CI).
av_SetAlpha()
BEGIN
  LOCAL saV, saL;
  saV := avAlpha;
  saL := avAl + ":";
  IF INPUT({saV}, "Significance level", {saL}, {"From 0.001 to 0.2; 0.05 is usual"}) == 1 THEN
    IF saV < 0.001 OR saV > 0.2 THEN
      MSGBOX(avAl + " must be from 0.001 to 0.2.");
      RETURN 0;
    END;
    avAlpha := saV;
    IF avODone == 1 THEN av_OCrit(); END;
  END;
  RETURN 0;
END;

// Arrow keys on a one-way page.  Returns an action (a page number) or 0.
av_OArrow(oaK)
BEGIN
  IF avScr == 4 AND (oaK == 7 OR oaK == 8) THEN
    // Dots: the previous / next pair
    IF oaK == 8 THEN avPIx := avPIx + 1; ELSE avPIx := avPIx - 1; END;
    IF avPIx > avNP THEN avPIx := 1; END;
    IF avPIx < 1 THEN avPIx := avNP; END;
    avSelA := avPI(avPIx); avSelB := avPJ(avPIx);
    RETURN 0;
  END;
  IF avScr == 3 AND (oaK == 2 OR oaK == 12) THEN
    // Pairs: move the highlight, scrolling to keep it on the page
    IF oaK == 12 THEN avPIx := MIN(avNP, avPIx + 1); ELSE avPIx := MAX(1, avPIx - 1); END;
    IF avPIx < avPTop THEN avPTop := avPIx; END;
    IF avPIx > avPTop + 9 THEN avPTop := avPIx - 9; END;
    avSelA := avPI(avPIx); avSelB := avPJ(avPIx);
    RETURN 0;
  END;
  IF oaK == 7 THEN
    IF avScr == 1 THEN RETURN 5; END;
    RETURN avScr - 1;
  END;
  IF oaK == 8 THEN
    IF avScr == 5 THEN RETURN 1; END;
    RETURN avScr + 1;
  END;
  RETURN 0;
END;

// A tap above the soft keys on a one-way page.  Returns an action or 0.
av_OTap()
BEGIN
  LOCAL otI, otW, otG;
  IF avScr == 3 THEN
    // the method tabs
    IF avTapY >= 18 AND avTapY <= 35 THEN
      avMeth := MAX(1, MIN(4, IP((avTapX - 4)/78) + 1));
      RETURN 0;
    END;
    // a pair's row: open it on the Dots page
    IF avTapY >= 71 AND avTapY < 182 THEN
      otI := avPTop + IP((avTapY - 71)/11);
      IF otI >= 1 AND otI <= avNP AND otI <= avPTop + 9 THEN
        avPIx := otI; avSelA := avPI(otI); avSelB := avPJ(otI);
        RETURN 4;
      END;
    END;
    RETURN 0;
  END;
  IF avScr == 4 THEN
    IF avTapY >= avDT AND avTapY <= avDB + 12 AND avTapX >= avDL AND avTapX <= avDR THEN
      otW := (avDR - avDL)/avOK;
      otG := MAX(1, MIN(avOK, IP((avTapX - avDL)/otW) + 1));
      IF avSelA == 0 OR avSelB > 0 THEN
        avSelA := otG; avSelB := 0; avPIx := 0;
      ELSE
        IF otG <> avSelA THEN avSelB := otG; avPIx := av_PairOf(avSelA, otG); END;
      END;
    END;
    RETURN 0;
  END;
  RETURN 0;
END;

// The one-way pages.  Soft keys, page 1: Table, Checks, Pairs, Dots, F,
// More; page 2: Method, Tukey p, Help, Menu, Back, More (the house
// "More" key: slot 6 on every page).  Actions: 1-5 the pages, 7 next
// method, 8 Tukey p, 9 help, 10 menu, 11 back, 12 previous method,
// 13 set alpha, 14 Tukey p for the selected pair.
av_OView(ovS)
BEGIN
  LOCAL ovK, ovA, ovGo, ovSl;
  IF avODone == 0 THEN
    IF av_OCompute() == 0 THEN RETURN 0; END;
  END;
  avScr := ovS; avSKP := 1; ovGo := 1;
  av_KDrain(); av_Drain();
  WHILE ovGo == 1 DO
    // beta 1a: Tukey is never started by opening a page.  When the
    // student picks Tukey and q crit is not known yet, a one-line warning
    // with the G1 estimate asks first; "Not now" leaves it uncomputed
    // (Menu > Tukey p computes it later).
    IF (avScr == 3 OR avScr == 4) AND avMeth == 1 AND avQc == -1 THEN
      IF avDf2 < 3 THEN
        avQc := -3;
      ELSE
        IF av_TukeyAsk("Tukey q crit", 14) == 1 THEN av_TukeyCrit(); ELSE avQc := -2; END;
      END;
    END;
    av_ODraw();
    IF avSKP == 1 THEN
      av_Show({"Table", "Checks", "Pairs", "Dots", "F", "More"});
    ELSE
      av_Show({"Method", "Tukey p", "Help", "Menu", "Back", "More"});
    END;
    ovK := av_Poll();
    ovA := 0;
    IF ovK == -1 THEN
      IF avTapY >= 218 THEN
        ovSl := MIN(6, IP(avTapX/53.4) + 1);
        IF ovSl == 6 THEN
          avSKP := 3 - avSKP;
        ELSE
          IF avSKP == 1 THEN ovA := ovSl; ELSE ovA := ovSl + 6; END;
        END;
      ELSE
        ovA := av_OTap();
      END;
    ELSE
      IF ovK == 4 THEN ovA := 11; END;
      IF ovK == 30 THEN ovA := 10; END;
      IF ovK == 50 THEN ovA := 7; END;
      IF ovK == 45 THEN ovA := 12; END;
      IF ovK == 7 OR ovK == 8 OR ovK == 2 OR ovK == 12 THEN ovA := av_OArrow(ovK); END;
    END;
    IF ovA == 10 THEN ovA := av_OMenu(); END;
    IF ovA >= 1 AND ovA <= 5 THEN avScr := ovA; END;
    IF ovA == 7 THEN
      avMeth := avMeth + 1;
      IF avMeth > 4 THEN avMeth := 1; END;
    END;
    IF ovA == 12 THEN
      avMeth := avMeth - 1;
      IF avMeth < 1 THEN avMeth := 4; END;
    END;
    IF ovA == 8 THEN
      av_TukeyP();
      IF avScr <> 4 THEN avScr := 3; END;
    END;
    IF ovA == 9 THEN
      IF avScr == 3 OR avScr == 4 THEN av_Help(3); ELSE av_Help(1); END;
    END;
    IF ovA == 13 THEN av_SetAlpha(); END;
    IF ovA == 14 THEN av_TukeyP1(avPIx); END;
    IF ovA == 11 THEN ovGo := 0; END;
  END;
  av_KDrain();
  RETURN 0;
END;

// Interaction plot geometry: the x level count and trace count, by Swap.
av_TNx()
BEGIN
  IF avTSwap == 0 THEN RETURN avTNA; END;
  RETURN avTNB;
END;

av_TNt()
BEGIN
  IF avTSwap == 0 THEN RETURN avTNB; END;
  RETURN avTNA;
END;

// The cell index for x level ix and trace it.
av_TCell(tlX, tlT)
BEGIN
  IF avTSwap == 0 THEN RETURN (tlX - 1)*avTNB + tlT; END;
  RETURN (tlT - 1)*avTNB + tlX;
END;

// The y range of the cell means, nice; kept in a list {lo, hi, step}.
av_TRange()
BEGIN
  LOCAL trLo, trHi, trS, trK;
  trLo := avTCM(1); trHi := avTCM(1);
  FOR trK FROM 1 TO SIZE(avTCM) DO
    trLo := MIN(trLo, avTCM(trK)); trHi := MAX(trHi, avTCM(trK));
  END;
  IF trHi - trLo <= 0 THEN trHi := trLo + 1; END;
  trS := av_Nice((trHi - trLo)/4);
  trLo := FLOOR(trLo/trS)*trS; trHi := CEILING(trHi/trS)*trS;
  IF trHi <= trLo THEN trHi := trLo + trS; END;
  RETURN {trLo, trHi, trS};
END;

// The y pixel of a value on the interaction plot (its box ends at 146,
// leaving room under it for the legend, the readout and two lines).
av_TY(tyV, tyLo, tyHi)
BEGIN
  RETURN 146 - IP((tyV - tyLo)/(tyHi - tyLo)*(146 - avDT) + 0.5);
END;

// 1 when two of the plotted lines cross (their order changes from one
// x level to another), else 0.
av_TCross()
BEGIN
  LOCAL tcT1, tcT2, tcX, tcD0, tcD1, tcR;
  tcR := 0;
  FOR tcT1 FROM 1 TO av_TNt() - 1 DO
    FOR tcT2 FROM tcT1 + 1 TO av_TNt() DO
      tcD0 := avTCM(av_TCell(1, tcT1)) - avTCM(av_TCell(1, tcT2));
      FOR tcX FROM 2 TO av_TNx() DO
        tcD1 := avTCM(av_TCell(tcX, tcT1)) - avTCM(av_TCell(tcX, tcT2));
        IF tcD0*tcD1 < 0 THEN tcR := 1; END;
        IF tcD1 <> 0 THEN tcD0 := tcD1; END;
      END;
    END;
  END;
  RETURN tcR;
END;

// "n 17-21" (or "n 5" when every cell has the same n).
av_TNs()
BEGIN
  LOCAL tnK, tnLo, tnHi;
  tnLo := avTCN(1); tnHi := avTCN(1);
  FOR tnK FROM 1 TO SIZE(avTCN) DO
    tnLo := MIN(tnLo, avTCN(tnK)); tnHi := MAX(tnHi, avTCN(tnK));
  END;
  IF tnLo == tnHi THEN RETURN "n " + av_I(tnLo); END;
  RETURN "n " + av_I(tnLo) + "-" + av_I(tnHi);
END;

// The x pixel of x level ix on the interaction plot.
av_TPx(tpX)
BEGIN
  RETURN IP(avDL + 20 + (tpX - 0.5)*(avDR - avDL - 40)/av_TNx());
END;

// The interaction plot: cell means joined by lines, one colour per
// level of the trace factor; the selected point ringed in red.
av_TDrawPlot()
BEGIN
  LOCAL tpR, tpV, tpY, tpX, tpT, tpK;
  LOCAL tpC, tpY2, tpNx, tpNt, tpDc, tpS;
  LOCAL tpXN, tpTN, tpXF, tpTF;
  tpNx := av_TNx(); tpNt := av_TNt();
  IF avTSwap == 0 THEN
    tpXN := avTAN; tpTN := avTBN; tpXF := avTFA; tpTF := avTFB;
  ELSE
    tpXN := avTBN; tpTN := avTAN; tpXF := avTFB; tpTF := avTFA;
  END;
  avTSx := MAX(1, MIN(tpNx, avTSx)); avTSt := MAX(1, MIN(tpNt, avTSt));
  av_Band("Interaction: means by " + tpXF);
  tpR := av_TRange();
  tpDc := MAX(0, -FLOOR(LOG(tpR(3))));
  tpV := tpR(1);
  WHILE tpV <= tpR(2) + tpR(3)/1000 DO
    tpY := av_TY(tpV, tpR(1), tpR(2));
    LINE_P(G1, avDL, tpY, avDR, tpY, avGRID);
    TEXTOUT_P(av_Fx(tpV, tpDc), G1, 2, tpY - 5, 1, avGRAY, 32);
    tpV := tpV + tpR(3);
  END;
  LINE_P(G1, avDL, avDT, avDL, 146, #000000h);
  LINE_P(G1, avDL, 146, avDR, 146, #000000h);
  tpS := MIN(90, IP((avDR - avDL - 40)/tpNx) - 4);
  FOR tpX FROM 1 TO tpNx DO
    TEXTOUT_P(tpXN(tpX), G1, av_TPx(tpX) - IP(tpS/2), 149, 1, #000000h, tpS);
  END;
  // ---- the lines (2 px), then the points ----
  FOR tpT FROM 1 TO tpNt DO
    tpC := avGCol(((tpT - 1) MOD 12) + 1);
    FOR tpX FROM 1 TO tpNx - 1 DO
      tpY := av_TY(avTCM(av_TCell(tpX, tpT)), tpR(1), tpR(2));
      tpY2 := av_TY(avTCM(av_TCell(tpX + 1, tpT)), tpR(1), tpR(2));
      LINE_P(G1, av_TPx(tpX), tpY, av_TPx(tpX + 1), tpY2, tpC);
      LINE_P(G1, av_TPx(tpX), tpY + 1, av_TPx(tpX + 1), tpY2 + 1, tpC);
    END;
  END;
  FOR tpT FROM 1 TO tpNt DO
    tpC := avGCol(((tpT - 1) MOD 12) + 1);
    FOR tpX FROM 1 TO tpNx DO
      tpY := av_TY(avTCM(av_TCell(tpX, tpT)), tpR(1), tpR(2));
      av_Mark(av_TPx(tpX), tpY, tpC);
    END;
  END;
  // the selected point: a red ring of 4 lines
  tpX := av_TPx(avTSx); tpY := av_TY(avTCM(av_TCell(avTSx, avTSt)), tpR(1), tpR(2));
  LINE_P(G1, tpX - 5, tpY - 5, tpX + 5, tpY - 5, avRED); LINE_P(G1, tpX - 5, tpY + 5, tpX + 5, tpY + 5, avRED);
  LINE_P(G1, tpX - 5, tpY - 5, tpX - 5, tpY + 5, avRED); LINE_P(G1, tpX + 5, tpY - 5, tpX + 5, tpY + 5, avRED);
  // ---- legend: the trace factor's levels ----
  tpS := 4;
  TEXTOUT_P(tpTF + ":", G1, tpS, 160, 1, #000000h, 120);
  tpS := tpS + IP(6.4*DIM(tpTF)) + 12;
  FOR tpT FROM 1 TO tpNt DO
    tpC := avGCol(((tpT - 1) MOD 12) + 1);
    IF tpS < 290 THEN
      RECT_P(G1, tpS, 162, tpS + 8, 168, tpC, tpC);
      TEXTOUT_P(av_Cut(tpTN(tpT), 70), G1, tpS + 11, 160, 1, tpC, 72);
    END;
    tpS := tpS + 16 + MIN(70, IP(6.4*DIM(tpTN(tpT))));
  END;
  tpK := av_TCell(avTSx, avTSt);
  av_Tx(tpXN(avTSx) + ", " + tpTN(avTSt) + ": " + avXb + " " + av_Gn(avTCM(tpK)) + ", n " + av_I(avTCN(tpK)) + ", s " + av_Gn(avTCS(tpK)), 4, 172, avRED);
  // beta 1a (second review, item 3): what parallel lines WOULD mean, and
  // what the test says about these lines
  IF avTP(3) < avAlpha THEN
    av_Tx("Interaction " + av_PE(avTP(3)) + ": evidence that", 4, 184, avRED);
    av_Tx("the lines are not parallel in the population.", 4, 195, avRED);
  ELSE
    IF av_TCross() == 1 THEN
      av_Tx("Parallel = no interaction. These lines cross,", 4, 184, avGRAY);
      av_Tx("but p = " + av_Fx(avTP(3), 3) + " (" + av_TNs() + ") cannot rule out chance.", 4, 195, avGRAY);
    ELSE
      av_Tx("Parallel lines would mean no interaction.", 4, 184, avGRAY);
      av_Tx("Here p = " + av_Fx(avTP(3), 3) + " (" + av_TNs() + "): chance not ruled out.", 4, 195, avGRAY);
    END;
  END;
  av_Foot("tap a point, " + avLR + " " + avUD + " = move, Swap = turn");
END;

// The two-way pages.  Soft keys: Table, Cells, Plot, F, Swap, Menu.
av_TView(tvS)
BEGIN
  LOCAL tvK, tvA, tvGo, tvSl, tvX, tvT;
  LOCAL tvD, tvBd, tvR, tvY, tvW, tvH;
  IF avTDone == 0 THEN
    IF av_TCompute() == 0 THEN RETURN 0; END;
  END;
  avScr := tvS; tvGo := 1;
  av_KDrain(); av_Drain();
  WHILE tvGo == 1 DO
    av_TDraw();
    av_Show({"Table", "Cells", "Plot", "F", "Swap", "Menu"});
    tvK := av_Poll();
    tvA := 0;
    IF tvK == -1 THEN
      IF avTapY >= 218 THEN
        tvSl := MIN(6, IP(avTapX/53.4) + 1);
        tvA := tvSl;
        IF tvSl == 6 THEN tvA := 10; END;
      ELSE
        IF avScr == 3 AND avTapY > avDT - 6 AND avTapY < 152 THEN
          // the nearest cell mean
          tvR := av_TRange(); tvBd := 9999;
          FOR tvX FROM 1 TO av_TNx() DO
            FOR tvT FROM 1 TO av_TNt() DO
              tvY := av_TY(avTCM(av_TCell(tvX, tvT)), tvR(1), tvR(2));
              tvD := ABS(avTapX - av_TPx(tvX)) + ABS(avTapY - tvY);
              IF tvD < tvBd THEN tvBd := tvD; avTSx := tvX; avTSt := tvT; END;
            END;
          END;
        END;
        IF avScr == 4 AND avTapY < 19 THEN tvA := 14; END;
        IF avScr == 2 AND avTapY >= 34 AND avTapX >= 64 THEN
          tvW := MIN(84, IP(252/(avTNB + 1))); tvH := MIN(26, IP(132/(avTNA + 1)));
          tvX := IP((avTapX - 64)/tvW) + 1; tvT := IP((avTapY - 34)/tvH) + 1;
          IF tvX >= 1 AND tvX <= avTNB AND tvT >= 1 AND tvT <= avTNA THEN avTCj := tvX; avTCi := tvT; END;
        END;
      END;
    ELSE
      IF tvK == 4 THEN tvA := 11; END;
      IF tvK == 30 THEN tvA := 10; END;
      IF avScr == 2 THEN
        IF tvK == 7 THEN avTCj := MAX(1, avTCj - 1); END;
        IF tvK == 8 THEN avTCj := MIN(avTNB, avTCj + 1); END;
        IF tvK == 2 THEN avTCi := MAX(1, avTCi - 1); END;
        IF tvK == 12 THEN avTCi := MIN(avTNA, avTCi + 1); END;
      END;
      IF avScr == 3 THEN
        IF tvK == 7 THEN avTSx := MAX(1, avTSx - 1); END;
        IF tvK == 8 THEN avTSx := MIN(av_TNx(), avTSx + 1); END;
        IF tvK == 2 THEN avTSt := MAX(1, avTSt - 1); END;
        IF tvK == 12 THEN avTSt := MIN(av_TNt(), avTSt + 1); END;
      END;
      IF avScr == 1 OR avScr == 4 THEN
        IF tvK == 7 THEN tvA := avScr - 1; IF tvA < 1 THEN tvA := 4; END; END;
        IF tvK == 8 THEN tvA := avScr + 1; IF tvA > 4 THEN tvA := 1; END; END;
      END;
      IF avScr == 4 AND tvK == 50 THEN tvA := 14; END;
      IF avScr == 4 AND tvK == 45 THEN tvA := 15; END;
    END;
    IF tvA == 10 THEN tvA := av_TMenu(); END;
    IF tvA >= 1 AND tvA <= 4 THEN avScr := tvA; END;
    IF tvA == 5 THEN
      avTSwap := 1 - avTSwap;
      tvX := avTSx; avTSx := avTSt; avTSt := tvX;
      avScr := 3;
    END;
    IF tvA == 14 THEN avTEff := avTEff + 1; IF avTEff > 3 THEN avTEff := 1; END; END;
    IF tvA == 15 THEN avTEff := avTEff - 1; IF avTEff < 1 THEN avTEff := 3; END; END;
    IF tvA == 9 THEN av_Help(4); END;
    IF tvA == 13 THEN av_SetAlpha(); END;
    IF tvA == 11 THEN tvGo := 0; END;
  END;
  av_KDrain();
  RETURN 0;
END;

// The fallback group picker: one group at a time, literal menus.
av_PickG2(p2T)
BEGIN
  LOCAL p2G, p2I;
  p2G := 1;
  WHILE p2G <= avOK DO
    p2I := 0;
    CHOOSE(p2I, p2T, {"This one: " + avON(p2G), "Next group", "Cancel"});
    IF p2I == 1 THEN RETURN p2G; END;
    IF p2I == 0 OR p2I == 3 THEN RETURN 0; END;
    p2G := p2G + 1;
    IF p2G > avOK THEN p2G := 1; END;
  END;
  RETURN 0;
END;

// Pick one group of the one-way data.  Returns its number or 0.
av_PickG(pgT)
BEGIN
  LOCAL pgI, pgL, pgG;
  pgL := {};
  FOR pgG FROM 1 TO avOK DO
    pgL := CONCAT(pgL, {avON(pgG) + " (n " + av_I(avRn(pgG)) + ", s " + av_Gn(SQRT(avRv(pgG))) + ")"});
  END;
  pgI := 0;
  // beta 1a (second review, item 7): a CHOOSE with a list VARIABLE has
  // not been seen on a calculator; if it raises an error, page through
  // the groups with literal three-item menus instead.
  IFERR
    CHOOSE(pgI, pgT, pgL);
  THEN
    pgI := av_PickG2(pgT);
  END;
  RETURN pgI;
END;

// The alternative, by CHOOSE.  Returns 1, 2, 3 or 0.

    avH1 + ": " + ptS + " < " + ptV,
    avH1 + ": " + ptS + " > " + ptV});
  RETURN ptI;
END;

// Set up a one-sigma problem.  Returns 1 when ready.
av_V1Ask()
BEGIN
  LOCAL vaI, vaS, vaN, vaS0, vaC, vaG;
  LOCAL vaL1, vaL2;
  vaI := 0;
  CHOOSE(vaI, "One " + avSg + ": the sample", {"Type s and n", "A group of the one-way data"});
  IF vaI == 0 THEN RETURN 0; END;
  IF vaI == 2 THEN
    IF avODone == 0 THEN
      IF av_OCompute() == 0 THEN RETURN 0; END;
    END;
    vaG := av_PickG("Which group?");
    IF vaG == 0 THEN RETURN 0; END;
    IF avRn(vaG) < 2 OR avRv(vaG) <= 0 THEN MSGBOX("That group needs n " + avGE + " 2 and s > 0."); RETURN 0; END;
    vaS := SQRT(avRv(vaG)); vaN := avRn(vaG);
    avVLbl := avOSh(vaG) + ": " + avOTtl;
  ELSE
    vaS := avVS1; vaN := avVN1;
    IF INPUT({vaS, vaN}, "One " + avSg + ": the sample", {"s:", "n:"}, {"Sample standard deviation (> 0)", "Sample size (a whole number, 2 or more)"}) == 0 THEN RETURN 0; END;
    IF vaS <= 0 OR vaN < 2 OR FP(vaN) <> 0 OR vaN > 1000000 THEN
      MSGBOX("s must be > 0 and n a whole number from 2 to 1000000.");
      RETURN 0;
    END;
    avVLbl := "Typed: s = " + av_Gn(vaS) + ", n = " + av_I(vaN) + " (illustrative numbers)";
  END;
  vaS0 := avVSg0; vaC := avVCL;
  vaL1 := avSg + avS0 + ":"; vaL2 := "Conf:";
  IF INPUT({vaS0, vaC}, "One " + avSg + ": test and CI", {vaL1, vaL2}, {"The population SD claimed by " + avH0 + " (> 0)", "Confidence level for the CI, 0.5 to 0.999"}) == 0 THEN RETURN 0; END;
  IF vaS0 <= 0 OR vaC < 0.5 OR vaC > 0.999 THEN
    MSGBOX(avSg + avS0 + " must be > 0 and the level from 0.5 to 0.999.");
    RETURN 0;
  END;
  avVMode := 1;
  vaI := av_PickAlt(av_Gn(vaS0));
  IF vaI == 0 THEN RETURN 0; END;
  avVAlt := vaI; avVS1 := vaS; avVN1 := vaN; avVSg0 := vaS0; avVCL := vaC;
  av_VCompute();
  RETURN 1;
END;

// Set up a two-sigma problem.  Returns 1 when ready.
av_V2Ask()
BEGIN
  LOCAL vbI, vbS1, vbN1, vbS2, vbN2, vbC;
  LOCAL vbG1, vbG2, vbL;
  vbI := 0;
  CHOOSE(vbI, "Two " + avSg + ": the samples", {"Type s and n for each", "Two groups of the one-way data"});
  IF vbI == 0 THEN RETURN 0; END;
  IF vbI == 2 THEN
    IF avODone == 0 THEN
      IF av_OCompute() == 0 THEN RETURN 0; END;
    END;
    vbG1 := av_PickG("First group (top of F)");
    IF vbG1 == 0 THEN RETURN 0; END;
    vbG2 := av_PickG("Second group (bottom of F)");
    IF vbG2 == 0 OR vbG2 == vbG1 THEN RETURN 0; END;
    IF avRn(vbG1) < 2 OR avRn(vbG2) < 2 OR avRv(vbG1) <= 0 OR avRv(vbG2) <= 0 THEN
      MSGBOX("Both groups need n " + avGE + " 2 and s > 0."); RETURN 0;
    END;
    vbS1 := SQRT(avRv(vbG1)); vbN1 := avRn(vbG1); vbS2 := SQRT(avRv(vbG2)); vbN2 := avRn(vbG2);
    avVLbl := avOSh(vbG1) + " / " + avOSh(vbG2) + ": " + avOTtl;
  ELSE
    vbS1 := avVS1; vbN1 := avVN1; vbS2 := avVS2; vbN2 := avVN2;
    IF INPUT({vbS1, vbN1, vbS2, vbN2}, "Two " + avSg + ": the samples", {"s1:", "n1:", "s2:", "n2:"},
       {"First sample's SD (> 0)", "First sample size (2 or more)", "Second sample's SD (> 0)", "Second sample size (2 or more)"}) == 0 THEN RETURN 0; END;
    IF vbS1 <= 0 OR vbS2 <= 0 OR vbN1 < 2 OR vbN2 < 2 OR FP(vbN1) <> 0 OR FP(vbN2) <> 0 OR MAX(vbN1, vbN2) > 1000000 THEN
      MSGBOX("Each s must be > 0 and each n a whole number from 2 to 1000000.");
      RETURN 0;
    END;
    avVLbl := "Typed (illustrative numbers)";
  END;
  vbC := avVCL; vbL := "Conf:";
  IF INPUT({vbC}, "Two " + avSg + ": the interval", {vbL}, {"Confidence level for the CI, 0.5 to 0.999"}) == 0 THEN RETURN 0; END;
  IF vbC < 0.5 OR vbC > 0.999 THEN MSGBOX("The level must be from 0.5 to 0.999."); RETURN 0; END;
  avVMode := 2;
  vbI := av_PickAlt("1");
  IF vbI == 0 THEN RETURN 0; END;
  avVAlt := vbI; avVS1 := vbS1; avVN1 := vbN1; avVS2 := vbS2; avVN2 := vbN2; avVCL := vbC;
  av_VCompute();
  RETURN 1;
END;

    IF vvK == -1 THEN
      IF avTapY >= 218 THEN
        vvSl := MIN(6, IP(avTapX/53.4) + 1);
        vvA := vvSl;
        IF vvSl == 5 THEN vvA := 9; END;
        IF vvSl == 6 THEN vvA := 10; END;
      END;
    ELSE
      IF vvK == 4 THEN vvA := 11; END;
      IF vvK == 30 THEN vvA := 10; END;
      IF vvK == 7 OR vvK == 8 THEN vvA := 3 - avScr; END;
    END;
    IF vvA == 10 THEN vvA := av_VMenu(); END;
    IF vvA == 1 OR vvA == 2 THEN avScr := vvA; END;
    IF vvA == 3 THEN av_V1Ask(); avScr := 1; END;
    IF vvA == 4 THEN av_V2Ask(); avScr := 1; END;
    IF vvA == 9 THEN av_Help(6); END;
    IF vvA == 13 THEN av_SetAlpha(); END;
    IF vvA == 11 THEN vvGo := 0; END;
  END;
  av_KDrain();
  RETURN 0;
END;

// List Di, or {} when it is empty, not a list, or not there yet (the
// first read is inside IFERR: Di does not exist until the app has been
// opened once).
av_ReadD(rdI)
BEGIN
  LOCAL rdL;
  rdL := {};
  IFERR
    IF rdI == 1 THEN rdL := D1; END;
    IF rdI == 2 THEN rdL := D2; END;
    IF rdI == 3 THEN rdL := D3; END;
    IF rdI == 4 THEN rdL := D4; END;
    IF rdI == 5 THEN rdL := D5; END;
    IF rdI == 6 THEN rdL := D6; END;
    IF rdI == 7 THEN rdL := D7; END;
    IF rdI == 8 THEN rdL := D8; END;
    IF rdI == 9 THEN rdL := D9; END;
  THEN
    rdL := {};
  END;
  IF TYPE(rdL) <> 6 THEN rdL := {}; END;
  RETURN rdL;
END;

// The distinct values of a list, sorted (the levels of a factor).
av_Levels(lvL)
BEGIN
  LOCAL lvS, lvR, lvI;
  lvS := SORT(lvL);
  lvR := {lvS(1)};
  FOR lvI FROM 2 TO SIZE(lvS) DO
    IF lvS(lvI) <> lvS(lvI - 1) THEN
      lvR := CONCAT(lvR, {lvS(lvI)});
      IF SIZE(lvR) > 6 THEN RETURN lvR; END;
    END;
  END;
  RETURN lvR;
END;

// A factor code as a label: whole codes as integers ("A=1", not
// "A=1.000"; second review item 9).
av_Lv(lbV)
BEGIN
  IF FP(lbV) == 0 THEN RETURN av_I(lbV); END;
  RETURN av_Gn(lbV);
END;

// The level number of a value.
av_LevOf(loV, loL)
BEGIN
  LOCAL loI;
  FOR loI FROM 1 TO SIZE(loL) DO
    IF loL(loI) == loV THEN RETURN loI; END;
  END;
  RETURN 0;
END;

// One-way from D1-D9: each non-empty list is a group.
av_MyOne()
BEGIN
  LOCAL moI, moL, moY, moG, moN, moS;
  LOCAL moJ, moK, moC;
  moY := {}; moN := {}; moS := {}; moC := {};
  FOR moI FROM 1 TO 9 DO
    moL := av_ReadD(moI);
    IF SIZE(moL) > 0 THEN
      moY := CONCAT(moY, moL);
      moC := CONCAT(moC, {SIZE(moL)});
      moN := CONCAT(moN, {"D" + av_I(moI)});
      moS := CONCAT(moS, {"D" + av_I(moI)});
    END;
  END;
  // the group of each value, filled in place (no growing list)
  moG := MAKELIST(0, moJ, 1, MAX(1, SIZE(moY)));
  moJ := 0;
  FOR moI FROM 1 TO SIZE(moC) DO
    FOR moK FROM 1 TO moC(moI) DO moJ := moJ + 1; moG(moJ) := moI; END;
  END;
  IF SIZE(moN) < 2 THEN
    MSGBOX("Put each group in its own list, D1 to D9, of the Statistics 1Var app: at least two lists.");
    RETURN 0;
  END;
  IF SIZE(moY) > 3000 THEN MSGBOX("More than 3000 values: too many for this program."); RETURN 0; END;
  avOY := moY; avOG := moG; avOK := SIZE(moN); avON := moN; avOSh := moS;
  avOTtl := "Your data (D1-D9)";
  avOSrc := "Statistics 1Var lists D1-D9";
  avOHas := 1; avODone := 0;
  IF av_OCompute() == 0 THEN RETURN 0; END;
  RETURN 1;
END;

// Two-way from D1 (y), D2 (A codes), D3 (B codes).
av_MyTwo()
BEGIN
  LOCAL mtY, mtA, mtB, mtLA, mtLB, mtI;
  LOCAL mtN, mtIA, mtIB;
  mtY := av_ReadD(1); mtA := av_ReadD(2); mtB := av_ReadD(3);
  mtN := SIZE(mtY);
  IF mtN > 3000 THEN MSGBOX("More than 3000 values: too many for this program."); RETURN 0; END;
  IF mtN < 4 OR SIZE(mtA) <> mtN OR SIZE(mtB) <> mtN THEN
    MSGBOX("Put y in D1, the A codes in D2 and the B codes in D3: three lists of the same length.");
    RETURN 0;
  END;
  mtLA := av_Levels(mtA); mtLB := av_Levels(mtB);
  IF SIZE(mtLA) < 2 OR SIZE(mtLB) < 2 OR SIZE(mtLA) > 6 OR SIZE(mtLB) > 6 THEN
    MSGBOX("Each factor needs 2 to 6 levels (D2 has " + av_I(SIZE(mtLA)) + ", D3 has " + av_I(SIZE(mtLB)) + ").");
    RETURN 0;
  END;
  mtIA := MAKELIST(0, mtI, 1, mtN); mtIB := MAKELIST(0, mtI, 1, mtN);
  avStop := 0;
  IF mtN > 500 THEN av_Busy("reading D1-D3", mtN); END;
  FOR mtI FROM 1 TO mtN DO
    mtIA(mtI) := av_LevOf(mtA(mtI), mtLA);
    mtIB(mtI) := av_LevOf(mtB(mtI), mtLB);
    IF mtN > 500 AND mtI MOD 100 == 0 THEN
      av_BusyStep(mtI, mtN);
      IF avStop == 1 THEN MSGBOX("Stopped (Esc): nothing was computed."); RETURN 0; END;
    END;
  END;
  avTY := mtY; avTA := mtIA; avTB := mtIB;
  avTNA := SIZE(mtLA); avTNB := SIZE(mtLB);
  avTAN := {}; avTBN := {};
  FOR mtI FROM 1 TO avTNA DO avTAN := CONCAT(avTAN, {"A=" + av_Lv(mtLA(mtI))}); END;
  FOR mtI FROM 1 TO avTNB DO avTBN := CONCAT(avTBN, {"B=" + av_Lv(mtLB(mtI))}); END;
  avTFA := "A (D2)"; avTFB := "B (D3)"; avTKA := "(D2)"; avTKB := "(D3)";
  avTTtl := "Your data: D1 by D2 " + avTm + " D3";
  avTSrc := "Statistics 1Var lists";
  avTHas := 1; avTDone := 0; avTSx := 1; avTSt := 1; avTEff := 3;
  IF av_TCompute() == 0 THEN RETURN 0; END;
  RETURN 1;
END;

// Blocks from D1-D9: each non-empty list a treatment, row i = block i.
av_MyBlock()
BEGIN
  LOCAL mbI, mbL, mbC, mbN, mbJ, mbY;
  mbC := {}; mbN := -1;
  FOR mbI FROM 1 TO 9 DO
    mbL := av_ReadD(mbI);
    IF SIZE(mbL) > 0 THEN
      IF mbN < 0 THEN mbN := SIZE(mbL); END;
      IF SIZE(mbL) <> mbN THEN
        MSGBOX("Every list must have the same length (one value per block): D" + av_I(mbI) + " differs.");
        RETURN 0;
      END;
      mbC := CONCAT(mbC, {mbI});
    END;
  END;
  IF SIZE(mbC) < 2 OR mbN < 2 THEN
    MSGBOX("Put each treatment in its own list (D1-D9), one row per block: at least 2 lists of 2 or more.");
    RETURN 0;
  END;
  IF mbN*SIZE(mbC) > 3000 THEN MSGBOX("More than 3000 values: too many for this program."); RETURN 0; END;
  // row-major: block j's values, treatment by treatment; each list is read once
  mbY := MAKELIST(0, mbJ, 1, mbN*SIZE(mbC));
  FOR mbI FROM 1 TO SIZE(mbC) DO
    mbL := av_ReadD(mbC(mbI));
    FOR mbJ FROM 1 TO mbN DO mbY((mbJ - 1)*SIZE(mbC) + mbI) := mbL(mbJ); END;
  END;
  avBY := mbY; avBNb := mbN; avBKt := SIZE(mbC);
  avBTN := {};
  FOR mbI FROM 1 TO avBKt DO avBTN := CONCAT(avBTN, {"D" + av_I(mbC(mbI))}); END;
  avBBl := "block"; avBTtl := "Your data (D1-D9)"; avBSrc := "Statistics 1Var lists";
  avBHas := 1; avBDone := 0; avBSel := 1;
  IF av_BCompute() == 0 THEN RETURN 0; END;
  RETURN 1;
END;
